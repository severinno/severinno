    83→    83→// ---------------------------------------------------------------------------
    84→    84→// Form model — extends the API payload with per-item providerId
    85→    85→// (the API needs providerId per item, plus an outer providerId for the
    86→    86→//  primary provider — we mirror this by sending the first item's provider
    87→    87→//  as the outer providerId).
    88→    88→// ---------------------------------------------------------------------------
    89→    89→
    90→    90→type QuoteItemForm = {
    91→    91→  providerId: string
    92→    92→  serviceId: string
    93→    93→  description: string
    94→    94→  quantity: number
    95→    95→  unit: ServiceUnit
    96→    96→  photos: string[]
    97→    97→}
    98→    98→
    99→    99→const quoteFormSchema = z.object({
   100→   100→  items: z
   101→   101→    .array(
   102→   102→      z.object({
   103→   103→        providerId: z.string().min(1, "Selecione um prestador"),
   104→   104→        serviceId: z.string().min(1, "Selecione um serviço"),
   105→   105→        description: z
   106→   106→          .string()
   107→   107→          .min(10, "Descreva com ao menos 10 caracteres")
   108→   108→          .max(400),
   109→   109→        quantity: z.coerce.number().min(0.01, "Quantidade inválida"),
   110→   110→        unit: z.enum([
   111→   111→          "UNIDADE",
   112→   112→          "METRO_LINEAR",
   113→   113→          "METRO_QUADRADO",
   114→   114→          "METRO_CUBICO",
   115→   115→        ]),
   116→   116→        photos: z.array(z.string()).max(4).default([]),
   117→   117→      }),
   118→   118→    )
   119→   119→    .min(1, "Adicione ao menos um item"),
   120→   120→  address: z.object({
   121→   121→    cep: z.string().min(8, "CEP inválido"),
   122→   122→    street: z.string().min(3, "Informe a rua"),
   123→   123→    number: z.string().min(1, "Informe o número"),
   124→   124→    complement: z.string().default(""),
   125→   125→    district: z.string().default(""),
   126→   126→    city: z.string().min(2, "Informe a cidade"),
   127→   127→    state: z.string().min(2, "UF"),
   128→   128→    lat: z.number().nullable().optional(),
   129→   129→    lng: z.number().nullable().optional(),
   130→   130→  }),
   131→   131→})
   132→   132→
   133→   133→type QuoteFormValues = z.infer<typeof quoteFormSchema>
   134→   134→

   144→   144→  unit,

   160→   160→export function QuoteModal() {
   161→   161→  const open = useUIStore((s) => s.quoteModal.open)
   162→   162→  const providerIdPreset = useUIStore((s) => s.quoteModal.providerId)
   163→   163→  const serviceIdPreset = useUIStore((s) => s.quoteModal.serviceId)
   164→   164→  const close = useUIStore((s) => s.closeQuote)
   165→   165→  const openAuth = useUIStore((s) => s.openAuth)
   166→   166→  const isMobile = useIsMobile()
   167→   167→
   168→   168→  const navigate = useViewStore((s) => s.navigate)
   169→   169→  const user = useAuthStore((s) => s.user)
   170→   170→
   171→   171→  const form = useForm<QuoteFormValues>({
   172→   172→    // Cast around Zod 4's `z.coerce.number().optional()` typing (widens
   173→   173→    // input to `unknown`, breaks Resolver inference).
   174→   174→    resolver: zodResolver(quoteFormSchema) as unknown as Resolver<QuoteFormValues>,
   175→   175→    defaultValues: {
   176→   176→      items: [emptyItem(providerIdPreset, serviceIdPreset)],
   177→   177→      address: emptyAddress,
   178→   178→    },
   179→   179→    mode: "onTouched",
   180→   180→  })
   181→   181→
   182→   182→  const { fields, append, remove } = useFieldArray({
   183→   183→    control: form.control,
   184→   184→    name: "items",
   185→   185→  })
   186→   186→
   187→   187→  // Reset preset when modal opens.
   188→   188→  React.useEffect(() => {
   189→   189→    if (open) {
   190→   190→      form.reset({
   191→   191→        items: [emptyItem(providerIdPreset, serviceIdPreset)],
   192→   192→        address: emptyAddress,
   193→   193→      })
   194→   194→    }
   195→   195→  }, [open, providerIdPreset, serviceIdPreset, form])
   196→   196→
   197→   197→  const [submitting, setSubmitting] = React.useState(false)
   198→   198→
   199→   199→  const onSubmit = async (values: QuoteFormValues) => {
   200→   200→    // Prevention heuristic — must be authenticated.
   201→   201→    if (!user) {
   202→   202→      toast.info("Faça cadastro gratuito para pedir orçamentos.")
   203→   203→      openAuth("register", "CLIENT")
   204→   204→      return
   205→   205→    }
   206→   206→
   207→   207→    // First item's provider = outer providerId (legacy API contract).
   208→   208→    const primary = values.items[0]
   209→   209→    if (!primary?.providerId) {
   210→   210→      toast.error("Selecione um prestador no primeiro item.")
   211→   211→      return
   212→   212→    }
   213→   213→
   214→   214→    setSubmitting(true)
   215→   215→    try {
   216→   216→      await apiPost("/api/quotes", {
   217→   217→        providerId: primary.providerId,
   218→   218→        items: values.items.map((it) => ({
   219→   219→          providerId: it.providerId,
   220→   220→          serviceId: it.serviceId,
   221→   221→          description: it.description,
   222→   222→          quantity: it.quantity,
   223→   223→          unit: it.unit,
   224→   224→          photos: it.photos,
   225→   225→        })),
   226→   226→        address: [
   227→   227→          values.address.street,
   228→   228→          values.address.number,
   229→   229→          values.address.complement,
   230→   230→          values.address.district,
   231→   231→        ]
   232→   232→          .filter(Boolean)
   233→   233→          .join(", "),
   234→   234→        cep: values.address.cep,
   235→   235→        lat: values.address.lat,
   236→   236→        lng: values.address.lng,
   237→   237→      })
   238→   238→      toast.success("Orçamento enviado! O prestador responderá em breve.")
   239→   239→      close()
   240→   240→      navigate("client.quotes")
   241→   241→    } catch (e) {
   242→   242→      const msg =
   243→   243→        (e as { message?: string })?.message ??
   244→   244→        "Não foi possível enviar o orçamento. Tente novamente."
   245→   245→      toast.error(msg)
   246→   246→    } finally {
   247→   247→      setSubmitting(false)
   248→   248→    }
   249→   249→  }
   250→   250→
   251→   251→  const content = (
   252→   252→    <FormBody
   253→   253→      form={form}
   254→   254→      fields={fields}
   255→   255→      append={append}
   256→   256→      remove={remove}
   257→   257→      submitting={submitting}
   258→   258→      onSubmit={form.handleSubmit(onSubmit)}
   259→   259→    />
   260→   260→  )
   261→   261→
   262→   262→  if (isMobile) {
   263→   263→    return (
   264→   264→      <Sheet open={open} onOpenChange={(o) => !o && close()}>
   265→   265→        <SheetContent
   266→   266→          side="bottom"
   267→   267→          className="h-[100dvh] max-h-[100dvh] w-full p-0 sm:max-w-full gap-0 flex flex-col"
   268→   268→        >
   269→   269→          <SheetHeader className="px-4 pt-4 pb-2 shrink-0">
   270→   270→            <SheetTitle className="flex items-center gap-2">
   271→   271→              <Wrench className="size-5 text-emerald-600" />
   272→   272→              Pedir orçamento
   273→   273→            </SheetTitle>
   274→   274→            <SheetDescription>
   275→   275→              Descreva o serviço e receba propostas de prestadores verificados.
   276→   276→            </SheetDescription>
   277→   277→          </SheetHeader>
   278→   278→          <div className="flex-1 overflow-hidden">{content}</div>
   279→   279→        </SheetContent>
   280→   280→      </Sheet>
   281→   281→    )
   282→   282→  }
   283→   283→
   284→   284→  return (
   285→   285→    <Dialog open={open} onOpenChange={(o) => !o && close()}>
   286→   286→      <DialogContent className="sm:max-w-2xl p-0 gap-0 overflow-hidden">
   287→   287→        <DialogHeader className="px-6 pt-6 pb-3 shrink-0 border-b">
   288→   288→          <DialogTitle className="flex items-center gap-2 text-lg">
   289→   289→            <Wrench className="size-5 text-emerald-600" />
   290→   290→            Pedir orçamento
   291→   291→          </DialogTitle>
   292→   292→          <DialogDescription>
   293→   293→            Solicite orçamentos de um ou mais serviços.
   294→   294→          </DialogDescription>
   295→   295→        </DialogHeader>
   296→   296→        {content}
   297→   297→      </DialogContent>
   298→   298→    </Dialog>
   299→   299→  )
   300→   300→}

   488→   488→  index,

   506→   506→        </span>
   507→   507→        {canRemove && (
   508→   508→          <button
   509→   509→            type="button"

   646→   646→function ProviderCombobox({
   647→   647→  value,
   648→   648→  onChange,
   649→   649→  error,
   650→   650→}: {
   651→   651→  value: string
   652→   652→  onChange: (providerId: string) => void
   653→   653→  error?: string
   654→   654→}) {
   655→   655→  const [open, setOpen] = React.useState(false)

   657→   657→
   658→   658→  const { data, isLoading } = useQuery({

   764→   764→function ServiceSelect({
   765→   765→  providerId,
   766→   766→  value,
   767→   767→  onChange,
   768→   768→  error,
   769→   769→}: {
   770→   770→  providerId: string
   771→   771→  value: string
   772→   772→  onChange: (service: ProviderService) => void
   773→   773→  error?: string
   774→   774→}) {
   775→   775→  const { data, isLoading } = useQuery({
   776→   776→    queryKey: ["services-by-provider", providerId],
   777→   777→    queryFn: () =>
   778→   778→      apiGet<ProviderService[]>("/api/services", { providerId }),
   779→   779→    enabled: !!providerId,
   780→   780→    staleTime: 30 * 1000,
   781→   781→  })
   782→   782→
   783→   783→  const services = data ?? []
   784→   784→  const selected = services.find((s) => s.id === value)
   785→   785→
   786→   786→  return (
   787→   787→    <div className="grid gap-1.5">
   788→   788→      <Label>Serviço</Label>
   789→   789→      <Select
   790→   790→        value={value}
   791→   791→        onValueChange={(v) => {
   792→   792→          const s = services.find((x) => x.id === v)
   793→   793→          if (s) onChange(s)
   794→   794→        }}
   795→   795→        disabled={!providerId}
   796→   796→      >
   797→   797→        <SelectTrigger className="w-full">
   798→   798→          <SelectValue
   799→   799→            placeholder={
   800→   800→              !providerId
   801→   801→                ? "Selecione um prestador primeiro"
   802→   802→                : isLoading
   803→   803→                  ? "Carregando..."
   804→   804→                  : "Selecionar serviço"
   805→   805→            }
   806→   806→          />
   807→   807→        </SelectTrigger>
   808→   808→        <SelectContent>
   809→   809→          {services.length === 0 && !isLoading && (
   810→   810→            <div className="px-2 py-3 text-center text-xs text-muted-foreground">
   811→   811→              Nenhum serviço disponível.
   812→   812→            </div>
   813→   813→          )}
   814→   814→          {services.map((s) => (
   815→   815→            <SelectItem key={s.id} value={s.id}>
   816→   816→              <span className="flex items-center justify-between gap-2 w-full">
   817→   817→                <span className="truncate">{s.title}</span>
   818→   818→                <span className="text-xs text-muted-foreground tabular-nums">
   819→   819→                  {s.basePrice.toLocaleString("pt-BR", {
   820→   820→                    style: "currency",
   821→   821→                    currency: "BRL",
   822→   822→                  })}
   823→   823→                  /{SERVICE_UNIT_SHORT[s.unit]}
   824→   824→                </span>
   825→   825→              </span>
   826→   826→            </SelectItem>
   827→   827→          ))}
   828→   828→        </SelectContent>
   829→   829→      </Select>
   830→   830→      {selected && (
   831→   831→        <p className="text-xs text-muted-foreground">
   832→   832→          {selected.description ?? "—"}
   833→   833→        </p>
   834→   834→      )}
   835→   835→      {error && <p className="text-xs text-destructive">{error}</p>}
   836→   836→    </div>
   837→   837→  )
   838→   838→}
