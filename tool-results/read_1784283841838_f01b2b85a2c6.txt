   148→function initials(name?: string | null): string {
   149→  if (!name) return "?"
   150→  return name
   151→    .split(" ")
   152→    .map((p) => p[0])
   153→    .filter(Boolean)
   154→    .slice(0, 2)
   155→    .join("")
   156→    .toUpperCase()
   157→}

   661→function NotificationsBell({
   662→  items,
   663→  unreadCount,
   664→  isLoading,
   665→  onMarkRead,
   666→  onMarkAllRead,
   667→  markingAll,
   668→}: {
   669→  items: NotificationItem[]
   670→  unreadCount: number
   671→  isLoading: boolean
   672→  onMarkRead: (id: string) => void
   673→  onMarkAllRead: (ids: string[]) => void
   674→  markingAll: boolean
   675→}) {
   676→  const unreadIds = React.useMemo(
   677→    () => items.filter((n) => !n.read).map((n) => n.id),
   678→    [items],
   679→  )
   680→  return (
   681→    <DropdownMenu>
   682→      <DropdownMenuTrigger asChild>
   683→        <Button
   684→          variant="ghost"
   685→          size="icon"
   686→          className="relative size-9 text-muted-foreground hover:text-foreground"
   687→          aria-label={`Notificações${
   688→            unreadCount > 0 ? ` (${unreadCount} não lidas)` : ""
   689→          }`}
   690→        >
   691→          <Bell className="size-[18px]" />
   692→          {unreadCount > 0 ? (
   693→            <span
   694→              className="absolute -right-0.5 -top-0.5 inline-flex h-4 min-w-4 items-center justify-center rounded-full bg-primary px-1 text-[10px] font-semibold text-primary-foreground ring-2 ring-background"
   695→              aria-hidden
   696→            >
   697→              {unreadCount > 9 ? "9+" : unreadCount}
   698→            </span>
   699→          ) : null}
   700→        </Button>
   701→      </DropdownMenuTrigger>
   702→      <DropdownMenuContent
   703→        align="end"
   704→        className="w-80 gap-0 p-0"
   705→      >
   706→        <div className="flex items-center justify-between border-b px-3 py-2.5">
   707→          <div className="flex items-center gap-2">
   708→            <p className="text-sm font-semibold">Notificações</p>
   709→            {unreadCount > 0 ? (
   710→              <Badge
   711→                variant="secondary"
   712→                className="h-5 bg-primary/10 px-1.5 text-[10px] font-semibold text-primary"
   713→              >
   714→                {unreadCount} nova{unreadCount > 1 ? "s" : ""}
   715→              </Badge>
   716→            ) : null}
   717→          </div>
   718→          {unreadCount > 0 ? (
   719→            <button
   720→              type="button"
   721→              onClick={() => onMarkAllRead(unreadIds)}
   722→              disabled={markingAll}
   723→              className="inline-flex items-center gap-1 text-[11px] font-medium text-primary transition-colors hover:text-primary/80 disabled:opacity-50"
   724→            >

   726→                <Loader2 className="size-3 animate-spin" />
   727→              ) : (
   728→                <CheckCheck className="size-3.5" />
   729→              )}
   730→              Marcar todas
   731→            </button>
   732→          ) : null}
   733→        </div>
   734→
   735→        <ScrollArea className="max-h-80">
   736→          {isLoading ? (
   737→            <div className="flex items-center justify-center gap-2 p-8 text-sm text-muted-foreground">
   738→              <Loader2 className="size-4 animate-spin" />
   739→              Carregando…
   740→            </div>
   741→          ) : items.length === 0 ? (
   742→            <div className="flex flex-col items-center gap-2 p-8 text-center">
   743→              <span className="flex size-12 items-center justify-center rounded-full bg-muted text-muted-foreground">
   744→                <Bell className="size-5" />
   745→              </span>
   746→              <p className="text-sm text-muted-foreground">
   747→                Você não tem notificações.
   748→              </p>
   749→            </div>
   750→          ) : (
   751→            <ul className="divide-y">
   752→              {items.map((n) => {
   753→                const typeLabel = NOTIFICATION_TYPE_LABELS[n.type]
   754→                return (
   755→                  <li
   756→                    key={n.id}
   757→                    className={cn(
   758→                      "relative flex gap-3 px-3 py-2.5 transition-colors hover:bg-accent/50",
   759→                      !n.read && "bg-primary/5",
   760→                    )}
   761→                  >
   762→                    <span
   763→                      className={cn(
   764→                        "mt-1.5 size-2 shrink-0 rounded-full",
   765→                        n.read ? "bg-transparent ring-1 ring-border" : "bg-primary",
   766→                      )}
   767→                      aria-hidden
   768→                    />
   769→                    <div className="min-w-0 flex-1">
   770→                      <div className="flex items-baseline justify-between gap-2">
   771→                        <p className="text-sm font-medium leading-tight">
   772→                          {n.title}
   773→                        </p>
   774→                        <span className="shrink-0 text-[10px] text-muted-foreground tabular-nums">
   775→                          {formatRelative(n.createdAt)}
   776→                        </span>
   777→                      </div>
   778→                      {n.body ? (
   779→                        <p className="mt-0.5 line-clamp-2 text-xs text-muted-foreground">
   780→                          {n.body}
   781→                        </p>
   782→                      ) : null}
   783→                      <div className="mt-1 flex items-center gap-2">
   784→                        {typeLabel ? (
   785→                          <Badge
   786→                            variant="outline"
   787→                            className="h-4 px-1.5 text-[10px] font-medium text-muted-foreground"
   788→                          >
   789→                            {typeLabel}
   790→                          </Badge>
   791→                        ) : null}
   792→                        {!n.read ? (
   793→                          <button
   794→                            type="button"
   795→                            onClick={() => onMarkRead(n.id)}
   796→                            className="inline-flex items-center gap-1 text-[10px] font-medium text-primary hover:underline"
   797→                          >
   798→                            <Check className="size-3" />
   799→                            Marcar como lida
   800→                          </button>
   801→                        ) : null}
   802→                      </div>
   803→                    </div>
   804→                  </li>
   805→                )
   806→              })}
   807→            </ul>
   808→          )}
   809→        </ScrollArea>
   810→      </DropdownMenuContent>
   811→    </DropdownMenu>
   812→  )
   813→}
