     3→/**
     4→ * AdminUsers — manage all personas (Clients, Providers, Admins).
     5→ *
     6→ * Data source: GET  /api/admin/users?role=&q=&page=
     7→ *              PATCH /api/admin/users/[id] { verified?, active?, role?, name?, ... }
     8→ *              DELETE /api/admin/users/[id]
     9→ *
    10→ * Nielsen heuristics covered (design system em admin-shared.tsx):
    11→ *   H1 — TableSkeleton / ErrorState / SavingPill inline (visibilidade de status)
    12→ *   H2 — "Filtro aplicado à página atual" (honestidade sobre filtro client-side)
    13→ *   H4 — RoleBadge / ActiveBadge / VerifiedBadge vindos do admin-shared
    14→ *   H5 — Switch + itens do ⋮ abrem ConfirmToggleDialog antes de aplicar
    15→ *   H5 — EditUserDialog mostra Alert amber ao rebaixar ADMIN (ação destrutiva)
    16→ *   H6 — Botão "Editar" visível na linha; ⋮ apenas para ações secundárias/destrutivas
    17→ *   H7 — Ordenação client-side por nome e data de cadastro
    18→ *   H8 — EmptyState / TableSkeleton / layout limpo sem poluição
    19→ *   H9 — Error banner dismissível + toast.error específico + ErrorState com retry
    20→ *   H10— Tooltips em todos os botões de ícone (⋮, Editar, Switch)
    21→ *
    22→ * Note: os filtros `verified` e `active` são aplicados client-side na página
    23→ * atual (a API ainda não suporta esses query params). Um hint amber "Filtro
    24→ * aplicado à página atual" é exibido quando ativo (H2 — honestidade).
    25→ *
    26→ * O tipo `AdminUser` é definido de forma IDÊNTICA em admin-providers.tsx (H4).
    27→ * Se alterar campos aqui, replique lá.
    28→ */

   120→// ---------------------------------------------------------------------------
   121→// Types — definidos IDÊNTICOS em admin-providers.tsx (H4 consistência).
   122→// ---------------------------------------------------------------------------
   123→type AdminUser = {
   124→  id: string
   125→  name: string
   126→  email: string
   127→  role: UserRole
   128→  cpfCnpj?: string | null
   129→  whatsapp?: string | null
   130→  phone?: string | null
   131→  avatarUrl?: string | null
   132→  city?: string | null
   133→  state?: string | null
   134→  bio?: string | null
   135→  verified: boolean
   136→  active: boolean
   137→  createdAt: string
   138→}
   139→
   140→type AdminUsersResponse = {
   141→  items: AdminUser[]
   142→  total: number
   143→  page: number
   144→  limit: number
   145→}
   146→
   147→type StatsResponse = {
   148→  usersByRole: Record<string, number>
   149→}
   150→
   151→type RoleFilter = "ALL" | UserRole
   152→type VerifiedFilter = "ALL" | "true" | "false"
   153→type ActiveFilter = "ALL" | "true" | "false"
   154→
   155→type SortKey = "name" | "createdAt"
   156→type SortDir = "asc" | "desc"
   157→type SortState = { key: SortKey; dir: SortDir } | null
   158→
   159→type PendingToggle = {
   160→  id: string
   161→  name: string
   162→  field: "verified" | "active"
   163→  currentValue: boolean
   164→} | null

   166→// ---------------------------------------------------------------------------
   167→// Main component
   168→// ---------------------------------------------------------------------------
   169→export function AdminUsers() {
   170→  const queryClient = useQueryClient()
   171→  const [role, setRole] = React.useState<RoleFilter>("ALL")
   172→  const [q, setQ] = React.useState("")
   173→  const [debouncedQ, setDebouncedQ] = React.useState("")
   174→  const [verified, setVerified] = React.useState<VerifiedFilter>("ALL")
   175→  const [active, setActive] = React.useState<ActiveFilter>("ALL")
   176→  const [page, setPage] = React.useState(1)
   177→  const [sort, setSort] = React.useState<SortState>(null)
   178→
   179→  const [editTarget, setEditTarget] = React.useState<AdminUser | null>(null)
   180→  const [deleteTarget, setDeleteTarget] = React.useState<AdminUser | null>(null)
   181→  const [pendingToggle, setPendingToggle] = React.useState<PendingToggle>(null)
   182→  const [patchingId, setPatchingId] = React.useState<string | null>(null)
   183→  const [errorBanner, setErrorBanner] = React.useState<string | null>(null)
   184→
   185→  const limit = 10
   186→
   187→  // Debounce search
   188→  React.useEffect(() => {
   189→    const t = setTimeout(() => {
   190→      setDebouncedQ(q.trim())
   191→      setPage(1)
   192→    }, 350)
   193→    return () => clearTimeout(t)
   194→  }, [q])
   195→
   196→  // Role counts (cached 60s, compartilhado com dashboard)
   197→  const { data: stats } = useQuery({
   198→    queryKey: ["admin", "stats"],
   199→    queryFn: () => apiGet<StatsResponse>("/api/admin/stats"),
   200→    staleTime: 60_000,
   201→  })
   202→
   203→  const { data, isLoading, isError, refetch } = useQuery({
   204→    queryKey: ["admin", "users", { role, debouncedQ, verified, active, page, limit }],
   205→    queryFn: () =>
   206→      apiGet<AdminUsersResponse>("/api/admin/users", {
   207→        ...(role !== "ALL" ? { role } : {}),
   208→        ...(debouncedQ ? { q: debouncedQ } : {}),
   209→        page,
   210→        limit,
   211→      }),
   212→    staleTime: 15_000,
   213→  })
   214→
   215→  const invalidate = () =>
   216→    queryClient.invalidateQueries({ queryKey: ["admin", "users"] })
   217→
   218→  const patchMutation = useMutation({
   219→    mutationFn: ({
   220→      id,
   221→      patch,
   222→    }: {
   223→      id: string
   224→      patch: Partial<AdminUser>
   225→    }) => apiPatch<{ user: AdminUser }>(`/api/admin/users/${id}`, patch),
   226→  })
   227→
   228→  const deleteMutation = useMutation({
   229→    mutationFn: (id: string) => apiDelete(`/api/admin/users/${id}`),
   230→  })
   231→
   232→  // Client-side filter for verified/active (API does not support these yet — H2)
   233→  const rawItems = data?.items ?? []
   234→  const filteredItems = React.useMemo(() => {
   235→    return rawItems.filter((u) => {
   236→      if (verified === "true" && !u.verified) return false
   237→      if (verified === "false" && u.verified) return false
   238→      if (active === "true" && !u.active) return false
   239→      if (active === "false" && u.active) return false
   240→      return true
   241→    })
   242→  }, [rawItems, verified, active])
   243→
   244→  const items = React.useMemo(() => {
   245→    if (!sort) return filteredItems
   246→    const sorted = [...filteredItems].sort((a, b) => {
   247→      let cmp = 0
   248→      if (sort.key === "name") cmp = a.name.localeCompare(b.name, "pt-BR")
   249→      else if (sort.key === "createdAt")
   250→        cmp = new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime()
   251→      return sort.dir === "asc" ? cmp : -cmp
   252→    })
   253→    return sorted
   254→  }, [filteredItems, sort])
   255→
   256→  const total = data?.total ?? 0
   257→  const totalPages = Math.max(1, Math.ceil(total / limit))
   258→
   259→  const clientFilterActive = verified !== "ALL" || active !== "ALL"
   260→
   261→  const activeFilterCount =
   262→    (role !== "ALL" ? 1 : 0) +
   263→    (debouncedQ ? 1 : 0) +
   264→    (verified !== "ALL" ? 1 : 0) +
   265→    (active !== "ALL" ? 1 : 0)
   266→
   267→  const clearFilters = () => {
   268→    setRole("ALL")
   269→    setQ("")
   270→    setDebouncedQ("")
   271→    setVerified("ALL")
   272→    setActive("ALL")
   273→    setSort(null)
   274→    setPage(1)
   275→  }
   276→
   277→  const toggleSort = (key: SortKey) => {
   278→    setSort((prev) => {
   279→      if (!prev || prev.key !== key) return { key, dir: "asc" }
   280→      if (prev.dir === "asc") return { key, dir: "desc" }
   281→      return null
   282→    })
   283→  }
   284→
   285→  const roleCounts = React.useMemo(() => {
   286→    const byRole = stats?.usersByRole ?? {}
   287→    return {
   288→      ALL:
   289→        (byRole.CLIENT ?? 0) + (byRole.PROVIDER ?? 0) + (byRole.ADMIN ?? 0),
   290→      CLIENT: byRole.CLIENT ?? 0,
   291→      PROVIDER: byRole.PROVIDER ?? 0,
   292→      ADMIN: byRole.ADMIN ?? 0,
   293→    }
   294→  }, [stats])
   295→
   296→  // ---- Mutation handlers ---------------------------------------------------
   297→
   298→  const handleToggleConfirm = () => {
   299→    if (!pendingToggle) return
   300→    const { id, field, currentValue } = pendingToggle
   301→    setPatchingId(id)
   302→    patchMutation.mutate(
   303→      { id, patch: { [field]: !currentValue } as Partial<AdminUser> },
   304→      {
   305→        onSuccess: () => {
   306→          invalidate()
   307→          toast.success(
   308→            field === "verified"
   309→              ? currentValue
   310→                ? "Verificação removida."
   311→                : "Usuário marcado como verificado."
   312→              : currentValue
   313→              ? "Usuário desativado."
   314→              : "Usuário ativado.",
   315→          )
   316→          setPendingToggle(null)
   317→          setPatchingId(null)
   318→        },
   319→        onError: (e: unknown) => {
   320→          const msg = errMsg(e, "Falha ao atualizar usuário.")
   321→          setErrorBanner(msg)
   322→          toast.error(msg)
   323→          setPendingToggle(null)
   324→          setPatchingId(null)
   325→        },
   326→      },
   327→    )
   328→  }
   329→
   330→  const handleEditSubmit = (patch: Partial<AdminUser>) => {
   331→    if (!editTarget) return
   332→    patchMutation.mutate(
   333→      { id: editTarget.id, patch },
   334→      {
   335→        onSuccess: () => {
   336→          invalidate()
   337→          toast.success("Usuário atualizado.")
   338→          setEditTarget(null)
   339→        },
   340→        onError: (e: unknown) => {
   341→          const msg = errMsg(e, "Falha ao atualizar usuário.")
   342→          setErrorBanner(msg)
   343→          toast.error(msg)
   344→        },
   345→      },
   346→    )
   347→  }
   348→
   349→  const handleDeleteConfirm = () => {
   350→    if (!deleteTarget) return
   351→    deleteMutation.mutate(deleteTarget.id, {
   352→      onSuccess: () => {
   353→        toast.success("Usuário excluído.")
   354→        invalidate()
   355→        setDeleteTarget(null)
   356→      },
   357→      onError: (e: unknown) => {
   358→        const msg = errMsg(e, "Não foi possível excluir o usuário.")
   359→        setErrorBanner(msg)
   360→        toast.error(msg)
   361→        setDeleteTarget(null)
   362→      },
   363→    })
   364→  }
   365→
   366→  const renderSortHeader = (label: string, sortKey: SortKey) => (
   367→    <button
   368→      type="button"
   369→      onClick={() => toggleSort(sortKey)}
   370→      className="inline-flex items-center gap-1 text-xs font-semibold uppercase tracking-wide text-muted-foreground transition-colors hover:text-foreground"
   371→    >
   372→      {label}
   373→      {sort?.key === sortKey ? (
   374→        sort.dir === "asc" ? (
   375→          <ChevronUp className="size-3" />
   376→        ) : (
   377→          <ChevronDown className="size-3" />
   378→        )
   379→      ) : (
   380→        <ArrowUpDown className="size-3 opacity-40" />
   381→      )}
   382→    </button>
   383→  )
   384→
   385→  return (
   386→    <div className="flex flex-col gap-4">
   387→      <PageSectionHeader
   388→        title="Usuários"
   389→        description="Gerencie clientes, prestadores e administradores da plataforma."
   390→      />
   391→
   392→      {/* Role tabs with counts */}
   393→      <Tabs
   394→        value={role}
   395→        onValueChange={(v) => {
   396→          setRole(v as RoleFilter)
   397→          setPage(1)
   398→        }}
   399→      >
   400→        <TabsList className="h-auto flex-wrap gap-1 bg-card p-1 shadow-sm">
   401→          {(
   402→            [
   403→              { value: "ALL", label: "Todos", icon: Users },
   404→              { value: "CLIENT", label: "Clientes", icon: CircleUser },
   405→              { value: "PROVIDER", label: "Prestadores", icon: HardHat },
   406→              { value: "ADMIN", label: "Administradores", icon: ShieldCheck },
   407→            ] as const
   408→          ).map((t) => (
   409→            <TabsTrigger
   410→              key={t.value}
   411→              value={t.value}
   412→              className="h-8 gap-1.5 rounded-md px-3 text-sm data-[state=active]:bg-primary data-[state=active]:text-primary-foreground"
   413→            >
   414→              <t.icon className="size-3.5" />
   415→              {t.label}
   416→              <span
   417→                className={cn(
   418→                  "ml-1 inline-flex h-5 min-w-5 items-center justify-center rounded-full px-1.5 text-[10px] font-semibold tabular-nums",
   419→                  role === t.value
   420→                    ? "bg-primary-foreground/20 text-primary-foreground"
   421→                    : "bg-muted text-muted-foreground",
   422→                )}
   423→              >
   424→                {roleCounts[t.value]}
   425→              </span>
   426→            </TabsTrigger>
   427→          ))}
   428→        </TabsList>
   429→      </Tabs>
   430→
   431→      {/* Error banner (dismissible) — H9 */}
   432→      {errorBanner ? (
   433→        <Alert variant="destructive">
   434→          <AlertTriangle className="size-4" />
   435→          <AlertTitle>Erro ao salvar</AlertTitle>
   436→          <AlertDescription>{errorBanner}</AlertDescription>
   437→          <button
   438→            type="button"
   439→            onClick={() => setErrorBanner(null)}
   440→            aria-label="Dispensar aviso"
   441→            className="absolute right-3 top-3 rounded-md p-1 text-current/70 transition-colors hover:text-current"
   442→          >
   443→            <X className="size-3.5" />
   444→          </button>
   445→        </Alert>
   446→      ) : null}
   447→
   448→      {/* Filter bar */}
   449→      <FilterBar
   450→        onClear={clearFilters}
   451→        activeCount={activeFilterCount}
   452→      >
   453→        <SearchInput
   454→          value={q}
   455→          onChange={setQ}
   456→          placeholder="Buscar por nome, e-mail ou cidade"
   457→          className="min-w-[200px] flex-1"
   458→        />
   459→        <Select
   460→          value={verified}
   461→          onValueChange={(v) => {
   462→            setVerified(v as VerifiedFilter)
   463→            setPage(1)
   464→          }}
   465→        >
   466→          <SelectTrigger className="h-9 w-auto min-w-[150px]">
   467→            <SelectValue placeholder="Verificação" />
   468→          </SelectTrigger>
   469→          <SelectContent>
   470→            <SelectItem value="ALL">Verificação: todas</SelectItem>
   471→            <SelectItem value="true">Apenas verificados</SelectItem>
   472→            <SelectItem value="false">Aguardando verificação</SelectItem>
   473→          </SelectContent>
   474→        </Select>
   475→        <Select
   476→          value={active}
   477→          onValueChange={(v) => {
   478→            setActive(v as ActiveFilter)
   479→            setPage(1)
   480→          }}
   481→        >
   482→          <SelectTrigger className="h-9 w-auto min-w-[140px]">
   483→            <SelectValue placeholder="Status" />
   484→          </SelectTrigger>
   485→          <SelectContent>
   486→            <SelectItem value="ALL">Status: todos</SelectItem>
   487→            <SelectItem value="true">Apenas ativos</SelectItem>
   488→            <SelectItem value="false">Apenas inativos</SelectItem>
   489→          </SelectContent>
   490→        </Select>
   491→      </FilterBar>
   492→
   493→      {/* Honest about client-side filter — H2 */}
   494→      {clientFilterActive ? (
   495→        <span className="-mt-2 inline-flex items-center gap-1 text-[11px] text-amber-700 dark:text-amber-300">
   496→          <ShieldQuestion className="size-3" />
   497→          Filtro de verificação/status aplicado apenas à página atual
   498→        </span>
   499→      ) : null}
   500→
   501→      {/* Table area: error / loading / empty / table */}
   502→      {isError ? (
   503→        <ErrorState
   504→          title="Não foi possível carregar os usuários"
   505→          description="Verifique sua conexão e tente novamente."
   506→          onRetry={() => refetch()}
   507→        />
   508→      ) : isLoading ? (
   509→        <TableSkeleton rows={8} cols={7} />
   510→      ) : items.length === 0 ? (
   511→        <EmptyState
   512→          icon={SearchX}
   513→          title="Nenhum usuário encontrado"
   514→          description="Ajuste os filtros de busca ou cadastre um novo usuário."
   515→          action={
   516→            activeFilterCount > 0 ? (
   517→              <Button
   518→                variant="outline"
   519→                size="sm"
   520→                onClick={clearFilters}
   521→                className="gap-1.5"
   522→              >
   523→                <X className="size-3.5" />
   524→                Limpar filtros
   525→              </Button>
   526→            ) : undefined
   527→          }
   528→        />
   529→      ) : (
   530→        <Card className="overflow-hidden">
   531→          <CardContent className="p-0">
   532→            <div className="overflow-x-auto">
   533→              <Table>
   534→                <TableHeader>
   535→                  <TableRow className="h-11 bg-muted/50 hover:bg-muted/50">
   536→                    <TableHead>{renderSortHeader("Usuário", "name")}</TableHead>
   537→                    <TableHead className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   538→                      Perfil
   539→                    </TableHead>
   540→                    <TableHead className="hidden text-xs font-semibold uppercase tracking-wide text-muted-foreground md:table-cell">
   541→                      Contato
   542→                    </TableHead>
   543→                    <TableHead className="hidden text-xs font-semibold uppercase tracking-wide text-muted-foreground lg:table-cell">
   544→                      Cidade
   545→                    </TableHead>
   546→                    <TableHead className="text-center text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   547→                      Verificado
   548→                    </TableHead>
   549→                    <TableHead className="text-center text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   550→                      Status
   551→                    </TableHead>
   552→                    <TableHead className="hidden sm:table-cell">
   553→                      {renderSortHeader("Criado em", "createdAt")}
   554→                    </TableHead>
   555→                    <TableHead className="text-right text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   556→                      Ações
   557→                    </TableHead>
   558→                  </TableRow>
   559→                </TableHeader>
   560→                <TableBody>
   561→                  {items.map((u) => {
   562→                    const isPatchingThis =
   563→                      patchingId === u.id && patchMutation.isPending
   564→                    const patchingField = isPatchingThis
   565→                      ? pendingToggle?.field ?? null
   566→                      : null
   567→                    return (
   568→                      <TableRow
   569→                        key={u.id}
   570→                        className="h-14 border-b transition-colors last:border-0 hover:bg-muted/30"
   571→                      >
   572→                        <TableCell className="px-4 py-3">
   573→                          <div className="flex items-center gap-2.5">
   574→                            <Avatar className="size-9 shrink-0">
   575→                              {u.avatarUrl ? (
   576→                                <AvatarImage src={u.avatarUrl} alt={u.name} />
   577→                              ) : null}
   578→                              <AvatarFallback className="bg-primary/10 text-[11px] font-semibold text-primary">
   579→                                {initials(u.name)}
   580→                              </AvatarFallback>
   581→                            </Avatar>
   582→                            <div className="min-w-0">
   583→                              <p className="truncate text-sm font-medium">
   584→                                {u.name}
   585→                              </p>
   586→                              <p className="truncate text-xs text-muted-foreground">
   587→                                {u.email}
   588→                              </p>
   589→                            </div>
   590→                          </div>
   591→                        </TableCell>
   592→                        <TableCell className="px-4 py-3">
   593→                          <RoleBadge role={u.role} />
   594→                        </TableCell>

   670→                                  onClick={() => setEditTarget(u)}

   754→

   756→      {!isError && !isLoading && items.length > 0 ? (

   770→        onOpenChange={(open) => !open && setEditTarget(null)}

   774→
