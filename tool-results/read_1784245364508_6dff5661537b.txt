   447→   447→                          {activeFilterCount > 0 ? (

   881→   881→function initials(name: string): string {
   882→   882→  if (!name) return "?"
   883→   883→  const parts = name.trim().split(/\s+/)
   884→   884→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase()
   885→   885→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase()
   886→   886→}

   888→   888→function errMsg(e: unknown, fallback: string): string {
   889→   889→  if (e && typeof e === "object" && "message" in e) {
   890→   890→    return String((e as { message?: unknown }).message ?? fallback)
   891→   891→  }
   892→   892→  return fallback
