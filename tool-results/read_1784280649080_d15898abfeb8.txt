   169→   169→export function AdminUsers() {
   170→   170→  const queryClient = useQueryClient()
   171→   171→  const [role, setRole] = React.useState<RoleFilter>("ALL")
   172→   172→  const [q, setQ] = React.useState("")
   173→   173→  const [debouncedQ, setDebouncedQ] = React.useState("")
   174→   174→  const [verified, setVerified] = React.useState<VerifiedFilter>("ALL")
   175→   175→  const [active, setActive] = React.useState<ActiveFilter>("ALL")
   176→   176→  const [page, setPage] = React.useState(1)
   177→   177→  const [sort, setSort] = React.useState<SortState>(null)
   178→   178→
   179→   179→  const [editTarget, setEditTarget] = React.useState<AdminUser | null>(null)
   180→   180→  const [deleteTarget, setDeleteTarget] = React.useState<AdminUser | null>(null)
   181→   181→  const [pendingToggle, setPendingToggle] = React.useState<PendingToggle>(null)
   182→   182→  const [patchingId, setPatchingId] = React.useState<string | null>(null)
   183→   183→  const [errorBanner, setErrorBanner] = React.useState<string | null>(null)
   184→   184→
   185→   185→  const limit = 10
   186→   186→
   187→   187→  // Debounce search
   188→   188→  React.useEffect(() => {
   189→   189→    const t = setTimeout(() => {
   190→   190→      setDebouncedQ(q.trim())
   191→   191→      setPage(1)
   192→   192→    }, 350)
   193→   193→    return () => clearTimeout(t)
   194→   194→  }, [q])
   195→   195→
   196→   196→  // Role counts (cached 60s, compartilhado com dashboard)
   197→   197→  const { data: stats } = useQuery({
   198→   198→    queryKey: ["admin", "stats"],
   199→   199→    queryFn: () => apiGet<StatsResponse>("/api/admin/stats"),
   200→   200→    staleTime: 60_000,
   201→   201→  })
   202→   202→
   203→   203→  const { data, isLoading, isError, refetch } = useQuery({
   204→   204→    queryKey: ["admin", "users", { role, debouncedQ, verified, active, page, limit }],
   205→   205→    queryFn: () =>
   206→   206→      apiGet<AdminUsersResponse>("/api/admin/users", {
   207→   207→        ...(role !== "ALL" ? { role } : {}),
   208→   208→        ...(debouncedQ ? { q: debouncedQ } : {}),
   209→   209→        page,
   210→   210→        limit,
   211→   211→      }),
   212→   212→    staleTime: 15_000,
   213→   213→  })
   214→   214→
   215→   215→  const invalidate = () =>
   216→   216→    queryClient.invalidateQueries({ queryKey: ["admin", "users"] })
   217→   217→
   218→   218→  const patchMutation = useMutation({
   219→   219→    mutationFn: ({
   220→   220→      id,
   221→   221→      patch,
   222→   222→    }: {
   223→   223→      id: string
   224→   224→      patch: Partial<AdminUser>
   225→   225→    }) => apiPatch<{ user: AdminUser }>(`/api/admin/users/${id}`, patch),
   226→   226→  })
   227→   227→
   228→   228→  const deleteMutation = useMutation({
   229→   229→    mutationFn: (id: string) => apiDelete(`/api/admin/users/${id}`),
   230→   230→  })
   231→   231→
   232→   232→  // Client-side filter for verified/active (API does not support these yet — H2)
   233→   233→  const rawItems = data?.items ?? []
   234→   234→  const filteredItems = React.useMemo(() => {
   235→   235→    return rawItems.filter((u) => {
   236→   236→      if (verified === "true" && !u.verified) return false
   237→   237→      if (verified === "false" && u.verified) return false
   238→   238→      if (active === "true" && !u.active) return false
   239→   239→      if (active === "false" && u.active) return false
   240→   240→      return true
   241→   241→    })
   242→   242→  }, [rawItems, verified, active])
   243→   243→
   244→   244→  const items = React.useMemo(() => {
   245→   245→    if (!sort) return filteredItems
   246→   246→    const sorted = [...filteredItems].sort((a, b) => {
   247→   247→      let cmp = 0
   248→   248→      if (sort.key === "name") cmp = a.name.localeCompare(b.name, "pt-BR")
   249→   249→      else if (sort.key === "createdAt")
   250→   250→        cmp = new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime()
   251→   251→      return sort.dir === "asc" ? cmp : -cmp
   252→   252→    })
   253→   253→    return sorted
   254→   254→  }, [filteredItems, sort])
   255→   255→
   256→   256→  const total = data?.total ?? 0
   257→   257→  const totalPages = Math.max(1, Math.ceil(total / limit))
   258→   258→
   259→   259→  const clientFilterActive = verified !== "ALL" || active !== "ALL"
   260→   260→
   261→   261→  const activeFilterCount =
   262→   262→    (role !== "ALL" ? 1 : 0) +
   263→   263→    (debouncedQ ? 1 : 0) +
   264→   264→    (verified !== "ALL" ? 1 : 0) +
   265→   265→    (active !== "ALL" ? 1 : 0)
   266→   266→
   267→   267→  const clearFilters = () => {
   268→   268→    setRole("ALL")
   269→   269→    setQ("")
   270→   270→    setDebouncedQ("")
   271→   271→    setVerified("ALL")
   272→   272→    setActive("ALL")
   273→   273→    setSort(null)
   274→   274→    setPage(1)
   275→   275→  }
   276→   276→
   277→   277→  const toggleSort = (key: SortKey) => {
   278→   278→    setSort((prev) => {
   279→   279→      if (!prev || prev.key !== key) return { key, dir: "asc" }
   280→   280→      if (prev.dir === "asc") return { key, dir: "desc" }
   281→   281→      return null
   282→   282→    })
   283→   283→  }
   284→   284→
   285→   285→  const roleCounts = React.useMemo(() => {
   286→   286→    const byRole = stats?.usersByRole ?? {}
   287→   287→    return {
   288→   288→      ALL:
   289→   289→        (byRole.CLIENT ?? 0) + (byRole.PROVIDER ?? 0) + (byRole.ADMIN ?? 0),
   290→   290→      CLIENT: byRole.CLIENT ?? 0,
   291→   291→      PROVIDER: byRole.PROVIDER ?? 0,
   292→   292→      ADMIN: byRole.ADMIN ?? 0,
   293→   293→    }
   294→   294→  }, [stats])
   295→   295→
   296→   296→  // ---- Mutation handlers ---------------------------------------------------
   297→   297→
   298→   298→  const handleToggleConfirm = () => {
   299→   299→    if (!pendingToggle) return
   300→   300→    const { id, field, currentValue } = pendingToggle
   301→   301→    setPatchingId(id)
   302→   302→    patchMutation.mutate(
   303→   303→      { id, patch: { [field]: !currentValue } as Partial<AdminUser> },
   304→   304→      {
   305→   305→        onSuccess: () => {
   306→   306→          invalidate()
   307→   307→          toast.success(
   308→   308→            field === "verified"
   309→   309→              ? currentValue
   310→   310→                ? "Verificação removida."
   311→   311→                : "Usuário marcado como verificado."
   312→   312→              : currentValue
   313→   313→              ? "Usuário desativado."
   314→   314→              : "Usuário ativado.",
   315→   315→          )
   316→   316→          setPendingToggle(null)
   317→   317→          setPatchingId(null)
   318→   318→        },
   319→   319→        onError: (e: unknown) => {
   320→   320→          const msg = errMsg(e, "Falha ao atualizar usuário.")
   321→   321→          setErrorBanner(msg)
   322→   322→          toast.error(msg)
   323→   323→          setPendingToggle(null)
   324→   324→          setPatchingId(null)
   325→   325→        },
   326→   326→      },
   327→   327→    )
   328→   328→  }
   329→   329→
   330→   330→  const handleEditSubmit = (patch: Partial<AdminUser>) => {
   331→   331→    if (!editTarget) return
   332→   332→    patchMutation.mutate(
   333→   333→      { id: editTarget.id, patch },
   334→   334→      {
   335→   335→        onSuccess: () => {
   336→   336→          invalidate()
   337→   337→          toast.success("Usuário atualizado.")
   338→   338→          setEditTarget(null)
   339→   339→        },
   340→   340→        onError: (e: unknown) => {
   341→   341→          const msg = errMsg(e, "Falha ao atualizar usuário.")
   342→   342→          setErrorBanner(msg)
   343→   343→          toast.error(msg)
   344→   344→        },
   345→   345→      },
   346→   346→    )
   347→   347→  }
   348→   348→
   349→   349→  const handleDeleteConfirm = () => {
   350→   350→    if (!deleteTarget) return
   351→   351→    deleteMutation.mutate(deleteTarget.id, {
   352→   352→      onSuccess: () => {
   353→   353→        toast.success("Usuário excluído.")
   354→   354→        invalidate()
   355→   355→        setDeleteTarget(null)
   356→   356→      },
   357→   357→      onError: (e: unknown) => {
   358→   358→        const msg = errMsg(e, "Não foi possível excluir o usuário.")
   359→   359→        setErrorBanner(msg)
   360→   360→        toast.error(msg)
   361→   361→        setDeleteTarget(null)
   362→   362→      },
   363→   363→    })
   364→   364→  }
   365→   365→
   366→   366→  const renderSortHeader = (label: string, sortKey: SortKey) => (
   367→   367→    <button
   368→   368→      type="button"
   369→   369→      onClick={() => toggleSort(sortKey)}
   370→   370→      className="inline-flex items-center gap-1 text-xs font-semibold uppercase tracking-wide text-muted-foreground transition-colors hover:text-foreground"
   371→   371→    >
   372→   372→      {label}
   373→   373→      {sort?.key === sortKey ? (
   374→   374→        sort.dir === "asc" ? (
   375→   375→          <ChevronUp className="size-3" />
   376→   376→        ) : (
   377→   377→          <ChevronDown className="size-3" />
   378→   378→        )
   379→   379→      ) : (
   380→   380→        <ArrowUpDown className="size-3 opacity-40" />
   381→   381→      )}
   382→   382→    </button>
   383→   383→  )
   384→   384→
   385→   385→  return (
   386→   386→    <div className="flex flex-col gap-4">
   387→   387→      <PageSectionHeader
   388→   388→        title="Usuários"
   389→   389→        description="Gerencie clientes, prestadores e administradores da plataforma."
   390→   390→      />
   391→   391→
   392→   392→      {/* Role tabs with counts */}
   393→   393→      <Tabs
   394→   394→        value={role}
   395→   395→        onValueChange={(v) => {
   396→   396→          setRole(v as RoleFilter)
   397→   397→          setPage(1)
   398→   398→        }}
   399→   399→      >
   400→   400→        <TabsList className="h-auto flex-wrap gap-1 bg-card p-1 shadow-sm">
   401→   401→          {(
   402→   402→            [
   403→   403→              { value: "ALL", label: "Todos", icon: Users },
   404→   404→              { value: "CLIENT", label: "Clientes", icon: CircleUser },
   405→   405→              { value: "PROVIDER", label: "Prestadores", icon: HardHat },
   406→   406→              { value: "ADMIN", label: "Administradores", icon: ShieldCheck },
   407→   407→            ] as const
   408→   408→          ).map((t) => (
   409→   409→            <TabsTrigger
   410→   410→              key={t.value}
   411→   411→              value={t.value}
   412→   412→              className="h-8 gap-1.5 rounded-md px-3 text-sm data-[state=active]:bg-primary data-[state=active]:text-primary-foreground"
   413→   413→            >
   414→   414→              <t.icon className="size-3.5" />
   415→   415→              {t.label}
   416→   416→              <span
   417→   417→                className={cn(
   418→   418→                  "ml-1 inline-flex h-5 min-w-5 items-center justify-center rounded-full px-1.5 text-[10px] font-semibold tabular-nums",
   419→   419→                  role === t.value
   420→   420→                    ? "bg-primary-foreground/20 text-primary-foreground"
   421→   421→                    : "bg-muted text-muted-foreground",
   422→   422→                )}
   423→   423→              >
   424→   424→                {roleCounts[t.value]}
   425→   425→              </span>
   426→   426→            </TabsTrigger>
   427→   427→          ))}
   428→   428→        </TabsList>
   429→   429→      </Tabs>
   430→   430→
   431→   431→      {/* Error banner (dismissible) — H9 */}
   432→   432→      {errorBanner ? (
   433→   433→        <Alert variant="destructive">
   434→   434→          <AlertTriangle className="size-4" />
   435→   435→          <AlertTitle>Erro ao salvar</AlertTitle>
   436→   436→          <AlertDescription>{errorBanner}</AlertDescription>
   437→   437→          <button
   438→   438→            type="button"
   439→   439→            onClick={() => setErrorBanner(null)}
   440→   440→            aria-label="Dispensar aviso"
   441→   441→            className="absolute right-3 top-3 rounded-md p-1 text-current/70 transition-colors hover:text-current"
   442→   442→          >
   443→   443→            <X className="size-3.5" />
   444→   444→          </button>
   445→   445→        </Alert>
   446→   446→      ) : null}
   447→   447→
   448→   448→      {/* Filter bar */}
   449→   449→      <FilterBar
   450→   450→        onClear={clearFilters}
   451→   451→        activeCount={activeFilterCount}
   452→   452→      >
   453→   453→        <SearchInput
   454→   454→          value={q}
   455→   455→          onChange={setQ}
   456→   456→          placeholder="Buscar por nome, e-mail ou cidade"
   457→   457→          className="min-w-[200px] flex-1"
   458→   458→        />
   459→   459→        <Select
   460→   460→          value={verified}
   461→   461→          onValueChange={(v) => {
   462→   462→            setVerified(v as VerifiedFilter)
   463→   463→            setPage(1)
   464→   464→          }}
   465→   465→        >
   466→   466→          <SelectTrigger className="h-9 w-auto min-w-[150px]">
   467→   467→            <SelectValue placeholder="Verificação" />
   468→   468→          </SelectTrigger>
   469→   469→          <SelectContent>
   470→   470→            <SelectItem value="ALL">Verificação: todas</SelectItem>
   471→   471→            <SelectItem value="true">Apenas verificados</SelectItem>
   472→   472→            <SelectItem value="false">Aguardando verificação</SelectItem>
   473→   473→          </SelectContent>
   474→   474→        </Select>
   475→   475→        <Select
   476→   476→          value={active}
   477→   477→          onValueChange={(v) => {
   478→   478→            setActive(v as ActiveFilter)
   479→   479→            setPage(1)
   480→   480→          }}
   481→   481→        >
   482→   482→          <SelectTrigger className="h-9 w-auto min-w-[140px]">
   483→   483→            <SelectValue placeholder="Status" />
   484→   484→          </SelectTrigger>
   485→   485→          <SelectContent>
   486→   486→            <SelectItem value="ALL">Status: todos</SelectItem>
   487→   487→            <SelectItem value="true">Apenas ativos</SelectItem>
   488→   488→            <SelectItem value="false">Apenas inativos</SelectItem>
   489→   489→          </SelectContent>
   490→   490→        </Select>
   491→   491→      </FilterBar>
   492→   492→
