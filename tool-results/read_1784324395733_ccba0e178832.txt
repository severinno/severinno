    80→export function ProviderProfileModal() {
    81→  const open = useUIStore((s) => s.providerModal.open)
    82→  const providerId = useUIStore((s) => s.providerModal.providerId)
    83→  const close = useUIStore((s) => s.closeProvider)
    84→  const openQuote = useUIStore((s) => s.openQuote)
    85→  const openBooking = useUIStore((s) => s.openBooking)
    86→  const addRecentlyViewed = useRecentlyViewedStore((s) => s.addView)
    87→  const isMobile = useIsMobile()
    88→
    89→  const query = useQuery({
    90→    queryKey: ["provider", providerId],
    91→    queryFn: () => apiGet<ProviderDetail>(`/api/providers/${providerId}`),
    92→    enabled: open && !!providerId,
    93→    staleTime: 60 * 1000,
    94→  })
    95→
    96→  const provider = query.data
    97→  const [favorited, setFavorited] = React.useState(false)
    98→
    99→  // Reset favorite state when provider changes.
   100→  React.useEffect(() => {
   101→    setFavorited(false)
   102→  }, [providerId])
   103→
   104→  // Track recently viewed (Nielsen H6 — recognition over recall).
   105→  // Only track once per provider when the modal opens with loaded data.
   106→  React.useEffect(() => {
   107→    if (open && provider) {
   108→      addRecentlyViewed(provider)
   109→    }
   110→  }, [open, provider, addRecentlyViewed])
   111→
   112→  const handleFavorite = async () => {
   113→    setFavorited((v) => !v)
   114→    toast.success(
   115→      favorited ? "Removido dos favoritos." : "Adicionado aos favoritos!",
   116→    )
   117→  }
   118→
   119→  const handleShare = async () => {
   120→    if (!provider) return
   121→    const url = `${window.location.origin}/?provider=${provider.id}`
   122→    try {
   123→      if (navigator.share) {
   124→        await navigator.share({
   125→          title: provider.name,
   126→          text: `Conheça ${provider.name} no Severinno`,
   127→          url,
   128→        })
   129→      } else {
   130→        await navigator.clipboard.writeText(url)
   131→        toast.success("Link copiado para a área de transferência.")
   132→      }
   133→    } catch {
   134→      // user dismissed share dialog — silent
   135→    }
   136→  }
   137→
   138→  const startQuote = (serviceId?: string) => {
   139→    if (!provider) return
   140→    close()
   141→    setTimeout(() => openQuote({ providerId: provider.id, serviceId }), 200)
   142→  }
   143→  const startBooking = (serviceId?: string) => {
   144→    if (!provider) return
   145→    close()
   146→    setTimeout(() => openBooking({ providerId: provider.id, serviceId }), 200)
   147→  }
   148→
   149→  const content = (
   150→    <ProfileBody
   151→      provider={provider}
   152→      loading={query.isLoading}
   153→      favorited={favorited}
   154→      onFavorite={handleFavorite}
   155→      onShare={handleShare}
   156→      onClose={close}
   157→      onQuote={startQuote}
   158→      onBooking={startBooking}
   159→    />
   160→  )
   161→
   162→  if (isMobile) {
   163→    return (
   164→      <Sheet open={open} onOpenChange={(o) => !o && close()}>
   165→        <SheetContent
   166→          side="bottom"
   167→          className="h-[100dvh] max-h-[100dvh] w-full p-0 sm:max-w-full gap-0 flex flex-col [&_[data-slot=sheet-close]]:hidden"
   168→        >
   169→          <SheetTitle className="sr-only">
   170→            {provider?.name ?? "Perfil do prestador"}
   171→          </SheetTitle>
   172→          <SheetDescription className="sr-only">
   173→            Detalhes do prestador de serviços.
   174→          </SheetDescription>
   175→          <div className="flex-1 overflow-hidden">{content}</div>
   176→        </SheetContent>
   177→      </Sheet>
   178→    )
   179→  }
   180→
   181→  return (
   182→    <Dialog open={open} onOpenChange={(o) => !o && close()}>
   183→      <DialogContent
   184→        className="sm:max-w-3xl p-0 gap-0 overflow-hidden"
   185→        showCloseButton={false}
   186→      >
   187→        <DialogTitle className="sr-only">
   188→          {provider?.name ?? "Perfil do prestador"}
   189→        </DialogTitle>
   190→        <DialogDescription className="sr-only">
   191→          Detalhes do prestador de serviços.
   192→        </DialogDescription>
   193→        {content}
   194→      </DialogContent>
   195→    </Dialog>
   196→  )
   197→}

   435→function ServicesTab({
   436→  services,
   437→  loading,
   438→  onQuote,
   439→  onBooking,
   440→}: {
   441→  services: ProviderService[]
   442→  loading: boolean
   443→  onQuote: (serviceId?: string) => void
   444→  onBooking: (serviceId?: string) => void
   445→}) {
   446→  if (loading) {
   447→    return (
   448→      <div className="space-y-3">
   449→        {Array.from({ length: 3 }).map((_, i) => (
   450→          <Skeleton key={i} className="h-20 w-full rounded-lg" />
   451→        ))}
   452→      </div>
   453→    )
   454→  }
   455→
   456→  if (services.length === 0) {
   457→    return (
   458→      <EmptyState
   459→        icon={Wrench}
   460→        title="Nenhum serviço cadastrado"
   461→        description="Este prestador ainda não publicou serviços."
   462→      />
   463→    )
   464→  }
   465→
   466→  // Group by category name
   467→  const groups = new Map<string, ProviderService[]>()
   468→  for (const s of services) {
   469→    const key = s.category?.name ?? "Outros"
   470→    const arr = groups.get(key) ?? []
   471→    arr.push(s)
   472→    groups.set(key, arr)
   473→  }
   474→  const groupEntries = Array.from(groups.entries())
   475→
   476→  return (
   477→    <Accordion
   478→      type="multiple"
   479→      defaultValue={groupEntries.map(([k]) => k)}
   480→      className="w-full"
   481→    >
   482→      {groupEntries.map(([categoryName, items]) => (
   483→        <AccordionItem key={categoryName} value={categoryName}>
   484→          <AccordionTrigger className="hover:no-underline">
   485→            <div className="flex w-full items-center justify-between pr-2">
   486→              <span className="font-medium">{categoryName}</span>
   487→              <Badge variant="secondary" className="mr-1">
   488→                {items.length}
   489→              </Badge>
   490→            </div>
   491→          </AccordionTrigger>
   492→          <AccordionContent>
   493→            <div className="grid gap-3">
   494→              {items.map((s) => (
   495→                <ServiceCard
   496→                  key={s.id}
   497→                  service={s}
   498→                  onQuote={() => onQuote(s.id)}
   499→                  onBooking={() => onBooking(s.id)}
   500→                />
   501→              ))}
   502→            </div>
   503→          </AccordionContent>
   504→        </AccordionItem>
   505→      ))}
   506→    </Accordion>
   507→  )
   508→}
   509→
   510→function ServiceCard({
   511→  service,
   512→  onQuote,
   513→  onBooking,
   514→}: {
   515→  service: ProviderService
   516→  onQuote: () => void
   517→  onBooking: () => void
   518→}) {
   519→  const photos = service.photos ?? []
   520→
   521→  return (
   522→    <motion.div
   523→      initial={{ opacity: 0, y: 4 }}
   524→      animate={{ opacity: 1, y: 0 }}
   525→      transition={{ duration: 0.15 }}
   526→      className="rounded-lg border bg-card overflow-hidden"
   527→    >
   528→      <div className="flex flex-col gap-3 p-3 sm:flex-row sm:items-start">
   529→        <div className="flex-1 min-w-0">
   530→          <h4 className="font-medium leading-tight">{service.title}</h4>
   531→          <p className="mt-0.5 text-sm text-muted-foreground line-clamp-2">
   532→            {service.description ?? "—"}
   533→          </p>
   534→          <p className="mt-2 text-emerald-700 font-semibold dark:text-emerald-400">
   535→            {formatBRL(service.basePrice)}{" "}
   536→            <span className="text-xs font-normal text-muted-foreground">
   537→              / {SERVICE_UNIT_LABELS[service.unit] ?? "un"}
   538→            </span>
   539→          </p>
   540→        </div>
   541→
   542→        {photos.length > 0 && (
   543→          <div className="w-full sm:w-32 shrink-0">
   544→            <Carousel
   545→              opts={{ loop: false, dragFree: false }}
   546→              className="w-full"
   547→            >
   548→              <CarouselContent>
   549→                {photos.map((url, i) => (
   550→                  <CarouselItem key={i} className="basis-full">
   551→                    <div className="aspect-video sm:aspect-square w-full overflow-hidden rounded-md bg-muted">
   552→                      <img
   553→                        src={url}
   554→                        alt={`${service.title} — foto ${i + 1}`}
   555→                        className="size-full object-cover"
   556→                        loading="lazy"
   557→                      />
   558→                    </div>
   559→                  </CarouselItem>
   560→                ))}
   561→              </CarouselContent>
   562→              {photos.length > 1 && (
   563→                <>
   564→                  <CarouselPrevious className="size-7 -left-2" />
   565→                  <CarouselNext className="size-7 -right-2" />
   566→                </>
   567→              )}
   568→            </Carousel>
   569→          </div>
   570→        )}
   571→      </div>
   572→
   573→      <Separator />
   574→      <div className="grid grid-cols-2 gap-2 p-2">
   575→        <Button
   576→          size="sm"
   577→          variant="ghost"
   578→          onClick={onQuote}
   579→          className="text-emerald-700 hover:bg-emerald-50 hover:text-emerald-800 dark:text-emerald-400 dark:hover:bg-emerald-950/40"
   580→        >
   581→          Orçamento
   582→          <ChevronRight className="size-3.5" />
   583→        </Button>
   584→        <Button
   585→          size="sm"
   586→          variant="ghost"
   587→          onClick={onBooking}
   588→          className="text-emerald-700 hover:bg-emerald-50 hover:text-emerald-800 dark:text-emerald-400 dark:hover:bg-emerald-950/40"
   589→        >
   590→          Agendar
   591→          <ChevronRight className="size-3.5" />
   592→        </Button>
   593→      </div>
   594→    </motion.div>
   595→  )
   596→}

   603→  provider,
   604→  loading,
   605→}: {

   608→}) {
   609→  if (loading) {
   610→    return (

   691→function ReviewsTab({
   692→  reviews,
   693→  rating,
   694→  reviewCount,
   695→  loading,
   696→}: {
   697→  reviews: ProviderDetail["reviews"]
   698→  rating?: number
   699→  reviewCount?: number
   700→  loading: boolean
   701→}) {
   702→  if (loading) {
   703→    return (
   704→      <div className="space-y-3">
   705→        <Skeleton className="h-24 w-full rounded-lg" />
   706→        {Array.from({ length: 3 }).map((_, i) => (
   707→          <Skeleton key={i} className="h-20 w-full rounded-lg" />
   708→        ))}
   709→      </div>
   710→    )
   711→  }
   712→
   713→  if (!reviews || reviews.length === 0) {
   714→    return (
   715→      <EmptyState
   716→        icon={Star}
   717→        title="Sem avaliações ainda"
   718→        description="Quando este prestador concluir serviços, as avaliações dos clientes aparecerão aqui."
   719→      />
   720→    )
   721→  }
   722→
   723→  // Distribution: count reviews per star (1..5)
   724→  const distribution = [5, 4, 3, 2, 1].map((star) => {
   725→    const count = reviews.filter((r) => Math.round(r.rating) === star).length
   726→    const pct = reviews.length > 0 ? (count / reviews.length) * 100 : 0
   727→    return { star, count, pct }
   728→  })
   729→  const avg = rating ?? reviews.reduce((s, r) => s + r.rating, 0) / reviews.length
   730→
   731→  return (
   732→    <div className="space-y-4">
   733→      {/* Summary */}
   734→      <div className="flex items-center gap-4 rounded-lg border bg-card p-4">
   735→        <div className="text-center">
   736→          <p className="text-3xl font-bold leading-none tabular-nums">
   737→            {avg.toFixed(1)}
   738→          </p>
   739→          <StarRatingDisplay
   740→            value={avg}
   741→            showCount={false}
   742→            size={12}
   743→            className="mt-1.5 justify-center"
   744→          />
   745→          <p className="mt-1 text-xs text-muted-foreground">
   746→            {reviewCount ?? reviews.length}{" "}
   747→            avaliação{(reviewCount ?? reviews.length) === 1 ? "" : "ões"}
   748→          </p>
   749→        </div>
   750→        <Separator orientation="vertical" className="h-16" />
   751→        <div className="flex-1 grid gap-1.5">
   752→          {distribution.map((d) => (
   753→            <div key={d.star} className="flex items-center gap-2 text-xs">
   754→              <span className="w-3 text-muted-foreground tabular-nums">{d.star}</span>
   755→              <Star className="size-3 fill-amber-400 text-amber-400" strokeWidth={0} />
   756→              <div className="relative h-1.5 flex-1 overflow-hidden rounded-full bg-muted">
   757→                <div
   758→                  className="absolute inset-y-0 left-0 rounded-full bg-amber-400"
   759→                  style={{ width: `${d.pct}%` }}
   760→                />
   761→              </div>
   762→              <span className="w-6 text-right text-muted-foreground tabular-nums">
   763→                {d.count}
   764→              </span>
   765→            </div>
   766→          ))}
   767→        </div>
   768→      </div>
   769→
   770→      {/* List */}
   771→      <div className="space-y-3">
   772→        {reviews.map((r) => (
   773→          <div key={r.id} className="rounded-lg border bg-card p-3">
   774→            <div className="flex items-center gap-3">
   775→              <Avatar className="size-9">
   776→                {r.author?.avatarUrl ? (
   777→                  <AvatarImage src={r.author.avatarUrl} alt={r.author.name} />
   778→                ) : null}
   779→                <AvatarFallback className="bg-emerald-100 text-emerald-700 text-xs dark:bg-emerald-950 dark:text-emerald-300">
   780→                  {r.author?.name?.[0]?.toUpperCase() ?? "?"}
   781→                </AvatarFallback>
   782→              </Avatar>
   783→              <div className="flex-1 min-w-0">
   784→                <p className="text-sm font-medium truncate">
   785→                  {r.author?.name ?? "Cliente"}
   786→                </p>
   787→                <div className="flex items-center gap-2">
   788→                  <StarRatingDisplay value={r.rating} size={12} showCount={false} />
   789→                  <span className="text-xs text-muted-foreground">
   790→                    {new Date(r.createdAt).toLocaleDateString("pt-BR")}
   791→                  </span>
   792→                </div>
   793→              </div>
   794→            </div>
   795→            {r.comment && (
   796→              <p className="mt-2 text-sm text-muted-foreground leading-relaxed">
   797→                {r.comment}
   798→              </p>
   799→            )}
   800→          </div>
   801→        ))}
   802→      </div>
   803→    </div>
   804→  )
   805→}

   811→function HoursTab({
   812→  availability,
   813→  loading,
   814→}: {
   815→  availability: ProviderDetail["availability"]
   816→  loading: boolean
   817→}) {
   818→  if (loading) {
   819→    return <Skeleton className="h-32 w-full rounded-lg" />
   820→  }
   821→
   822→  // Build full week, even days without availability.
   823→  const byDay = new Map<number, { start: string; end: string }[]>()
   824→  for (const a of availability ?? []) {
   825→    const arr = byDay.get(a.dayOfWeek) ?? []
   826→    arr.push({ start: a.startTime, end: a.endTime })
   827→    byDay.set(a.dayOfWeek, arr)
   828→  }
   829→
   830→  // Today (for highlighting open/closed)
   831→  const today = new Date().getDay()
   832→
   833→  return (
   834→    <div className="rounded-lg border overflow-hidden">
   835→      <Table>
   836→        <TableHeader>
   837→          <TableRow>
   838→            <TableHead className="w-1/3">Dia</TableHead>
   839→            <TableHead>Horários</TableHead>
   840→            <TableHead className="w-24 text-right">Status</TableHead>
   841→          </TableRow>
   842→        </TableHeader>
   843→        <TableBody>
   844→          {WEEKDAYS.map((day, i) => {
   845→            const slots = byDay.get(i) ?? []
   846→            const open = slots.length > 0
   847→            const isToday = i === today
   848→            return (
   849→              <TableRow key={day} className={isToday ? "bg-emerald-50/40 dark:bg-emerald-950/20" : ""}>
   850→                <TableCell className="font-medium">
   851→                  <span className="hidden sm:inline">{day}</span>
   852→                  <span className="sm:hidden">{WEEKDAYS_SHORT[i]}</span>
   853→                  {isToday && (
   854→                    <span className="ml-1.5 text-[10px] font-semibold uppercase text-emerald-700 dark:text-emerald-400">
   855→                      hoje
   856→                    </span>
   857→                  )}
   858→                </TableCell>
   859→                <TableCell>
   860→                  {slots.length === 0 ? (
   861→                    <span className="text-muted-foreground text-xs">—</span>
   862→                  ) : (
   863→                    <div className="flex flex-wrap gap-1.5">
   864→                      {slots.map((s, idx) => (
   865→                        <Badge
   866→                          key={idx}
   867→                          variant="secondary"
   868→                          className="font-mono text-xs"
   869→                        >
   870→                          {s.start}–{s.end}
   871→                        </Badge>
   872→                      ))}
   873→                    </div>
   874→                  )}
   875→                </TableCell>
   876→                <TableCell className="text-right">
   877→                  <Badge
   878→                    variant={open ? "default" : "outline"}
   879→                    className={cn(
   880→                      open
   881→                        ? "bg-emerald-100 text-emerald-700 hover:bg-emerald-100 dark:bg-emerald-950/50 dark:text-emerald-300"
   882→                        : "text-muted-foreground",
   883→                    )}
   884→                  >
   885→                    {open ? "Aberto" : "Fechado"}
   886→                  </Badge>
   887→                </TableCell>
   888→              </TableRow>
   889→            )
   890→          })}
   891→        </TableBody>
   892→      </Table>
   893→    </div>
   894→  )
   895→}
