   141→const WELCOME_TOAST_KEY = "severinno-welcome-seen"
   142→
   143→function WelcomeToast({ onDismiss }: { onDismiss: () => void }) {
   144→  const [providerCount, setProviderCount] = React.useState<number | null>(null)
   145→
   146→  React.useEffect(() => {
   147→    apiGet<ProvidersCountResponse>("/api/providers", { limit: 1 })
   148→      .then((data) => {
   149→        if (data && typeof data.total === "number") {
   150→          setProviderCount(data.total)
   151→        }
   152→      })
   153→      .catch(() => {
   154→        // Fallback count
   155→        setProviderCount(120)
   156→      })
   157→  }, [])
   158→
   159→  // Auto-dismiss after 8 seconds
   160→  React.useEffect(() => {
   161→    const t = window.setTimeout(onDismiss, 8000)
   162→    return () => window.clearTimeout(t)
   163→  }, [onDismiss])
   164→
   165→  return (
   166→    <motion.div
   167→      initial={{ y: -100, opacity: 0 }}
   168→      animate={{ y: 0, opacity: 1 }}
   169→      exit={{ y: -100, opacity: 0 }}
   170→      transition={{ type: "spring", stiffness: 300, damping: 30 }}
   171→      className="fixed left-1/2 top-4 z-[60] -translate-x-1/2"
   172→    >
   173→      <div className="flex items-center gap-3 rounded-2xl border border-emerald-200/60 bg-gradient-to-r from-emerald-50 to-teal-50 px-5 py-3 shadow-lg shadow-emerald-500/10 dark:border-emerald-800/40 dark:from-emerald-950/90 dark:to-teal-950/90 dark:shadow-emerald-500/5">
   174→        <span className="flex size-8 shrink-0 items-center justify-center rounded-xl bg-gradient-to-br from-emerald-500 to-teal-500 shadow-md shadow-emerald-500/20">
   175→          <Sparkles className="size-4 text-white" />
   176→        </span>
   177→        <div className="min-w-0">
   178→          <p className="text-sm font-semibold text-emerald-800 dark:text-emerald-200">
   179→            Bem-vindo ao Severinno!
   180→          </p>
   181→          <p className="text-xs text-emerald-600 dark:text-emerald-400">
   182→            {providerCount !== null
   183→              ? `${providerCount} prestadores disponíveis na sua região`
   184→              : "Carregando…"}
   185→          </p>
   186→        </div>
   187→        <button
   188→          type="button"
   189→          onClick={onDismiss}
   190→          className="ml-2 flex size-6 shrink-0 items-center justify-center rounded-full text-emerald-500 transition-colors hover:bg-emerald-200/50 hover:text-emerald-700 dark:hover:bg-emerald-800/50 dark:hover:text-emerald-300"
   191→          aria-label="Dispensar notificação"
   192→        >
   193→          <X className="size-3.5" />
   194→        </button>
   195→      </div>
   196→    </motion.div>
   197→  )
   198→}
