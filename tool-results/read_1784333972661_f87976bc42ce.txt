   304→   304→    } catch (e) {

   324→   324→            provider={provider}
   325→   325→            loading={providerQuery.isLoading || servicesQuery.isLoading}
   326→   326→          />
   327→   327→        )
   328→   328→      case 2:
   329→   329→        return (
   330→   330→          <Step2Details
   331→   331→            state={state}
   332→   332→            set={set}
   333→   333→            provider={provider}
   334→   334→            selectedService={selectedService}
   335→   335→          />
   336→   336→        )
   337→   337→      case 3:
   338→   338→        return (
   339→   339→          <Step3Payment
   340→   340→            state={state}
   341→   341→            set={set}
   342→   342→            markTouched={markTouched}
   343→   343→            touched={touched}
   344→   344→            selectedService={selectedService}
   345→   345→          />
   346→   346→        )
   347→   347→      case 4:
   348→   348→        return (
   349→   349→          <Step4Confirmation
   350→   350→            state={state}
   351→   351→            provider={provider}
   352→   352→            selectedService={selectedService}
   353→   353→            goToStep={setStep}
   354→   354→          />

   356→   356→      default:
   357→   357→        return null

   359→   359→  })()

   392→   392→            <SheetTitle className="flex items-center gap-2 text-base">
   393→   393→              <CalendarDays className="size-4 text-emerald-600" />

   430→   430→function Step1Schedule({
   431→   431→  state,
   432→   432→  set,
   433→   433→  availability,
   434→   434→  selectedService,
   435→   435→  provider,
   436→   436→  loading,
   437→   437→}: {
   438→   438→  state: BookingFormState
   439→   439→  set: <K extends keyof BookingFormState>(
   440→   440→    key: K,
   441→   441→    value: BookingFormState[K],
   442→   442→  ) => void
   443→   443→  availability: ProviderDetail["availability"]
   444→   444→  selectedService?: ProviderService
   445→   445→  provider?: ProviderDetail
   446→   446→  loading: boolean
   447→   447→}) {
   448→   448→  // Generate slot list for selected date
   449→   449→  const slots = React.useMemo(() => {
   450→   450→    if (!state.date) return [] as { label: string; value: string }[]
   451→   451→    const dow = state.date.getDay()
   452→   452→    const dayBlocks = (availability ?? [])
   453→   453→      .filter((a) => a.dayOfWeek === dow)
   454→   454→      .sort((a, b) => a.startTime.localeCompare(b.startTime))
   455→   455→    if (dayBlocks.length === 0) return []
   456→   456→    const out: { label: string; value: string }[] = []
   457→   457→    for (const block of dayBlocks) {
   458→   458→      const [sh, sm] = block.startTime.split(":").map(Number)
   459→   459→      const [eh, em] = block.endTime.split(":").map(Number)
   460→   460→      let cur = sh * 60 + sm
   461→   461→      const end = eh * 60 + em
   462→   462→      while (cur + 60 <= end) {
   463→   463→        const h = Math.floor(cur / 60)
   464→   464→        const m = cur % 60
   465→   465→        const hhmm = `${String(h).padStart(2, "0")}:${String(m).padStart(2, "0")}`
   466→   466→        out.push({ label: formatHHmm(hhmm), value: hhmm })
   467→   467→        cur += 60
   468→   468→      }
   469→   469→    }
   470→   470→    return out
   471→   471→  }, [state.date, availability])
   472→   472→
   473→   473→  const today = new Date()
   474→   474→  today.setHours(0, 0, 0, 0)
   475→   475→
   476→   476→  if (loading) {
   477→   477→    return (
   478→   478→      <div className="flex items-center justify-center py-10">
   479→   479→        <Loader2 className="size-5 animate-spin text-emerald-600" />
   480→   480→      </div>
   481→   481→    )
   482→   482→  }
   483→   483→
   484→   484→  return (
   485→   485→    <div className="grid gap-4">
   486→   486→      <StepHeader
   487→   487→        icon={CalendarDays}
   488→   488→        title="Escolha a data e horário"
   489→   489→        description="Selecione o melhor dia e horário para o serviço."
   490→   490→      />
   491→   491→
   492→   492→      {/* Service + provider info banner — Nielsen #1: visibility */}
   493→   493→      {selectedService && (
   494→   494→        <InfoCard variant="emerald">
   495→   495→          <div className="flex items-center gap-3">
   496→   496→            <Avatar className="size-9 rounded-md">
   497→   497→              {provider?.avatarUrl ? (
   498→   498→                <AvatarImage src={provider.avatarUrl} alt={provider.name} />
   499→   499→              ) : null}
   500→   500→              <AvatarFallback className="rounded-md bg-emerald-100 text-emerald-700 text-xs dark:bg-emerald-950 dark:text-emerald-300">
   501→   501→                {provider?.name?.[0]?.toUpperCase() ?? "?"}
   502→   502→              </AvatarFallback>
   503→   503→            </Avatar>
   504→   504→            <div className="flex-1 min-w-0">
   505→   505→              <p className="text-sm font-medium truncate">{selectedService.title}</p>
   506→   506→              <p className="text-xs text-emerald-700 dark:text-emerald-400 font-medium">
   507→   507→                {formatBRL(selectedService.basePrice)} /{" "}
   508→   508→                {SERVICE_UNIT_LABELS[selectedService.unit] ?? "un"}
   509→   509→              </p>
