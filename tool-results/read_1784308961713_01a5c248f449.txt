   268→function ParallaxBlobs({ scrollYProgress }: { scrollYProgress: ReturnType<typeof useScroll>["scrollYProgress"] }) {
   269→  const y1 = useTransform(scrollYProgress, [0, 1], [0, -60])
   270→  const y2 = useTransform(scrollYProgress, [0, 1], [0, 40])
   271→  const y3 = useTransform(scrollYProgress, [0, 1], [0, -30])

   273→  return (
   274→    <>
   275→      <motion.div
   276→        aria-hidden
   277→        className="absolute -top-32 right-0 size-80 rounded-full bg-emerald-200/40 blur-3xl dark:bg-emerald-800/20"
   278→        style={{ y: y1 }}
   279→      />
   280→      <motion.div
   281→        aria-hidden
   282→        className="absolute -bottom-24 -left-16 size-96 rounded-full bg-teal-200/30 blur-3xl dark:bg-teal-800/15"

   284→      />

   290→    </>
