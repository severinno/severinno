   118→   118→export function ProviderProfileModal() {
   119→   119→  const open = useUIStore((s) => s.providerModal.open)
   120→   120→  const providerId = useUIStore((s) => s.providerModal.providerId)
   121→   121→  const close = useUIStore((s) => s.closeProvider)
   122→   122→  const openQuote = useUIStore((s) => s.openQuote)
   123→   123→  const openBooking = useUIStore((s) => s.openBooking)
   124→   124→  const addRecentlyViewed = useRecentlyViewedStore((s) => s.addView)
   125→   125→  const isMobile = useIsMobile()
   126→   126→
   127→   127→  const query = useQuery({
   128→   128→    queryKey: ["provider", providerId],
   129→   129→    queryFn: () => apiGet<ProviderDetail>(`/api/providers/${providerId}`),
   130→   130→    enabled: open && !!providerId,
   131→   131→    staleTime: 60 * 1000,
   132→   132→  })
   133→   133→
   134→   134→  const provider = query.data
   135→   135→  const [favorited, setFavorited] = React.useState(false)
   136→   136→
   137→   137→  React.useEffect(() => { setFavorited(false) }, [providerId])
   138→   138→
   139→   139→  React.useEffect(() => {
   140→   140→    if (open && provider) addRecentlyViewed(provider)
   141→   141→  }, [open, provider, addRecentlyViewed])
   142→   142→
   143→   143→  const handleFavorite = async () => {
   144→   144→    setFavorited((v) => !v)
   145→   145→    toast.success(favorited ? "Removido dos favoritos." : "Adicionado aos favoritos!")
   146→   146→  }
   147→   147→
   148→   148→  const handleShare = async () => {
   149→   149→    if (!provider) return
   150→   150→    const url = `${window.location.origin}/?provider=${provider.id}`
   151→   151→    try {
   152→   152→      if (navigator.share) {
   153→   153→        await navigator.share({ title: provider.name, text: `Conheça ${provider.name} no Severinno`, url })
   154→   154→      } else {
   155→   155→        await navigator.clipboard.writeText(url)
   156→   156→        toast.success("Link copiado para a área de transferência.")
   157→   157→      }
   158→   158→    } catch { /* user dismissed */ }
   159→   159→  }
   160→   160→
   161→   161→  const startQuote = (serviceId?: string) => {
   162→   162→    if (!provider) return
   163→   163→    close()
   164→   164→    setTimeout(() => openQuote({ providerId: provider.id, serviceId }), 200)
   165→   165→  }
   166→   166→  const startBooking = (serviceId?: string) => {
   167→   167→    if (!provider) return
   168→   168→    close()
   169→   169→    setTimeout(() => openBooking({ providerId: provider.id, serviceId }), 200)
   170→   170→  }
   171→   171→
   172→   172→  const content = (
   173→   173→    <ProfileBody
   174→   174→      provider={provider}
   175→   175→      loading={query.isLoading}
   176→   176→      favorited={favorited}
   177→   177→      onFavorite={handleFavorite}
   178→   178→      onShare={handleShare}
   179→   179→      onClose={close}
   180→   180→      onQuote={startQuote}
   181→   181→      onBooking={startBooking}
   182→   182→    />
   183→   183→  )
   184→   184→
   185→   185→  if (isMobile) {
   186→   186→    return (
   187→   187→      <Sheet open={open} onOpenChange={(o) => !o && close()}>
   188→   188→        <SheetContent side="bottom" className="h-[100dvh] max-h-[100dvh] w-full p-0 sm:max-w-full gap-0 flex flex-col [&_[data-slot=sheet-close]]:hidden">
   189→   189→          <SheetTitle className="sr-only">{provider?.name ?? "Perfil do prestador"}</SheetTitle>
   190→   190→          <SheetDescription className="sr-only">Detalhes do prestador de serviços.</SheetDescription>
   191→   191→          <div className="flex-1 overflow-hidden">{content}</div>
   192→   192→        </SheetContent>
   193→   193→      </Sheet>
   194→   194→    )
   195→   195→  }
   196→   196→
   197→   197→  return (
   198→   198→    <Dialog open={open} onOpenChange={(o) => !o && close()}>
   199→   199→      <DialogContent className="sm:max-w-3xl p-0 gap-0 overflow-hidden" showCloseButton={false}>
   200→   200→        <DialogTitle className="sr-only">{provider?.name ?? "Perfil do prestador"}</DialogTitle>
   201→   201→        <DialogDescription className="sr-only">Detalhes do prestador de serviços.</DialogDescription>
   202→   202→        {content}
   203→   203→      </DialogContent>
   204→   204→    </Dialog>
   205→   205→  )
   206→   206→}

   840→   840→  const byDay = new Map<number, { start: string; end: string }[]>()
   841→   841→  for (const a of availability ?? []) {
   842→   842→    const arr = byDay.get(a.dayOfWeek) ?? []
   843→   843→    arr.push({ start: a.startTime, end: a.endTime })
   844→   844→    byDay.set(a.dayOfWeek, arr)
   845→   845→  }
   846→   846→
   847→   847→  const today = new Date().getDay()
   848→   848→
   849→   849→  // Check if currently open
   850→   850→  const now = new Date()
   851→   851→  const todaySlots = byDay.get(today) ?? []
   852→   852→  const currentMinutes = now.getHours() * 60 + now.getMinutes()
   853→   853→  const currentlyOpen = todaySlots.some((s) => {
   854→   854→    const [sh, sm] = s.start.split(":").map(Number)
   855→   855→    const [eh, em] = s.end.split(":").map(Number)
   856→   856→    return currentMinutes >= sh * 60 + sm && currentMinutes <= eh * 60 + em
   857→   857→  })
   858→   858→
   859→   859→  return (
   860→   860→    <div className="space-y-4">
   861→   861→      {/* Current status indicator */}
   862→   862→      <div className={cn(
   863→   863→        "inline-flex items-center gap-2 rounded-full px-4 py-2 text-sm font-semibold",
   864→   864→        currentlyOpen
   865→   865→          ? "bg-emerald-50 text-emerald-700 ring-1 ring-emerald-200 dark:bg-emerald-950/30 dark:text-emerald-300 dark:ring-emerald-800"
   866→   866→          : "bg-red-50 text-red-700 ring-1 ring-red-200 dark:bg-red-950/30 dark:text-red-300 dark:ring-red-800",
   867→   867→      )}>
   868→   868→        <span className="relative flex size-2">
   869→   869→          {currentlyOpen && <span className="absolute inline-flex size-full animate-ping rounded-full bg-emerald-400 opacity-75" />}
   870→   870→          <span className={cn("relative inline-flex size-2 rounded-full", currentlyOpen ? "bg-emerald-500" : "bg-red-500")} />
   871→   871→        </span>
   872→   872→        {currentlyOpen ? "Aberto agora" : "Fechado agora"}
   873→   873→      </div>
   874→   874→
   875→   875→      {/* Weekly schedule */}
   876→   876→      <div className="rounded-xl border overflow-hidden ring-1 ring-black/[0.03]">
   877→   877→        <Table>
   878→   878→          <TableHeader>
   879→   879→            <TableRow>
   880→   880→              <TableHead className="w-1/3">Dia</TableHead>
   881→   881→              <TableHead>Horários</TableHead>
   882→   882→              <TableHead className="w-24 text-right">Status</TableHead>
   883→   883→            </TableRow>
   884→   884→          </TableHeader>
   885→   885→          <TableBody>
   886→   886→            {WEEKDAYS.map((day, i) => {
   887→   887→              const slots = byDay.get(i) ?? []
   888→   888→              const open = slots.length > 0
   889→   889→              const isToday = i === today
   890→   890→              return (
   891→   891→                <TableRow key={day} className={cn("transition-colors", isToday && "bg-emerald-50/50 dark:bg-emerald-950/20")}>
   892→   892→                  <TableCell className="font-medium">

   895→   895→                    {isToday && (
   896→   896→                      <span className="ml-1.5 inline-flex items-center gap-1 rounded-full bg-emerald-100 px-1.5 py-0.5 text-[10px] font-bold uppercase text-emerald-700 dark:bg-emerald-900/50 dark:text-emerald-300">
   897→   897→                        hoje
   898→   898→                      </span>
   899→   899→                    )}
   900→   900→                  </TableCell>
   901→   901→                  <TableCell>
   902→   902→                    {slots.length === 0 ? (
   903→   903→                      <span className="text-muted-foreground text-xs">—</span>
   904→   904→                    ) : (
   905→   905→                      <div className="flex flex-wrap gap-1.5">
   906→   906→                        {slots.map((s, idx) => (
   907→   907→                          <Badge key={idx} variant="secondary" className="font-mono text-xs rounded-md">
   908→   908→                            {s.start}–{s.end}
   909→   909→                          </Badge>
   910→   910→                        ))}
   911→   911→                      </div>
   912→   912→                    )}
   913→   913→                  </TableCell>
   914→   914→                  <TableCell className="text-right">
   915→   915→                    <Badge
   916→   916→                      variant={open ? "default" : "outline"}
   917→   917→                      className={cn(
   918→   918→                        "text-[11px]",
   919→   919→                        open
   920→   920→                          ? "bg-emerald-100 text-emerald-700 hover:bg-emerald-100 dark:bg-emerald-950/50 dark:text-emerald-300"
   921→   921→                          : "text-muted-foreground",
   922→   922→                      )}
   923→   923→                    >
   924→   924→                      {open ? "Aberto" : "Fechado"}
   925→   925→                    </Badge>
   926→   926→                  </TableCell>
   927→   927→                </TableRow>
   928→   928→              )
   929→   929→            })}
   930→   930→          </TableBody>
   931→   931→        </Table>
   932→   932→      </div>
   933→   933→    </div>
   934→   934→  )
   935→   935→}
