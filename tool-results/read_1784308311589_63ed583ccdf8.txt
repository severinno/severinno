   206→function getTint(slug: string): CategoryTint {
   207→  let hash = 0
   208→  for (let i = 0; i < slug.length; i++) {

   388→function resolveMeta(c: Category) {
   389→  if (c.slug && CATEGORY_META[c.slug]) return CATEGORY_META[c.slug]!
   390→  if (c.icon && CATEGORY_META[c.icon]) return CATEGORY_META[c.icon]!
   391→  // Best-effort slug substring match
   392→  for (const [slug, meta] of Object.entries(CATEGORY_META)) {
   393→    if (c.slug.includes(slug) || c.name.toLowerCase().includes(slug)) {
   394→      return meta
   395→    }
   396→  }
   397→  return null
   398→}

   405→function getProviderCount(categoryId: string): number {
   406→  let hash = 0
   407→  for (let i = 0; i < categoryId.length; i++) {
   408→    hash = (hash * 31 + categoryId.charCodeAt(i)) | 0
   409→  }
   410→  return (Math.abs(hash) % 45) + 3 // 3–47 range
   411→}

   520→  // Update scroll arrow visibility

   554→    (e: React.KeyboardEvent, idx: number) => {

   558→      else if (e.key === "ArrowLeft") nextIdx = idx - 1
   559→      else if (e.key === "ArrowDown") nextIdx = idx + cols
   560→      else if (e.key === "ArrowUp") nextIdx = idx - cols
   561→      else return

   564→      if (nextIdx >= 0 && nextIdx < visibleCategories.length) {

   569→    },

   571→  )

   573→  const scrollBy = (direction: "left" | "right") => {

   594→      )}
   595→    >
