   102→type QuoteItemForm = {
   103→  providerId: string
   104→  serviceId: string
   105→  description: string
   106→  quantity: number
   107→  unit: ServiceUnit
   108→  photos: string[]

   110→
   111→const quoteFormSchema = z.object({
   112→  items: z
   113→    .array(
   114→      z.object({
   115→        providerId: z.string().min(1, "Selecione um prestador"),
   116→        serviceId: z.string().min(1, "Selecione um serviço"),
   117→        description: z
   118→          .string()
   119→          .min(10, "Descreva com ao menos 10 caracteres")
   120→          .max(400),
   121→        quantity: z.coerce.number().min(0.01, "Quantidade inválida"),
   122→        unit: z.enum([
   123→          "UNIDADE",
   124→          "METRO_LINEAR",
   125→          "METRO_QUADRADO",
   126→          "METRO_CUBICO",
   127→        ]),
   128→        photos: z.array(z.string()).max(4).default([]),
   129→      }),
   130→    )
   131→    .min(1, "Adicione ao menos um item"),
   132→  address: z.object({
   133→    cep: z.string().min(8, "CEP inválido"),
   134→    street: z.string().min(3, "Informe a rua"),
   135→    number: z.string().min(1, "Informe o número"),
   136→    complement: z.string().default(""),
   137→    district: z.string().default(""),
   138→    city: z.string().min(2, "Informe a cidade"),
   139→    state: z.string().min(2, "UF"),
   140→    lat: z.number().nullable().optional(),
   141→    lng: z.number().nullable().optional(),
   142→  }),
   143→})

   176→export function QuoteModal() {
   177→  const open = useUIStore((s) => s.quoteModal.open)
   178→  const providerIdPreset = useUIStore((s) => s.quoteModal.providerId)
   179→  const serviceIdPreset = useUIStore((s) => s.quoteModal.serviceId)
   180→  const close = useUIStore((s) => s.closeQuote)
   181→  const openAuth = useUIStore((s) => s.openAuth)
   182→  const isMobile = useIsMobile()
   183→
   184→  const navigate = useViewStore((s) => s.navigate)
   185→  const user = useAuthStore((s) => s.user)
   186→
   187→  const form = useForm<QuoteFormValues>({
   188→    resolver: zodResolver(quoteFormSchema) as unknown as Resolver<QuoteFormValues>,
   189→    defaultValues: {
   190→      items: [emptyItem(providerIdPreset, serviceIdPreset)],
   191→      address: emptyAddress,
   192→    },
   193→    mode: "onTouched",
   194→  })
   195→
   196→  // Reset preset when modal opens.
   197→  React.useEffect(() => {
   198→    if (open) {
   199→      form.reset({
   200→        items: [emptyItem(providerIdPreset, serviceIdPreset)],
   201→        address: emptyAddress,
   202→      })
   203→      setStep(1)
   204→    }
   205→  }, [open, providerIdPreset, serviceIdPreset, form])
   206→
   207→  const [step, setStep] = React.useState<Step>(1)
   208→  const [submitting, setSubmitting] = React.useState(false)
   209→
   210→  const onSubmit = async (values: QuoteFormValues) => {
   211→    if (!user) {
   212→      toast.info("Faça cadastro gratuito para pedir orçamentos.")
   213→      openAuth("register", "CLIENT")
   214→      return
   215→    }
   216→
   217→    const primary = values.items[0]
   218→    if (!primary?.providerId) {
   219→      toast.error("Selecione um prestador no primeiro item.")
   220→      return
   221→    }
   222→
   223→    setSubmitting(true)
   224→    try {
   225→      await apiPost("/api/quotes", {
   226→        providerId: primary.providerId,
   227→        items: values.items.map((it) => ({
   228→          providerId: it.providerId,
   229→          serviceId: it.serviceId,
   230→          description: it.description,
   231→          quantity: it.quantity,
   232→          unit: it.unit,
   233→          photos: it.photos,
   234→        })),
   235→        address: [
   236→          values.address.street,
   237→          values.address.number,
   238→          values.address.complement,
   239→          values.address.district,
   240→        ]
   241→          .filter(Boolean)
   242→          .join(", "),
   243→        cep: values.address.cep,
   244→        lat: values.address.lat,
   245→        lng: values.address.lng,
   246→      })
   247→      toast.success("Orçamento enviado! O prestador responderá em breve.")
   248→      close()
   249→      navigate("client.quotes")
   250→    } catch (e) {
   251→      const msg =
   252→        (e as { message?: string })?.message ??
   253→        "Não foi possível enviar o orçamento. Tente novamente."
   254→      toast.error(msg)
   255→    } finally {
   256→      setSubmitting(false)
   257→    }
   258→  }
   259→
   260→  const content = (
   261→    <WizardBody
   262→      step={step}
   263→      setStep={setStep}
   264→      form={form}
   265→      submitting={submitting}
   266→      onSubmit={form.handleSubmit(onSubmit)}
   267→    />
   268→  )
   269→
   270→  if (isMobile) {
   271→    return (
   272→      <Sheet open={open} onOpenChange={(o) => !o && close()}>
   273→        <SheetContent
   274→          side="bottom"
   275→          className="h-[100dvh] max-h-[100dvh] w-full p-0 sm:max-w-full gap-0 flex flex-col"
   276→        >
   277→          <SheetHeader className="px-4 pt-4 pb-2 shrink-0">
   278→            <SheetTitle className="flex items-center gap-2">
   279→              <Wrench className="size-5 text-emerald-600" />
   280→              Pedir orçamento
   281→            </SheetTitle>
   282→            <SheetDescription>
   283→              Descreva o serviço e receba propostas de prestadores verificados.
   284→            </SheetDescription>
   285→          </SheetHeader>
   286→          <div className="flex-1 overflow-hidden">{content}</div>
   287→        </SheetContent>
   288→      </Sheet>
   289→    )
   290→  }
   291→
   292→  return (
   293→    <Dialog open={open} onOpenChange={(o) => !o && close()}>
   294→      <DialogContent className="sm:max-w-2xl p-0 gap-0 overflow-hidden">
   295→        <DialogHeader className="px-6 pt-6 pb-3 shrink-0 border-b">
   296→          <DialogTitle className="flex items-center gap-2 text-lg">
   297→            <Wrench className="size-5 text-emerald-600" />
   298→            Pedir orçamento
   299→          </DialogTitle>
   300→          <DialogDescription>
   301→            Descreva o serviço e receba propostas de prestadores verificados.
   302→          </DialogDescription>
   303→        </DialogHeader>
   304→        {content}
   305→      </DialogContent>
   306→    </Dialog>
   307→  )
   308→}

   314→function WizardBody({
   315→  step,
   316→  setStep,
   317→  form,
   318→  submitting,
   319→  onSubmit,
   320→}: {
   321→  step: Step
   322→  setStep: React.Dispatch<React.SetStateAction<Step>>
   323→  form: UseFormReturn<QuoteFormValues>
   324→  submitting: boolean
   325→  onSubmit: () => void
   326→}) {
   327→  const items = form.watch("items")
   328→  const address = form.watch("address")
   329→  const errors = form.formState.errors
   330→  const user = useAuthStore((s) => s.user)
   331→  const openAuth = useUIStore((s) => s.openAuth)
   332→
   333→  // ── Step validation ───────────────────────────────────────────────────
   334→  const item0 = items[0]
   335→  const item0Errors = errors.items?.[0]
   336→
   337→  const step1Valid = !!item0?.providerId && !!item0?.serviceId
   338→  const step2Valid =
   339→    !!item0?.description && item0.description.length >= 10 && item0?.quantity > 0
   340→  const step3Valid =
   341→    !!address.cep && address.cep.replace(/\D/g, "").length >= 8 &&
   342→    !!address.street && !!address.number && !!address.city && !!address.state
   343→
   344→  const stepValidMap: Record<Step, boolean> = {
   345→    1: step1Valid,
   346→    2: step2Valid,
   347→    3: step3Valid,
   348→    4: true, // review step always valid (shows summary)
   349→  }
   350→
   351→  const progressPct = (step / 4) * 100
   352→
   353→  // ── Navigation ────────────────────────────────────────────────────────
   354→  const handleNext = () => {
   355→    // Validate current step before advancing
   356→    if (step === 1 && !step1Valid) {
   357→      form.trigger("items.0.providerId")
   358→      form.trigger("items.0.serviceId")
   359→      return
   360→    }
   361→    if (step === 2 && !step2Valid) {
   362→      form.trigger("items.0.description")
   363→      form.trigger("items.0.quantity")
   364→      return
   365→    }
   366→    if (step === 3 && !step3Valid) {
   367→      form.trigger("address")
   368→      return
   369→    }
   370→    setStep((s) => Math.min(4, s + 1) as Step)
   371→  }
   372→
   373→  const handleBack = () => setStep((s) => Math.max(1, s - 1) as Step)
   374→
   375→  const goToStep = (target: Step) => {
   376→    // Allow going back to any previous step freely
   377→    if (target < step) {
   378→      setStep(target)
   379→      return
   380→    }
   381→    // Going forward requires all intermediate steps to be valid
   382→    for (let i = 1; i < target; i++) {
   383→      if (!stepValidMap[i as Step]) {
   384→        toast.error("Complete os passos anteriores primeiro.")
   385→        return
   386→      }
   387→    }
   388→    setStep(target)
   389→  }
   390→
   391→  return (
   392→    <div className="flex h-full flex-col">
   393→      {/* Stepper */}
   394→      <div className="shrink-0 border-b px-4 sm:px-6 py-3">
   395→        <div className="flex items-center justify-between">
   396→          {STEPS.map((s, i) => {
   397→            const active = step === s.id
   398→            const done = step > s.id
   399→            return (
   400→              <React.Fragment key={s.id}>
   401→                <button
   402→                  type="button"
   403→                  onClick={() => goToStep(s.id)}
   404→                  className={cn(
   405→                    "flex items-center gap-1.5 text-xs sm:text-sm transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring rounded",
   406→                    active
   407→                      ? "text-emerald-700 dark:text-emerald-400 font-medium"
   408→                      : done
   409→                        ? "text-emerald-600 cursor-pointer"
   410→                        : "text-muted-foreground",
   411→                  )}
   412→                >
   413→                  <span
   414→                    className={cn(
   415→                      "inline-flex size-7 items-center justify-center rounded-full border-2 text-xs font-bold transition-colors",
   416→                      active && "border-emerald-600 bg-emerald-600 text-white",
   417→                      done && "border-emerald-600 bg-emerald-600 text-white cursor-pointer",
   418→                      !active && !done && "border-muted-foreground/30 text-muted-foreground",
   419→                    )}
   420→                  >
   421→                    {done ? <Check className="size-3.5" /> : s.id}
   422→                  </span>
   423→                  <span className="hidden sm:inline">{s.label}</span>
   424→                </button>
   425→                {i < STEPS.length - 1 && (
   426→                  <div className="flex-1 h-0.5 bg-muted-foreground/20 mx-1.5 sm:mx-2 relative">
   427→                    <div
   428→                      className="absolute inset-0 bg-emerald-600 transition-transform origin-left"
   429→                      style={{
   430→                        transform: step > s.id ? "scaleX(1)" : "scaleX(0)",
   431→                      }}
   432→                    />
   433→                  </div>
   434→                )}
   435→              </React.Fragment>
   436→            )
   437→          })}
   438→        </div>
   439→        <Progress
   440→          value={progressPct}
   441→          className="mt-2 h-1 bg-emerald-100 dark:bg-emerald-950/40"
   442→        />
   443→      </div>
   444→
   445→      {/* Auth gate — shown on Step 1 */}
   446→      {step === 1 && !user && (
   447→        <div className="shrink-0 px-4 sm:px-6 pt-3">
   448→          <div className="flex items-start gap-3 rounded-lg border border-amber-200 bg-amber-50 p-3 text-sm dark:border-amber-900/50 dark:bg-amber-950/30">
   449→            <LogIn className="size-5 shrink-0 text-amber-600" />
   450→            <div className="flex-1">
   451→              <p className="font-medium text-amber-900 dark:text-amber-200">
   452→                Faça cadastro gratuito para pedir orçamentos
   453→              </p>
   454→              <p className="mt-0.5 text-xs text-amber-800 dark:text-amber-300">
   455→                Seus dados ficam salvos para acompanhar as respostas dos
   456→                prestadores.
   457→              </p>
   458→            </div>
   459→            <Button
   460→              type="button"
   461→              size="sm"
   462→              onClick={() => openAuth("register", "CLIENT")}
   463→              className="shrink-0 bg-amber-600 hover:bg-amber-700 text-white"
   464→            >
   465→              Entrar / Cadastrar
   466→            </Button>
   467→          </div>
   468→        </div>
   469→      )}
   470→
   471→      {/* Step content */}
   472→      <ScrollArea className="flex-1">
   473→        <div className="px-4 sm:px-6 py-4">
   474→          <AnimatePresence mode="wait">
   475→            <motion.div
   476→              key={step}
   477→              initial={{ opacity: 0, x: 10 }}
   478→              animate={{ opacity: 1, x: 0 }}
   479→              exit={{ opacity: 0, x: -10 }}
   480→              transition={{ duration: 0.18 }}
   481→            >
   482→              {step === 1 && (
   483→                <Step1Service form={form} />
   484→              )}
   485→              {step === 2 && (
   486→                <Step2Details form={form} />
   487→              )}
   488→              {step === 3 && (
   489→                <Step3Address form={form} />
   490→              )}
   491→              {step === 4 && (
   492→                <Step4Review
   493→                  form={form}
   494→                  goToStep={goToStep}
   495→                />
   496→              )}
   497→            </motion.div>
   498→          </AnimatePresence>
   499→        </div>
   500→      </ScrollArea>
   501→
   502→      {/* Sticky footer nav */}
   503→      <div className="shrink-0 border-t bg-background/95 backdrop-blur px-4 sm:px-6 py-3">
   504→        <div className="flex items-center justify-between gap-3">
   505→          <Button
   506→            type="button"
   507→            variant="ghost"
   508→            onClick={handleBack}
   509→            disabled={step === 1 || submitting}
   510→            className="text-muted-foreground hover:text-foreground"
   511→          >
   512→            Voltar
   513→          </Button>
   514→          {step < 4 ? (
   515→            <Button
   516→              type="button"
   517→              onClick={handleNext}
   518→              className="h-11 bg-emerald-600 hover:bg-emerald-700"
   519→              disabled={!stepValidMap[step]}
   520→            >
   521→              Continuar
   522→              <ChevronRight className="size-4" />
   523→            </Button>
   524→          ) : (
   525→            <Button
   526→              type="button"
   527→              onClick={onSubmit}
   528→              disabled={submitting}
   529→              className="h-11 bg-emerald-600 hover:bg-emerald-700"
   530→            >
   531→              {submitting && <Loader2 className="size-4 animate-spin" />}
   532→              <Send className="size-4" />
   533→              Enviar orçamento
   534→            </Button>
   535→          )}
   536→        </div>
   537→      </div>
   538→    </div>
   539→  )
   540→}

   594→/** Compact info card showing selected service details */
   595→function ServiceInfoCard({ providerId, serviceId }: { providerId: string; serviceId: string }) {
   596→  const { data: services } = useQuery({
   597→    queryKey: ["services-by-provider", providerId],
   598→    queryFn: () =>
   599→      apiGet<ProviderService[]>("/api/services", { providerId }),
   600→    enabled: !!providerId,
   601→    staleTime: 30 * 1000,
   602→  })
   603→
   604→  const selected = services?.find((s) => s.id === serviceId)
   605→
   606→  if (!selected) return null
   607→
   608→  return (
   609→    <div className="rounded-lg border bg-emerald-50/50 dark:bg-emerald-950/20 p-3">
   610→      <div className="flex items-start justify-between gap-2">
   611→        <div className="min-w-0">
   612→          <p className="text-sm font-medium text-emerald-800 dark:text-emerald-300 truncate">
   613→            {selected.title}
   614→          </p>
   615→          {selected.description && (
   616→            <p className="text-xs text-muted-foreground mt-0.5 line-clamp-2">
   617→              {selected.description}
   618→            </p>
   619→          )}
   620→        </div>
   621→        <Badge variant="outline" className="shrink-0 border-emerald-500 text-emerald-700 text-xs">
   622→          {selected.basePrice.toLocaleString("pt-BR", {
   623→            style: "currency",
