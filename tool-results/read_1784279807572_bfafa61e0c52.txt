   161→export function AdminDashboard({
   162→  onNavigate,
   163→}: {
   164→  onNavigate: (view: string) => void
   165→}) {
   166→  const [range, setRange] = React.useState<DateRange>("30d")
   167→
   168→  const { data, isLoading, isError, isFetching, refetch, dataUpdatedAt } =
   169→    useQuery({
   170→      queryKey: ["admin", "stats"],
   171→      queryFn: () => apiGet<AdminStats>("/api/admin/stats"),
   172→      staleTime: 30_000,
   173→    })
   174→
   175→  // H9 — erro com retry actionável (não mais linha muted)
   176→  if (isError) {
   177→    return (
   178→      <ErrorState
   179→        title="Não foi possível carregar as estatísticas"
   180→        description="Verifique sua conexão e se você está autenticado como administrador."
   181→        onRetry={() => void refetch()}
   182→      />
   183→    )
   184→  }
   185→
   186→  // H1 — skeleton que espelha o layout final (não um placeholder genérico)
   187→  if (isLoading || !data) {
   188→    return <DashboardSkeleton />
   189→  }
   190→
   191→  const totalUsers = Object.values(data.usersByRole).reduce((a, b) => a + b, 0)
   192→  const totalBookings = Object.values(data.bookingsByStatus).reduce(
   193→    (a, b) => a + b,
   194→    0,
   195→  )
   196→  const completed = data.bookingsByStatus.COMPLETED ?? 0
   197→  const pending = data.bookingsByStatus.PENDING ?? 0
   198→  const inProgress = data.bookingsByStatus.IN_PROGRESS ?? 0
   199→  const verifiedProviders = data.providers
   200→  const totalProviders = data.usersByRole.PROVIDER ?? 0
   201→  const verifyRate =
   202→    totalProviders > 0
   203→      ? Math.round((verifiedProviders / totalProviders) * 100)
   204→      : 0
   205→
   206→  // H4 — donut usa bookingTone() do admin-shared (mesma source of truth do badge)
   207→  const statusRows = BOOKING_STATUS_ORDER.map((s) => ({
   208→    status: s,
   209→    label: BOOKING_STATUS_LABELS[s],
   210→    count: data.bookingsByStatus[s] ?? 0,
   211→    tone: bookingTone(s),
   212→    fill: TONE_HEX[bookingTone(s)],
   213→  })).filter((r) => r.count > 0)
   214→  const chartTotal = statusRows.reduce((a, r) => a + r.count, 0)
   215→
   216→  const freshnessDate = dataUpdatedAt ? new Date(dataUpdatedAt) : null
   217→
   218→  return (
   219→    <div className="flex flex-col gap-6">
   220→      {/* H1 + H5 — cabeçalho com frescor, refresh e seletor de período */}
   221→      <PageSectionHeader
   222→        title="Visão geral"
   223→        description="Métricas e atividade da plataforma em tempo real."
   224→        action={
   225→          <div className="flex flex-wrap items-center gap-2">
   226→            <FreshnessLabel updatedAt={freshnessDate} />
   227→            <Button
   228→              type="button"
   229→              variant="outline"
   230→              size="sm"
   231→              onClick={() => void refetch()}
   232→              disabled={isFetching}
   233→              className="h-9 gap-1.5"
   234→            >
   235→              <RefreshCw
   236→                className={cn("size-3.5", isFetching && "animate-spin")}
   237→              />
   238→              Atualizar
   239→            </Button>
   240→            <Select
   241→              value={range}
   242→              onValueChange={(v) => setRange(v as DateRange)}
   243→            >
   244→              <SelectTrigger size="sm" className="h-9 w-[130px]">
   245→                <SelectValue />
   246→              </SelectTrigger>
   247→              <SelectContent>
   248→                {DATE_RANGE_OPTIONS.map((opt) => (
   249→                  <SelectItem key={opt.value} value={opt.value}>
   250→                    {opt.label}
   251→                  </SelectItem>
   252→                ))}
   253→              </SelectContent>
   254→            </Select>
   255→          </div>
   256→        }
   257→      />
   258→
   259→      {/* H8 — KPI grid enxuto: 4 primários (não 5+4) */}
   260→      <section
   261→        aria-label="Indicadores principais"
   262→        className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4"
   263→      >
   264→        <KpiCard
   265→          icon={Users}
   266→          label="Total de usuários"
   267→          value={totalUsers.toLocaleString("pt-BR")}
   268→          breakdown={`${data.usersByRole.CLIENT ?? 0} clientes · ${
   269→            data.usersByRole.PROVIDER ?? 0
   270→          } prestadores · ${data.usersByRole.ADMIN ?? 0} admins`}
   271→        />
   272→        <KpiCard
   273→          icon={DollarSign}
   274→          label="Receita total"
   275→          value={formatBRL(data.revenue.total)}
   276→          breakdown={`${data.revenue.paymentsPaid} pagamentos confirmados`}
   277→        />
   278→        <KpiCard
   279→          icon={CalendarCheck}
   280→          label="Agendamentos"
   281→          value={totalBookings.toLocaleString("pt-BR")}
   282→          breakdown={`${pending} pendentes · ${inProgress} em andamento · ${completed} concluídos`}
   283→        />
   284→        <KpiCard
   285→          icon={BadgeCheck}
   286→          label="Prestadores verificados"
   287→          value={verifiedProviders.toLocaleString("pt-BR")}
   288→          breakdown={`${verifiedProviders} de ${totalProviders} total · taxa ${verifyRate}%`}
   289→        />
   290→      </section>
   291→
   292→      {/* H8 — 2 charts mais úteis (não 4) */}
   293→      <section className="grid grid-cols-1 gap-4 lg:grid-cols-2">
   294→        <Card className="rounded-xl shadow-sm">
   295→          <CardContent className="p-5">
   296→            <ChartHeader
   297→              icon={CalendarCheck}
   298→              title="Agendamentos por status"
   299→              subtitle="Distribuição atual"
   300→            />
   301→            <BookingsByStatusChart rows={statusRows} total={chartTotal} />
   302→          </CardContent>
   303→        </Card>
   304→
   305→        <Card className="rounded-xl shadow-sm">
   306→          <CardContent className="p-5">
   307→            <ChartHeader
   308→              icon={TrendingUp}
   309→              title="Receita por mês"
   310→              subtitle="Últimos 6 meses — pagamentos confirmados"
   311→            />
   312→            <RevenueChart recentBookings={data.recentBookings} />
   313→          </CardContent>
   314→        </Card>
   315→      </section>
   316→
   317→      {/* H1 + H6 — atividade recente em 2 colunas, com linhas clicáveis */}
   318→      <section className="grid grid-cols-1 gap-4 lg:grid-cols-2">
   319→        <RecentBookingsCard
   320→          bookings={data.recentBookings}
   321→          onNavigate={onNavigate}
   322→        />
   323→        <TopProvidersCard
   324→          providers={data.topProviders}
   325→          onNavigate={onNavigate}
   326→        />
   327→      </section>
   328→
   329→      {/* H7 + H6 — atalhos compactos com Kbd mnemônico casado à sidebar */}
   330→      <section
   331→        aria-label="Atalhos"
   332→        className="grid grid-cols-2 gap-3 sm:grid-cols-4"
   333→      >
   334→        <QuickLink
   335→          label="Gerenciar usuários"
   336→          hint="Clientes, prestadores, admins"
   337→          icon={Users}
   338→          kbd="3"
   339→          onClick={() => onNavigate("admin.users")}
   340→        />
   341→        <QuickLink
   342→          label="Gerenciar serviços"
   343→          hint="Catálogo global"
   344→          icon={Wrench}
   345→          kbd="5"
   346→          onClick={() => onNavigate("admin.services")}
   347→        />
   348→        <QuickLink
   349→          label="Ver agendamentos"
   350→          hint="Calendário e status"
   351→          icon={CalendarCheck}
   352→          kbd="6"
   353→          onClick={() => onNavigate("admin.bookings")}
   354→        />
   355→        <QuickLink
   356→          label="Configurações"
   357→          hint="Variáveis e ajustes"
   358→          icon={SettingsIcon}
   359→          kbd="7"
   360→          onClick={() => onNavigate("admin.settings")}
   361→        />
   362→      </section>
   363→
   364→      {/* H7 — dica visual de atalho (mnemônico da sidebar) */}
   365→      <div className="flex flex-wrap items-center gap-1.5 rounded-lg border bg-muted/30 px-3 py-2 text-xs text-muted-foreground">
   366→        <span>Dica: use a navegação à esquerda para acessar outras seções.</span>
   367→        <Kbd>1</Kbd>
   368→        <Kbd>2</Kbd>
   369→        <Kbd>3</Kbd>
   370→        <span className="text-muted-foreground/70">…</span>
   371→      </div>
   372→    </div>
   373→  )
   374→}

   700→function RevenueChart({
   701→  recentBookings,
   702→}: {
   703→  recentBookings: AdminStats["recentBookings"]
   704→}) {
   705→  // Série dos últimos 6 meses sintetizada a partir de recentBookings pagos.
   706→  // Substituir por endpoint dedicado quando disponível.
   707→  const months: { label: string; revenue: number }[] = []
   708→  const now = new Date()
   709→  for (let i = 5; i >= 0; i--) {
   710→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1)
   711→    const label = d.toLocaleDateString("pt-BR", {
   712→      month: "short",
   713→      year: "2-digit",
   714→    })
   715→    months.push({ label, revenue: 0 })
   716→  }
   717→  for (const b of recentBookings) {
   718→    if (b.paymentStatus !== "PAID") continue
   719→    const created = new Date(b.createdAt)
   720→    const diffMonths =
   721→      (now.getFullYear() - created.getFullYear()) * 12 +
   722→      (now.getMonth() - created.getMonth())
   723→    const idx = months.length - 1 - diffMonths
   724→    if (idx >= 0 && idx < months.length) {
   725→      months[idx].revenue += b.amount
   726→    }
   727→  }
   728→
   729→  return (
   730→    <div className="h-[200px] w-full">
   731→      <ResponsiveContainer width="100%" height="100%">
   732→        <BarChart data={months} margin={{ left: 0, right: 8, top: 8, bottom: 0 }}>
   733→          <CartesianGrid
   734→            vertical={false}
   735→            stroke="var(--border)"
   736→            strokeDasharray="3 3"
   737→          />
   738→          <XAxis
   739→            dataKey="label"
   740→            tickLine={false}
   741→            axisLine={false}
   742→            tick={{ fontSize: 11, fill: "var(--muted-foreground)" }}
   743→          />
   744→          <YAxis
   745→            tickLine={false}
   746→            axisLine={false}
   747→            width={56}
   748→            tickFormatter={(v) =>
   749→              v >= 1000 ? `R$ ${(v / 1000).toFixed(1)}k` : `R$ ${v}`
   750→            }
   751→            tick={{ fontSize: 11, fill: "var(--muted-foreground)" }}
   752→          />
   753→          <RTooltip
   754→            cursor={{ fill: "var(--accent)", opacity: 0.5 }}
   755→            formatter={(v: number) => [formatBRL(v), "Receita"]}
   756→            contentStyle={CHART_TOOLTIP_STYLE}
   757→          />
   758→          <Bar
   759→            dataKey="revenue"
   760→            fill="var(--primary)"
   761→            radius={[4, 4, 0, 0]}
   762→            barSize={28}
   763→          />
   764→        </BarChart>
   765→      </ResponsiveContainer>
   766→    </div>
   767→  )
   768→}
