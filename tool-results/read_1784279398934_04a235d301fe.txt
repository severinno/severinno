   147→   147→function initials(name?: string | null): string {
   148→   148→  if (!name) return "?"
   149→   149→  return name
   150→   150→    .split(" ")
   151→   151→    .map((p) => p[0])
   152→   152→    .filter(Boolean)
   153→   153→    .slice(0, 2)
   154→   154→    .join("")
   155→   155→    .toUpperCase()
   156→   156→}
