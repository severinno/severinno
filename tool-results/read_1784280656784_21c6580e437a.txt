     3→/**
     4→ * AdminServices — all services across providers (read-mostly).
     5→ *
     6→ * Data source: GET  /api/admin/services (admin-only, returns ALL incl. inactive)
     7→ * Toggle active: PATCH /api/services/[id] { active }  (admin override)
     8→ * Delete:        DELETE /api/services/[id]            (admin override)
     9→ *
    10→ * Nielsen heuristics covered (design system em admin-shared.tsx):
    11→ *   H1 — TableSkeleton / ErrorState / SavingPill inline / Pagination + ResultCount
    12→ *        (a API devolve TODOS os serviços de uma vez — paginação é client-side
    13→ *         e dá feedback visível do "onde estou" no conjunto)
    14→ *   H2 — Labels pt-BR no domínio do produto (preço, unidade, categoria)
    15→ *   H3 — ConfirmToggleDialog antes de Ativar/Desativar (controle/liberdade)
    16→ *   H4 — StatusBadge (única source of truth p/ tons) substitui badges inline
    17→ *        de categoria; usa o mesmo tom por nível (emerald/teal/zinc) em todo o app
    18→ *   H5 — Switch de Ativo NÃO é instantâneo: abre ConfirmToggleDialog antes do PATCH
    19→ *        Delete continua confirmado via ConfirmDialog (destructive)
    20→ *   H6 — Botão "Ver prestador" visível em cada linha (ação primária de recognition)
    21→ *        ⋮ apenas para ações secundárias (Ativar/Desativar, Excluir)
    22→ *   H7 — Colunas Preço e Criado ordenáveis (client-side, indicador chevron)
    23→ *        Paginação permite pular páginas (não rolagem infinita opaca)
    24→ *   H8 — Categoria: badge ÚNICO da folha + tooltip com caminho completo
    25→ *        "Pai › Filha › Sub" (substitui 3 badges + chevrons poluídos)
    26→ *   H9 — ErrorState com onRetry=refetch; toast.error específico em mutations
    27→ *   H10— Tooltips em Switch, ⋮ e "Ver prestador"; caption de filtros visível
    28→ */

   162→// ---------------------------------------------------------------------------
   163→// Category tone by level (H4 — UMA source of truth, reaproveita StatusBadge)
   164→// level 0 (Pai) → emerald | level 1 (Filha) → teal | level ≥2 (Sub) → zinc
   165→// ---------------------------------------------------------------------------
   166→function categoryTone(level: number | undefined): StatusTone {
   167→  if (!level || level <= 0) return "emerald"
   168→  if (level === 1) return "teal"
   169→  return "zinc"
   170→}

   174→// ---------------------------------------------------------------------------
   175→export function AdminServices() {
   176→  const queryClient = useQueryClient()
   177→  const openProvider = useUIStore((s) => s.openProvider)
   178→
   179→  const [q, setQ] = React.useState("")
   180→  const [debouncedQ, setDebouncedQ] = React.useState("")
   181→  const [active, setActive] = React.useState<ActiveFilter>("ALL")
   182→  const [categoryFilter, setCategoryFilter] = React.useState<string>("ALL")
   183→  const [providerFilter, setProviderFilter] = React.useState<string>("ALL")
   184→  const [page, setPage] = React.useState(1)
   185→  const [sort, setSort] = React.useState<SortState>(null)
   186→
   187→  const [deleteTarget, setDeleteTarget] = React.useState<AdminService | null>(
   188→    null,
   189→  )
   190→  const [pendingToggle, setPendingToggle] = React.useState<PendingToggle>(null)
   191→  const [patchingId, setPatchingId] = React.useState<string | null>(null)
   192→  const [errorBanner, setErrorBanner] = React.useState<string | null>(null)
   193→
   194→  const limit = 10
   195→
   196→  React.useEffect(() => {
   197→    const t = setTimeout(() => {
   198→      setDebouncedQ(q.trim())
   199→      setPage(1)
   200→    }, 350)
   201→    return () => clearTimeout(t)
   202→  }, [q])
   203→
   204→  // Full category list (incl. parentId/level — usado p/ resolver caminho)
   205→  const { data: categories } = useQuery({
   206→    queryKey: ["categories", "all-flat"],
   207→    queryFn: () => apiGet<CategoryOption[]>("/api/categories"),
   208→    staleTime: 60_000,
   209→  })
   210→
   211→  // Provider options para o filtro (leve)
   212→  const { data: providersList } = useQuery({
   213→    queryKey: ["admin", "providers", "options"],
   214→    queryFn: () =>
   215→      apiGet<{ items: ProviderOption[]; total: number }>(
   216→        "/api/admin/users",
   217→        { role: "PROVIDER", limit: 100 },
   218→      ),
   219→    staleTime: 60_000,
   220→  })
   221→
   222→  // API devolve TODOS os serviços de uma vez — filtros server-side, ordenação
   223→  // e paginação client-side (H1 + H7).
   224→  const { data, isLoading, isError, refetch } = useQuery({
   225→    queryKey: [
   226→      "admin",
   227→      "services",
   228→      { debouncedQ, active, categoryFilter, providerFilter },
   229→    ],
   230→    queryFn: () =>
   231→      apiGet<AdminServicesResponse>("/api/admin/services", {
   232→        ...(debouncedQ ? { q: debouncedQ } : {}),
   233→        ...(active !== "ALL" ? { active } : {}),
   234→        ...(categoryFilter !== "ALL" ? { categoryId: categoryFilter } : {}),
   235→        ...(providerFilter !== "ALL" ? { providerId: providerFilter } : {}),
   236→      }),
   237→    staleTime: 15_000,
   238→  })
   239→
   240→  const invalidate = () =>
   241→    queryClient.invalidateQueries({ queryKey: ["admin", "services"] })
   242→
   243→  const patchMutation = useMutation({
   244→    mutationFn: ({
   245→      id,
   246→      patch,
   247→    }: {
   248→      id: string
   249→      patch: { active?: boolean }
   250→    }) => apiPatch(`/api/services/${id}`, patch),
   251→  })
   252→
   253→  const deleteMutation = useMutation({
   254→    mutationFn: (id: string) => apiDelete(`/api/services/${id}`),
   255→  })
   256→
   257→  // ---- Derived: sort + paginate client-side --------------------------------
   258→  const allItems = data?.items ?? []
   259→  const total = allItems.length
   260→
   261→  const sortedItems = React.useMemo(() => {
   262→    if (!sort) return allItems
   263→    const copy = [...allItems]
   264→    copy.sort((a, b) => {
   265→      let cmp = 0
   266→      if (sort.key === "basePrice") {
   267→        cmp = a.basePrice - b.basePrice
   268→      } else {
   269→        cmp =
   270→          new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime()
   271→      }
   272→      return sort.dir === "asc" ? cmp : -cmp
   273→    })
   274→    return copy
   275→  }, [allItems, sort])
   276→
   277→  const totalPages = Math.max(1, Math.ceil(sortedItems.length / limit))
   278→  const safePage = Math.min(page, totalPages)
   279→  const pageItems = sortedItems.slice(
   280→    (safePage - 1) * limit,
   281→    safePage * limit,
   282→  )
   283→
   284→  const activeFilterCount =
   285→    (debouncedQ ? 1 : 0) +
   286→    (active !== "ALL" ? 1 : 0) +
   287→    (categoryFilter !== "ALL" ? 1 : 0) +
   288→    (providerFilter !== "ALL" ? 1 : 0)
   289→
   290→  const clearFilters = () => {
   291→    setQ("")
   292→    setDebouncedQ("")
   293→    setActive("ALL")
   294→    setCategoryFilter("ALL")
   295→    setProviderFilter("ALL")
   296→    setSort(null)
   297→    setPage(1)
   298→  }
   299→
   300→  const toggleSort = (key: SortKey) => {
   301→    setSort((prev) => {
   302→      if (!prev || prev.key !== key) return { key, dir: "asc" }
   303→      if (prev.dir === "asc") return { key, dir: "desc" }
   304→      return null
   305→    })
   306→    setPage(1)
   307→  }
   308→
   309→  // ---- Mutation handlers ---------------------------------------------------
   310→
   311→  const handleToggleConfirm = () => {
   312→    if (!pendingToggle) return
   313→    const { id, currentValue } = pendingToggle
   314→    setPatchingId(id)
   315→    patchMutation.mutate(
   316→      { id, patch: { active: !currentValue } },
   317→      {
   318→        onSuccess: () => {
   319→          invalidate()
   320→          toast.success(
   321→            currentValue ? "Serviço desativado." : "Serviço ativado.",
   322→          )
   323→          setPendingToggle(null)
   324→          setPatchingId(null)
   325→        },
   326→        onError: (e: unknown) => {
   327→          const msg = errMsg(e, "Falha ao atualizar serviço.")
   328→          setErrorBanner(msg)
   329→          toast.error(msg)
   330→          setPendingToggle(null)
   331→          setPatchingId(null)
   332→        },
   333→      },
   334→    )
   335→  }
   336→
   337→  const handleDeleteConfirm = () => {
   338→    if (!deleteTarget) return
   339→    deleteMutation.mutate(deleteTarget.id, {
   340→      onSuccess: () => {
   341→        toast.success("Serviço excluído.")
   342→        invalidate()
   343→        setDeleteTarget(null)
   344→      },
   345→      onError: (e: unknown) => {
   346→        const msg = errMsg(e, "Não foi possível excluir o serviço.")
   347→        setErrorBanner(msg)
   348→        toast.error(msg)
   349→        setDeleteTarget(null)
   350→      },
   351→    })
   352→  }
   353→
   354→  const renderSortHeader = (
   355→    label: string,
   356→    sortKey: SortKey,
   357→    align: "left" | "right" = "left",
   358→  ) => (
   359→    <button
   360→      type="button"
   361→      onClick={() => toggleSort(sortKey)}
   362→      className={cn(
   363→        "inline-flex items-center gap-1 text-xs font-semibold uppercase tracking-wide text-muted-foreground transition-colors hover:text-foreground",
   364→        align === "right" && "flex-row-reverse",
   365→      )}
   366→    >
   367→      {label}
   368→      {sort?.key === sortKey ? (
   369→        sort.dir === "asc" ? (
   370→          <ChevronUp className="size-3" />
   371→        ) : (
   372→          <ChevronDown className="size-3" />
   373→        )
   374→      ) : (
   375→        <ArrowUpDown className="size-3 opacity-40" />
   376→      )}
   377→    </button>
   378→  )
   379→
   380→  return (
   381→    <div className="flex flex-col gap-4">
   382→      <PageSectionHeader
   383→        title="Serviços"
   384→        description="Gerencie o catálogo de serviços oferecidos pelos prestadores."
   385→      />
   386→
   387→      {/* Error banner (dismissible) — H9 */}
   388→      {errorBanner ? (
   389→        <Alert variant="destructive">
   390→          <AlertTriangle className="size-4" />
   391→          <AlertTitle>Erro ao salvar</AlertTitle>
   392→          <AlertDescription>{errorBanner}</AlertDescription>
   393→          <button
   394→            type="button"
   395→            onClick={() => setErrorBanner(null)}
   396→            aria-label="Dispensar aviso"
   397→            className="absolute right-3 top-3 rounded-md p-1 text-current/70 transition-colors hover:text-current"
   398→          >
   399→            <X className="size-3.5" />
   400→          </button>
   401→        </Alert>
   402→      ) : null}
   403→
   404→      {/* Filter bar (H4 consistência) */}
   405→      <FilterBar
   406→        onClear={clearFilters}
   407→        activeCount={activeFilterCount}
   408→        resultCount={total}
   409→        resultLabel={
   410→          total === 1 ? "serviço encontrado" : "serviços encontrados"
   411→        }
   412→      >
   413→        <SearchInput
   414→          value={q}
   415→          onChange={setQ}
   416→          placeholder="Buscar por título ou descrição"
   417→          className="min-w-[200px] flex-1"
   418→        />
   419→        <Select
   420→          value={categoryFilter}
   421→          onValueChange={(v) => {
   422→            setCategoryFilter(v)
   423→            setPage(1)
   424→          }}
   425→        >
   426→          <SelectTrigger className="h-9 w-auto min-w-[160px]">
   427→            <SelectValue placeholder="Categoria" />
   428→          </SelectTrigger>
   429→          <SelectContent>
   430→            <SelectItem value="ALL">Todas as categorias</SelectItem>
   431→            {(categories ?? [])
   432→              .slice()
   433→              .sort(
   434→                (a, b) =>
   435→                  a.level - b.level ||
   436→                  (a.order ?? 0) - (b.order ?? 0) ||
   437→                  a.name.localeCompare(b.name),
   438→              )
   439→              .map((c) => (
   440→                <SelectItem key={c.id} value={c.id}>
   441→                  <span className="text-muted-foreground">
   442→                    {"—".repeat(c.level)}
   443→                  </span>{" "}
   444→                  {c.name}
   445→                  <span className="ml-1.5 text-[10px] uppercase tracking-wide text-muted-foreground">
   446→                    {c.level === 0
   447→                      ? "pai"
   448→                      : c.level === 1
   449→                        ? "filha"
   450→                        : "sub"}
   451→                  </span>
   452→                </SelectItem>
   453→              ))}
   454→          </SelectContent>
   455→        </Select>
   456→        <Select
   457→          value={providerFilter}
   458→          onValueChange={(v) => {
   459→            setProviderFilter(v)
   460→            setPage(1)
   461→          }}
   462→        >
   463→          <SelectTrigger className="h-9 w-auto min-w-[160px]">
   464→            <SelectValue placeholder="Prestador" />
   465→          </SelectTrigger>
   466→          <SelectContent>
   467→            <SelectItem value="ALL">Todos os prestadores</SelectItem>
   468→            {(providersList?.items ?? []).map((p) => (
   469→              <SelectItem key={p.id} value={p.id}>
   470→                {p.name}
   471→              </SelectItem>
   472→            ))}
   473→          </SelectContent>
   474→        </Select>
   475→        <Select
   476→          value={active}
   477→          onValueChange={(v) => {
   478→            setActive(v as ActiveFilter)
   479→            setPage(1)
   480→          }}
   481→        >
   482→          <SelectTrigger className="h-9 w-auto min-w-[140px]">
   483→            <SelectValue placeholder="Status" />
   484→          </SelectTrigger>
   485→          <SelectContent>
   486→            <SelectItem value="ALL">Status: todos</SelectItem>
   487→            <SelectItem value="true">Apenas ativos</SelectItem>
   488→            <SelectItem value="false">Apenas inativos</SelectItem>
   489→          </SelectContent>
   490→        </Select>
   491→      </FilterBar>
   492→
   493→      {/* Table area: error / loading / empty / table */}
   494→      {isError ? (
   495→        <ErrorState
   496→          title="Não foi possível carregar os serviços"
   497→          description="Verifique sua conexão e tente novamente."
   498→          onRetry={() => refetch()}
   499→        />
   500→      ) : isLoading ? (
   501→        <TableSkeleton rows={8} cols={6} />
   502→      ) : pageItems.length === 0 ? (
   503→        <EmptyState
   504→          icon={SearchX}
   505→          title="Nenhum serviço encontrado"
   506→          description="Ajuste os filtros de busca ou aguarde novos cadastros."
   507→          action={
   508→            activeFilterCount > 0 ? (
   509→              <Button
   510→                variant="outline"
   511→                size="sm"
   512→                onClick={clearFilters}
   513→                className="gap-1.5"
   514→              >
   515→                <X className="size-3.5" />
   516→                Limpar filtros
   517→              </Button>
   518→            ) : undefined
   519→          }
   520→        />
   521→      ) : (
   522→        <Card className="overflow-hidden">
   523→          <CardContent className="p-0">
   524→            <div className="overflow-x-auto">
   525→              <Table>
   526→                <TableHeader>
   527→                  <TableRow className="h-11 bg-muted/50 hover:bg-muted/50">
   528→                    <TableHead className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   529→                      Serviço
   530→                    </TableHead>
   531→                    <TableHead className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   532→                      Prestador
   533→                    </TableHead>
   534→                    <TableHead className="hidden text-xs font-semibold uppercase tracking-wide text-muted-foreground md:table-cell">
   535→                      Categoria
   536→                    </TableHead>
   537→                    <TableHead className="text-right">
   538→                      {renderSortHeader("Preço", "basePrice", "right")}
   539→                    </TableHead>
   540→                    <TableHead className="text-center text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   541→                      Ativo
   542→                    </TableHead>
   543→                    <TableHead className="hidden sm:table-cell">
   544→                      {renderSortHeader("Criado", "createdAt")}
   545→                    </TableHead>
   546→                    <TableHead className="text-right text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   547→                      Ações
   548→                    </TableHead>
   549→                  </TableRow>
   550→                </TableHeader>
   551→                <TableBody>
   552→                  {pageItems.map((s) => {
   553→                    const path = resolveCategoryPath(s.category, categories)
   554→                    const leaf = path.length > 0 ? path[path.length - 1] : null
   555→                    const pathStr = path.map((c) => c.name).join(" › ")
   556→                    const photo = firstPhoto(s.photos)
   557→                    const isPatchingThis =
   558→                      patchingId === s.id && patchMutation.isPending
   559→                    return (
   560→                      <TableRow
   561→                        key={s.id}
   562→                        className={cn(
   563→                          "h-14 border-b transition-colors last:border-0 hover:bg-muted/30",
   564→                          !s.active && "opacity-70",
   565→                        )}
   566→                      >
   567→                        <TableCell className="px-4 py-3">
   568→                          <div className="flex items-center gap-2.5">
   569→                            <div className="flex size-10 shrink-0 items-center justify-center overflow-hidden rounded-md border bg-muted">
   570→                              {photo ? (
   571→                                <img
   572→                                  src={photo}
   573→                                  alt={s.title}
   574→                                  className="size-full object-cover"
   575→                                />
   576→                              ) : (
   577→                                <ImageIcon className="size-4 text-muted-foreground" />
   578→                              )}
   579→                            </div>
   580→                            <div className="min-w-0">
   581→                              <p className="truncate text-sm font-medium">
   582→                                {s.title}
   583→                              </p>
   584→                              <p className="truncate text-xs text-muted-foreground">
   585→                                {s.description.slice(0, 80)}
   586→                                {s.description.length > 80 ? "…" : ""}
   587→                              </p>
   588→                            </div>
   589→                          </div>
   590→                        </TableCell>
   591→                        <TableCell className="px-4 py-3">
   592→                          <div className="flex items-center gap-2">
   593→                            <Avatar className="size-7 shrink-0">
   594→                              {s.provider.avatarUrl ? (
   595→                                <AvatarImage
   596→                                  src={s.provider.avatarUrl}
   597→                                  alt={s.provider.name}
   598→                                />
   599→                              ) : null}
   600→                              <AvatarFallback className="bg-primary/10 text-[10px] font-semibold text-primary">
   601→                                {initials(s.provider.name)}

   815→/** Resolve a cadeia pai › filha › sub a partir da categoria folha. */
   816→function resolveCategoryPath(
   817→  leaf: CategoryOption | null | undefined,
   818→  all: CategoryOption[] | undefined,
   819→): CategoryOption[] {
   820→  if (!leaf || !all) return []
   821→  const byId = new Map(all.map((c) => [c.id, c]))
   822→  const path: CategoryOption[] = []
   823→  let current: CategoryOption | undefined = leaf
   824→  let safety = 0
   825→  while (current && safety < 5) {
   826→    path.unshift(current)
   827→    current = current.parentId ? byId.get(current.parentId) : undefined
   828→    safety++
   829→  }
   830→  return path
   831→}
