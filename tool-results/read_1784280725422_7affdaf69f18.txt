     4→     4→ * AdminTaxonomy — gerenciador da árvore de categorias em 3 níveis.
     5→     5→ *
     6→     6→ * Heurísticas de Nielsen aplicadas neste redesign:
     7→     7→ *   H1  Visibilidade do status  → PageSectionHeader + contagem + SavingPill
     8→     8→ *   H2  Correspondência c/ mundo real → labels pt-BR; remoção do drag-handle
     9→     9→ *        visual-only (substituído por botões explícitos "Mover p/ cima/baixo")
    10→    10→ *   H3  Controle e liberdade    → ConfirmDialog antes de excluir; toggle c/ confirmação
    11→    11→ *   H4  Consistência            → StatusBadge (ÚNICA source of truth) p/ nível e estado
    12→    12→ *   H5  Prevenção de erros      → Switch não é instantâneo: abre ConfirmToggleDialog
    13→    13→ *   H6  Reconhecimento          → Select de ícones com preview; tooltips em icon-buttons
    14→    14→ *   H7  Eficiência              → Botões Mover p/ cima/baixo (disabled c/ tooltip honesto)
    15→    15→ *   H8  Minimalismo             → hierarquia clara, retirada de badge "serviço(s)" sem dados
    16→    16→ *   H9  Recuperar erros         → ErrorState c/ retry; toast específico p/ 409
    17→    17→ *   H10 Ajuda e documentação    → Info tooltip no campo Slug
    18→    18→ *
    19→    19→ * APIs (relativas, mantidas):
    20→    20→ *   GET    /api/categories
    21→    21→ *   POST   /api/categories            { name, slug, parentId?, level, icon?, order? }
    22→    22→ *   PATCH  /api/categories/[id]       { name?, slug?, parentId?, level?, icon?, order?, active? }
    23→    23→ *   DELETE /api/categories/[id]       (409 se tiver filhos ou serviços)
    24→    24→ *
    25→    25→ * Observação H7: a API atual não expõe endpoint de reordenação. Para não enganar
    26→    26→ * o administrador (H2), mantemos os botões "Mover p/ cima/baixo" visíveis mas
    27→    27→ * desabilitados com tooltip explicativo — em vez de um "drag handle" mudo que

   200→   200→type PendingToggle = {

   202→   202→  name: string

   205→   205→
   206→   206→export function AdminTaxonomy() {
   207→   207→  const queryClient = useQueryClient()
   208→   208→  const [createOpen, setCreateOpen] = React.useState(false)
   209→   209→  const [editTarget, setEditTarget] = React.useState<CategoryNode | null>(null)
   210→   210→  const [deleteTarget, setDeleteTarget] = React.useState<CategoryNode | null>(
   211→   211→    null,
   212→   212→  )
   213→   213→  const [expanded, setExpanded] = React.useState<Set<string>>(new Set())
   214→   214→  const [pendingToggle, setPendingToggle] = React.useState<PendingToggle>(null)
   215→   215→
   216→   216→  const { data: flat, isLoading, isError, refetch } = useQuery({
   217→   217→    queryKey: ["categories", "all"],
   218→   218→    queryFn: () => apiGet<CategoryRow[]>("/api/categories"),
   219→   219→    staleTime: 30_000,
   220→   220→  })
   221→   221→
   222→   222→  // Build the tree (from a flat list)
   223→   223→  const tree = React.useMemo<CategoryNode[]>(() => {
   224→   224→    if (!flat) return []
   225→   225→    const byParent = new Map<string | null, CategoryNode[]>()
   226→   226→    for (const c of flat) {
   227→   227→      const key = c.parentId ?? null
   228→   228→      const arr = byParent.get(key) ?? []
   229→   229→      arr.push({ ...c, children: [] })
   230→   230→      byParent.set(key, arr)
   231→   231→    }
   232→   232→    const roots = byParent.get(null) ?? []
   233→   233→    const build = (nodes: CategoryNode[]): CategoryNode[] =>
   234→   234→      nodes
   235→   235→        .sort(
   236→   236→          (a, b) =>
   237→   237→            (a.order ?? 0) - (b.order ?? 0) || a.name.localeCompare(b.name),
   238→   238→        )
   239→   239→        .map((n) => {
   240→   240→          const children = byParent.get(n.id) ?? []
   241→   241→          n.children = build(children)
   242→   242→          return n
   243→   243→        })
   244→   244→    return build(roots)
   245→   245→  }, [flat])
   246→   246→
   247→   247→  // H1/H8 — service count por categoria não está disponível no MVP (o endpoint
   248→   248→  // /api/categories não retorna contagem de serviços). Em vez de manter um
   249→   249→  // badge que nunca mostra nada (H1 violado), simplesmente NÃO renderizamos.
   250→   250→  // TODO(N1/Svcs): quando /api/services?categoryId=X existir, alimentar aqui.
   251→   251→
   252→   252→  // Auto-expand all level-0 nodes on first load
   253→   253→  React.useEffect(() => {
   254→   254→    if (tree.length && expanded.size === 0) {
   255→   255→      setExpanded(new Set(tree.map((n) => n.id)))
   256→   256→    }
   257→   257→  }, [tree, expanded.size])
   258→   258→
   259→   259→  const invalidate = () =>
   260→   260→    queryClient.invalidateQueries({ queryKey: ["categories"] })
   261→   261→
   262→   262→  const createMutation = useMutation({
   263→   263→    mutationFn: (payload: CategoryPayload) =>
   264→   264→      apiPost<{ category: CategoryRow }>("/api/categories", payload),
   265→   265→    onSuccess: () => {
   266→   266→      toast.success("Categoria criada com sucesso.")
   267→   267→      invalidate()
   268→   268→      setCreateOpen(false)
   269→   269→    },
   270→   270→    onError: (e: unknown) => {
   271→   271→      toast.error(errMsg(e, "Não foi possível criar a categoria."))
   272→   272→    },
   273→   273→  })
   274→   274→
   275→   275→  const patchMutation = useMutation({
   276→   276→    mutationFn: ({ id, patch }: { id: string; patch: Partial<CategoryPayload> }) =>
   277→   277→      apiPatch<{ category: CategoryRow }>(`/api/categories/${id}`, patch),
   278→   278→    onSuccess: () => {
   279→   279→      toast.success("Categoria atualizada.")
   280→   280→      invalidate()
   281→   281→      setEditTarget(null)
   282→   282→    },
   283→   283→    onError: (e: unknown) => {
   284→   284→      toast.error(errMsg(e, "Não foi possível atualizar a categoria."))
   285→   285→    },
   286→   286→  })
   287→   287→
   288→   288→  const toggleActiveMutation = useMutation({
   289→   289→    mutationFn: ({ id, active }: { id: string; active: boolean }) =>
   290→   290→      apiPatch<{ category: CategoryRow }>(`/api/categories/${id}`, { active }),
   291→   291→    onSuccess: (_d, vars) => {
   292→   292→      toast.success(
   293→   293→        vars.active
   294→   294→          ? "Categoria ativada."
   295→   295→          : "Categoria desativada.",
   296→   296→      )
   297→   297→      invalidate()
   298→   298→      setPendingToggle(null)
   299→   299→    },
   300→   300→    onError: (e: unknown) => {
   301→   301→      toast.error(errMsg(e, "Falha ao alternar o estado ativo."))
   302→   302→      setPendingToggle(null)
   303→   303→    },
   304→   304→  })
   305→   305→
   306→   306→  const deleteMutation = useMutation({
   307→   307→    mutationFn: (id: string) => apiDelete(`/api/categories/${id}`),
   308→   308→    onSuccess: () => {
   309→   309→      toast.success("Categoria excluída.")
   310→   310→      invalidate()
   311→   311→      setDeleteTarget(null)
   312→   312→    },
   313→   313→    onError: (e: unknown) => {
   314→   314→      const msg = errMsg(e, "Não foi possível excluir.")
   315→   315→      // H9 — mensagem específica para 409 (vínculos existentes)
   316→   316→      toast.error(msg, {
   317→   317→        description:
   318→   318→          "Remova os vínculos (filhos ou serviços) antes de tentar novamente.",
   319→   319→      })
   320→   320→    },
   321→   321→  })
   322→   322→
   323→   323→  const toggleExpand = (id: string) =>
   324→   324→    setExpanded((prev) => {
   325→   325→      const next = new Set(prev)
   326→   326→      if (next.has(id)) next.delete(id)
   327→   327→      else next.add(id)
   328→   328→      return next
   329→   329→    })
   330→   330→
   331→   331→  // H5 — handler do confirm-toggle
   332→   332→  const handleToggleConfirm = () => {
   333→   333→    if (!pendingToggle) return
   334→   334→    toggleActiveMutation.mutate({
   335→   335→      id: pendingToggle.id,
   336→   336→      active: !pendingToggle.currentValue,
   337→   337→    })
   338→   338→  }
   339→   339→
   340→   340→  const totalNodes = flat?.length ?? 0
   341→   341→  const expandAll = () => {
   342→   342→    if (!flat) return
   343→   343→    setExpanded(new Set(flat.map((c) => c.id)))
   344→   344→  }
   345→   345→  const collapseAll = () => setExpanded(new Set())
   346→   346→
   347→   347→  return (
   348→   348→    <div className="flex flex-col gap-4">
   349→   349→      <PageSectionHeader
   350→   350→        title="Taxonomia de categorias"
   351→   351→        description="Organize a árvore em até 3 níveis: pai, filha e subcategoria."
   352→   352→      />
   353→   353→
   354→   354→      {/* Toolbar — H8 minimalismo, agrupa contagem + ações de árvore */}
   355→   355→      <div className="flex flex-wrap items-center justify-between gap-3 rounded-xl border bg-card p-3 shadow-sm">
   356→   356→        <div className="flex flex-wrap items-center gap-2">
   357→   357→          <span className="inline-flex items-center gap-1.5 rounded-full border bg-muted/40 px-2.5 py-1 text-xs font-medium text-muted-foreground">
   358→   358→            <FolderTree className="size-3.5 text-primary" />
   359→   359→            {totalNodes} {totalNodes === 1 ? "categoria" : "categorias"}
   360→   360→          </span>
   361→   361→          <span className="hidden items-center gap-1.5 text-xs text-muted-foreground sm:flex">
   362→   362→            Níveis:
   363→   363→            {([0, 1, 2] as const).map((lv) => (
   364→   364→              <StatusBadge key={lv} tone={LEVEL_META[lv].tone}>
   365→   365→                <span className={cn("size-1.5 rounded-full", LEVEL_META[lv].dot)} />
   366→   366→                {LEVEL_META[lv].label}
   367→   367→              </StatusBadge>
   368→   368→            ))}
   369→   369→          </span>
   370→   370→        </div>
   371→   371→        <div className="flex flex-wrap items-center gap-2">
   372→   372→          <Button
   373→   373→            variant="outline"
   374→   374→            size="sm"
   375→   375→            onClick={expandAll}
   376→   376→            disabled={!flat || flat.length === 0}
   377→   377→            className="h-8 gap-1.5"
   378→   378→          >
   379→   379→            <ChevronsUpDown className="size-3.5" />
   380→   380→            Expandir tudo
   381→   381→          </Button>
   382→   382→          <Button
   383→   383→            variant="outline"
   384→   384→            size="sm"
   385→   385→            onClick={collapseAll}
   386→   386→            disabled={expanded.size === 0}
   387→   387→            className="h-8 gap-1.5"
   388→   388→          >
   389→   389→            <ChevronsDownUp className="size-3.5" />
   390→   390→            Recolher tudo
   391→   391→          </Button>
   392→   392→          <Button
   393→   393→            size="sm"
   394→   394→            onClick={() => setCreateOpen(true)}
   395→   395→            className="h-8 gap-1.5"
   396→   396→          >
   397→   397→            <Plus className="size-4" />
   398→   398→            Nova categoria
   399→   399→          </Button>
   400→   400→        </div>
   401→   401→      </div>
   402→   402→
   403→   403→      {/* H9 — error recovery */}
   404→   404→      {isError ? (
   405→   405→        <ErrorRetry onRetry={() => refetch()} />
   406→   406→      ) : (
   407→   407→        <Card className="overflow-hidden">
   408→   408→          <CardContent className="p-0">
   409→   409→            {isLoading ? (
   410→   410→              <div className="flex flex-col gap-2 p-4">
   411→   411→                {Array.from({ length: 4 }).map((_, i) => (
   412→   412→                  <Skeleton key={i} className="h-14 w-full" />
   413→   413→                ))}
   414→   414→              </div>
   415→   415→            ) : tree.length === 0 ? (
   416→   416→              <div className="flex flex-col items-center gap-3 px-4 py-16 text-center">
   417→   417→                <div className="flex size-16 items-center justify-center rounded-full bg-muted text-muted-foreground">
   418→   418→                  <FolderTree className="size-7" />
   419→   419→                </div>
   420→   420→                <div>
   421→   421→                  <p className="text-lg font-semibold">
   422→   422→                    Nenhuma categoria cadastrada
   423→   423→                  </p>
   424→   424→                  <p className="mt-0.5 text-sm text-muted-foreground">
   425→   425→                    Crie a primeira categoria pai para iniciar a taxonomia.
   426→   426→                  </p>
   427→   427→                </div>
   428→   428→                <Button
   429→   429→                  size="sm"
   430→   430→                  onClick={() => setCreateOpen(true)}
   431→   431→                  className="gap-1.5"
   432→   432→                >
   433→   433→                  <Plus className="size-4" />
   434→   434→                  Criar primeira categoria
   435→   435→                </Button>
   436→   436→              </div>
   437→   437→            ) : (
   438→   438→              <ul className="flex flex-col" role="tree">
   439→   439→                {tree.map((node) => (
   440→   440→                  <TreeNode
   441→   441→                    key={node.id}
   442→   442→                    node={node}
   443→   443→                    level={0}
   444→   444→                    expanded={expanded}
   445→   445→                    onToggleExpand={toggleExpand}
   446→   446→                    onEdit={(n) => setEditTarget(n)}
   447→   447→                    onDelete={(n) => setDeleteTarget(n)}
   448→   448→                    onRequestToggleActive={(n) =>
   449→   449→                      setPendingToggle({
   450→   450→                        id: n.id,
   451→   451→                        name: n.name,
   452→   452→                        currentValue: n.active,
   453→   453→                      })
   454→   454→                    }
   455→   455→                  />
   456→   456→                ))}
   457→   457→              </ul>
   458→   458→            )}
   459→   459→          </CardContent>
   460→   460→        </Card>
   461→   461→      )}
   462→   462→
   463→   463→      {/* Create dialog */}
   464→   464→      <CategoryDialog
   465→   465→        open={createOpen}
   466→   466→        onOpenChange={setCreateOpen}
   467→   467→        title="Nova categoria"
   468→   468→        description="Defina o nome, slug e o pai. O nível é determinado automaticamente pelo pai escolhido."
   469→   469→        allCategories={(flat as CategoryRow[] | undefined) ?? []}
