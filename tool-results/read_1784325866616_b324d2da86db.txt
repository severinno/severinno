   116→export function BookingModal() {
   117→  const open = useUIStore((s) => s.bookingModal.open)
   118→  const providerIdPreset = useUIStore((s) => s.bookingModal.providerId)
   119→  const serviceIdPreset = useUIStore((s) => s.bookingModal.serviceId)
   120→  const close = useUIStore((s) => s.closeBooking)
   121→  const openAuth = useUIStore((s) => s.openAuth)
   122→  const isMobile = useIsMobile()
   123→  const navigate = useViewStore((s) => s.navigate)
   124→  const user = useAuthStore((s) => s.user)
   125→
   126→  const [step, setStep] = React.useState<Step>(1)
   127→  const [state, setState] = React.useState<BookingFormState>(initialState())
   128→  const [submitting, setSubmitting] = React.useState(false)
   129→
   130→  // Reset when modal opens.
   131→  React.useEffect(() => {
   132→    if (open) {
   133→      setStep(1)
   134→      setState(initialState())
   135→    }
   136→  }, [open])
   137→
   138→  // Fetch provider + services
   139→  const providerQuery = useQuery({
   140→    queryKey: ["provider", providerIdPreset],
   141→    queryFn: () =>
   142→      apiGet<ProviderDetail>(`/api/providers/${providerIdPreset}`),
   143→    enabled: open && !!providerIdPreset,
   144→    staleTime: 60 * 1000,
   145→  })
   146→
   147→  const servicesQuery = useQuery({
   148→    queryKey: ["services-by-provider", providerIdPreset],
   149→    queryFn: () =>
   150→      apiGet<ProviderService[]>("/api/services", {
   151→        providerId: providerIdPreset,
   152→      }),
   153→    enabled: open && !!providerIdPreset,
   154→    staleTime: 60 * 1000,
   155→  })
   156→
   157→  const provider = providerQuery.data
   158→  const services = servicesQuery.data ?? []
   159→  const selectedService =
   160→    services.find((s) => s.id === serviceIdPreset) ?? services[0]
   161→
   162→  // Pre-fill quantity from service default unit
   163→  React.useEffect(() => {
   164→    if (open && selectedService) {
   165→      setState((s) => ({ ...s, quantity: 1 }))
   166→    }
   167→  }, [open, selectedService])
   168→
   169→  const set = <K extends keyof BookingFormState>(
   170→    key: K,
   171→    value: BookingFormState[K],
   172→  ) => setState((s) => ({ ...s, [key]: value }))
   173→
   174→  // Step validation
   175→  const step1Valid = !!state.date && !!state.time
   176→  const step2Valid =
   177→    !!state.address.cep &&
   178→    state.address.cep.replace(/\D/g, "").length === 8 &&
   179→    !!state.address.street &&
   180→    !!state.address.number &&
   181→    !!state.address.city &&
   182→    !!state.address.state
   183→  const step3Valid =
   184→    state.paymentMethod === "PIX" ||
   185→    (state.paymentMethod === "CARD" &&
   186→      state.cardName.length >= 3 &&
   187→      state.cardNumber.replace(/\s/g, "").length >= 13 &&
   188→      /^\d{2}\/\d{2}$/.test(state.cardExpiry) &&
   189→      /^\d{3,4}$/.test(state.cardCvv))
   190→
   191→  const handleNext = () => {
   192→    if (step === 1 && !step1Valid) {
   193→      toast.error("Selecione data e horário para continuar.")
   194→      return
   195→    }
   196→    if (step === 2 && !step2Valid) {
   197→      toast.error("Preencha o endereço completo.")
   198→      return
   199→    }
   200→    setStep((s) => Math.min(3, s + 1) as Step)
   201→  }
   202→  const handleBack = () => setStep((s) => Math.max(1, s - 1) as Step)
   203→
   204→  const handleSubmit = async () => {
   205→    if (!user) {
   206→      toast.info("Faça cadastro gratuito para agendar serviços.")
   207→      openAuth("register", "CLIENT")
   208→      return
   209→    }
   210→    if (!provider || !selectedService || !state.date || !state.time) {
   211→      toast.error("Dados incompletos. Revise o agendamento.")
   212→      return
   213→    }
   214→    if (!step3Valid) {
   215→      toast.error("Verifique os dados de pagamento.")
   216→      return
   217→    }
   218→
   219→    // Combine date + "HH:mm" into ISO
   220→    const [h, m] = state.time.split(":").map(Number)
   221→    const scheduledAt = new Date(state.date)
   222→    scheduledAt.setHours(h ?? 0, m ?? 0, 0, 0)
   223→
   224→    const amount =
   225→      (selectedService.basePrice || 0) * (state.quantity || 1)
   226→
   227→    setSubmitting(true)
   228→    try {
   229→      await apiPost("/api/bookings", {
   230→        providerId: provider.id,
   231→        serviceId: selectedService.id,
   232→        scheduledAt: scheduledAt.toISOString(),
   233→        address: [
   234→          state.address.street,
   235→          state.address.number,
   236→          state.address.complement,
   237→          state.address.district,
   238→        ]
   239→          .filter(Boolean)
   240→          .join(", "),
   241→        cep: state.address.cep,
   242→        lat: state.address.lat,
   243→        lng: state.address.lng,
   244→        amount,
   245→        paymentMethod: state.paymentMethod,
   246→        notes: state.notes,
   247→        quantity: state.quantity,
   248→        unit: selectedService.unit,
   249→      })
   250→      toast.success("Agendamento confirmado! Acompanhe em seus agendamentos.")
   251→      close()
   252→      navigate("client.bookings")
   253→    } catch (e) {
   254→      const msg =
   255→        (e as { message?: string })?.message ??
   256→        "Não foi possível concluir o agendamento. Tente novamente."
   257→      toast.error(msg)
   258→    } finally {
   259→      setSubmitting(false)
   260→    }
   261→  }
   262→
   263→  const content = (
   264→    <BookingBody
   265→      step={step}
   266→      state={state}
   267→      set={set}
   268→      provider={provider}
   269→      services={services}
   270→      selectedService={selectedService}
   271→      loading={providerQuery.isLoading || servicesQuery.isLoading}
   272→      onNext={handleNext}
   273→      onBack={handleBack}
   274→      onSubmit={handleSubmit}
   275→      submitting={submitting}
   276→      step1Valid={step1Valid}
   277→      step2Valid={step2Valid}
   278→      step3Valid={step3Valid}
   279→    />
   280→  )
   281→
   282→  if (isMobile) {
   283→    return (
   284→      <Sheet open={open} onOpenChange={(o) => !o && close()}>
   285→        <SheetContent
   286→          side="bottom"
   287→          className="h-[100dvh] max-h-[100dvh] w-full p-0 sm:max-w-full gap-0 flex flex-col"
   288→        >
   289→          <SheetHeader className="px-4 pt-4 pb-2 shrink-0">
   290→            <SheetTitle className="flex items-center gap-2">
   291→              <CalendarDays className="size-5 text-emerald-600" />
   292→              Agendar serviço
   293→            </SheetTitle>
   294→            <SheetDescription>
   295→              Escolha data, detalhes e pagamento.
   296→            </SheetDescription>
   297→          </SheetHeader>
   298→          <div className="flex-1 overflow-hidden">{content}</div>
   299→        </SheetContent>
   300→      </Sheet>
   301→    )
   302→  }
   303→
   304→  return (
   305→    <Dialog open={open} onOpenChange={(o) => !o && close()}>
   306→      <DialogContent className="sm:max-w-lg p-0 gap-0 overflow-hidden">
   307→        <DialogHeader className="px-6 pt-6 pb-3 shrink-0 border-b">
   308→          <DialogTitle className="flex items-center gap-2 text-lg">
   309→            <CalendarDays className="size-5 text-emerald-600" />
   310→            Agendar serviço
   311→          </DialogTitle>
   312→          <DialogDescription>
   313→            Escolha data, detalhes e pagamento.
   314→          </DialogDescription>
   315→        </DialogHeader>
   316→        {content}
   317→      </DialogContent>
   318→    </Dialog>
   319→  )
   320→}

   326→function BookingBody({
   327→  step,
   328→  state,
   329→  set,
   330→  provider,
   331→  services,
   332→  selectedService,
   333→  loading,
   334→  onNext,
   335→  onBack,
   336→  onSubmit,
   337→  submitting,
   338→  step1Valid,
   339→  step2Valid,
   340→  step3Valid,
   341→}: {
   342→  step: Step
   343→  state: BookingFormState
   344→  set: <K extends keyof BookingFormState>(
   345→    key: K,
   346→    value: BookingFormState[K],
   347→  ) => void
   348→  provider?: ProviderDetail
   349→  services: ProviderService[]
   350→  selectedService?: ProviderService
   351→  loading: boolean
   352→  onNext: () => void
   353→  onBack: () => void
   354→  onSubmit: () => void
   355→  submitting: boolean
   356→  step1Valid: boolean
   357→  step2Valid: boolean
   358→  step3Valid: boolean
   359→}) {
   360→  const progressPct = (step / 3) * 100
   361→
   362→  return (
   363→    <div className="flex h-full flex-col">
   364→      {/* Stepper */}
   365→      <div className="shrink-0 border-b px-4 sm:px-6 py-3">
   366→        <div className="flex items-center justify-between">
   367→          {STEPS.map((s, i) => {
   368→            const active = step === s.id
   369→            const done = step > s.id
   370→            const Icon = s.icon
   371→            return (
   372→              <React.Fragment key={s.id}>
   373→                <div
   374→                  className={cn(
   375→                    "flex items-center gap-2 text-xs sm:text-sm transition-colors",
   376→                    active
   377→                      ? "text-emerald-700 dark:text-emerald-400 font-medium"
   378→                      : done
   379→                        ? "text-emerald-600"
   380→                        : "text-muted-foreground",
   381→                  )}
   382→                >
   383→                  <span
   384→                    className={cn(
   385→                      "inline-flex size-7 items-center justify-center rounded-full border-2 transition-colors",
   386→                      active &&
   387→                        "border-emerald-600 bg-emerald-600 text-white",
   388→                      done && "border-emerald-600 bg-emerald-600 text-white",
   389→                      !active && !done && "border-muted-foreground/30",
   390→                    )}
   391→                  >
   392→                    {done ? (
   393→                      <Check className="size-3.5" />
   394→                    ) : (
   395→                      <Icon className="size-3.5" />
   396→                    )}
   397→                  </span>
   398→                  <span className="hidden sm:inline">{s.label}</span>
   399→                </div>
   400→                {i < STEPS.length - 1 && (
   401→                  <div className="flex-1 h-0.5 bg-muted-foreground/20 mx-2 relative">
   402→                    <div
   403→                      className="absolute inset-0 bg-emerald-600 transition-transform origin-left"
   404→                      style={{
   405→                        transform:
   406→                          step > s.id ? "scaleX(1)" : "scaleX(0)",
   407→                      }}
   408→                    />
   409→                  </div>
   410→                )}
   411→              </React.Fragment>
   412→            )
   413→          })}
   414→        </div>
   415→        <Progress
   416→          value={progressPct}
   417→          className="mt-2 h-1 bg-emerald-100 dark:bg-emerald-950/40"
   418→        />
   419→      </div>
   420→
   421→      {/* Step content */}
   422→      <ScrollArea className="flex-1">
   423→        <div className="px-4 sm:px-6 py-4">
   424→          {loading ? (
   425→            <div className="flex items-center justify-center py-12">
   426→              <Loader2 className="size-6 animate-spin text-emerald-600" />
   427→            </div>
   428→          ) : (
   429→            <AnimatePresence mode="wait">
   430→              <motion.div
   431→                key={step}
   432→                initial={{ opacity: 0, x: 10 }}
   433→                animate={{ opacity: 1, x: 0 }}
   434→                exit={{ opacity: 0, x: -10 }}
   435→                transition={{ duration: 0.18 }}
   436→              >
   437→                {step === 1 && (
   438→                  <Step1Schedule
   439→                    state={state}
   440→                    set={set}
   441→                    availability={provider?.availability ?? []}
   442→                    selectedService={selectedService}
   443→                  />
   444→                )}
   445→                {step === 2 && (
   446→                  <Step2Details
   447→                    state={state}
   448→                    set={set}
   449→                    provider={provider}
   450→                    selectedService={selectedService}
   451→                  />
   452→                )}
   453→                {step === 3 && (
   454→                  <Step3Payment
   455→                    state={state}
   456→                    set={set}
   457→                    selectedService={selectedService}
   458→                  />
   459→                )}
   460→              </motion.div>
   461→            </AnimatePresence>
   462→          )}
   463→        </div>
   464→      </ScrollArea>
   465→
   466→      {/* Footer nav */}
   467→      <div className="shrink-0 border-t bg-background/95 backdrop-blur px-4 sm:px-6 py-3">
   468→        <div className="flex items-center justify-between gap-3">
   469→          <Button
   470→            type="button"
   471→            variant="ghost"
   472→            onClick={onBack}
   473→            disabled={step === 1 || submitting}
   474→            className="text-muted-foreground hover:text-foreground"
   475→          >
   476→            Voltar
   477→          </Button>
   478→          {step < 3 ? (
   479→            <Button
   480→              type="button"
   481→              onClick={onNext}
   482→              className="h-11 bg-emerald-600 hover:bg-emerald-700"
   483→              disabled={
   484→                (step === 1 && !step1Valid) ||
   485→                (step === 2 && !step2Valid)
   486→              }
   487→            >
   488→              Continuar
   489→            </Button>
   490→          ) : (
   491→            <Button
   492→              type="button"
   493→              onClick={onSubmit}
   494→              disabled={submitting || !step3Valid}
   495→              className="h-11 bg-emerald-600 hover:bg-emerald-700"
   496→            >
   497→              {submitting && <Loader2 className="size-4 animate-spin" />}
   498→              <ShieldCheck className="size-4" />
   499→              Confirmar e agendar
   500→            </Button>
   501→          )}
   502→        </div>
   503→      </div>
   504→    </div>
   505→  )
   506→}

   512→function Step1Schedule({
   513→  state,
   514→  set,
   515→  availability,
   516→  selectedService,
   517→}: {
   518→  state: BookingFormState
   519→  set: <K extends keyof BookingFormState>(
   520→    key: K,
   521→    value: BookingFormState[K],
   522→  ) => void
   523→  availability: ProviderDetail["availability"]
   524→  selectedService?: ProviderService
   525→}) {
   526→  // Generate slot list for selected date
   527→  const slots = React.useMemo(() => {
   528→    if (!state.date) return [] as { label: string; value: string }[]
   529→    const dow = state.date.getDay()
   530→    const dayBlocks = (availability ?? [])
   531→      .filter((a) => a.dayOfWeek === dow)
   532→      .sort((a, b) => a.startTime.localeCompare(b.startTime))
   533→    if (dayBlocks.length === 0) return []
   534→    const out: { label: string; value: string }[] = []
   535→    for (const block of dayBlocks) {
   536→      const [sh, sm] = block.startTime.split(":").map(Number)
   537→      const [eh, em] = block.endTime.split(":").map(Number)
   538→      let cur = sh * 60 + sm
   539→      const end = eh * 60 + em
   540→      while (cur + 60 <= end) {
   541→        const h = Math.floor(cur / 60)
   542→        const m = cur % 60
   543→        const hhmm = `${String(h).padStart(2, "0")}:${String(m).padStart(2, "0")}`
   544→        out.push({ label: hhmm, value: hhmm })
   545→        cur += 60
   546→      }
   547→    }
   548→    return out
   549→  }, [state.date, availability])
   550→
   551→  const today = new Date()
   552→  today.setHours(0, 0, 0, 0)
   553→
   554→  return (
   555→    <div className="grid gap-5">
   556→      {selectedService && (
   557→        <div className="rounded-lg border bg-muted/30 p-3">
   558→          <p className="text-xs text-muted-foreground">Serviço selecionado</p>
   559→          <p className="font-medium">{selectedService.title}</p>
   560→          <p className="text-sm text-emerald-700 dark:text-emerald-400 font-medium">
   561→            {formatBRL(selectedService.basePrice)} /{" "}
   562→            {SERVICE_UNIT_LABELS[selectedService.unit] ?? "un"}
   563→          </p>
   564→        </div>
   565→      )}
   566→
   567→      <div className="grid gap-3">
   568→        <h3 className="text-sm font-semibold flex items-center gap-1.5">
   569→          <CalendarDays className="size-4 text-emerald-600" />
   570→          Escolha a data
   571→        </h3>
   572→        <div className="flex justify-center">
   573→          <Calendar
   574→            mode="single"
   575→            locale={ptBR}
   576→            selected={state.date}
   577→            onSelect={(d) => {
   578→              set("date", d)
   579→              set("time", undefined)
   580→            }}
   581→            disabled={(d) => d < today}
   582→            className="rounded-md border"
   583→          />
   584→        </div>
   585→      </div>
   586→
   587→      <div className="grid gap-3">
   588→        <h3 className="text-sm font-semibold flex items-center gap-1.5">
   589→          <Clock className="size-4 text-emerald-600" />
   590→          Horário disponível
   591→        </h3>
   592→        {!state.date ? (
   593→          <p className="text-sm text-muted-foreground py-6 text-center">
   594→            Selecione uma data para ver os horários.
   595→          </p>
   596→        ) : slots.length === 0 ? (
   597→          <div className="rounded-lg border border-dashed bg-muted/30 p-6 text-center">
   598→            <CalendarOff className="mx-auto size-8 text-muted-foreground/60" />
   599→            <p className="mt-2 text-sm font-medium">
   600→              Prestador não atende neste dia
   601→            </p>
   602→            <p className="text-xs text-muted-foreground mt-1">
   603→              {WEEKDAYS_SHORT[state.date.getDay()]} — sem expediente.
   604→            </p>
   605→          </div>
   606→        ) : (
   607→          <div className="grid grid-cols-3 gap-2 sm:grid-cols-5">
   608→            {slots.map((s) => {
   609→              const active = state.time === s.value
   610→              return (
   611→                <button
   612→                  key={s.value}
   613→                  type="button"
   614→                  onClick={() => set("time", s.value)}
   615→                  className={cn(
   616→                    "rounded-lg border p-2 text-sm text-center font-medium transition-all",
   617→                    active
   618→                      ? "border-primary bg-primary/10 text-primary shadow-sm"
   619→                      : "hover:border-primary/40 hover:bg-accent/40",
   620→                  )}
   621→                >
   622→                  {s.label}
   623→                </button>
   624→              )
   625→            })}
   626→          </div>
   627→        )}
   628→      </div>
   629→    </div>
   630→  )

   637→function Step2Details({
   638→  state,
   639→  set,
   640→  provider,
   641→  selectedService,
   642→}: {
   643→  state: BookingFormState
   644→  set: <K extends keyof BookingFormState>(

   646→    value: BookingFormState[K],
   647→  ) => void
   648→  provider?: ProviderDetail
   649→  selectedService?: ProviderService
   650→}) {

   655→          const [h, m] = state.time.split(":").map(Number)

   659→      : null

   665→        <div className="flex items-center gap-3">
   666→          <Avatar className="size-10 rounded-lg">

   669→            ) : null}
   670→            <AvatarFallback className="rounded-lg bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300">

   674→          <div className="flex-1 min-w-0">
   675→            <p className="font-medium truncate">{provider?.name}</p>
   676→            <p className="text-xs text-muted-foreground truncate">
   677→              {selectedService?.title}
   678→            </p>
   679→          </div>
   680→        </div>
   681→        <Separator className="my-3" />
   682→        <div className="grid grid-cols-2 gap-2 text-sm">
   683→          <div>
   684→            <p className="text-xs text-muted-foreground">Data</p>
   685→            <p className="font-medium">
   686→              {scheduledAt ? formatDate(scheduledAt) : "—"}
   687→            </p>
   688→          </div>
   689→          <div>
   690→            <p className="text-xs text-muted-foreground">Horário</p>
   691→            <p className="font-medium">
   692→              {scheduledAt ? formatTime(scheduledAt) : "—"}
   693→            </p>
   694→          </div>
   695→          <div>
   696→            <p className="text-xs text-muted-foreground">Valor base</p>
   697→            <p className="font-medium text-emerald-700 dark:text-emerald-400">
   698→              {selectedService
   699→                ? formatBRL(selectedService.basePrice)
   700→                : "—"}
   701→            </p>
   702→          </div>
   703→          <div>
   704→            <p className="text-xs text-muted-foreground">Unidade</p>

   730→                (selectedService.basePrice || 0) * (state.quantity || 1),

   732→            </strong>

   746→          value={state.address}
   747→          onChange={(v) => set("address", v)}
   748→        />
   749→      </div>

   751→      <Separator />
   752→
   753→      {/* Notes */}
   754→      <div className="grid gap-1.5">

   769→// Step 3 — Payment

   771→
   772→function Step3Payment({
   773→  state,
   774→  set,
   775→  selectedService,
   776→}: {
   777→  state: BookingFormState
   778→  set: <K extends keyof BookingFormState>(
   779→    key: K,
   780→    value: BookingFormState[K],
   781→  ) => void
   782→  selectedService?: ProviderService
   783→}) {
   784→  const amount =
   785→    (selectedService?.basePrice ?? 0) * (state.quantity || 1)
   786→  const fees = 0
   787→  const total = amount + fees
   788→  const [paid, setPaid] = React.useState(false)
   789→
   790→  return (
   791→    <div className="grid gap-5">
   792→      {/* Amount summary */}
   793→      <div className="rounded-lg border bg-muted/30 p-3 sm:p-4">
   794→        <div className="flex items-center justify-between text-sm">
   795→          <span className="text-muted-foreground">
   796→            {selectedService?.title} × {state.quantity}
   797→          </span>
   798→          <span className="font-medium">{formatBRL(amount)}</span>
   799→        </div>
   800→        <div className="mt-1 flex items-center justify-between text-sm">
   801→          <span className="text-muted-foreground">Taxa de serviço</span>
   802→          <span className="font-medium">{formatBRL(fees)}</span>
   803→        </div>
   804→        <Separator className="my-2" />
   805→        <div className="flex items-center justify-between">
   806→          <span className="font-medium">Total</span>
   807→          <span className="text-lg font-bold text-emerald-700 dark:text-emerald-400">
   808→            {formatBRL(total)}
   809→          </span>
   810→        </div>
   811→      </div>
   812→
   813→      {/* Payment method */}
   814→      <div className="grid gap-3">
   815→        <h3 className="text-sm font-semibold">Forma de pagamento</h3>
   816→        <RadioGroup
   817→          value={state.paymentMethod}
   818→          onValueChange={(v) => set("paymentMethod", v as "CARD" | "PIX")}
   819→          className="grid grid-cols-2 gap-2"
   820→        >
   821→          <PaymentOption
   822→            value="PIX"
   823→            title="PIX"
   824→            description="Aprovação imediata"
   825→            icon={QrCode}
   826→            selected={state.paymentMethod === "PIX"}
   827→          />
   828→          <PaymentOption
   829→            value="CARD"
   830→            title="Cartão"
   831→            description="Crédito"
   832→            icon={CreditCard}
   833→            selected={state.paymentMethod === "CARD"}
   834→          />
   835→        </RadioGroup>
   836→      </div>
   837→
   838→      {/* Payment details */}
   839→      {state.paymentMethod === "CARD" ? (
   840→        <div className="grid gap-3 rounded-xl border bg-card p-4">
   841→          <div className="flex items-center justify-between">
   842→            <p className="text-sm font-medium">Dados do cartão</p>
   843→            <Badge variant="outline" className="border-amber-500 text-amber-700">
   844→              Demonstração — não processa pagamento real
   845→            </Badge>
   846→          </div>
   847→          <div className="grid gap-1.5">
   848→            <Label htmlFor="cardName">Nome impresso</Label>
   849→            <Input
   850→              id="cardName"
   851→              placeholder="NOME COMO NO CARTÃO"
   852→              value={state.cardName}
   853→              onChange={(e) => set("cardName", e.target.value.toUpperCase())}
   854→            />
   855→          </div>
   856→          <div className="grid gap-1.5">
   857→            <Label htmlFor="cardNumber">Número</Label>
   858→            <Input
   859→              id="cardNumber"
   860→              inputMode="numeric"
   861→              placeholder="0000 0000 0000 0000"
   862→              value={state.cardNumber}
   863→              onChange={(e) => {
   864→                const digits = e.target.value.replace(/\D/g, "").slice(0, 16)
   865→                const parts = digits.match(/.{1,4}/g)
   866→                set("cardNumber", parts ? parts.join(" ") : "")
   867→              }}
   868→            />
   869→            <p className="text-xs text-muted-foreground">
   870→              Formato: 4 grupos de 4 dígitos.
   871→            </p>
   872→          </div>
   873→          <div className="grid grid-cols-2 gap-3">
   874→            <div className="grid gap-1.5">
   875→              <Label htmlFor="cardExpiry">Validade</Label>
   876→              <Input
   877→                id="cardExpiry"
   878→                inputMode="numeric"
   879→                placeholder="MM/AA"
   880→                maxLength={5}
   881→                value={state.cardExpiry}
   882→                onChange={(e) => {
   883→                  let v = e.target.value.replace(/\D/g, "").slice(0, 4)
   884→                  if (v.length >= 3) v = `${v.slice(0, 2)}/${v.slice(2)}`
   885→                  set("cardExpiry", v)
   886→                }}
   887→              />
   888→            </div>
   889→            <div className="grid gap-1.5">
   890→              <Label htmlFor="cardCvv">CVV</Label>
   891→              <Input
   892→                id="cardCvv"
   893→                inputMode="numeric"
   894→                placeholder="123"
   895→                maxLength={4}
   896→                value={state.cardCvv}
   897→                onChange={(e) =>
   898→                  set(
   899→                    "cardCvv",
   900→                    e.target.value.replace(/\D/g, "").slice(0, 4),
   901→                  )
   902→                }
   903→              />
   904→            </div>
   905→          </div>
   906→          <p className="text-xs text-muted-foreground flex items-start gap-1.5">
   907→            <ShieldCheck className="size-3.5 mt-0.5 text-emerald-600" />
   908→            Ambiente de demonstração — não use dados reais. Nenhum pagamento
   909→            será efetivado.
   910→          </p>
   911→        </div>
   912→      ) : (
   913→        <div className="grid gap-3 rounded-xl border bg-card p-4">
   914→          <p className="text-sm font-medium">Pague com PIX</p>
   915→          <div className="flex flex-col items-center gap-3 py-3">
   916→            <div className="rounded-lg bg-slate-100 p-8 text-center dark:bg-slate-800/50">
   917→              <QrCode className="size-24 text-slate-500 dark:text-slate-300" />
   918→            </div>
   919→            <p className="text-xs text-muted-foreground text-center max-w-xs">
   920→              Escaneie o QR code ou copie a chave abaixo para pagar.
   921→            </p>
   922→            <Button
   923→              type="button"
   924→              variant="outline"
   925→              size="sm"
   926→              onClick={() => {
   927→                navigator.clipboard
   928→                  .writeText("severinno@exemplo.com")
   929→                  .catch(() => {})
   930→                toast.success("Chave PIX copiada!")
   931→              }}
   932→            >
   933→              Copiar chave PIX
   934→            </Button>
   935→          </div>
   936→          <Button
   937→            type="button"
   938→            variant="outline"
   939→            className="border-emerald-500 text-emerald-700 hover:bg-emerald-50 dark:text-emerald-400 dark:hover:bg-emerald-950/40"
   940→            onClick={() => {
   941→              setPaid(true)
   942→              toast.success("Pagamento confirmado. Conclua o agendamento.")
   943→            }}
   944→            disabled={paid}
   945→          >
   946→            {paid ? (
   947→              <>
   948→                <Check className="size-4" /> Pagamento confirmado
   949→              </>
   950→            ) : (
   951→              "Já paguei"
   952→            )}
   953→          </Button>
   954→        </div>
   955→      )}
   956→    </div>
   957→  )
   958→}
