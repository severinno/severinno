   131→// ---------------------------------------------------------------------------

   134→
   135→function cardNameValid(v: string) { return v.trim().length >= 3 }
   136→function cardNumberValid(v: string) { return v.replace(/\s/g, "").length >= 13 }
   137→function cardExpiryValid(v: string) { return /^\d{2}\/\d{2}$/.test(v) }
   138→function cardCvvValid(v: string) { return /^\d{3,4}$/.test(v) }

   144→export function BookingModal() {
   145→  const open = useUIStore((s) => s.bookingModal.open)
   146→  const providerIdPreset = useUIStore((s) => s.bookingModal.providerId)
   147→  const serviceIdPreset = useUIStore((s) => s.bookingModal.serviceId)
   148→  const close = useUIStore((s) => s.closeBooking)
   149→  const openAuth = useUIStore((s) => s.openAuth)
   150→  const isMobile = useIsMobile()
   151→  const navigate = useViewStore((s) => s.navigate)
   152→  const user = useAuthStore((s) => s.user)
   153→
   154→  const [step, setStep] = React.useState<Step>(1)
   155→  const [state, setState] = React.useState<BookingFormState>(initialState())
   156→  const [submitting, setSubmitting] = React.useState(false)
   157→  const [touched, setTouched] = React.useState<Set<string>>(new Set())
   158→
   159→  // Reset when modal opens
   160→  React.useEffect(() => {
   161→    if (open) {
   162→      setStep(1)
   163→      setState(initialState())
   164→      setTouched(new Set())
   165→    }
   166→  }, [open])
   167→
   168→  // Fetch provider + services
   169→  const providerQuery = useQuery({
   170→    queryKey: ["provider", providerIdPreset],
   171→    queryFn: () =>
   172→      apiGet<ProviderDetail>(`/api/providers/${providerIdPreset}`),
   173→    enabled: open && !!providerIdPreset,
   174→    staleTime: 60 * 1000,
   175→  })
   176→
   177→  const servicesQuery = useQuery({
   178→    queryKey: ["services-by-provider", providerIdPreset],
   179→    queryFn: () =>
   180→      apiGet<ProviderService[]>("/api/services", { providerId: providerIdPreset }),
   181→    enabled: open && !!providerIdPreset,
   182→    staleTime: 60 * 1000,
   183→  })
   184→
   185→  const provider = providerQuery.data
   186→  const services = servicesQuery.data ?? []
   187→  const selectedService =
   188→    services.find((s) => s.id === serviceIdPreset) ?? services[0]
   189→
   190→  React.useEffect(() => {
   191→    if (open && selectedService) {
   192→      setState((s) => ({ ...s, quantity: 1 }))
   193→    }
   194→  }, [open, selectedService])
   195→
   196→  const set = <K extends keyof BookingFormState>(
   197→    key: K,
   198→    value: BookingFormState[K],
   199→  ) => setState((s) => ({ ...s, [key]: value }))
   200→
   201→  const markTouched = (field: string) =>
   202→    setTouched((prev) => new Set(prev).add(field))
   203→
   204→  // ── Step validation ──
   205→  const step1Valid = !!state.date && !!state.time
   206→  const step2Valid =
   207→    !!state.address.cep &&
   208→    state.address.cep.replace(/\D/g, "").length === 8 &&
   209→    !!state.address.street &&
   210→    !!state.address.number &&
   211→    !!state.address.city &&
   212→    !!state.address.state
   213→  const step3Valid =
   214→    state.paymentMethod === "PIX" ||
   215→    (state.paymentMethod === "CARD" &&
   216→      cardNameValid(state.cardName) &&
   217→      cardNumberValid(state.cardNumber) &&
   218→      cardExpiryValid(state.cardExpiry) &&
   219→      cardCvvValid(state.cardCvv))
   220→  // Step 4 is always valid (it's a review)
   221→
   222→  const validSteps: Record<number, boolean> = {
   223→    1: step1Valid,
   224→    2: step2Valid,
   225→    3: step3Valid,
   226→    4: true,
   227→  }
   228→
   229→  // ── Navigation ──
   230→  const handleStepClick = (target: Step) => {
   231→    if (target < step) {
   232→      setStep(target)
   233→      return
   234→    }
   235→    for (let i = 1; i < target; i++) {
   236→      if (!validSteps[i as Step]) {
   237→        toast.error("Complete os passos anteriores primeiro.")
   238→        return
   239→      }
   240→    }
   241→    setStep(target)
   242→  }
   243→
   244→  const handleNext = () => {
   245→    if (step === 1 && !step1Valid) {
   246→      toast.error("Selecione data e horário para continuar.")
   247→      return
   248→    }
   249→    if (step === 2 && !step2Valid) {
   250→      toast.error("Preencha o endereço completo.")
   251→      return
   252→    }
   253→    if (step === 3 && !step3Valid) {
   254→      toast.error("Verifique os dados de pagamento.")
   255→      return
   256→    }
   257→    setStep((s) => Math.min(4, s + 1) as Step)
   258→  }
   259→
   260→  const handleBack = () => setStep((s) => Math.max(1, s - 1) as Step)
   261→
   262→  const handleSubmit = async () => {
   263→    if (!user) {
   264→      toast.info("Faça cadastro gratuito para agendar serviços.")
   265→      openAuth("register", "CLIENT")
   266→      return
   267→    }
   268→    if (!provider || !selectedService || !state.date || !state.time) {
   269→      toast.error("Dados incompletos. Revise o agendamento.")
   270→      return
   271→    }
   272→
   273→    const [h, m] = state.time.split(":").map(Number)
   274→    const scheduledAt = new Date(state.date)
   275→    scheduledAt.setHours(h ?? 0, m ?? 0, 0, 0)
   276→    const amount = (selectedService.basePrice || 0) * (state.quantity || 1)
   277→
   278→    setSubmitting(true)
   279→    try {
   280→      await apiPost("/api/bookings", {
   281→        providerId: provider.id,
   282→        serviceId: selectedService.id,
   283→        scheduledAt: scheduledAt.toISOString(),
   284→        address: [
   285→          state.address.street,
   286→          state.address.number,
   287→          state.address.complement,
   288→          state.address.district,
   289→        ]
   290→          .filter(Boolean)
   291→          .join(", "),
   292→        cep: state.address.cep,
   293→        lat: state.address.lat,
   294→        lng: state.address.lng,
   295→        amount,
   296→        paymentMethod: state.paymentMethod,
   297→        notes: state.notes,
   298→        quantity: state.quantity,
   299→        unit: selectedService.unit,
   300→      })
   301→      toast.success("Agendamento confirmado! Acompanhe em seus agendamentos.")
   302→      close()
   303→      navigate("client.bookings")
   304→    } catch (e) {
   305→      const msg =
   306→        (e as { message?: string })?.message ??
   307→        "Não foi possível concluir o agendamento. Tente novamente."
   308→      toast.error(msg)
   309→    } finally {
   310→      setSubmitting(false)
   311→    }
   312→  }
   313→
   314→  // ── Step content ──
   315→  const stepContent = (() => {
   316→    switch (step) {
   317→      case 1:
   318→        return (
   319→          <Step1Schedule
   320→            state={state}
   321→            set={set}
   322→            availability={provider?.availability ?? []}
   323→            selectedService={selectedService}
   324→            provider={provider}
   325→            loading={providerQuery.isLoading || servicesQuery.isLoading}
   326→          />
   327→        )
   328→      case 2:
   329→        return (
   330→          <Step2Details
   331→            state={state}
   332→            set={set}
   333→            provider={provider}
   334→            selectedService={selectedService}
   335→          />
   336→        )
   337→      case 3:
   338→        return (
   339→          <Step3Payment
   340→            state={state}
   341→            set={set}
   342→            markTouched={markTouched}
   343→            touched={touched}
   344→            selectedService={selectedService}
   345→          />
   346→        )
   347→      case 4:
   348→        return (
   349→          <Step4Confirmation
   350→            state={state}
   351→            provider={provider}
   352→            selectedService={selectedService}
   353→            goToStep={setStep}
   354→          />
   355→        )
   356→      default:
   357→        return null
   358→    }
   359→  })()
   360→
   361→  const wizardBody = (
   362→    <StepWizard
   363→      steps={STEPS}
   364→      currentStep={step}
   365→      validSteps={validSteps}
   366→      onStepClick={handleStepClick}
   367→      onBack={handleBack}
   368→      onNext={handleNext}
   369→      onSubmit={handleSubmit}
   370→      submitting={submitting}
   371→      currentStepValid={validSteps[step] ?? true}
   372→      submitLabel={
   373→        <>
   374→          <Send className="size-3.5" />
   375→          Confirmar agendamento
   376→        </>
   377→      }
   378→    >
   379→      {stepContent}
   380→    </StepWizard>
   381→  )
   382→
   383→  if (isMobile) {
   384→    return (
   385→      <Sheet open={open} onOpenChange={(o) => !o && close()}>
   386→        <SheetContent
   387→          side="bottom"
   388→          className="h-[100dvh] max-h-[100dvh] w-full p-0 sm:max-w-full gap-0 flex flex-col"
   389→          onInteractOutside={(e) => e.preventDefault()}
   390→        >
   391→          <SheetHeader className="px-4 pt-4 pb-2 shrink-0">
   392→            <SheetTitle className="flex items-center gap-2 text-base">
   393→              <CalendarDays className="size-4 text-emerald-600" />
   394→              Agendar serviço
   395→            </SheetTitle>
   396→            <SheetDescription className="text-xs">
   397→              Escolha data, detalhes e pagamento
   398→            </SheetDescription>
   399→          </SheetHeader>
   400→          <div className="flex-1 overflow-hidden">{wizardBody}</div>
   401→        </SheetContent>
   402→      </Sheet>
   403→    )
   404→  }
   405→
   406→  return (
   407→    <Dialog open={open} onOpenChange={(o) => !o && close()}>
   408→      <DialogContent className="sm:max-w-lg p-0 gap-0 overflow-hidden" onInteractOutside={(e) => e.preventDefault()}>
   409→        {/* Nielsen #3: User control — prevent accidental close during wizard */}
   410→        <DialogHeader className="px-5 pt-5 pb-2 shrink-0 border-b">
   411→          <DialogTitle className="flex items-center gap-2 text-base">
   412→            <CalendarDays className="size-4 text-emerald-600" />
   413→            Agendar serviço
   414→          </DialogTitle>
   415→          <DialogDescription className="text-xs">
   416→            Escolha data, detalhes e pagamento
   417→          </DialogDescription>
   418→        </DialogHeader>
   419→        {wizardBody}
   420→      </DialogContent>
   421→    </Dialog>
   422→  )
   423→}

   425→// ---------------------------------------------------------------------------
   426→// Step 1 — Agenda (Date & Time)
   427→// Nielsen #6: Recognition over recall — visible calendar, time slot buttons

   429→
   430→function Step1Schedule({
   431→  state,
   432→  set,
   433→  availability,
   434→  selectedService,
   435→  provider,
   436→  loading,
   437→}: {
   438→  state: BookingFormState
   439→  set: <K extends keyof BookingFormState>(
   440→    key: K,
   441→    value: BookingFormState[K],
   442→  ) => void
   443→  availability: ProviderDetail["availability"]
   444→  selectedService?: ProviderService
   445→  provider?: ProviderDetail
   446→  loading: boolean
   447→}) {
   448→  // Generate slot list for selected date
   449→  const slots = React.useMemo(() => {
   450→    if (!state.date) return [] as { label: string; value: string }[]
   451→    const dow = state.date.getDay()
   452→    const dayBlocks = (availability ?? [])
   453→      .filter((a) => a.dayOfWeek === dow)
   454→      .sort((a, b) => a.startTime.localeCompare(b.startTime))
   455→    if (dayBlocks.length === 0) return []
   456→    const out: { label: string; value: string }[] = []
   457→    for (const block of dayBlocks) {
   458→      const [sh, sm] = block.startTime.split(":").map(Number)
   459→      const [eh, em] = block.endTime.split(":").map(Number)
   460→      let cur = sh * 60 + sm
   461→      const end = eh * 60 + em
   462→      while (cur + 60 <= end) {
   463→        const h = Math.floor(cur / 60)
   464→        const m = cur % 60
   465→        const hhmm = `${String(h).padStart(2, "0")}:${String(m).padStart(2, "0")}`
   466→        out.push({ label: formatHHmm(hhmm), value: hhmm })
   467→        cur += 60
   468→      }
   469→    }
   470→    return out
   471→  }, [state.date, availability])
   472→
   473→  const today = new Date()
   474→  today.setHours(0, 0, 0, 0)
   475→
   476→  if (loading) {
   477→    return (
   478→      <div className="flex items-center justify-center py-10">
   479→        <Loader2 className="size-5 animate-spin text-emerald-600" />
   480→      </div>
   481→    )
   482→  }
   483→
   484→  return (
   485→    <div className="grid gap-4">
   486→      <StepHeader
   487→        icon={CalendarDays}
   488→        title="Escolha a data e horário"
   489→        description="Selecione o melhor dia e horário para o serviço."
   490→      />
   491→
   492→      {/* Service + provider info banner — Nielsen #1: visibility */}
   493→      {selectedService && (
   494→        <InfoCard variant="emerald">
   495→          <div className="flex items-center gap-3">
   496→            <Avatar className="size-9 rounded-md">
   497→              {provider?.avatarUrl ? (
   498→                <AvatarImage src={provider.avatarUrl} alt={provider.name} />
   499→              ) : null}
   500→              <AvatarFallback className="rounded-md bg-emerald-100 text-emerald-700 text-xs dark:bg-emerald-950 dark:text-emerald-300">
   501→                {provider?.name?.[0]?.toUpperCase() ?? "?"}
   502→              </AvatarFallback>
   503→            </Avatar>
   504→            <div className="flex-1 min-w-0">
   505→              <p className="text-sm font-medium truncate">{selectedService.title}</p>
   506→              <p className="text-xs text-emerald-700 dark:text-emerald-400 font-medium">
   507→                {formatBRL(selectedService.basePrice)} /{" "}
   508→                {SERVICE_UNIT_LABELS[selectedService.unit] ?? "un"}
   509→              </p>
   510→            </div>
   511→            {state.date && state.time && (
   512→              <CheckCircle2 className="size-4 text-emerald-600 shrink-0" />
   513→            )}
   514→          </div>
   515→        </InfoCard>
   516→      )}
   517→
   518→      {/* Calendar */}
   519→      <div>
   520→        <p className="text-xs font-medium text-muted-foreground mb-1.5 flex items-center gap-1">
   521→          <CalendarDays className="size-3.5 text-emerald-600" />
   522→          Data
   523→        </p>
   524→        <div className="flex justify-center">
   525→          <Calendar
   526→            mode="single"
   527→            locale={ptBR}
   528→            selected={state.date}
   529→            onSelect={(d) => {
   530→              set("date", d)
   531→              set("time", undefined)
   532→            }}
   533→            disabled={(d) => d < today}
   534→            className="rounded-md border"
   535→          />
   536→        </div>
   537→      </div>
   538→
   539→      {/* Time slots — Nielsen #6: recognition over recall */}
   540→      <div>
   541→        <p className="text-xs font-medium text-muted-foreground mb-1.5 flex items-center gap-1">
   542→          <Clock className="size-3.5 text-emerald-600" />
   543→          Horários disponíveis
   544→        </p>
   545→        {!state.date ? (
   546→          <p className="text-xs text-muted-foreground py-4 text-center">
   547→            Selecione uma data para ver os horários
   548→          </p>
   549→        ) : slots.length === 0 ? (
   550→          <div className="rounded-lg border border-dashed bg-muted/30 p-4 text-center">
   551→            <CalendarOff className="mx-auto size-6 text-muted-foreground/50" />
   552→            <p className="mt-1.5 text-xs font-medium">
   553→              Sem horários neste dia
   554→            </p>
   555→            <p className="text-[11px] text-muted-foreground">
   556→              {WEEKDAYS_SHORT[state.date.getDay()]} — fora do expediente
   557→            </p>
   558→          </div>
   559→        ) : (
   560→          <div className="grid grid-cols-4 gap-1.5 sm:grid-cols-5">
   561→            {slots.map((s) => {
   562→              const active = state.time === s.value
   563→              return (
   564→                <button
   565→                  key={s.value}
   566→                  type="button"
   567→                  onClick={() => set("time", s.value)}
   568→                  className={cn(
   569→                    "rounded-md border px-1 py-1.5 text-xs text-center font-medium transition-all",
   570→                    active
   571→                      ? "border-emerald-600 bg-emerald-600 text-white shadow-sm"
   572→                      : "hover:border-emerald-400 hover:bg-emerald-50 dark:hover:bg-emerald-950/30",
   573→                  )}
   574→                >
   575→                  {s.label}
   576→                </button>
   577→              )
   578→            })}
   579→          </div>
   580→        )}
   581→      </div>
   582→    </div>
   583→  )
   584→}

   586→// ---------------------------------------------------------------------------
   587→// Step 2 — Detalhes (Address + Quantity + Notes)
   588→// Nielsen #5: Error prevention — CEP auto-fill, GPS
   589→// Nielsen #8: Minimalist — clean form layout
   590→// ---------------------------------------------------------------------------
   591→
   592→function Step2Details({
   593→  state,
   594→  set,
   595→  provider,
   596→  selectedService,
   597→}: {
   598→  state: BookingFormState
   599→  set: <K extends keyof BookingFormState>(
   600→    key: K,
   601→    value: BookingFormState[K],
   602→  ) => void
   603→  provider?: ProviderDetail
   604→  selectedService?: ProviderService
   605→}) {
   606→  const scheduledAt =
   607→    state.date && state.time
   608→      ? (() => {
   609→          const d = new Date(state.date)
   610→          const [h, m] = state.time.split(":").map(Number)
   611→          d.setHours(h ?? 0, m ?? 0, 0, 0)
   612→          return d
   613→        })()
   614→      : null
   615→
   616→  const estimatedTotal = (selectedService?.basePrice ?? 0) * (state.quantity || 1)
   617→
   618→  return (
   619→    <div className="grid gap-4">
   620→      <StepHeader
   621→        icon={MapPin}
   622→        title="Detalhes do agendamento"
   623→        description="Informe a quantidade, endereço e observações."
   624→      />
