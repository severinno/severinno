    80→    80→export function ProviderProfileModal() {
    81→    81→  const open = useUIStore((s) => s.providerModal.open)
    82→    82→  const providerId = useUIStore((s) => s.providerModal.providerId)
    83→    83→  const close = useUIStore((s) => s.closeProvider)
    84→    84→  const openQuote = useUIStore((s) => s.openQuote)
    85→    85→  const openBooking = useUIStore((s) => s.openBooking)
    86→    86→  const addRecentlyViewed = useRecentlyViewedStore((s) => s.addView)
    87→    87→  const isMobile = useIsMobile()
    88→    88→
    89→    89→  const query = useQuery({
    90→    90→    queryKey: ["provider", providerId],
    91→    91→    queryFn: () => apiGet<ProviderDetail>(`/api/providers/${providerId}`),
    92→    92→    enabled: open && !!providerId,
    93→    93→    staleTime: 60 * 1000,
    94→    94→  })
    95→    95→
    96→    96→  const provider = query.data
    97→    97→  const [favorited, setFavorited] = React.useState(false)
    98→    98→
    99→    99→  // Reset favorite state when provider changes.
   100→   100→  React.useEffect(() => {
   101→   101→    setFavorited(false)
   102→   102→  }, [providerId])
   103→   103→
   104→   104→  // Track recently viewed (Nielsen H6 — recognition over recall).
   105→   105→  // Only track once per provider when the modal opens with loaded data.
   106→   106→  React.useEffect(() => {
   107→   107→    if (open && provider) {
   108→   108→      addRecentlyViewed(provider)
   109→   109→    }
   110→   110→  }, [open, provider, addRecentlyViewed])
   111→   111→
   112→   112→  const handleFavorite = async () => {
   113→   113→    setFavorited((v) => !v)
   114→   114→    toast.success(
   115→   115→      favorited ? "Removido dos favoritos." : "Adicionado aos favoritos!",
   116→   116→    )
   117→   117→  }
   118→   118→
   119→   119→  const handleShare = async () => {
   120→   120→    if (!provider) return
   121→   121→    const url = `${window.location.origin}/?provider=${provider.id}`
   122→   122→    try {
   123→   123→      if (navigator.share) {
   124→   124→        await navigator.share({
   125→   125→          title: provider.name,
   126→   126→          text: `Conheça ${provider.name} no Severinno`,
   127→   127→          url,
   128→   128→        })
   129→   129→      } else {
   130→   130→        await navigator.clipboard.writeText(url)
   131→   131→        toast.success("Link copiado para a área de transferência.")
   132→   132→      }
   133→   133→    } catch {
   134→   134→      // user dismissed share dialog — silent
   135→   135→    }
   136→   136→  }
   137→   137→
   138→   138→  const startQuote = (serviceId?: string) => {
   139→   139→    if (!provider) return
   140→   140→    close()
   141→   141→    setTimeout(() => openQuote({ providerId: provider.id, serviceId }), 200)
   142→   142→  }
   143→   143→  const startBooking = (serviceId?: string) => {
   144→   144→    if (!provider) return
   145→   145→    close()
   146→   146→    setTimeout(() => openBooking({ providerId: provider.id, serviceId }), 200)
   147→   147→  }
   148→   148→
   149→   149→  const content = (
   150→   150→    <ProfileBody
   151→   151→      provider={provider}
   152→   152→      loading={query.isLoading}
   153→   153→      favorited={favorited}
   154→   154→      onFavorite={handleFavorite}
   155→   155→      onShare={handleShare}
   156→   156→      onClose={close}
   157→   157→      onQuote={startQuote}
   158→   158→      onBooking={startBooking}
   159→   159→    />
   160→   160→  )
   161→   161→
   162→   162→  if (isMobile) {
   163→   163→    return (
   164→   164→      <Sheet open={open} onOpenChange={(o) => !o && close()}>
   165→   165→        <SheetContent
   166→   166→          side="bottom"
   167→   167→          className="h-[100dvh] max-h-[100dvh] w-full p-0 sm:max-w-full gap-0 flex flex-col [&_[data-slot=sheet-close]]:hidden"
   168→   168→        >
   169→   169→          <SheetTitle className="sr-only">
   170→   170→            {provider?.name ?? "Perfil do prestador"}
   171→   171→          </SheetTitle>
   172→   172→          <SheetDescription className="sr-only">
   173→   173→            Detalhes do prestador de serviços.
   174→   174→          </SheetDescription>
   175→   175→          <div className="flex-1 overflow-hidden">{content}</div>
   176→   176→        </SheetContent>
   177→   177→      </Sheet>
   178→   178→    )
   179→   179→  }
   180→   180→
   181→   181→  return (
   182→   182→    <Dialog open={open} onOpenChange={(o) => !o && close()}>
   183→   183→      <DialogContent
   184→   184→        className="sm:max-w-3xl p-0 gap-0 overflow-hidden"
   185→   185→        showCloseButton={false}
   186→   186→      >
   187→   187→        <DialogTitle className="sr-only">
   188→   188→          {provider?.name ?? "Perfil do prestador"}
   189→   189→        </DialogTitle>
   190→   190→        <DialogDescription className="sr-only">
   191→   191→          Detalhes do prestador de serviços.
   192→   192→        </DialogDescription>
   193→   193→        {content}
   194→   194→      </DialogContent>
   195→   195→    </Dialog>
   196→   196→  )
   197→   197→}

   435→   435→function ServicesTab({
   436→   436→  services,
   437→   437→  loading,
   438→   438→  onQuote,
   439→   439→  onBooking,
   440→   440→}: {
   441→   441→  services: ProviderService[]
   442→   442→  loading: boolean
   443→   443→  onQuote: (serviceId?: string) => void
   444→   444→  onBooking: (serviceId?: string) => void
   445→   445→}) {
   446→   446→  if (loading) {
   447→   447→    return (
   448→   448→      <div className="space-y-3">
   449→   449→        {Array.from({ length: 3 }).map((_, i) => (
   450→   450→          <Skeleton key={i} className="h-20 w-full rounded-lg" />
   451→   451→        ))}
   452→   452→      </div>
   453→   453→    )
   454→   454→  }
   455→   455→
   456→   456→  if (services.length === 0) {
   457→   457→    return (
   458→   458→      <EmptyState
   459→   459→        icon={Wrench}
   460→   460→        title="Nenhum serviço cadastrado"
   461→   461→        description="Este prestador ainda não publicou serviços."
   462→   462→      />
   463→   463→    )
   464→   464→  }
   465→   465→
   466→   466→  // Group by category name
   467→   467→  const groups = new Map<string, ProviderService[]>()
   468→   468→  for (const s of services) {
   469→   469→    const key = s.category?.name ?? "Outros"
   470→   470→    const arr = groups.get(key) ?? []
   471→   471→    arr.push(s)
   472→   472→    groups.set(key, arr)
   473→   473→  }
   474→   474→  const groupEntries = Array.from(groups.entries())
   475→   475→
   476→   476→  return (
   477→   477→    <Accordion
   478→   478→      type="multiple"
   479→   479→      defaultValue={groupEntries.map(([k]) => k)}
   480→   480→      className="w-full"
   481→   481→    >
   482→   482→      {groupEntries.map(([categoryName, items]) => (
   483→   483→        <AccordionItem key={categoryName} value={categoryName}>
   484→   484→          <AccordionTrigger className="hover:no-underline">
   485→   485→            <div className="flex w-full items-center justify-between pr-2">
   486→   486→              <span className="font-medium">{categoryName}</span>
   487→   487→              <Badge variant="secondary" className="mr-1">
   488→   488→                {items.length}
   489→   489→              </Badge>
   490→   490→            </div>
   491→   491→          </AccordionTrigger>
   492→   492→          <AccordionContent>
   493→   493→            <div className="grid gap-3">
   494→   494→              {items.map((s) => (
