   570→function Step1Schedule({
   571→  state,
   572→  set,
   573→  availability,
   574→  selectedService,
   575→}: {
   576→  state: BookingFormState
   577→  set: <K extends keyof BookingFormState>(
   578→    key: K,
   579→    value: BookingFormState[K],
   580→  ) => void
   581→  availability: ProviderDetail["availability"]
   582→  selectedService?: ProviderService
   583→}) {
   584→  // Generate slot list for selected date
   585→  const slots = React.useMemo(() => {
   586→    if (!state.date) return [] as { label: string; value: string }[]
   587→    const dow = state.date.getDay()
   588→    const dayBlocks = (availability ?? [])
   589→      .filter((a) => a.dayOfWeek === dow)
   590→      .sort((a, b) => a.startTime.localeCompare(b.startTime))
   591→    if (dayBlocks.length === 0) return []
   592→    const out: { label: string; value: string }[] = []
   593→    for (const block of dayBlocks) {
   594→      const [sh, sm] = block.startTime.split(":").map(Number)
   595→      const [eh, em] = block.endTime.split(":").map(Number)
   596→      let cur = sh * 60 + sm
   597→      const end = eh * 60 + em
   598→      while (cur + 60 <= end) {
   599→        const h = Math.floor(cur / 60)
   600→        const m = cur % 60
   601→        const hhmm = `${String(h).padStart(2, "0")}:${String(m).padStart(2, "0")}`
   602→        out.push({ label: formatHHmm(hhmm), value: hhmm })
   603→        cur += 60
   604→      }
   605→    }
   606→    return out
   607→  }, [state.date, availability])
   608→
   609→  const today = new Date()
   610→  today.setHours(0, 0, 0, 0)
   611→
   612→  return (
   613→    <div className="grid gap-3">
   614→      {/* Service info banner */}
   615→      {selectedService && (
   616→        <div className="flex items-center gap-2.5 rounded-lg border bg-emerald-50/60 dark:bg-emerald-950/30 px-3 py-2">
   617→          <div className="flex-1 min-w-0">
   618→            <p className="text-sm font-medium truncate">{selectedService.title}</p>
   619→            <p className="text-xs text-emerald-700 dark:text-emerald-400 font-medium">
   620→              {formatBRL(selectedService.basePrice)} /{" "}
   621→              {SERVICE_UNIT_LABELS[selectedService.unit] ?? "un"}
   622→            </p>
   623→          </div>
   624→          {state.date && state.time && (
   625→            <CheckCircle2 className="size-4 text-emerald-600 shrink-0" />
   626→          )}
   627→        </div>
   628→      )}
   629→
   630→      {/* Calendar */}
   631→      <div>
   632→        <p className="text-xs font-medium text-muted-foreground mb-1.5 flex items-center gap-1">
   633→          <CalendarDays className="size-3.5 text-emerald-600" />
   634→          Escolha a data
   635→        </p>
   636→        <div className="flex justify-center">
   637→          <Calendar
   638→            mode="single"
   639→            locale={ptBR}
   640→            selected={state.date}
   641→            onSelect={(d) => {
   642→              set("date", d)
   643→              set("time", undefined)
   644→            }}
   645→            disabled={(d) => d < today}
   646→            className="rounded-md border"
   647→          />
   648→        </div>
   649→      </div>
   650→
   651→      {/* Time slots */}
   652→      <div>
   653→        <p className="text-xs font-medium text-muted-foreground mb-1.5 flex items-center gap-1">
   654→          <Clock className="size-3.5 text-emerald-600" />
   655→          Horários disponíveis
   656→        </p>
   657→        {!state.date ? (
   658→          <p className="text-xs text-muted-foreground py-4 text-center">
   659→            Selecione uma data para ver os horários
   660→          </p>
   661→        ) : slots.length === 0 ? (
   662→          <div className="rounded-lg border border-dashed bg-muted/30 p-4 text-center">
   663→            <CalendarOff className="mx-auto size-6 text-muted-foreground/50" />
   664→            <p className="mt-1.5 text-xs font-medium">
   665→              Sem horários neste dia
   666→            </p>
   667→            <p className="text-[11px] text-muted-foreground">
   668→              {WEEKDAYS_SHORT[state.date.getDay()]} — fora do expediente
   669→            </p>
   670→          </div>
   671→        ) : (
   672→          <div className="grid grid-cols-4 gap-1.5 sm:grid-cols-5">
   673→            {slots.map((s) => {
   674→              const active = state.time === s.value
   675→              return (
   676→                <button
   677→                  key={s.value}
   678→                  type="button"
   679→                  onClick={() => set("time", s.value)}
   680→                  className={cn(
   681→                    "rounded-md border px-1 py-1.5 text-xs text-center font-medium transition-all",
   682→                    active
   683→                      ? "border-emerald-600 bg-emerald-600 text-white shadow-sm"
   684→                      : "hover:border-emerald-400 hover:bg-emerald-50 dark:hover:bg-emerald-950/30",
   685→                  )}
   686→                >
   687→                  {s.label}
   688→                </button>
   689→              )
   690→            })}
   691→          </div>
   692→        )}
   693→      </div>
   694→    </div>
   695→  )
   696→}
