   113→type QuoteItemForm = {
   114→  providerId: string
   115→  serviceId: string
   116→  description: string
   117→  quantity: number
   118→  unit: ServiceUnit
   119→  photos: string[]
   120→}
   121→
   122→const quoteFormSchema = z.object({
   123→  items: z
   124→    .array(
   125→      z.object({
   126→        providerId: z.string().min(1, "Selecione um prestador"),
   127→        serviceId: z.string().min(1, "Selecione um serviço"),
   128→        description: z
   129→          .string()
   130→          .min(10, "Descreva com ao menos 10 caracteres")
   131→          .max(400),
   132→        quantity: z.coerce.number().min(0.01, "Quantidade inválida"),
   133→        unit: z.enum([
   134→          "UNIDADE",
   135→          "METRO_LINEAR",
   136→          "METRO_QUADRADO",
   137→          "METRO_CUBICO",
   138→        ]),
   139→        photos: z.array(z.string()).max(4).default([]),
   140→      }),
   141→    )
   142→    .min(1, "Adicione ao menos um item"),
   143→  address: z.object({
   144→    cep: z.string().min(8, "CEP inválido"),
   145→    street: z.string().min(3, "Informe a rua"),
   146→    number: z.string().min(1, "Informe o número"),
   147→    complement: z.string().default(""),
   148→    district: z.string().default(""),
   149→    city: z.string().min(2, "Informe a cidade"),
   150→    state: z.string().min(2, "UF"),
   151→    lat: z.number().nullable().optional(),
   152→    lng: z.number().nullable().optional(),
   153→  }),
   154→})
   155→
   156→type QuoteFormValues = z.infer<typeof quoteFormSchema>
   157→

   159→  providerId?: string,

   161→  unit: ServiceUnit = "UNIDADE",

   164→  serviceId: serviceId ?? "",
   165→  description: "",
   166→  quantity: 1,
   167→  unit,
   168→  photos: [],
   169→})
   170→
   171→const emptyAddress: AddressFormValue = {
   172→  cep: "",
   173→  street: "",
   174→  number: "",
   175→  complement: "",
   176→  district: "",
   177→  city: "",
   178→  state: "",
   179→  lat: null,
   180→  lng: null,

   184→// Main modal

   186→
   187→export function QuoteModal() {
   188→  const open = useUIStore((s) => s.quoteModal.open)
   189→  const providerIdPreset = useUIStore((s) => s.quoteModal.providerId)
   190→  const serviceIdPreset = useUIStore((s) => s.quoteModal.serviceId)
   191→  const close = useUIStore((s) => s.closeQuote)
   192→  const openAuth = useUIStore((s) => s.openAuth)
   193→  const isMobile = useIsMobile()
   194→  const navigate = useViewStore((s) => s.navigate)
   195→  const user = useAuthStore((s) => s.user)
   196→
   197→  const form = useForm<QuoteFormValues>({
   198→    resolver: zodResolver(quoteFormSchema) as unknown as Resolver<QuoteFormValues>,
   199→    defaultValues: {
   200→      items: [emptyItem(providerIdPreset, serviceIdPreset)],
   201→      address: emptyAddress,
   202→    },
   203→    mode: "onTouched",
   204→  })
   205→
   206→  const [step, setStep] = React.useState<Step>(1)
   207→  const [submitting, setSubmitting] = React.useState(false)
   208→
   209→  // Reset when modal opens
   210→  React.useEffect(() => {
   211→    if (open) {
   212→      form.reset({
   213→        items: [emptyItem(providerIdPreset, serviceIdPreset)],
   214→        address: emptyAddress,
   215→      })
   216→      setStep(1)
   217→    }
   218→  }, [open, providerIdPreset, serviceIdPreset, form])
   219→
   220→  // ── Step validation map ──
   221→  const items = form.watch("items")
   222→  const address = form.watch("address")
   223→  const errors = form.formState.errors
   224→  const item0 = items[0]
   225→  const item0Errors = errors.items?.[0]
   226→
   227→  const step1Valid = !!item0?.providerId
   228→  const step2Valid = !!item0?.serviceId
   229→  const step3Valid =
   230→    !!item0?.description && item0.description.length >= 10 && item0?.quantity > 0
   231→  const step4Valid =
   232→    !!address.cep && address.cep.replace(/\D/g, "").length >= 8 &&
   233→    !!address.street && !!address.number && !!address.city && !!address.state
   234→
   235→  const validSteps: Record<number, boolean> = {
   236→    1: step1Valid,
   237→    2: step2Valid,
   238→    3: step3Valid,
   239→    4: step4Valid,
   240→    5: true,
   241→  }
   242→
   243→  // ── Navigation ──
   244→  const handleStepClick = (target: Step) => {
   245→    if (target < step) {
   246→      setStep(target)
   247→      return
   248→    }
   249→    for (let i = 1; i < target; i++) {
   250→      if (!validSteps[i as Step]) {
   251→        toast.error("Complete os passos anteriores primeiro.")
   252→        return
   253→      }
   254→    }
   255→    setStep(target)
   256→  }
   257→
   258→  const handleNext = () => {
   259→    if (step === 1 && !step1Valid) {
   260→      form.trigger("items.0.providerId")
   261→      return
   262→    }
   263→    if (step === 2 && !step2Valid) {
   264→      form.trigger("items.0.serviceId")
   265→      return
   266→    }
   267→    if (step === 3 && !step3Valid) {
   268→      form.trigger("items.0.description")
   269→      form.trigger("items.0.quantity")
   270→      return
   271→    }
   272→    if (step === 4 && !step4Valid) {
   273→      form.trigger("address")
   274→      return
   275→    }
   276→    setStep((s) => Math.min(5, s + 1) as Step)
   277→  }
   278→
   279→  const handleBack = () => setStep((s) => Math.max(1, s - 1) as Step)
   280→
   281→  const onSubmit = async (values: QuoteFormValues) => {
   282→    if (!user) {
   283→      toast.info("Faça cadastro gratuito para pedir orçamentos.")
   284→      openAuth("register", "CLIENT")
   285→      return
   286→    }
   287→    const primary = values.items[0]
   288→    if (!primary?.providerId) {
   289→      toast.error("Selecione um prestador.")
   290→      return
   291→    }
   292→    setSubmitting(true)
   293→    try {
   294→      await apiPost("/api/quotes", {
   295→        providerId: primary.providerId,
   296→        items: values.items.map((it) => ({
   297→          providerId: it.providerId,
   298→          serviceId: it.serviceId,
   299→          description: it.description,
   300→          quantity: it.quantity,
   301→          unit: it.unit,
   302→          photos: it.photos,
   303→        })),
   304→        address: [
   305→          values.address.street,
   306→          values.address.number,
   307→          values.address.complement,
   308→          values.address.district,
   309→        ]
   310→          .filter(Boolean)
   311→          .join(", "),
   312→        cep: values.address.cep,
   313→        lat: values.address.lat,
   314→        lng: values.address.lng,
   315→      })
   316→      toast.success("Orçamento enviado! O prestador responderá em breve.")
   317→      close()
   318→      navigate("client.quotes")
   319→    } catch (e) {
   320→      const msg =
   321→        (e as { message?: string })?.message ??
   322→        "Não foi possível enviar o orçamento. Tente novamente."
   323→      toast.error(msg)
   324→    } finally {
   325→      setSubmitting(false)
   326→    }
   327→  }
   328→
   329→  // ── Step content ──
   330→  const stepContent = (() => {
   331→    switch (step) {
   332→      case 1:
   333→        return <Step1Provider form={form} />
   334→      case 2:
   335→        return <Step2Service form={form} />
   336→      case 3:
   337→        return <Step3Details form={form} />
   338→      case 4:
   339→        return <Step4Address form={form} />
   340→      case 5:
   341→        return <Step5Review form={form} goToStep={setStep} />
   342→      default:
   343→        return null
   344→    }
   345→  })()
   346→
   347→  const wizardBody = (
   348→    <StepWizard
   349→      steps={STEPS}
   350→      currentStep={step}
   351→      validSteps={validSteps}
   352→      onStepClick={handleStepClick}
   353→      onBack={handleBack}
   354→      onNext={handleNext}
   355→      onSubmit={form.handleSubmit(onSubmit)}
   356→      submitting={submitting}
   357→      currentStepValid={validSteps[step] ?? true}
   358→      submitLabel={
   359→        <>
   360→          <Send className="size-3.5" />
   361→          Enviar orçamento
   362→        </>
   363→      }
   364→    >
   365→      {/* Auth gate — Nielsen #5: error prevention (remind user to login) */}
   366→      {step === 1 && !user && (
   367→        <div className="mb-4 flex items-start gap-3 rounded-lg border border-amber-200 bg-amber-50 p-3 text-sm dark:border-amber-900/50 dark:bg-amber-950/30">
   368→          <LogIn className="size-5 shrink-0 text-amber-600" />
   369→          <div className="flex-1">
   370→            <p className="font-medium text-amber-900 dark:text-amber-200">
   371→              Faça cadastro gratuito para pedir orçamentos
   372→            </p>
   373→            <p className="mt-0.5 text-xs text-amber-800 dark:text-amber-300">
   374→              Seus dados ficam salvos para acompanhar as respostas.
   375→            </p>
   376→          </div>
   377→          <Button
   378→            type="button"
   379→            size="sm"
   380→            onClick={() => openAuth("register", "CLIENT")}
   381→            className="shrink-0 bg-amber-600 hover:bg-amber-700 text-white h-8"
   382→          >
   383→            Entrar
   384→          </Button>
   385→        </div>
   386→      )}
   387→      {stepContent}
   388→    </StepWizard>
   389→  )
   390→
   391→  if (isMobile) {
   392→    return (
   393→      <Sheet open={open} onOpenChange={(o) => !o && close()}>
   394→        <SheetContent
   395→          side="bottom"
   396→          className="h-[100dvh] max-h-[100dvh] w-full p-0 sm:max-w-full gap-0 flex flex-col"
   397→          onInteractOutside={(e) => e.preventDefault()}
   398→        >
   399→          <SheetHeader className="px-4 pt-4 pb-2 shrink-0">
   400→            <SheetTitle className="flex items-center gap-2 text-base">
   401→              <Wrench className="size-4 text-emerald-600" />
   402→              Pedir orçamento
   403→            </SheetTitle>
   404→            <SheetDescription className="text-xs">
   405→              Descreva o serviço e receba propostas de prestadores verificados.
   406→            </SheetDescription>
   407→          </SheetHeader>
   408→          <div className="flex-1 overflow-hidden">{wizardBody}</div>
   409→        </SheetContent>
   410→      </Sheet>
   411→    )
   412→  }
   413→
   414→  return (
   415→    <Dialog open={open} onOpenChange={(o) => !o && close()}>
   416→      <DialogContent className="sm:max-w-2xl p-0 gap-0 overflow-hidden" onInteractOutside={(e) => e.preventDefault()}>
   417→        {/* Nielsen #3: User control — prevent accidental close during wizard */}
   418→        <DialogHeader className="px-5 pt-5 pb-2 shrink-0 border-b">
   419→          <DialogTitle className="flex items-center gap-2 text-base">
   420→            <Wrench className="size-4 text-emerald-600" />
   421→            Pedir orçamento
   422→          </DialogTitle>
   423→          <DialogDescription className="text-xs">
   424→            Descreva o serviço e receba propostas de prestadores verificados.
   425→          </DialogDescription>
   426→        </DialogHeader>
   427→        {wizardBody}
   428→      </DialogContent>
   429→    </Dialog>
   430→  )
   431→}

   468→function SelectedProviderCard({ providerId }: { providerId: string }) {
   469→  const { data } = useQuery({
   470→    queryKey: ["providers-options", ""],
   471→    queryFn: () =>
   472→      apiGet<PagedResult<ProviderCard>>("/api/providers", { limit: 50 }),
   473→    staleTime: 30 * 1000,
   474→  })
   475→
   476→  const provider = data?.items?.find((p) => p.id === providerId)
   477→  if (!provider) return null
   478→
   479→  return (
   480→    <InfoCard variant="emerald">
   481→      <div className="flex items-center gap-3">
   482→        <Avatar className="size-10 rounded-md">
   483→          {provider.avatarUrl ? (
   484→            <AvatarImage src={provider.avatarUrl} alt={provider.name} />
   485→          ) : null}
   486→          <AvatarFallback className="rounded-md bg-emerald-100 text-emerald-700 text-sm dark:bg-emerald-950 dark:text-emerald-300">
   487→            {provider.name?.[0]?.toUpperCase() ?? "?"}
   488→          </AvatarFallback>
   489→        </Avatar>
   490→        <div className="flex-1 min-w-0">
   491→          <p className="text-sm font-medium truncate">{provider.name}</p>
   492→          <p className="text-xs text-muted-foreground truncate">
   493→            {provider.city ?? "—"}
   494→            {provider.verified && " · Verificado ✓"}
   495→          </p>
   496→        </div>
   497→        {provider.verified && (
   498→          <Badge variant="outline" className="border-emerald-500 text-emerald-700 text-[10px] shrink-0">
   499→            Verificado
   500→          </Badge>
   501→        )}
   502→      </div>
   503→    </InfoCard>
   504→  )
   505→}

   547→function ServiceInfoCard({ providerId, serviceId }: { providerId: string; serviceId: string }) {
   548→  const { data: services } = useQuery({
   549→    queryKey: ["services-by-provider", providerId],
   550→    queryFn: () =>
   551→      apiGet<ProviderService[]>("/api/services", { providerId }),
   552→    enabled: !!providerId,
   553→    staleTime: 30 * 1000,
   554→  })
   555→
   556→  const selected = services?.find((s) => s.id === serviceId)
   557→  if (!selected) return null
   558→
   559→  return (
   560→    <InfoCard variant="emerald">
   561→      <div className="flex items-start justify-between gap-2">
   562→        <div className="min-w-0">
   563→          <p className="text-sm font-medium text-emerald-800 dark:text-emerald-300 truncate">
   564→            {selected.title}
   565→          </p>
   566→          {selected.description && (
   567→            <p className="text-xs text-muted-foreground mt-0.5 line-clamp-2">
   568→              {selected.description}
   569→            </p>
   570→          )}
   571→        </div>
   572→        <Badge variant="outline" className="shrink-0 border-emerald-500 text-emerald-700 text-xs">
   573→          {selected.basePrice.toLocaleString("pt-BR", {
   574→            style: "currency",
   575→            currency: "BRL",
   576→          })}
   577→          /{SERVICE_UNIT_SHORT[selected.unit]}
   578→        </Badge>
   579→      </div>
   580→    </InfoCard>
   581→  )
   582→}

   585→// Step 3 — Detalhes (Service Details)

   587→// Nielsen #8: Minimalist — clean form, focused on description

   589→
   590→function Step3Details({ form }: { form: UseFormReturn<QuoteFormValues> }) {
   591→  const item = form.watch("items.0")
   592→  const error = form.formState.errors.items?.[0]
   593→  const descLen = item?.description?.length ?? 0
   594→
   595→  return (
   596→    <div className="grid gap-4">
   597→      <StepHeader
   598→        icon={Pencil}
   599→        title="Descreva o serviço"
   600→        description="Quanto mais detalhes, mais preciso será o orçamento."
   601→      />
   602→
   603→      {/* Description — Nielsen #5: error prevention with counter */}
   604→      <div className="grid gap-1.5">
   605→        <div className="flex items-center justify-between">
   606→          <Label htmlFor="item-0-desc">Descrição do serviço</Label>
   607→          <span className={cn(
   608→            "text-xs tabular-nums",
   609→            descLen < 10 ? "text-destructive" : "text-muted-foreground",
   610→          )}>
   611→            {descLen}/400
   612→          </span>
   613→        </div>
   614→        <Textarea
   615→          id="item-0-desc"
   616→          placeholder="Ex.: Instalar tomada na parede da sala, fio visível. Preciso de 3 pontos novos."
