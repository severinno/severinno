     1→"use client"

     3→/**
     4→ * AdminUsers — manage all personas (Clients, Providers, Admins).
     5→ *
     6→ * Data source: GET /api/admin/users?role=&q=&page=
     7→ *              PATCH /api/admin/users/[id] { verified?, active?, role?, name?, ... }
     8→ *              DELETE /api/admin/users/[id]
     9→ *
    10→ * Note: the `verified` and `active` filters are applied client-side on the
    11→ * current page's items (the API does not yet support them as query params).
    12→ * A small "Filtro aplicado à página atual" hint is shown when active.
    13→ */
    14→
    15→import * as React from "react"
    16→import {
    17→  Search,
    18→  MoreHorizontal,
    19→  Pencil,
    20→  Trash2,
    21→  BadgeCheck,
    22→  Loader2,
    23→  ChevronLeft,
    24→  ChevronRight,
    25→  ShieldAlert,
    26→  X,
    27→  Users,
    28→  User,
    29→  HardHat,
    30→  ShieldCheck,
    31→  ShieldQuestion,
    32→  CircleUser,
    33→} from "lucide-react"
    34→import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query"
    35→import { toast } from "sonner"
    36→
    37→import { apiGet, apiPatch, apiDelete } from "@/lib/api"
    38→import {
    39→  ROLE_LABELS,
    40→  type UserRole,
    41→} from "@/lib/constants"
    42→import { formatDate } from "@/lib/format"
    43→import { cn } from "@/lib/utils"
    44→import { Avatar, AvatarFallback, AvatarImage } from "@/components/ui/avatar"
    45→import { Badge } from "@/components/ui/badge"
    46→import { Button } from "@/components/ui/button"

    50→} from "@/components/ui/card"
    51→import {

    57→  DialogTitle,
    58→} from "@/components/ui/dialog"
    59→import {

    68→} from "@/components/ui/alert-dialog"

    73→  DropdownMenuLabel,
    74→  DropdownMenuSeparator,
    75→  DropdownMenuTrigger,
    76→} from "@/components/ui/dropdown-menu"

    85→} from "@/components/ui/select"

    98→// ---------------------------------------------------------------------------
    99→// Types
   100→// ---------------------------------------------------------------------------
   101→type AdminUser = {
   102→  id: string
   103→  name: string
   104→  email: string
   105→  role: UserRole
   106→  cpfCnpj?: string | null
   107→  whatsapp?: string | null
   108→  phone?: string | null
   109→  avatarUrl?: string | null
   110→  city?: string | null
   111→  state?: string | null
   112→  verified: boolean
   113→  active: boolean
   114→  createdAt: string
   115→}
   116→
   117→type AdminUsersResponse = {
   118→  items: AdminUser[]
   119→  total: number
   120→  page: number
   121→  limit: number
   122→}
   123→
   124→type RoleFilter = "ALL" | UserRole
   125→type VerifiedFilter = "ALL" | "true" | "false"
   126→type ActiveFilter = "ALL" | "true" | "false"
   127→
   128→type StatsResponse = {
   129→  usersByRole: Record<string, number>
   130→}
   131→
   132→// ---------------------------------------------------------------------------
   133→// Main component
   134→// ---------------------------------------------------------------------------
   135→export function AdminUsers() {
   136→  const queryClient = useQueryClient()
   137→  const [role, setRole] = React.useState<RoleFilter>("ALL")
   138→  const [q, setQ] = React.useState("")
   139→  const [debouncedQ, setDebouncedQ] = React.useState("")
   140→  const [verified, setVerified] = React.useState<VerifiedFilter>("ALL")
   141→  const [active, setActive] = React.useState<ActiveFilter>("ALL")
   142→  const [page, setPage] = React.useState(1)
   143→
   144→  const [editTarget, setEditTarget] = React.useState<AdminUser | null>(null)
   145→  const [deleteTarget, setDeleteTarget] = React.useState<AdminUser | null>(null)
   146→
   147→  const limit = 10
   148→
   149→  // Debounce search
   150→  React.useEffect(() => {
   151→    const t = setTimeout(() => {
   152→      setDebouncedQ(q.trim())
   153→      setPage(1)
   154→    }, 350)
   155→    return () => clearTimeout(t)
   156→  }, [q])
   157→
   158→  // Lightweight role counts for the pill tabs (cached 60s, shared with dashboard)
   159→  const { data: stats } = useQuery({
   160→    queryKey: ["admin", "stats"],
   161→    queryFn: () => apiGet<StatsResponse>("/api/admin/stats"),
   162→    staleTime: 60_000,
   163→  })
   164→
   165→  const { data, isLoading, isError } = useQuery({
   166→    queryKey: ["admin", "users", { role, debouncedQ, verified, active, page, limit }],
   167→    queryFn: () =>
   168→      apiGet<AdminUsersResponse>("/api/admin/users", {
   169→        ...(role !== "ALL" ? { role } : {}),
   170→        ...(debouncedQ ? { q: debouncedQ } : {}),
   171→        page,
   172→        limit,
   173→      }),
   174→    staleTime: 15_000,
   175→  })
   176→
   177→  const invalidate = () =>
   178→    queryClient.invalidateQueries({ queryKey: ["admin", "users"] })
   179→
   180→  const patchMutation = useMutation({
   181→    mutationFn: ({
   182→      id,
   183→      patch,
   184→    }: {
   185→      id: string
   186→      patch: Partial<AdminUser>
   187→    }) => apiPatch<{ user: AdminUser }>(`/api/admin/users/${id}`, patch),
   188→    onSuccess: () => {
   189→      invalidate()
   190→    },
   191→    onError: (e: unknown) => {
   192→      toast.error(errMsg(e, "Falha ao atualizar usuário."))
   193→    },
   194→  })
   195→
   196→  const deleteMutation = useMutation({
   197→    mutationFn: (id: string) => apiDelete(`/api/admin/users/${id}`),
   198→    onSuccess: () => {
   199→      toast.success("Usuário excluído.")
   200→      invalidate()
   201→      setDeleteTarget(null)
   202→    },
   203→    onError: (e: unknown) => {
   204→      toast.error(errMsg(e, "Não foi possível excluir o usuário."))
   205→    },
   206→  })
   207→
   208→  // Client-side filter for verified/active (API does not support these yet)
   209→  const rawItems = data?.items ?? []
   210→  const items = React.useMemo(() => {
   211→    return rawItems.filter((u) => {
   212→      if (verified === "true" && !u.verified) return false
   213→      if (verified === "false" && u.verified) return false
   214→      if (active === "true" && !u.active) return false
   215→      if (active === "false" && u.active) return false
   216→      return true
   217→    })
   218→  }, [rawItems, verified, active])
   219→
   220→  const total = data?.total ?? 0
   221→  const totalPages = Math.max(1, Math.ceil(total / limit))
   222→
   223→  const clientFilterActive = verified !== "ALL" || active !== "ALL"
   224→
   225→  const activeFilterCount =
   226→    (role !== "ALL" ? 1 : 0) +
   227→    (debouncedQ ? 1 : 0) +
   228→    (verified !== "ALL" ? 1 : 0) +
   229→    (active !== "ALL" ? 1 : 0)
   230→
   231→  const clearFilters = () => {
   232→    setRole("ALL")
   233→    setQ("")
   234→    setDebouncedQ("")
   235→    setVerified("ALL")
   236→    setActive("ALL")
   237→    setPage(1)
   238→  }
   239→
   240→  const roleCounts = React.useMemo(() => {
   241→    const byRole = stats?.usersByRole ?? {}
   242→    return {
   243→      ALL:
   244→        (byRole.CLIENT ?? 0) + (byRole.PROVIDER ?? 0) + (byRole.ADMIN ?? 0),
   245→      CLIENT: byRole.CLIENT ?? 0,
   246→      PROVIDER: byRole.PROVIDER ?? 0,
   247→      ADMIN: byRole.ADMIN ?? 0,
   248→    }
   249→  }, [stats])
   250→
   251→  return (
   252→    <div className="flex flex-col gap-4">
   253→      {/* Pill segmented control — role tabs with counts */}
   254→      <Tabs
   255→        value={role}
   256→        onValueChange={(v) => {
   257→          setRole(v as RoleFilter)
   258→          setPage(1)
   259→        }}
   260→      >
   261→        <TabsList className="h-auto flex-wrap gap-1 bg-card p-1 shadow-sm">
   262→          {(
   263→            [
   264→              { value: "ALL", label: "Todos", icon: Users },
   265→              { value: "CLIENT", label: "Clientes", icon: CircleUser },
   266→              { value: "PROVIDER", label: "Prestadores", icon: HardHat },
   267→              { value: "ADMIN", label: "Administradores", icon: ShieldCheck },
   268→            ] as const
   269→          ).map((t) => (
   270→            <TabsTrigger
   271→              key={t.value}
   272→              value={t.value}
   273→              className="h-8 gap-1.5 rounded-md px-3 text-sm data-[state=active]:bg-primary data-[state=active]:text-primary-foreground"
   274→            >
   275→              <t.icon className="size-3.5" />
   276→              {t.label}
   277→              <span
   278→                className={cn(
   279→                  "ml-1 inline-flex h-5 min-w-5 items-center justify-center rounded-full px-1.5 text-[10px] font-semibold tabular-nums",
   280→                  role === t.value
   281→                    ? "bg-primary-foreground/20 text-primary-foreground"
   282→                    : "bg-muted text-muted-foreground",
   283→                )}
   284→              >
   285→                {roleCounts[t.value]}
   286→              </span>
   287→            </TabsTrigger>
   288→          ))}
   289→        </TabsList>
   290→      </Tabs>
   291→
   292→      {/* Filter bar */}
   293→      <div className="flex flex-wrap items-center gap-3 rounded-xl border bg-card p-3 shadow-sm">
   294→        <div className="relative min-w-[200px] flex-1">
   295→          <Search className="absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
   296→          <Input
   297→            id="u-search"
   298→            value={q}
   299→            onChange={(e) => setQ(e.target.value)}
   300→            placeholder="Buscar por nome, e-mail ou cidade"
   301→            className="h-9 pl-8"
   302→          />
   303→        </div>
   304→        <Select
   305→          value={verified}
   306→          onValueChange={(v) => {
   307→            setVerified(v as VerifiedFilter)
   308→            setPage(1)
   309→          }}
   310→        >
   311→          <SelectTrigger className="h-9 w-auto min-w-[150px]">
   312→            <SelectValue placeholder="Verificação" />
   313→          </SelectTrigger>
   314→          <SelectContent>
   315→            <SelectItem value="ALL">Verificação: todas</SelectItem>
   316→            <SelectItem value="true">Apenas verificados</SelectItem>
   317→            <SelectItem value="false">Aguardando verificação</SelectItem>
   318→          </SelectContent>
   319→        </Select>
   320→        <Select
   321→          value={active}
   322→          onValueChange={(v) => {
   323→            setActive(v as ActiveFilter)
   324→            setPage(1)
   325→          }}
   326→        >
   327→          <SelectTrigger className="h-9 w-auto min-w-[140px]">
   328→            <SelectValue placeholder="Status" />
   329→          </SelectTrigger>
   330→          <SelectContent>
   331→            <SelectItem value="ALL">Status: todos</SelectItem>
   332→            <SelectItem value="true">Apenas ativos</SelectItem>
   333→            <SelectItem value="false">Apenas inativos</SelectItem>
   334→          </SelectContent>
   335→        </Select>
   336→        <Button
   337→          variant="ghost"
   338→          size="sm"
   339→          onClick={clearFilters}
   340→          disabled={activeFilterCount === 0}
   341→          className="h-9 gap-1.5 text-muted-foreground"
   342→        >
   343→          <X className="size-3.5" />
   344→          Limpar filtros
   345→        </Button>
   346→        {activeFilterCount > 0 ? (
   347→          <Badge
   348→            variant="outline"
   349→            className="gap-1 border-primary/30 bg-primary/5 text-primary"
   350→          >
   351→            <span className="size-1.5 rounded-full bg-primary" />
   352→            {activeFilterCount} filtro(s) ativo(s)
   353→          </Badge>
   354→        ) : null}
   355→      </div>
   356→
   357→      {/* Result count */}
   358→      <div className="flex flex-wrap items-center justify-between gap-2">
   359→        <p className="text-xs text-muted-foreground">
   360→          {total > 0
   361→            ? `Mostrando ${(page - 1) * limit + 1}–${Math.min(
   362→                page * limit,
   363→                total,
   364→              )} de ${total.toLocaleString("pt-BR")} usuário(s)`
   365→            : "Nenhum usuário"}
   366→        </p>
   367→        {clientFilterActive ? (
   368→          <span className="inline-flex items-center gap-1 text-[11px] text-amber-700 dark:text-amber-300">
   369→            <ShieldQuestion className="size-3" />
   370→            Filtro aplicado à página atual
   371→          </span>
   372→        ) : null}
   373→      </div>
   374→
   375→      {/* Table */}
   376→      <Card className="overflow-hidden">
   377→        <CardContent className="p-0">
   378→          {isError ? (
   379→            <div className="flex flex-col items-center gap-3 px-4 py-16 text-center">
   380→              <div className="flex size-14 items-center justify-center rounded-full bg-rose-50 text-rose-600 dark:bg-rose-950/30 dark:text-rose-300">
   381→                <ShieldAlert className="size-6" />
   382→              </div>
   383→              <div>
   384→                <p className="text-sm font-semibold">
   385→                  Não foi possível carregar os usuários
   386→                </p>
   387→                <p className="mt-0.5 text-xs text-muted-foreground">
   388→                  Verifique sua conexão e tente novamente.
   389→                </p>
   390→              </div>
   391→            </div>
   392→          ) : (
   393→            <div className="overflow-x-auto">
   394→              <Table>
   395→                <TableHeader>
   396→                  <TableRow className="h-11 bg-muted/50 hover:bg-muted/50">
   397→                    <TableHead className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   398→                      Usuário
   399→                    </TableHead>
   400→                    <TableHead className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   401→                      Perfil
   402→                    </TableHead>
   403→                    <TableHead className="hidden text-xs font-semibold uppercase tracking-wide text-muted-foreground md:table-cell">
   404→                      Contato
   405→                    </TableHead>
   406→                    <TableHead className="hidden text-xs font-semibold uppercase tracking-wide text-muted-foreground lg:table-cell">
   407→                      Cidade
   408→                    </TableHead>
   409→                    <TableHead className="text-center text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   410→                      Verificado
   411→                    </TableHead>
   412→                    <TableHead className="text-center text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   413→                      Status
   414→                    </TableHead>
   415→                    <TableHead className="hidden text-xs font-semibold uppercase tracking-wide text-muted-foreground sm:table-cell">
   416→                      Criado em
   417→                    </TableHead>
   418→                    <TableHead className="text-right text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   419→                      Ações
   420→                    </TableHead>
   421→                  </TableRow>
   422→                </TableHeader>
   423→                <TableBody>
   424→                  {isLoading ? (
   425→                    Array.from({ length: 5 }).map((_, i) => (
   426→                      <TableRow key={i} className="h-14">
   427→                        <TableCell colSpan={8}>
   428→                          <Skeleton className="h-8 w-full" />
   429→                        </TableCell>
   430→                      </TableRow>
   431→                    ))
   432→                  ) : items.length === 0 ? (
   433→                    <TableRow className="h-14 hover:bg-transparent">
   434→                      <TableCell colSpan={8} className="py-12">
   435→                        <div className="flex flex-col items-center gap-3 text-center">
   436→                          <div className="flex size-16 items-center justify-center rounded-full bg-muted text-muted-foreground">
   437→                            <User className="size-7" />
   438→                          </div>
   439→                          <div>
   440→                            <p className="text-base font-semibold">
   441→                              Nenhum usuário encontrado
   442→                            </p>
   443→                            <p className="mt-0.5 text-sm text-muted-foreground">
   444→                              Ajuste os filtros ou cadastre um novo usuário.
   445→                            </p>
   446→                          </div>
   447→                          {activeFilterCount > 0 ? (
   448→                            <Button
   449→                              variant="outline"
   450→                              size="sm"
   451→                              onClick={clearFilters}
   452→                              className="gap-1.5"
   453→                            >
   454→                              <X className="size-3.5" />
   455→                              Limpar filtros
   456→                            </Button>
   457→                          ) : null}
   458→                        </div>
   459→                      </TableCell>
   460→                    </TableRow>
   461→                  ) : (
   462→                    items.map((u) => (
   463→                      <TableRow
   464→                        key={u.id}
   465→                        className="h-14 border-b transition-colors last:border-0 hover:bg-muted/30"
   466→                      >
   467→                        <TableCell className="px-4 py-3">
   468→                          <div className="flex items-center gap-2.5">
   469→                            <Avatar className="size-9 shrink-0">
   470→                              {u.avatarUrl ? (
   471→                                <AvatarImage
   472→                                  src={u.avatarUrl}
   473→                                  alt={u.name}
   474→                                />
   475→                              ) : null}
   476→                              <AvatarFallback className="bg-primary/10 text-[11px] font-semibold text-primary">
   477→                                {initials(u.name)}
   478→                              </AvatarFallback>
   479→                            </Avatar>
   480→                            <div className="min-w-0">
   481→                              <p className="truncate text-sm font-medium">
   482→                                {u.name}
   483→                              </p>
   484→                              <p className="truncate text-xs text-muted-foreground">
   485→                                {u.email}
   486→                              </p>
   487→                            </div>
   488→                          </div>
   489→                        </TableCell>
   490→                        <TableCell className="px-4 py-3">
   491→                          <RoleBadge role={u.role} />
   492→                        </TableCell>
   493→                        <TableCell className="hidden px-4 py-3 md:table-cell">
   494→                          <div className="flex flex-col text-xs">
   495→                            {u.whatsapp ? (
   496→                              <span className="text-foreground/80">
   497→                                {u.whatsapp}
   498→                              </span>
   499→                            ) : (
   500→                              <span className="text-muted-foreground">—</span>
   501→                            )}
   502→                            {u.cpfCnpj ? (
   503→                              <span className="font-mono text-muted-foreground">
   504→                                {u.cpfCnpj}
   505→                              </span>
   506→                            ) : (
   507→                              <span className="text-muted-foreground">—</span>
   508→                            )}
   509→                          </div>
   510→                        </TableCell>
   511→                        <TableCell className="hidden px-4 py-3 text-xs text-muted-foreground lg:table-cell">
   512→                          {u.city ? (
   513→                            <span className="text-foreground/80">
   514→                              {u.city}
   515→                              {u.state ? `/${u.state}` : ""}
   516→                            </span>
   517→                          ) : (
   518→                            <span className="text-muted-foreground">—</span>
   519→                          )}
   520→                        </TableCell>
   521→                        <TableCell className="px-4 py-3 text-center">
   522→                          <Switch
   523→                            checked={u.verified}
   524→                            onCheckedChange={(v) => {
   525→                              patchMutation.mutate({
   526→                                id: u.id,
   527→                                patch: { verified: v },
   528→                              })
   529→                              toast.message(
   530→                                v
   531→                                  ? "Usuário marcado como verificado."
   532→                                  : "Verificação removida.",
   533→                              )
   534→                            }}
   535→                            disabled={patchMutation.isPending}
   536→                            aria-label="Alternar verificação"
   537→                            className="data-[state=checked]:bg-emerald-600"
   538→                          />
   539→                        </TableCell>
   540→                        <TableCell className="px-4 py-3 text-center">
   541→                          {u.active ? (
   542→                            <Badge
   543→                              variant="outline"
   544→                              className="gap-1 border-emerald-200 bg-emerald-50 text-emerald-700 dark:border-emerald-900/40 dark:bg-emerald-950/30 dark:text-emerald-300"
   545→                            >
   546→                              <span className="size-1.5 rounded-full bg-emerald-500" />
   547→                              Ativo
   548→                            </Badge>
   549→                          ) : (
   550→                            <Badge
   551→                              variant="outline"
   552→                              className="gap-1 border-slate-200 bg-slate-50 text-slate-600 dark:border-slate-700/40 dark:bg-slate-800/40 dark:text-slate-300"
   553→                            >
   554→                              <span className="size-1.5 rounded-full bg-slate-400" />
   555→                              Inativo
   556→                            </Badge>
   557→                          )}
   558→                        </TableCell>
   559→                        <TableCell className="hidden px-4 py-3 text-xs text-muted-foreground tabular-nums sm:table-cell">
   560→                          {formatDate(u.createdAt)}
   561→                        </TableCell>
   562→                        <TableCell className="px-4 py-3 text-right">
   563→                          <DropdownMenu>
   564→                            <DropdownMenuTrigger asChild>
   565→                              <Button
   566→                                variant="ghost"
   567→                                size="icon"
   568→                                className="size-8"
   569→                                aria-label="Ações"
   570→                              >
   571→                                <MoreHorizontal className="size-4" />
   572→                              </Button>
   573→                            </DropdownMenuTrigger>
   574→                            <DropdownMenuContent align="end">
   575→                              <DropdownMenuLabel>{u.name}</DropdownMenuLabel>
   576→                              <DropdownMenuSeparator />
   577→                              <DropdownMenuItem
   578→                                onClick={() => setEditTarget(u)}
   579→                                className="gap-2"
   580→                              >
   581→                                <Pencil className="size-3.5" />
   582→                                Editar
   583→                              </DropdownMenuItem>
   584→                              <DropdownMenuItem
   585→                                onClick={() =>
   586→                                  patchMutation.mutate({
   587→                                    id: u.id,
   588→                                    patch: { verified: !u.verified },
   589→                                  })
   590→                                }
   591→                                className="gap-2"
   592→                              >
   593→                                <BadgeCheck className="size-3.5" />
   594→                                {u.verified
   595→                                  ? "Remover verificação"
   596→                                  : "Marcar verificado"}
   597→                              </DropdownMenuItem>
   598→                              <DropdownMenuItem
   599→                                onClick={() =>
   600→                                  patchMutation.mutate({
   601→                                    id: u.id,
   602→                                    patch: { active: !u.active },
   603→                                  })
   604→                                }
   605→                                className="gap-2"
   606→                              >
   607→                                <User className="size-3.5" />
   608→                                {u.active ? "Desativar" : "Ativar"}
   609→                              </DropdownMenuItem>
   610→                              <DropdownMenuSeparator />

   612→                                onClick={() => setDeleteTarget(u)}

   615→                                <Trash2 className="size-3.5" />

   618→                            </DropdownMenuContent>

   620→                        </TableCell>
   621→                      </TableRow>
   622→                    ))
   623→                  )}
   624→                </TableBody>
   625→              </Table>
   626→            </div>
   627→          )}
   628→
   629→          {/* Pagination — hidden when only 1 page */}
   630→          {totalPages > 1 ? (
   631→            <div className="flex flex-col items-center justify-between gap-2 border-t px-4 py-3 sm:flex-row">
   632→              <p className="text-xs text-muted-foreground tabular-nums">

   638→                  size="sm"

   646→                <Button
   647→                  variant="outline"
   648→                  size="sm"

   652→                >
   653→                  Próxima
   654→                  <ChevronRight className="size-4" />
   655→                </Button>

   657→            </div>

   659→        </CardContent>
   660→      </Card>
   661→
   662→      {/* Edit dialog */}
   663→      <EditUserDialog
   664→        user={editTarget}
   665→        onOpenChange={(open) => !open && setEditTarget(null)}
   666→        submitting={patchMutation.isPending}
   667→        onSubmit={(patch) => {
   668→          if (!editTarget) return
   669→          patchMutation.mutate(
   670→            { id: editTarget.id, patch },
   671→            {
   672→              onSuccess: () => {

   881→function initials(name: string): string {
   882→  if (!name) return "?"
   883→  const parts = name.trim().split(/\s+/)
   884→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase()
   885→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase()
   886→}
   887→
   888→function errMsg(e: unknown, fallback: string): string {
   889→  if (e && typeof e === "object" && "message" in e) {
   890→    return String((e as { message?: unknown }).message ?? fallback)
   891→  }
   892→  return fallback
   893→}
