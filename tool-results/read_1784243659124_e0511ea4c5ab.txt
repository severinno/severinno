     3→/**
     4→ * DashboardShell — reusable sidebar + topbar layout for all panels
     5→ * (client / provider / admin).
     6→ *
     7→ * Props:
     8→ *  - navItems:        primary navigation (label, icon, view, optional badge)

    17→ *  - children:        main content
    18→ *
    19→ * The shell DOES NOT include the marketing footer — instead it renders a
    20→ * thin copyright bar at the bottom of the main content area.

   147→function initials(name?: string | null): string {
   148→  if (!name) return "?"
   149→  return name
   150→    .split(" ")
   151→    .map((p) => p[0])
   152→    .filter(Boolean)
   153→    .slice(0, 2)
   154→    .join("")
   155→    .toUpperCase()
   156→}

   391→                              : "text-foreground hover:bg-accent hover:text-accent-foreground",
   392→                          )}
   393→                        >
   394→                          {active ? (
   395→                            <span
   396→                              aria-hidden
   397→                              className="absolute left-0 top-1/2 h-5 w-1 -translate-y-1/2 rounded-r-full bg-primary"
   398→                            />
   399→                          ) : null}
   400→                          <Icon className="size-4 shrink-0" />
   401→                          <span className="truncate">{item.label}</span>
   402→                          {item.badge != null && item.badge !== 0 ? (
   403→                            <span className="ml-auto inline-flex h-5 min-w-5 items-center justify-center rounded-full bg-primary px-1.5 text-[10px] font-semibold text-primary-foreground">
   404→                              {item.badge}
   405→                            </span>
   406→                          ) : null}
   407→                        </button>
   408→                      )
   409→                    })}
   410→                  </nav>
   411→                </ScrollArea>
   412→                <div className="border-t p-3">
   413→                  <div className="flex items-center gap-2.5 rounded-lg border bg-muted/30 p-2">
   414→                    <Avatar className="size-9">
   415→                      {user?.avatarUrl ? (
   416→                        <AvatarImage
   417→                          src={user.avatarUrl}
   418→                          alt={user.name ?? ""}
   419→                        />
   420→                      ) : null}
   421→                      <AvatarFallback className="bg-primary text-xs font-semibold text-primary-foreground">
   422→                        {initials(user?.name)}
   423→                      </AvatarFallback>
   424→                    </Avatar>
   425→                    <div className="min-w-0 flex-1">
   426→                      <p className="truncate text-sm font-medium leading-tight">
   427→                        {user?.name ?? "Visitante"}

   429→                      {user?.role ? (
   430→                        <Badge
   431→                          variant="secondary"
   432→                          className="mt-0.5 h-4 px-1.5 text-[10px] font-medium"
   433→                        >
   434→                          {ROLE_LABELS[user.role]}

   436→                      ) : null}
   437→                    </div>
   438→                    <Button
   439→                      variant="ghost"
   440→                      size="icon"
   441→                      className="size-8 shrink-0 text-muted-foreground hover:text-foreground"
   442→                      onClick={() => {
   443→                        setMobileOpen(false)
   444→                        logout()
   445→                      }}
   446→                      aria-label="Sair"
   447→                    >
   448→                      <LogOut className="size-4" />
   449→                    </Button>
   450→                  </div>
   451→                </div>
   452→              </div>
   453→            </SheetContent>
   454→          </Sheet>
   455→
   456→          {/* Main inset */}
   457→          <SidebarInset>
   458→            {/* Topbar */}
   459→            <header className="sticky top-0 z-30 flex h-14 items-center gap-2 border-b bg-background/80 px-3 backdrop-blur supports-[backdrop-filter]:bg-background/65 md:px-4">
   460→              <Button
   461→                variant="ghost"
   462→                size="icon"
   463→                className="size-9 md:hidden"
   464→                onClick={() => setMobileOpen(true)}
   465→                aria-label="Abrir menu"
   466→              >
   467→                <Menu className="size-5" />
   468→              </Button>
   469→              <SidebarTrigger className="hidden size-9 md:flex" />
   470→
   471→              <div className="min-w-0 flex-1">
   472→                {breadcrumbs && breadcrumbs.length > 0 ? (

   491→                          <ChevronRight className="size-3" />

   494→                    ))}
   495→                  </nav>

   497→                <h1 className="truncate text-base font-semibold tracking-tight text-foreground md:text-lg">
   498→                  {title}
   499→                </h1>
   500→                {subtitle ? (
   501→                  <p className="hidden truncate text-xs text-muted-foreground sm:block">
   502→                    {subtitle}
   503→                  </p>
   504→                ) : null}
   505→              </div>
   506→
   507→              {/* Location chip */}
   508→              <Button
   509→                variant="ghost"
   510→                size="sm"
   511→                onClick={handleLocate}
   512→                disabled={locating || geoStatus === "locating"}

   525→              </Button>
   526→
   527→              {/* Theme toggle */}
   528→              <Button
   529→                variant="ghost"
   530→                size="icon"

   542→              {/* Notifications */}

   547→                onMarkRead={(id) => markReadMutation.mutate(id)}

   552→              {/* User dropdown */}

   558→                    aria-label="Menu da conta"
   559→                  >
   560→                    <Avatar className="size-9">
   561→                      {user?.avatarUrl ? (
   562→                        <AvatarImage
   563→                          src={user.avatarUrl}
   564→                          alt={user.name ?? ""}
   565→                        />
   566→                      ) : null}
   567→                      <AvatarFallback className="bg-primary text-xs font-semibold text-primary-foreground">
   568→                        {initials(user?.name)}
   569→                      </AvatarFallback>
   570→                    </Avatar>
   571→                  </button>
   572→                </DropdownMenuTrigger>
   573→                <DropdownMenuContent align="end" className="w-60">
   574→                  <DropdownMenuLabel className="flex flex-col gap-1">
   575→                    <span className="truncate">{user?.name}</span>
   576→                    <span className="truncate text-xs font-normal text-muted-foreground">
   577→                      {user?.email}
   578→                    </span>
   579→                    {user?.role ? (
   580→                      <Badge
   581→                        variant="secondary"
   582→                        className="mt-1 w-fit text-[10px]"
   583→                      >
   584→                        {ROLE_LABELS[user.role]}
   585→                      </Badge>
   586→                    ) : null}
   587→                  </DropdownMenuLabel>
   588→                  <DropdownMenuSeparator />

   594→                    <LayoutDashboard className="size-4" />
   595→                    Meu painel

   597→                  <DropdownMenuItem
   598→                    onSelect={() => {

   601→                  >
   602→                    <MapPin className="size-4" />
   603→                    Voltar à vitrine

   648→function NotificationsBell({
   649→  items,
   650→  unreadCount,
   651→  isLoading,
   652→  onMarkRead,
   653→  onMarkAllRead,
   654→  markingAll,
   655→}: {
   656→  items: NotificationItem[]
   657→  unreadCount: number
   658→  isLoading: boolean
   659→  onMarkRead: (id: string) => void
   660→  onMarkAllRead: (ids: string[]) => void
   661→  markingAll: boolean
   662→}) {
   663→  const unreadIds = React.useMemo(
   664→    () => items.filter((n) => !n.read).map((n) => n.id),
   665→    [items],
   666→  )
   667→  return (
   668→    <DropdownMenu>
   669→      <DropdownMenuTrigger asChild>
   670→        <Button
   671→          variant="ghost"
   672→          size="icon"
   673→          className="relative size-9 text-muted-foreground hover:text-foreground"
   674→          aria-label={`Notificações${
   675→            unreadCount > 0 ? ` (${unreadCount} não lidas)` : ""
   676→          }`}
   677→        >
   678→          <Bell className="size-5" />
   679→          {unreadCount > 0 ? (
   680→            <span
   681→              className="absolute -right-0.5 -top-0.5 inline-flex h-4 min-w-4 items-center justify-center rounded-full bg-primary px-1 text-[10px] font-semibold text-primary-foreground ring-2 ring-background"

   685→            </span>
   686→          ) : null}
   687→        </Button>
   688→      </DropdownMenuTrigger>
   689→      <DropdownMenuContent
   690→        align="end"
   691→        className="w-80 gap-0 p-0"
   692→      >
   693→        <div className="flex items-center justify-between border-b px-3 py-2.5">
   694→          <div className="flex items-center gap-2">
   695→            <p className="text-sm font-semibold">Notificações</p>

   697→              <Badge
   698→                variant="secondary"
   699→                className="h-5 bg-primary/10 px-1.5 text-[10px] font-semibold text-primary"
   700→              >
   701→                {unreadCount} nova{unreadCount > 1 ? "s" : ""}
   702→              </Badge>
   703→            ) : null}
   704→          </div>
   705→          {unreadCount > 0 ? (
   706→            <button
   707→              type="button"
   708→              onClick={() => onMarkAllRead(unreadIds)}
   709→              disabled={markingAll}
   710→              className="inline-flex items-center gap-1 text-[11px] font-medium text-primary transition-colors hover:text-primary/80 disabled:opacity-50"

   713→                <Loader2 className="size-3 animate-spin" />
   714→              ) : (
   715→                <CheckCheck className="size-3.5" />
   716→              )}
   717→              Marcar todas
   718→            </button>
   719→          ) : null}

   721→
   722→        <ScrollArea className="max-h-80">
   723→          {isLoading ? (
   724→            <div className="flex items-center justify-center gap-2 p-8 text-sm text-muted-foreground">
   725→              <Loader2 className="size-4 animate-spin" />
   726→              Carregando…
   727→            </div>
   728→          ) : items.length === 0 ? (
   729→            <div className="flex flex-col items-center gap-2 p-8 text-center">
   730→              <span className="flex size-12 items-center justify-center rounded-full bg-muted text-muted-foreground">
   731→                <Bell className="size-5" />
   732→              </span>
   733→              <p className="text-sm text-muted-foreground">
   734→                Você não tem notificações.
   735→              </p>

   738→            <ul className="divide-y">

   740→                const typeLabel = NOTIFICATION_TYPE_LABELS[n.type]
   741→                return (

   743→                    key={n.id}
   744→                    className={cn(
   745→                      "relative flex gap-3 px-3 py-2.5 transition-colors hover:bg-accent/50",

   747→                    )}

   749→                    <span
   750→                      className={cn(
   751→                        "mt-1.5 size-2 shrink-0 rounded-full",
   752→                        n.read ? "bg-transparent ring-1 ring-border" : "bg-primary",
   753→                      )}
   754→                      aria-hidden
   755→                    />
   756→                    <div className="min-w-0 flex-1">
   757→                      <div className="flex items-baseline justify-between gap-2">
   758→                        <p className="text-sm font-medium leading-tight">

   762→                          {formatRelative(n.createdAt)}

   767→                          {n.body}

   769→                      ) : null}
   770→                      <div className="mt-1 flex items-center gap-2">
   771→                        {typeLabel ? (
   772→                          <Badge
   773→                            variant="outline"
   774→                            className="h-4 px-1.5 text-[10px] font-medium text-muted-foreground"
   775→                          >
   776→                            {typeLabel}
   777→                          </Badge>
   778→                        ) : null}
   779→                        {!n.read ? (
   780→                          <button
   781→                            type="button"
   782→                            onClick={() => onMarkRead(n.id)}
   783→                            className="inline-flex items-center gap-1 text-[10px] font-medium text-primary hover:underline"

   785→                            <Check className="size-3" />
   786→                            Marcar como lida
   787→                          </button>
   788→                        ) : null}
   789→                      </div>
   790→                    </div>
   791→                  </li>
   792→                )
   793→              })}
   794→            </ul>
   795→          )}
   796→        </ScrollArea>
   797→      </DropdownMenuContent>
   798→    </DropdownMenu>
   799→  )

   844→export function StatCard({
   845→  icon: Icon,
   846→  label,
   847→  value,
   848→  hint,
   849→  tone = "primary",
   850→  index = 0,
   851→  trend,

   855→  value: React.ReactNode

   857→  tone?: "primary" | "amber" | "sky" | "rose" | "zinc"
