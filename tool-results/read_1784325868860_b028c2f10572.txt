   116→   116→export function BookingModal() {
   117→   117→  const open = useUIStore((s) => s.bookingModal.open)
   118→   118→  const providerIdPreset = useUIStore((s) => s.bookingModal.providerId)
   119→   119→  const serviceIdPreset = useUIStore((s) => s.bookingModal.serviceId)
   120→   120→  const close = useUIStore((s) => s.closeBooking)
   121→   121→  const openAuth = useUIStore((s) => s.openAuth)
   122→   122→  const isMobile = useIsMobile()
   123→   123→  const navigate = useViewStore((s) => s.navigate)
   124→   124→  const user = useAuthStore((s) => s.user)
   125→   125→
   126→   126→  const [step, setStep] = React.useState<Step>(1)
   127→   127→  const [state, setState] = React.useState<BookingFormState>(initialState())
   128→   128→  const [submitting, setSubmitting] = React.useState(false)
   129→   129→
   130→   130→  // Reset when modal opens.
   131→   131→  React.useEffect(() => {
   132→   132→    if (open) {
   133→   133→      setStep(1)
   134→   134→      setState(initialState())
   135→   135→    }
   136→   136→  }, [open])
   137→   137→
   138→   138→  // Fetch provider + services
   139→   139→  const providerQuery = useQuery({
   140→   140→    queryKey: ["provider", providerIdPreset],
   141→   141→    queryFn: () =>
   142→   142→      apiGet<ProviderDetail>(`/api/providers/${providerIdPreset}`),
   143→   143→    enabled: open && !!providerIdPreset,
   144→   144→    staleTime: 60 * 1000,
   145→   145→  })
   146→   146→
   147→   147→  const servicesQuery = useQuery({
   148→   148→    queryKey: ["services-by-provider", providerIdPreset],
   149→   149→    queryFn: () =>
   150→   150→      apiGet<ProviderService[]>("/api/services", {
   151→   151→        providerId: providerIdPreset,
   152→   152→      }),
   153→   153→    enabled: open && !!providerIdPreset,
   154→   154→    staleTime: 60 * 1000,
   155→   155→  })
   156→   156→
   157→   157→  const provider = providerQuery.data
   158→   158→  const services = servicesQuery.data ?? []
   159→   159→  const selectedService =
   160→   160→    services.find((s) => s.id === serviceIdPreset) ?? services[0]
   161→   161→
   162→   162→  // Pre-fill quantity from service default unit
   163→   163→  React.useEffect(() => {
   164→   164→    if (open && selectedService) {
   165→   165→      setState((s) => ({ ...s, quantity: 1 }))
   166→   166→    }
   167→   167→  }, [open, selectedService])
   168→   168→
   169→   169→  const set = <K extends keyof BookingFormState>(
   170→   170→    key: K,
   171→   171→    value: BookingFormState[K],
   172→   172→  ) => setState((s) => ({ ...s, [key]: value }))
   173→   173→
   174→   174→  // Step validation
   175→   175→  const step1Valid = !!state.date && !!state.time
   176→   176→  const step2Valid =
   177→   177→    !!state.address.cep &&
   178→   178→    state.address.cep.replace(/\D/g, "").length === 8 &&
   179→   179→    !!state.address.street &&
   180→   180→    !!state.address.number &&
   181→   181→    !!state.address.city &&
   182→   182→    !!state.address.state
   183→   183→  const step3Valid =
   184→   184→    state.paymentMethod === "PIX" ||
   185→   185→    (state.paymentMethod === "CARD" &&
   186→   186→      state.cardName.length >= 3 &&
   187→   187→      state.cardNumber.replace(/\s/g, "").length >= 13 &&
   188→   188→      /^\d{2}\/\d{2}$/.test(state.cardExpiry) &&
   189→   189→      /^\d{3,4}$/.test(state.cardCvv))
   190→   190→
   191→   191→  const handleNext = () => {
   192→   192→    if (step === 1 && !step1Valid) {
   193→   193→      toast.error("Selecione data e horário para continuar.")
   194→   194→      return
   195→   195→    }
   196→   196→    if (step === 2 && !step2Valid) {
   197→   197→      toast.error("Preencha o endereço completo.")
   198→   198→      return
   199→   199→    }
   200→   200→    setStep((s) => Math.min(3, s + 1) as Step)
   201→   201→  }
   202→   202→  const handleBack = () => setStep((s) => Math.max(1, s - 1) as Step)
   203→   203→
   204→   204→  const handleSubmit = async () => {
   205→   205→    if (!user) {
   206→   206→      toast.info("Faça cadastro gratuito para agendar serviços.")
   207→   207→      openAuth("register", "CLIENT")
   208→   208→      return
   209→   209→    }
   210→   210→    if (!provider || !selectedService || !state.date || !state.time) {
   211→   211→      toast.error("Dados incompletos. Revise o agendamento.")
   212→   212→      return
   213→   213→    }
   214→   214→    if (!step3Valid) {
   215→   215→      toast.error("Verifique os dados de pagamento.")
   216→   216→      return
   217→   217→    }
   218→   218→
   219→   219→    // Combine date + "HH:mm" into ISO
   220→   220→    const [h, m] = state.time.split(":").map(Number)
   221→   221→    const scheduledAt = new Date(state.date)
   222→   222→    scheduledAt.setHours(h ?? 0, m ?? 0, 0, 0)
   223→   223→
   224→   224→    const amount =
   225→   225→      (selectedService.basePrice || 0) * (state.quantity || 1)
   226→   226→
   227→   227→    setSubmitting(true)
   228→   228→    try {
   229→   229→      await apiPost("/api/bookings", {
   230→   230→        providerId: provider.id,
   231→   231→        serviceId: selectedService.id,
   232→   232→        scheduledAt: scheduledAt.toISOString(),
   233→   233→        address: [
   234→   234→          state.address.street,
   235→   235→          state.address.number,
   236→   236→          state.address.complement,
   237→   237→          state.address.district,
   238→   238→        ]
   239→   239→          .filter(Boolean)
   240→   240→          .join(", "),
   241→   241→        cep: state.address.cep,
   242→   242→        lat: state.address.lat,
   243→   243→        lng: state.address.lng,
   244→   244→        amount,
   245→   245→        paymentMethod: state.paymentMethod,
   246→   246→        notes: state.notes,
   247→   247→        quantity: state.quantity,
   248→   248→        unit: selectedService.unit,
   249→   249→      })
   250→   250→      toast.success("Agendamento confirmado! Acompanhe em seus agendamentos.")
   251→   251→      close()
   252→   252→      navigate("client.bookings")
   253→   253→    } catch (e) {
   254→   254→      const msg =
   255→   255→        (e as { message?: string })?.message ??
   256→   256→        "Não foi possível concluir o agendamento. Tente novamente."
   257→   257→      toast.error(msg)
   258→   258→    } finally {
   259→   259→      setSubmitting(false)
   260→   260→    }
   261→   261→  }
   262→   262→
   263→   263→  const content = (
   264→   264→    <BookingBody
   265→   265→      step={step}
   266→   266→      state={state}
   267→   267→      set={set}
   268→   268→      provider={provider}
   269→   269→      services={services}
   270→   270→      selectedService={selectedService}
   271→   271→      loading={providerQuery.isLoading || servicesQuery.isLoading}
   272→   272→      onNext={handleNext}
   273→   273→      onBack={handleBack}
   274→   274→      onSubmit={handleSubmit}
   275→   275→      submitting={submitting}
   276→   276→      step1Valid={step1Valid}
   277→   277→      step2Valid={step2Valid}
   278→   278→      step3Valid={step3Valid}
   279→   279→    />
   280→   280→  )
   281→   281→
   282→   282→  if (isMobile) {
   283→   283→    return (
   284→   284→      <Sheet open={open} onOpenChange={(o) => !o && close()}>
   285→   285→        <SheetContent
   286→   286→          side="bottom"
   287→   287→          className="h-[100dvh] max-h-[100dvh] w-full p-0 sm:max-w-full gap-0 flex flex-col"
   288→   288→        >
   289→   289→          <SheetHeader className="px-4 pt-4 pb-2 shrink-0">
   290→   290→            <SheetTitle className="flex items-center gap-2">
   291→   291→              <CalendarDays className="size-5 text-emerald-600" />
   292→   292→              Agendar serviço
   293→   293→            </SheetTitle>
   294→   294→            <SheetDescription>
   295→   295→              Escolha data, detalhes e pagamento.
   296→   296→            </SheetDescription>
   297→   297→          </SheetHeader>
   298→   298→          <div className="flex-1 overflow-hidden">{content}</div>
   299→   299→        </SheetContent>
   300→   300→      </Sheet>
   301→   301→    )
   302→   302→  }
   303→   303→
   304→   304→  return (
   305→   305→    <Dialog open={open} onOpenChange={(o) => !o && close()}>
   306→   306→      <DialogContent className="sm:max-w-lg p-0 gap-0 overflow-hidden">
   307→   307→        <DialogHeader className="px-6 pt-6 pb-3 shrink-0 border-b">
   308→   308→          <DialogTitle className="flex items-center gap-2 text-lg">
   309→   309→            <CalendarDays className="size-5 text-emerald-600" />
   310→   310→            Agendar serviço
   311→   311→          </DialogTitle>
   312→   312→          <DialogDescription>
   313→   313→            Escolha data, detalhes e pagamento.
   314→   314→          </DialogDescription>
   315→   315→        </DialogHeader>
   316→   316→        {content}
   317→   317→      </DialogContent>
   318→   318→    </Dialog>
   319→   319→  )
   320→   320→}

   326→   326→function BookingBody({
   327→   327→  step,
   328→   328→  state,
   329→   329→  set,
   330→   330→  provider,
   331→   331→  services,
   332→   332→  selectedService,
   333→   333→  loading,
   334→   334→  onNext,
   335→   335→  onBack,
   336→   336→  onSubmit,
   337→   337→  submitting,
   338→   338→  step1Valid,
   339→   339→  step2Valid,
   340→   340→  step3Valid,
   341→   341→}: {
   342→   342→  step: Step
   343→   343→  state: BookingFormState
   344→   344→  set: <K extends keyof BookingFormState>(
   345→   345→    key: K,
   346→   346→    value: BookingFormState[K],
   347→   347→  ) => void
   348→   348→  provider?: ProviderDetail
   349→   349→  services: ProviderService[]
   350→   350→  selectedService?: ProviderService
   351→   351→  loading: boolean
   352→   352→  onNext: () => void
   353→   353→  onBack: () => void
   354→   354→  onSubmit: () => void
   355→   355→  submitting: boolean
   356→   356→  step1Valid: boolean
   357→   357→  step2Valid: boolean
   358→   358→  step3Valid: boolean
   359→   359→}) {
   360→   360→  const progressPct = (step / 3) * 100
   361→   361→
   362→   362→  return (
   363→   363→    <div className="flex h-full flex-col">
   364→   364→      {/* Stepper */}
   365→   365→      <div className="shrink-0 border-b px-4 sm:px-6 py-3">
   366→   366→        <div className="flex items-center justify-between">
   367→   367→          {STEPS.map((s, i) => {
   368→   368→            const active = step === s.id
   369→   369→            const done = step > s.id
   370→   370→            const Icon = s.icon
   371→   371→            return (
   372→   372→              <React.Fragment key={s.id}>
   373→   373→                <div
   374→   374→                  className={cn(
   375→   375→                    "flex items-center gap-2 text-xs sm:text-sm transition-colors",
   376→   376→                    active
   377→   377→                      ? "text-emerald-700 dark:text-emerald-400 font-medium"
   378→   378→                      : done
   379→   379→                        ? "text-emerald-600"
   380→   380→                        : "text-muted-foreground",
   381→   381→                  )}
   382→   382→                >
   383→   383→                  <span
   384→   384→                    className={cn(
   385→   385→                      "inline-flex size-7 items-center justify-center rounded-full border-2 transition-colors",
   386→   386→                      active &&
   387→   387→                        "border-emerald-600 bg-emerald-600 text-white",
   388→   388→                      done && "border-emerald-600 bg-emerald-600 text-white",
   389→   389→                      !active && !done && "border-muted-foreground/30",
   390→   390→                    )}
   391→   391→                  >
   392→   392→                    {done ? (
   393→   393→                      <Check className="size-3.5" />
   394→   394→                    ) : (
   395→   395→                      <Icon className="size-3.5" />
   396→   396→                    )}
   397→   397→                  </span>
   398→   398→                  <span className="hidden sm:inline">{s.label}</span>
   399→   399→                </div>
   400→   400→                {i < STEPS.length - 1 && (
   401→   401→                  <div className="flex-1 h-0.5 bg-muted-foreground/20 mx-2 relative">
   402→   402→                    <div
   403→   403→                      className="absolute inset-0 bg-emerald-600 transition-transform origin-left"
   404→   404→                      style={{
   405→   405→                        transform:
   406→   406→                          step > s.id ? "scaleX(1)" : "scaleX(0)",
   407→   407→                      }}
   408→   408→                    />
   409→   409→                  </div>
   410→   410→                )}
   411→   411→              </React.Fragment>
   412→   412→            )
   413→   413→          })}
   414→   414→        </div>
   415→   415→        <Progress
   416→   416→          value={progressPct}
   417→   417→          className="mt-2 h-1 bg-emerald-100 dark:bg-emerald-950/40"
   418→   418→        />
   419→   419→      </div>
   420→   420→
   421→   421→      {/* Step content */}
   422→   422→      <ScrollArea className="flex-1">
   423→   423→        <div className="px-4 sm:px-6 py-4">
   424→   424→          {loading ? (
   425→   425→            <div className="flex items-center justify-center py-12">
   426→   426→              <Loader2 className="size-6 animate-spin text-emerald-600" />
   427→   427→            </div>
   428→   428→          ) : (
   429→   429→            <AnimatePresence mode="wait">
   430→   430→              <motion.div
   431→   431→                key={step}
   432→   432→                initial={{ opacity: 0, x: 10 }}
   433→   433→                animate={{ opacity: 1, x: 0 }}
   434→   434→                exit={{ opacity: 0, x: -10 }}
   435→   435→                transition={{ duration: 0.18 }}
   436→   436→              >
   437→   437→                {step === 1 && (
   438→   438→                  <Step1Schedule
   439→   439→                    state={state}
   440→   440→                    set={set}
   441→   441→                    availability={provider?.availability ?? []}
   442→   442→                    selectedService={selectedService}
   443→   443→                  />
   444→   444→                )}
   445→   445→                {step === 2 && (
   446→   446→                  <Step2Details
   447→   447→                    state={state}
   448→   448→                    set={set}
   449→   449→                    provider={provider}
   450→   450→                    selectedService={selectedService}
   451→   451→                  />
   452→   452→                )}
   453→   453→                {step === 3 && (
   454→   454→                  <Step3Payment
   455→   455→                    state={state}
   456→   456→                    set={set}
   457→   457→                    selectedService={selectedService}
   458→   458→                  />
   459→   459→                )}
   460→   460→              </motion.div>
   461→   461→            </AnimatePresence>
   462→   462→          )}
   463→   463→        </div>
   464→   464→      </ScrollArea>
   465→   465→
   466→   466→      {/* Footer nav */}
   467→   467→      <div className="shrink-0 border-t bg-background/95 backdrop-blur px-4 sm:px-6 py-3">
   468→   468→        <div className="flex items-center justify-between gap-3">
   469→   469→          <Button
   470→   470→            type="button"
   471→   471→            variant="ghost"
   472→   472→            onClick={onBack}
   473→   473→            disabled={step === 1 || submitting}
   474→   474→            className="text-muted-foreground hover:text-foreground"
   475→   475→          >
   476→   476→            Voltar
   477→   477→          </Button>
   478→   478→          {step < 3 ? (
   479→   479→            <Button
   480→   480→              type="button"
   481→   481→              onClick={onNext}
   482→   482→              className="h-11 bg-emerald-600 hover:bg-emerald-700"
   483→   483→              disabled={
   484→   484→                (step === 1 && !step1Valid) ||
   485→   485→                (step === 2 && !step2Valid)
   486→   486→              }
   487→   487→            >
   488→   488→              Continuar
   489→   489→            </Button>
   490→   490→          ) : (
   491→   491→            <Button
   492→   492→              type="button"
   493→   493→              onClick={onSubmit}
   494→   494→              disabled={submitting || !step3Valid}
   495→   495→              className="h-11 bg-emerald-600 hover:bg-emerald-700"
   496→   496→            >
   497→   497→              {submitting && <Loader2 className="size-4 animate-spin" />}
   498→   498→              <ShieldCheck className="size-4" />
   499→   499→              Confirmar e agendar
   500→   500→            </Button>
   501→   501→          )}
   502→   502→        </div>
   503→   503→      </div>
   504→   504→    </div>
   505→   505→  )
   506→   506→}

   512→   512→function Step1Schedule({
   513→   513→  state,
   514→   514→  set,
   515→   515→  availability,
   516→   516→  selectedService,
   517→   517→}: {
   518→   518→  state: BookingFormState
   519→   519→  set: <K extends keyof BookingFormState>(
   520→   520→    key: K,
   521→   521→    value: BookingFormState[K],
   522→   522→  ) => void
   523→   523→  availability: ProviderDetail["availability"]
