   121→
   122→  // Carousel API

   124→  const [current, setCurrent] = React.useState(0)

   176→      <div aria-hidden className="pointer-events-none absolute inset-0 overflow-hidden">

   179→          <Quote className="size-32" />

   182→        <div
   183→          className="absolute inset-0 opacity-[0.03] dark:opacity-[0.05]"
   184→          style={{

   186→              "radial-gradient(circle, currentColor 1px, transparent 1px)",
   187→            backgroundSize: "32px 32px",
   188→          }}
   189→        />
   190→        {/* Gradient orb decoration */}
   191→        <div className="absolute -top-24 -right-24 size-96 rounded-full bg-gradient-to-br from-emerald-100/40 to-teal-50/20 blur-3xl dark:from-emerald-950/30 dark:to-teal-950/20" />
   192→        <div className="absolute -bottom-16 -left-16 size-72 rounded-full bg-gradient-to-tr from-amber-100/30 to-emerald-50/20 blur-3xl dark:from-amber-950/20 dark:to-emerald-950/10" />
   193→      </div>
   194→
   195→      <div
   196→        ref={ref}
   197→        onMouseEnter={handleMouseEnter}
   198→        onMouseLeave={handleMouseLeave}
   199→        className="relative"

   201→        {/* Header */}

   214→          </h2>

   218→              : "Avaliações de clientes após a conclusão do serviço."}
   219→          </p>
   220→        </motion.header>
   221→
   222→        {/* Summary stats + star distribution */}
   223→        {total > 0 && (
   224→          <motion.div
   225→            initial={{ opacity: 0, y: 12 }}
   226→            animate={visible ? { opacity: 1, y: 0 } : {}}
   227→            transition={{ duration: 0.4, delay: 0.1 }}
   228→            className="mb-8 mx-auto max-w-xl"
   229→          >
   230→            <div className="flex flex-col sm:flex-row items-center gap-6 sm:gap-8">
   231→              {/* Average rating hero — large & prominent */}
   232→              <div className="flex flex-col items-center sm:items-end gap-1 shrink-0">
   233→                <div className="flex items-baseline gap-1.5">
   234→                  <span className="text-5xl font-extrabold tabular-nums tracking-tighter bg-gradient-to-br from-emerald-600 to-teal-600 bg-clip-text text-transparent dark:from-emerald-400 dark:to-teal-400">
   235→                    {avgRating.toFixed(1)}

   266→                  </Tooltip>
   267→                </TooltipProvider>

   278→                      onClick={() =>

   282→                      }
   283→                      className={cn(
   284→                        "group flex items-center gap-2 w-full rounded px-1 py-0.5 transition-colors text-left",
   285→                        ratingFilter === String(starVal)
   286→                          ? "bg-emerald-50 dark:bg-emerald-950/30"
   287→                          : "hover:bg-muted/60",
   288→                      )}
   289→                      aria-label={`Filtrar por ${starVal} estrelas: ${count} avaliações`}
   290→                    >
   291→                      <span className="text-xs font-medium tabular-nums w-3 text-right shrink-0">
   292→                        {starVal}

   294→                      <Star className="size-3 fill-amber-400 text-amber-400 shrink-0" />

   297→                          initial={{ width: 0 }}
   298→                          animate={visible ? { width: `${pct}%` } : { width: 0 }}
   299→                          transition={{ duration: 0.6, delay: 0.2 + (5 - starVal) * 0.06 }}
   300→                          className={cn(
   301→                            "h-full rounded-full",
   302→                            ratingFilter === String(starVal)
   303→                              ? "bg-emerald-500"
   304→                              : "bg-gradient-to-r from-amber-400 to-amber-500",
   305→                          )}
   306→                        />
   307→                      </div>
   308→                      <span className="text-[10px] text-muted-foreground tabular-nums w-6 text-right shrink-0">
   309→                        {count}
   310→                      </span>
   311→                    </button>
   312→                  )
   313→                })}
   314→              </div>
   315→            </div>
   316→          </motion.div>
   317→        )}
   318→
   319→        {/* Rating filter chips */}
   320→        {reviews.length > 0 && (
   321→          <motion.div
   322→            initial={{ opacity: 0 }}
   323→            animate={visible ? { opacity: 1 } : {}}
   324→            transition={{ duration: 0.3, delay: 0.15 }}
   325→            className="mb-6 flex items-center justify-center gap-2"
   326→          >
   327→            {(
   328→              [
   329→                { key: "all", label: "Todas" },
   330→                { key: "5", label: "5 estrelas" },
   331→                { key: "4", label: "4 estrelas" },
   332→              ] as const
   333→            ).map((f) => (
   334→              <button
   335→                key={f.key}
   336→                onClick={() => setRatingFilter(f.key)}
   337→                className={cn(
   338→                  "rounded-full px-3 py-1 text-xs font-medium transition-all",
   339→                  ratingFilter === f.key
   340→                    ? "bg-emerald-600 text-white shadow-sm"
   341→                    : "bg-muted text-muted-foreground hover:bg-muted/80 hover:text-foreground",
   342→                )}
   343→                aria-pressed={ratingFilter === f.key}
   344→              >
   345→                {f.label}
   346→              </button>
   347→            ))}
   348→          </motion.div>
   349→        )}
   350→
   351→        {/* Loading state */}
   352→        {isLoading ? (
   353→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   354→            {Array.from({ length: 3 }).map((_, i) => (
   355→              <ReviewSkeleton key={i} />
   356→            ))}
   357→          </div>
   358→        ) : isError ? (
   359→          <ErrorState onRetry={() => refetch()} />
   360→        ) : filteredReviews.length > 0 ? (
   361→          /* Carousel */
   362→          <div className="relative">
   363→            <Carousel
   364→              setApi={setApi}
   365→              opts={{
   366→                align: "start",
   367→                loop: true,
   368→              }}
   369→              plugins={[autoplayRef.current]}
   370→              className="w-full"
   371→            >
   372→              <CarouselContent className="-ml-4">
   373→                {filteredReviews.map((review, idx) => (
   374→                  <CarouselItem
   375→                    key={review.id}
   376→                    className="pl-4 basis-full sm:basis-1/2 lg:basis-1/3"
   377→                  >
   378→                    <motion.div
   379→                      initial={{ opacity: 0, y: 24 }}
   380→                      animate={visible ? { opacity: 1, y: 0 } : {}}
   381→                      transition={{ duration: 0.45, delay: idx * 0.06 }}
   382→                      className="h-full"
   383→                    >
   384→                      <ReviewCard review={review} />
   385→                    </motion.div>
   386→                  </CarouselItem>
   387→                ))}
   388→              </CarouselContent>
   389→
   390→              {/* Prev / Next arrows — circular with gradient */}
   391→              <div className="pointer-events-none absolute inset-y-0 left-0 hidden items-center lg:flex">
   392→                <button
   393→                  className="pointer-events-auto -ml-3 size-11 rounded-full flex items-center justify-center bg-gradient-to-br from-emerald-500 to-teal-600 text-white shadow-lg shadow-emerald-500/25 transition-all duration-200 hover:scale-110 hover:shadow-xl hover:shadow-emerald-500/30 active:scale-95"
   394→                  onClick={() => api?.scrollPrev()}
   395→                  aria-label="Avaliação anterior"
   396→                >
   397→                  <ChevronLeft className="size-5" />
   398→                </button>
   399→              </div>
   400→              <div className="pointer-events-none absolute inset-y-0 right-0 hidden items-center lg:flex">
   401→                <button
   402→                  className="pointer-events-auto -mr-3 size-11 rounded-full flex items-center justify-center bg-gradient-to-br from-emerald-500 to-teal-600 text-white shadow-lg shadow-emerald-500/25 transition-all duration-200 hover:scale-110 hover:shadow-xl hover:shadow-emerald-500/30 active:scale-95"
   403→                  onClick={() => api?.scrollNext()}
   404→                  aria-label="Próxima avaliação"
   405→                >
   406→                  <ChevronRight className="size-5" />
   407→                </button>
   408→              </div>
   409→            </Carousel>
   410→
   411→            {/* Dot indicators + slide counter */}
   412→            <div className="mt-6 flex flex-col items-center gap-2">
   413→              {count > 1 && (
   414→                <div
   415→                  className="flex items-center justify-center gap-2"
   416→                  role="tablist"
   417→                  aria-label="Navegação do carrossel"
   418→                >
   419→                  {Array.from({ length: count }).map((_, i) => (
   420→                    <button
   421→                      key={i}
   422→                      onClick={() => api?.scrollTo(i)}
   423→                      className={cn(
   424→                        "rounded-full transition-all duration-300",
   425→                        i === current
   426→                          ? "bg-emerald-500 h-3 w-6 shadow-sm shadow-emerald-500/30"
   427→                          : "size-2 bg-muted-foreground/25 hover:bg-muted-foreground/50",
   428→                      )}
   429→                      role="tab"
   430→                      aria-selected={i === current}
   431→                      aria-label={`Ir para avaliação ${i + 1}`}
   432→                    />
   433→                  ))}
   434→                </div>
   435→              )}
   436→
   437→              {/* Slide X of Y counter for accessibility */}
   438→              {totalSlides > 0 && (
   439→                <p
   440→                  className="text-[11px] font-medium text-muted-foreground tabular-nums"
   441→                  aria-live="polite"
   442→                  aria-atomic
   443→                >
   444→                  {visibleSlideNumber} de {totalSlides}
   445→                </p>
   446→              )}
   447→            </div>
   448→
   449→            {/* Pause indicator */}
   450→            <AnimatePresence>
   451→              {isPaused && count > 1 && (
   452→                <motion.div
   453→                  initial={{ opacity: 0, y: 4 }}
   454→                  animate={{ opacity: 1, y: 0 }}
   455→                  exit={{ opacity: 0, y: 4 }}
   456→                  className="mt-2 flex items-center justify-center gap-1.5 text-[11px] text-muted-foreground"
   457→                >
   458→                  <span className="size-1.5 rounded-full bg-amber-400 animate-pulse" />
   459→                  Pausado — passe o mouse para pausar
   460→                </motion.div>
   461→              )}
   462→            </AnimatePresence>
   463→          </div>
   464→        ) : (
   465→          <EmptyTestimonials />
   466→        )}
   467→      </div>
   468→    </section>
   469→  )
   470→}

   478→
   479→  return (
   480→    <div className="group relative flex h-full flex-col gap-3 rounded-xl border bg-card shadow-sm transition-all duration-300 hover:-translate-y-1 hover:shadow-lg hover:border-emerald-200 dark:hover:border-emerald-800/50 overflow-hidden">
   481→      {/* Left accent stripe — gradient border */}
   482→      <div
   483→        aria-hidden
   484→        className="absolute inset-y-0 left-0 w-1 bg-gradient-to-b from-emerald-400 via-teal-500 to-emerald-600 transition-all duration-300 group-hover:w-1.5 group-hover:from-emerald-300 group-hover:via-teal-400 group-hover:to-emerald-500"
   485→      />
   486→
   487→      <div className="flex flex-col gap-3 p-5 pl-6 sm:p-5 sm:pl-7">

   489→        <div className="flex items-center justify-between gap-2">
   490→          {/* Star rating with glow */}
   491→          <div className="flex items-center gap-0.5">
   492→            {stars.map((filled, i) => (
   493→              <Star
   494→                key={i}
   495→                className={cn(
   496→                  "size-4 transition-all duration-200",
   497→                  filled
   498→                    ? "fill-amber-400 text-amber-400 drop-shadow-[0_0_6px_rgba(251,191,36,0.35)] group-hover:drop-shadow-[0_0_8px_rgba(251,191,36,0.5)]"
   499→                    : "fill-muted text-muted-foreground/30",
   500→                )}
   501→              />
   502→            ))}
   503→          </div>
   504→
   505→          {/* Verified review badge */}
   506→          <span className="inline-flex items-center gap-1 rounded-full bg-emerald-50 px-2 py-0.5 text-[10px] font-semibold text-emerald-700 ring-1 ring-emerald-200 dark:bg-emerald-950/40 dark:text-emerald-300 dark:ring-emerald-800/50">
   507→            <ShieldCheck className="size-3" />
   508→            <span className="hidden sm:inline">Verificada</span>
   509→          </span>
   510→        </div>
   511→
   512→        {/* Quote with large opening mark */}
   513→        <div className="relative">
   514→          <div
   515→            aria-hidden
   516→            className="absolute -top-2 -left-2 flex size-10 items-center justify-center rounded-lg bg-gradient-to-br from-emerald-100 to-teal-100 dark:from-emerald-900/40 dark:to-teal-900/40"

   521→            <p className="line-clamp-3 pl-10 text-sm italic leading-relaxed text-foreground/80">
   522→              {review.comment}
   523→            </p>
   524→          ) : (
   525→            <p className="pl-10 text-sm italic text-muted-foreground">
   526→              Sem comentário escrito.
   527→            </p>
   528→          )}
   529→        </div>
   530→
   531→        {/* Service type badge */}
