     3→/**
     4→ * AdminTaxonomy — the flagship 3-level category tree manager.
     5→ *
     6→ * Visualization:
     7→ *   Level 0 (Categoria pai)    — emerald
     8→ *   Level 1 (Categoria filha)  — teal
     9→ *   Level 2 (Subcategoria)     — lime
    10→ *
    11→ * Features:
    12→ *   - Expandable/collapsible tree (Accordion-style with custom tree UX)
    13→ *   - Per-node: edit, toggle active, delete (with 409 guard)
    14→ *   - "Nova categoria" dialog: name, slug (auto-generated), parent select,
    15→ *     icon (lucide name), order
    16→ *   - Inline edit dialog
    17→ *   - Service count per category
    18→ *   - Visual hierarchy with indentation + connecting lines + level badges
    19→ *
    20→ * APIs (relative):
    21→ *   GET    /api/categories
    22→ *   POST   /api/categories            { name, slug, parentId?, level, icon?, order? }
    23→ *   PATCH  /api/categories/[id]       { name?, slug?, parentId?, level?, icon?, order?, active? }
    24→ *   DELETE /api/categories/[id]       (409 if has children/services)
    25→ */

    85→/**
    86→ * The /api/categories endpoint returns the full Prisma row (includes `active`,
    87→ * `order`, `parentId`, `level`, `icon`, `createdAt`, `updatedAt`) — but the
    88→ * client-side `Category` type from `@/lib/api` is narrower. Extend it locally
    89→ * so we can read the extra fields.
    90→ */
    91→type CategoryRow = Category & {
    92→  active: boolean
    93→  order: number
    94→}
    95→
    96→type CategoryNode = Omit<CategoryRow, "children"> & {
    97→  children?: CategoryNode[]
    98→  serviceCount?: number
    99→  childrenCount?: number
   100→}
   101→
   102→type LevelMeta = {
   103→  label: string
   104→  badgeClass: string
   105→  rowAccent: string
   106→  dot: string
   107→}
   108→
   109→const LEVEL_META: Record<number, LevelMeta> = {
   110→  0: {

   114→    rowAccent: "border-l-2 border-emerald-500/60",

   116→  },

   121→    rowAccent: "border-l-2 border-teal-500/60",
   122→    dot: "bg-teal-500",
   123→  },
   124→  2: {
   125→    label: "Subcategoria",
   126→    badgeClass:
   127→      "bg-slate-200 text-slate-700 dark:bg-slate-700/40 dark:text-slate-200",

   133→// ---------------------------------------------------------------------------
   134→// Main component
   135→// ---------------------------------------------------------------------------
   136→export function AdminTaxonomy() {
   137→  const queryClient = useQueryClient()
   138→  const [createOpen, setCreateOpen] = React.useState(false)
   139→  const [editTarget, setEditTarget] = React.useState<CategoryNode | null>(null)
   140→  const [deleteTarget, setDeleteTarget] = React.useState<CategoryNode | null>(null)
   141→  const [expanded, setExpanded] = React.useState<Set<string>>(new Set())
   142→
   143→  const { data: flat, isLoading, isError } = useQuery({
   144→    queryKey: ["categories", "all"],
   145→    queryFn: () => apiGet<CategoryRow[]>("/api/categories"),
   146→    staleTime: 30_000,
   147→  })
   148→
   149→  // Build the tree + counts (from a flat list)
   150→  const tree = React.useMemo<CategoryNode[]>(() => {
   151→    if (!flat) return []
   152→    const byParent = new Map<string | null, CategoryNode[]>()
   153→    for (const c of flat) {
   154→      const key = c.parentId ?? null
   155→      const arr = byParent.get(key) ?? []
   156→      arr.push({ ...c, children: [] })
   157→      byParent.set(key, arr)
   158→    }
   159→    const roots = byParent.get(null) ?? []
   160→    const build = (nodes: CategoryNode[]): CategoryNode[] =>
   161→      nodes
   162→        .sort((a, b) => (a.order ?? 0) - (b.order ?? 0) || a.name.localeCompare(b.name))
   163→        .map((n) => {
   164→          const children = byParent.get(n.id) ?? []
   165→          n.children = build(children)
   166→          return n
   167→        })
   168→    return build(roots)
   169→  }, [flat])
   170→
   171→  const serviceCounts = React.useMemo(() => {
   172→    // Service count is fetched per-category lazily via a separate query —
   173→    // but for MVP we only have the flat /api/categories (no counts). We'll
   174→    // render counts if present on the node (admin extensions may add later).
   175→    return new Map<string, number>()
   176→  }, [])
   177→
   178→  // Auto-expand all level-0 nodes on first load
   179→  React.useEffect(() => {
   180→    if (tree.length && expanded.size === 0) {
   181→      setExpanded(new Set(tree.map((n) => n.id)))
   182→    }
   183→  }, [tree, expanded.size])
   184→
   185→  const invalidate = () =>
   186→    queryClient.invalidateQueries({ queryKey: ["categories"] })
   187→
   188→  const createMutation = useMutation({
   189→    mutationFn: (payload: CategoryPayload) =>
   190→      apiPost<{ category: CategoryRow }>("/api/categories", payload),
   191→    onSuccess: () => {
   192→      toast.success("Categoria criada com sucesso.")
   193→      invalidate()
   194→      setCreateOpen(false)
   195→    },
   196→    onError: (e: unknown) => {
   197→      toast.error(errMsg(e, "Não foi possível criar a categoria."))
   198→    },
   199→  })
   200→
   201→  const patchMutation = useMutation({
   202→    mutationFn: ({ id, patch }: { id: string; patch: Partial<CategoryPayload> }) =>
   203→      apiPatch<{ category: CategoryRow }>(`/api/categories/${id}`, patch),
   204→    onSuccess: () => {
   205→      toast.success("Categoria atualizada.")
   206→      invalidate()
   207→      setEditTarget(null)
   208→    },
   209→    onError: (e: unknown) => {
   210→      toast.error(errMsg(e, "Não foi possível atualizar a categoria."))
   211→    },
   212→  })
   213→
   214→  const toggleActiveMutation = useMutation({
   215→    mutationFn: ({ id, active }: { id: string; active: boolean }) =>
   216→      apiPatch<{ category: CategoryRow }>(`/api/categories/${id}`, { active }),
   217→    onSuccess: () => {
   218→      invalidate()
   219→    },
   220→    onError: (e: unknown) => {
   221→      toast.error(errMsg(e, "Falha ao alternar o estado ativo."))
   222→    },
   223→  })
   224→
   225→  const deleteMutation = useMutation({
   226→    mutationFn: (id: string) => apiDelete(`/api/categories/${id}`),
   227→    onSuccess: () => {
   228→      toast.success("Categoria excluída.")
   229→      invalidate()
   230→      setDeleteTarget(null)
   231→    },
   232→    onError: (e: unknown) => {
   233→      const msg = errMsg(e, "Não foi possível excluir.")
   234→      // Special 409 message
   235→      toast.error(msg, {
   236→        description:
   237→          "Remova os vínculos (filhos ou serviços) antes de tentar novamente.",
   238→      })
   239→    },
   240→  })
   241→
   242→  const toggleExpand = (id: string) =>
   243→    setExpanded((prev) => {
   244→      const next = new Set(prev)
   245→      if (next.has(id)) next.delete(id)
   246→      else next.add(id)
   247→      return next
   248→    })
   249→
   250→  if (isError) {
   251→    return (
   252→      <Card>
   253→        <CardContent className="py-12 text-center text-sm text-muted-foreground">
   254→          Não foi possível carregar a árvore de categorias.
   255→        </CardContent>
   256→      </Card>
   257→    )
   258→  }
   259→
   260→  const totalNodes = flat?.length ?? 0
   261→  const expandAll = () => {
   262→    if (!flat) return
   263→    setExpanded(new Set(flat.map((c) => c.id)))
   264→  }
   265→  const collapseAll = () => setExpanded(new Set())
   266→
   267→  return (
   268→    <div className="flex flex-col gap-4">
   269→      {/* Toolbar */}
   270→      <div className="flex flex-wrap items-center justify-between gap-3 rounded-xl border bg-card p-3 shadow-sm">
   271→        <div className="flex flex-wrap items-center gap-2">
   272→          <span className="inline-flex items-center gap-1.5 rounded-full border bg-muted/40 px-2.5 py-1 text-xs font-medium text-muted-foreground">
   273→            <FolderTree className="size-3.5 text-primary" />
   274→            {totalNodes} categoria(s)
   275→          </span>
   276→          <span className="hidden items-center gap-1.5 text-xs text-muted-foreground sm:flex">
   277→            Níveis:
   278→            <Badge
   279→              variant="secondary"
   280→              className="gap-1 bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-200"
   281→            >
   282→              <span className="size-1.5 rounded-full bg-emerald-500" />
   283→              Pai
   284→            </Badge>
   285→            <Badge
   286→              variant="secondary"
   287→              className="gap-1 bg-teal-100 text-teal-800 dark:bg-teal-900/40 dark:text-teal-200"
   288→            >
   289→              <span className="size-1.5 rounded-full bg-teal-500" />
   290→              Filha
   291→            </Badge>
   292→            <Badge
   293→              variant="secondary"
   294→              className="gap-1 bg-slate-200 text-slate-700 dark:bg-slate-700/40 dark:text-slate-200"
   295→            >
   296→              <span className="size-1.5 rounded-full bg-slate-400" />
   297→              Sub
   298→            </Badge>
   299→          </span>
   300→        </div>
   301→        <div className="flex flex-wrap items-center gap-2">
   302→          <Button
   303→            variant="outline"
   304→            size="sm"
   305→            onClick={expandAll}
   306→            disabled={!flat || flat.length === 0}
   307→            className="h-8 gap-1.5"
   308→          >
   309→            <ChevronsUpDown className="size-3.5" />
   310→            Expandir tudo
   311→          </Button>
   312→          <Button
   313→            variant="outline"
   314→            size="sm"
   315→            onClick={collapseAll}
   316→            disabled={expanded.size === 0}
   317→            className="h-8 gap-1.5"
   318→          >
   319→            <ChevronsDownUp className="size-3.5" />
   320→            Recolher tudo
   321→          </Button>
   322→          <Button
   323→            size="sm"
   324→            onClick={() => setCreateOpen(true)}
   325→            className="h-8 gap-1.5"
   326→          >
   327→            <Plus className="size-4" />
   328→            Nova categoria
   329→          </Button>
   330→        </div>
   331→      </div>
   332→
   333→      {/* Tree */}
   334→      <Card className="overflow-hidden">
   335→        <CardContent className="p-0">
   336→          {isLoading ? (
   337→            <div className="flex flex-col gap-2 p-4">
   338→              {Array.from({ length: 4 }).map((_, i) => (
   339→                <Skeleton key={i} className="h-14 w-full" />
   340→              ))}
   341→            </div>
   342→          ) : tree.length === 0 ? (
   343→            <div className="flex flex-col items-center gap-3 px-4 py-16 text-center">
   344→              <div className="flex size-16 items-center justify-center rounded-full bg-muted text-muted-foreground">
   345→                <FolderTree className="size-7" />
   346→              </div>
   347→              <div>
   348→                <p className="text-lg font-semibold">Nenhuma categoria cadastrada</p>
   349→                <p className="mt-0.5 text-sm text-muted-foreground">
   350→                  Crie a primeira categoria pai para iniciar a taxonomia.
   351→                </p>
   352→              </div>
   353→              <Button size="sm" onClick={() => setCreateOpen(true)} className="gap-1.5">
   354→                <Plus className="size-4" />
   355→                Criar primeira categoria
   356→              </Button>
   357→            </div>
   358→          ) : (
   359→            <ul className="flex flex-col" role="tree">
   360→              {tree.map((node) => (
   361→                <TreeNode
   362→                  key={node.id}
   363→                  node={node}
   364→                  level={0}
   365→                  expanded={expanded}
   366→                  onToggleExpand={toggleExpand}
   367→                  onEdit={(n) => setEditTarget(n)}
   368→                  onDelete={(n) => setDeleteTarget(n)}
   369→                  onToggleActive={(n, active) =>
   370→                    toggleActiveMutation.mutate({ id: n.id, active })
   371→                  }
   372→                  serviceCounts={serviceCounts}
   373→                />
   374→              ))}
   375→            </ul>
   376→          )}
   377→        </CardContent>
   378→      </Card>
   379→
   380→      {/* Create dialog */}
   381→      <CategoryDialog
   382→        open={createOpen}
   383→        onOpenChange={setCreateOpen}
   384→        title="Nova categoria"
   385→        description="Defina o nome, slug e o pai. O nível é determinado automaticamente pelo pai escolhido."
   386→        allCategories={(flat as CategoryRow[] | undefined) ?? []}
   387→        submitting={createMutation.isPending}
   388→        onSubmit={(p) => createMutation.mutate(p)}
   389→      />
   390→
   391→      {/* Edit dialog */}
   392→      <CategoryDialog
   393→        open={!!editTarget}
   394→        onOpenChange={(open) => !open && setEditTarget(null)}
   395→        title="Editar categoria"
   396→        description="Altere os campos abaixo. Mudar o pai pode alterar o nível da categoria."
   397→        allCategories={(flat as CategoryRow[] | undefined) ?? []}
   398→        initial={editTarget ?? undefined}
   399→        submitting={patchMutation.isPending}
   400→        onSubmit={(p) => {
   401→          if (!editTarget) return
   402→          patchMutation.mutate({ id: editTarget.id, patch: p })
   403→        }}
   404→      />
   405→
   406→      {/* Delete confirmation */}
   407→      <AlertDialog
   408→        open={!!deleteTarget}
   409→        onOpenChange={(open) => !open && setDeleteTarget(null)}
   410→      >
   411→        <AlertDialogContent>
   412→          <AlertDialogHeader>
   413→            <AlertDialogTitle>Excluir categoria?</AlertDialogTitle>
   414→            <AlertDialogDescription>
   415→              Você está prestes a excluir{" "}
   416→              <strong className="text-foreground">
   417→                {deleteTarget?.name}
   418→              </strong>
   419→              . Esta ação não pode ser desfeita. Categorias com filhos ou
   420→              serviços vinculados não podem ser excluídas.
   421→            </AlertDialogDescription>
   422→          </AlertDialogHeader>
   423→          <AlertDialogFooter>
   424→            <AlertDialogCancel disabled={deleteMutation.isPending}>
   425→              Cancelar
   426→            </AlertDialogCancel>
   427→            <AlertDialogAction
   428→              onClick={() => deleteTarget && deleteMutation.mutate(deleteTarget.id)}
   429→              disabled={deleteMutation.isPending}
   430→              className="gap-1.5 bg-red-600 hover:bg-red-700 focus-visible:ring-red-600"
   431→            >
   432→              {deleteMutation.isPending ? (
   433→                <Loader2 className="size-4 animate-spin" />
   434→              ) : (
   435→                <Trash2 className="size-4" />
   436→              )}
   437→              Excluir
   438→            </AlertDialogAction>
   439→          </AlertDialogFooter>
   440→        </AlertDialogContent>
   441→      </AlertDialog>
   442→    </div>
   443→  )
   444→}

   446→// ---------------------------------------------------------------------------
   447→// TreeNode — recursive
   448→// ---------------------------------------------------------------------------
   449→function TreeNode({
   450→  node,
   451→  level,
   452→  expanded,
   453→  onToggleExpand,
   454→  onEdit,
   455→  onDelete,
   456→  onToggleActive,
   457→  serviceCounts,
   458→}: {
   459→  node: CategoryNode
   460→  level: number
   461→  expanded: Set<string>
   462→  onToggleExpand: (id: string) => void
   463→  onEdit: (n: CategoryNode) => void
   464→  onDelete: (n: CategoryNode) => void
   465→  onToggleActive: (n: CategoryNode, active: boolean) => void
   466→  serviceCounts: Map<string, number>
   467→}) {
   468→  const meta = LEVEL_META[level] ?? LEVEL_META[0]
   469→  const hasChildren = !!node.children && node.children.length > 0
   470→  const isOpen = expanded.has(node.id)
   471→  const services = serviceCounts.get(node.id) ?? 0
   472→
   473→  return (
   474→    <li
   475→      role="treeitem"
   476→      aria-expanded={hasChildren ? isOpen : undefined}
   477→      aria-selected={false}
   478→    >
   479→      <div
   480→        className={cn(
   481→          "group flex items-center gap-2 border-b px-3 py-2.5 transition-colors hover:bg-muted/30 last:border-0",
   482→          meta.rowAccent,
   483→          !node.active && "opacity-60",
   484→        )}
   485→        style={{ paddingLeft: `${12 + level * 22}px` }}
   486→      >
   487→        {/* Connecting line (visual only, for nested levels) */}
   488→        {level > 0 ? (
   489→          <span
   490→            aria-hidden
   491→            className="absolute left-0 top-0 h-full border-l border-dashed border-border"
   492→            style={{ marginLeft: `${12 + (level - 1) * 22 + 8}px` }}
   493→          />
   494→        ) : null}
   495→
   496→        {/* Expand toggle */}
   497→        <button
   498→          type="button"
   499→          onClick={() => hasChildren && onToggleExpand(node.id)}
   500→          aria-label={isOpen ? "Recolher" : "Expandir"}
   501→          disabled={!hasChildren}
   502→          className={cn(
   503→            "flex size-6 shrink-0 items-center justify-center rounded-md text-muted-foreground transition-colors",
   504→            hasChildren
   505→              ? "hover:bg-accent hover:text-foreground"
   506→              : "cursor-default opacity-30",
   507→          )}
   508→        >
   509→          {hasChildren ? (
   510→            <ChevronRight
   511→              className={cn(
   512→                "size-4 transition-transform",
   513→                isOpen && "rotate-90",
   514→              )}
   515→            />
   516→          ) : (
   517→            <span className="size-1.5 rounded-full bg-current opacity-40" />
   518→          )}
   519→        </button>
   520→
   521→        {/* Drag handle (visual only) */}
   522→        <GripVertical
   523→          className="size-4 shrink-0 cursor-grab text-muted-foreground/40 transition-colors group-hover:text-muted-foreground/70"
   524→          aria-hidden
   525→        />
   526→
   527→        {/* Name + level badge */}
   528→        <div className="min-w-0 flex-1">
   529→          <div className="flex flex-wrap items-center gap-2">
   530→            <span className="truncate text-sm font-medium">{node.name}</span>
   531→            <Badge
   532→              variant="secondary"
   533→              className={cn(
   534→                "h-5 gap-1 px-2 text-[10px] font-medium uppercase tracking-wide",
   535→                meta.badgeClass,
   536→              )}
   537→            >
   538→              <span className={cn("size-1.5 rounded-full", meta.dot)} />
   539→              {meta.label}
   540→            </Badge>
   541→            {!node.active ? (
   542→              <Badge
   543→                variant="outline"
   544→                className="h-5 gap-1 px-2 text-[10px] font-medium text-muted-foreground"
   545→              >
   546→                Inativa
   547→              </Badge>
   548→            ) : null}
   549→            {services > 0 ? (
   550→              <Badge
   551→                variant="outline"
   552→                className="h-5 gap-1 px-2 text-[10px] font-medium"
   553→              >
   554→                <Wrench className="size-3" />
   555→                {services} serviço(s)
   556→              </Badge>
   557→            ) : null}
   558→            {hasChildren ? (
   559→              <Badge
   560→                variant="outline"
   561→                className="h-5 gap-1 px-2 text-[10px] font-medium text-muted-foreground"
   562→              >
   563→                {node.children!.length} filha(s)
   564→              </Badge>
   565→            ) : null}
   566→          </div>
   567→          <p className="mt-0.5 truncate font-mono text-[11px] text-muted-foreground">
   568→            /{node.slug}
   569→            {node.icon ? ` · ícone: ${node.icon}` : ""}
   570→            {typeof node.order === "number" ? ` · ordem ${node.order}` : ""}
   571→          </p>
   572→        </div>
   573→
   574→        {/* Active toggle */}
   575→        <div className="flex items-center gap-1.5 pr-1">
   576→          <Label
   577→            htmlFor={`active-${node.id}`}
   578→            className="hidden text-[11px] text-muted-foreground sm:block"
   579→          >
   580→            Ativa
   581→          </Label>
   582→          <Switch
   583→            id={`active-${node.id}`}
   584→            checked={node.active}
   585→            onCheckedChange={(v) => onToggleActive(node, v)}
   586→            aria-label="Ativar ou desativar categoria"
   587→          />
   588→        </div>
   589→
   590→        {/* Actions */}
   591→        <div className="flex items-center gap-1">
   592→          <Button
   593→            variant="ghost"
   594→            size="icon"
   595→            className="size-7 text-muted-foreground hover:text-primary"
   596→            onClick={() => onEdit(node)}
   597→            aria-label="Editar categoria"
   598→          >
   599→            <Pencil className="size-3.5" />
   600→          </Button>
   601→          <Button
   602→            variant="ghost"
   603→            size="icon"
   604→            className="size-7 text-muted-foreground hover:text-red-600"
   605→            onClick={() => onDelete(node)}
   606→            aria-label="Excluir categoria"
   607→          >
   608→            <Trash2 className="size-3.5" />
   609→          </Button>
   610→        </div>
   611→      </div>
   612→
   613→      {/* Children */}
   614→      {hasChildren && isOpen ? (
   615→        <ul role="group">
   616→          {node.children!.map((child) => (
   617→            <TreeNode
   618→              key={child.id}
   619→              node={child}
   620→              level={level + 1}
   621→              expanded={expanded}
   622→              onToggleExpand={onToggleExpand}
   623→              onEdit={onEdit}
   624→              onDelete={onDelete}
   625→              onToggleActive={onToggleActive}
   626→              serviceCounts={serviceCounts}
   627→            />
   628→          ))}
   629→        </ul>
   630→      ) : null}
   631→    </li>
   632→  )
   633→}

   648→function slugify(value: string): string {
   649→  return value
   650→    .toLowerCase()
   651→    .normalize("NFD")
   652→    .replace(/[\u0300-\u036f]/g, "")
   653→    .replace(/[^a-z0-9]+/g, "-")
   654→    .replace(/^-+|-+$/g, "")
   655→    .slice(0, 80)
   656→}

   658→function CategoryDialog({
   659→  open,
   660→  onOpenChange,
   661→  title,
   662→  description,
   663→  allCategories,
   664→  initial,
   665→  submitting,
   666→  onSubmit,
   667→}: {
   668→  open: boolean
   669→  onOpenChange: (open: boolean) => void
   670→  title: string
   671→  description: string
   672→  allCategories: CategoryRow[]
   673→  initial?: CategoryNode
   674→  submitting: boolean
   675→  onSubmit: (payload: CategoryPayload) => void
   676→}) {
   677→  const [name, setName] = React.useState("")
   678→  const [slug, setSlug] = React.useState("")
   679→  const [slugTouched, setSlugTouched] = React.useState(false)
   680→  const [parentId, setParentId] = React.useState<string>("__none__")
   681→  const [icon, setIcon] = React.useState("")
   682→  const [order, setOrder] = React.useState(0)
   683→  const [active, setActive] = React.useState(true)
   684→
   685→  // Reset form when opening
   686→  React.useEffect(() => {
   687→    if (!open) return
   688→    if (initial) {
   689→      setName(initial.name)
   690→      setSlug(initial.slug)
   691→      setSlugTouched(true)
   692→      setParentId(initial.parentId ?? "__none__")
   693→      setIcon(initial.icon ?? "")
   694→      setOrder(initial.order ?? 0)
   695→      setActive(initial.active)
   696→    } else {
   697→      setName("")
   698→      setSlug("")
   699→      setSlugTouched(false)
   700→      setParentId("__none__")
   701→      setIcon("")
   702→      setOrder(0)
   703→      setActive(true)
   704→    }
   705→  }, [open, initial])
   706→
   707→  // Auto-generate slug from name unless user edited it manually
   708→  React.useEffect(() => {
   709→    if (!slugTouched) setSlug(slugify(name))
   710→  }, [name, slugTouched])
   711→
   712→  // Determine the level from the chosen parent
   713→  const parent = allCategories.find((c) => c.id === parentId)
   714→  const level = parent ? Math.min(2, parent.level + 1) : 0
   715→
   716→  // Candidates for parent (any node whose level allows a child at our intended level)
   717→  // - To create a PAI (level 0): no parent.
   718→  // - To create a FILHA (level 1): parent must be level 0.
   719→  // - To create a SUB (level 2): parent must be level 1.
   720→  const parentCandidates = allCategories.filter(
   721→    (c) => c.level < 2 && c.id !== initial?.id,
   722→  )
   723→
   724→  const canSubmit =
   725→    name.trim().length >= 2 &&
   726→    /^[a-z0-9-]+$/.test(slug) &&
   727→    slug.length >= 2 &&
   728→    !submitting
   729→
   730→  const handleSubmit = (e: React.FormEvent) => {
   731→    e.preventDefault()
   732→    if (!canSubmit) return
   733→    onSubmit({
   734→      name: name.trim(),
   735→      slug,
   736→      parentId: parentId === "__none__" ? null : parentId,
   737→      level,
   738→      icon: icon.trim() || "",
   739→      order: Number.isFinite(order) ? order : 0,
   740→      active,
   741→    })
   742→  }
   743→
   744→  return (
   745→    <Dialog open={open} onOpenChange={onOpenChange}>
   746→      <DialogContent className="sm:max-w-lg">
   747→        <DialogHeader>
   748→          <DialogTitle>{title}</DialogTitle>
   749→          <DialogDescription>{description}</DialogDescription>
   750→        </DialogHeader>
   751→
   752→        <form onSubmit={handleSubmit} className="flex flex-col gap-4 py-1">
   753→          {/* Name + slug */}
   754→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   755→            <div className="flex flex-col gap-1.5">
   756→              <Label htmlFor="cat-name">
   757→                Nome <span className="text-red-500">*</span>
   758→              </Label>
   759→              <Input
   760→                id="cat-name"
   761→                value={name}
   762→                onChange={(e) => setName(e.target.value)}
   763→                placeholder="Ex.: Reparos"
   764→                required
   765→                maxLength={80}
   766→              />
   767→            </div>
   768→            <div className="flex flex-col gap-1.5">
   769→              <Label htmlFor="cat-slug">
   770→                Slug <span className="text-red-500">*</span>
   771→              </Label>
   772→              <Input
   773→                id="cat-slug"
   774→                value={slug}
   775→                onChange={(e) => {
   776→                  setSlug(e.target.value)
   777→                  setSlugTouched(true)
   778→                }}
   779→                placeholder="reparos"
   780→                required
   781→                maxLength={80}
   782→              />
   783→              <p className="text-[11px] text-muted-foreground">
   784→                Apenas letras minúsculas, números e hífens.
   785→              </p>
   786→            </div>
   787→          </div>
   788→
   789→          {/* Parent + level preview */}
   790→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   791→            <div className="flex flex-col gap-1.5">
   792→              <Label htmlFor="cat-parent">Categoria pai</Label>
   793→              <Select value={parentId} onValueChange={setParentId}>
   794→                <SelectTrigger id="cat-parent">
   795→                  <SelectValue placeholder="Nenhuma (categoria pai)" />
   796→                </SelectTrigger>
   797→                <SelectContent>
   798→                  <SelectItem value="__none__">
   799→                    Nenhuma (categoria pai)
   800→                  </SelectItem>
   801→                  {parentCandidates.map((c) => (
   802→                    <SelectItem key={c.id} value={c.id}>
   803→                      {"—".repeat(c.level)} {c.name}{" "}
   804→                      <span className="text-muted-foreground">
   805→                        ({LEVEL_META[c.level]?.label})
   806→                      </span>
   807→                    </SelectItem>
   808→                  ))}
   809→                </SelectContent>
   810→              </Select>
   811→            </div>
   812→            <div className="flex flex-col gap-1.5">
   813→              <Label>Nível resultante</Label>
   814→              <div className="flex h-9 items-center gap-2 rounded-md border bg-muted/40 px-3">
   815→                <Badge
   816→                  variant="secondary"
   817→                  className={LEVEL_META[level]?.badgeClass}
   818→                >
   819→                  {LEVEL_META[level]?.label}
   820→                </Badge>
   821→                <span className="text-xs text-muted-foreground">
   822→                  Nível {level}
   823→                </span>
   824→              </div>
   825→            </div>
   826→          </div>
   827→
   828→          {/* Icon + order */}
   829→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   830→            <div className="flex flex-col gap-1.5">
   831→              <Label htmlFor="cat-icon">Ícone (opcional)</Label>
   832→              <Input
   833→                id="cat-icon"
   834→                value={icon}
   835→                onChange={(e) => setIcon(e.target.value)}
   836→                placeholder="Ex.: Wrench (lucide)"
   837→                maxLength={40}
   838→              />
   839→              <p className="text-[11px] text-muted-foreground">
   840→                Nome de um ícone lucide-react.
   841→              </p>
   842→            </div>
   843→            <div className="flex flex-col gap-1.5">
   844→              <Label htmlFor="cat-order">Ordem</Label>
   845→              <Input
   846→                id="cat-order"
   847→                type="number"
   848→                value={order}
   849→                onChange={(e) => setOrder(Number(e.target.value))}
   850→                min={0}
   851→              />
   852→            </div>
   853→          </div>
   854→
   855→          {/* Active toggle */}
   856→          <div className="flex items-center justify-between rounded-md border p-3">
   857→            <div>
   858→              <Label
   859→                htmlFor="cat-active"
   860→                className="text-sm font-medium"
   861→              >
   862→                Ativa
   863→              </Label>
   864→              <p className="text-[11px] text-muted-foreground">
   865→                Categorias inativas não aparecem na vitrine pública.
   866→              </p>
   867→            </div>
   868→            <Switch
   869→              id="cat-active"
   870→              checked={active}
   871→              onCheckedChange={setActive}
   872→            />
   873→          </div>
   874→
   875→          <DialogFooter>
   876→            <Button
   877→              type="button"
   878→              variant="outline"
   879→              onClick={() => onOpenChange(false)}
   880→              disabled={submitting}
   881→            >
   882→              Cancelar
   883→            </Button>
   884→            <Button type="submit" disabled={!canSubmit} className="gap-1.5">
   885→              {submitting ? (
   886→                <Loader2 className="size-4 animate-spin" />
   887→              ) : (
   888→                <Plus className="size-4" />
   889→              )}
   890→              {initial ? "Salvar alterações" : "Criar categoria"}
   891→            </Button>
   892→          </DialogFooter>
   893→        </form>
   894→      </DialogContent>
   895→    </Dialog>
   896→  )
   897→}

   902→function errMsg(e: unknown, fallback: string): string {

   904→    return String((e as { message?: unknown }).message ?? fallback)
   905→  }
