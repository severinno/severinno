     4→ * DashboardShell — reusable sidebar + topbar layout for all panels
     5→ * (client / provider / admin).
     6→ *

     8→ *  - navItems:        primary navigation (label, icon, view, optional badge)

    19→ * The shell DOES NOT include the marketing footer — instead it renders a
    20→ * thin copyright bar at the bottom of the main content area.

   147→function initials(name?: string | null): string {
   148→  if (!name) return "?"
   149→  return name
   150→    .split(" ")
   151→    .map((p) => p[0])
   152→    .filter(Boolean)
   153→    .slice(0, 2)
   154→    .join("")
   155→    .toUpperCase()
   156→}

   406→                            <span
   407→                              aria-hidden
   408→                              className="absolute left-0 top-1/2 h-5 w-1 -translate-y-1/2 rounded-r-full bg-primary"
   409→                            />
   410→                          ) : null}
   411→                          <Icon className="size-4 shrink-0" />
   412→                          <span className="truncate">{item.label}</span>
   413→                          {item.badge != null && item.badge !== 0 ? (
   414→                            <span className="ml-auto inline-flex h-5 min-w-5 items-center justify-center rounded-full bg-primary px-1.5 text-[10px] font-semibold text-primary-foreground">
   415→                              {item.badge}
   416→                            </span>

   423→                <div className="border-t p-3">
   424→                  <div className="flex items-center gap-2.5 rounded-lg border bg-muted/30 p-2">
   425→                    <Avatar className="size-9">
   426→                      {user?.avatarUrl ? (
   427→                        <AvatarImage
   428→                          src={user.avatarUrl}
   429→                          alt={user.name ?? ""}
   430→                        />
   431→                      ) : null}
   432→                      <AvatarFallback className="bg-primary text-xs font-semibold text-primary-foreground">
   433→                        {initials(user?.name)}
   434→                      </AvatarFallback>

   436→                    <div className="min-w-0 flex-1">

   438→                        {user?.name ?? "Visitante"}

   446→                        </Badge>
   447→                      ) : null}
   448→                    </div>
   449→                    <Button
   450→                      variant="ghost"
   451→                      size="icon"
   452→                      className="size-8 shrink-0 text-muted-foreground hover:text-foreground"
   453→                      onClick={() => {
   454→                        setMobileOpen(false)

   456→                      }}
   457→                      aria-label="Sair"
   458→                    >
   459→                      <LogOut className="size-4" />
   460→                    </Button>

   462→                </div>
   463→              </div>
   464→            </SheetContent>

   473→                size="icon"
   474→                className="size-9 md:hidden"

   518→              {/* Location chip */}

   521→                size="sm"

   563→              {/* User dropdown */}

   565→                <DropdownMenuTrigger asChild>

   568→                    className="flex items-center gap-1.5 rounded-full p-0.5 outline-none transition hover:bg-accent focus-visible:ring-2 focus-visible:ring-ring"
   569→                    aria-label="Menu da conta"
   570→                  >
   571→                    <Avatar className="size-9">
   572→                      {user?.avatarUrl ? (
   573→                        <AvatarImage
   574→                          src={user.avatarUrl}
   575→                          alt={user.name ?? ""}
   576→                        />
   577→                      ) : null}
   578→                      <AvatarFallback className="bg-primary text-xs font-semibold text-primary-foreground">
   579→                        {initials(user?.name)}
   580→                      </AvatarFallback>
   581→                    </Avatar>
   582→                  </button>
   583→                </DropdownMenuTrigger>
   584→                <DropdownMenuContent align="end" className="w-60">
   585→                  <DropdownMenuLabel className="flex flex-col gap-1">
   586→                    <span className="truncate">{user?.name}</span>

   588→                      {user?.email}
   589→                    </span>
   590→                    {user?.role ? (
   591→                      <Badge
   592→                        variant="secondary"
   593→                        className="mt-1 w-fit text-[10px]"
   594→                      >
   595→                        {ROLE_LABELS[user.role]}
   596→                      </Badge>
   597→                    ) : null}
   598→                  </DropdownMenuLabel>
   599→                  <DropdownMenuSeparator />
   600→                  <DropdownMenuItem

   659→function NotificationsBell({
   660→  items,
   661→  unreadCount,
   662→  isLoading,
   663→  onMarkRead,
   664→  onMarkAllRead,
   665→  markingAll,
   666→}: {
   667→  items: NotificationItem[]
   668→  unreadCount: number
   669→  isLoading: boolean
   670→  onMarkRead: (id: string) => void
   671→  onMarkAllRead: (ids: string[]) => void
   672→  markingAll: boolean
   673→}) {
   674→  const unreadIds = React.useMemo(
   675→    () => items.filter((n) => !n.read).map((n) => n.id),
   676→    [items],
   677→  )
   678→  return (
   679→    <DropdownMenu>
   680→      <DropdownMenuTrigger asChild>
   681→        <Button
   682→          variant="ghost"
   683→          size="icon"
   684→          className="relative size-9 text-muted-foreground hover:text-foreground"
   685→          aria-label={`Notificações${
   686→            unreadCount > 0 ? ` (${unreadCount} não lidas)` : ""
   687→          }`}
   688→        >
   689→          <Bell className="size-5" />
   690→          {unreadCount > 0 ? (
   691→            <span
   692→              className="absolute -right-0.5 -top-0.5 inline-flex h-4 min-w-4 items-center justify-center rounded-full bg-primary px-1 text-[10px] font-semibold text-primary-foreground ring-2 ring-background"
   693→              aria-hidden
   694→            >
   695→              {unreadCount > 9 ? "9+" : unreadCount}
   696→            </span>
   697→          ) : null}
   698→        </Button>
   699→      </DropdownMenuTrigger>
   700→      <DropdownMenuContent
   701→        align="end"
   702→        className="w-80 gap-0 p-0"
   703→      >
   704→        <div className="flex items-center justify-between border-b px-3 py-2.5">
   705→          <div className="flex items-center gap-2">
   706→            <p className="text-sm font-semibold">Notificações</p>
   707→            {unreadCount > 0 ? (
   708→              <Badge
   709→                variant="secondary"
   710→                className="h-5 bg-primary/10 px-1.5 text-[10px] font-semibold text-primary"
   711→              >
   712→                {unreadCount} nova{unreadCount > 1 ? "s" : ""}
   713→              </Badge>
   714→            ) : null}
   715→          </div>
   716→          {unreadCount > 0 ? (
   717→            <button
   718→              type="button"
   719→              onClick={() => onMarkAllRead(unreadIds)}
   720→              disabled={markingAll}
   721→              className="inline-flex items-center gap-1 text-[11px] font-medium text-primary transition-colors hover:text-primary/80 disabled:opacity-50"
   722→            >
   723→              {markingAll ? (
   724→                <Loader2 className="size-3 animate-spin" />
   725→              ) : (
   726→                <CheckCheck className="size-3.5" />
   727→              )}
   728→              Marcar todas
   729→            </button>
   730→          ) : null}
   731→        </div>
   732→
   733→        <ScrollArea className="max-h-80">
   734→          {isLoading ? (
   735→            <div className="flex items-center justify-center gap-2 p-8 text-sm text-muted-foreground">
   736→              <Loader2 className="size-4 animate-spin" />
   737→              Carregando…
   738→            </div>
   739→          ) : items.length === 0 ? (
   740→            <div className="flex flex-col items-center gap-2 p-8 text-center">
   741→              <span className="flex size-12 items-center justify-center rounded-full bg-muted text-muted-foreground">
   742→                <Bell className="size-5" />
   743→              </span>
   744→              <p className="text-sm text-muted-foreground">
   745→                Você não tem notificações.
   746→              </p>
   747→            </div>
   748→          ) : (
   749→            <ul className="divide-y">
   750→              {items.map((n) => {
   751→                const typeLabel = NOTIFICATION_TYPE_LABELS[n.type]
   752→                return (
   753→                  <li
   754→                    key={n.id}
   755→                    className={cn(
   756→                      "relative flex gap-3 px-3 py-2.5 transition-colors hover:bg-accent/50",
   757→                      !n.read && "bg-primary/5",
   758→                    )}
   759→                  >
   760→                    <span
   761→                      className={cn(
   762→                        "mt-1.5 size-2 shrink-0 rounded-full",
   763→                        n.read ? "bg-transparent ring-1 ring-border" : "bg-primary",
   764→                      )}
   765→                      aria-hidden
   766→                    />
   767→                    <div className="min-w-0 flex-1">
   768→                      <div className="flex items-baseline justify-between gap-2">
   769→                        <p className="text-sm font-medium leading-tight">
   770→                          {n.title}
   771→                        </p>
   772→                        <span className="shrink-0 text-[10px] text-muted-foreground tabular-nums">
   773→                          {formatRelative(n.createdAt)}
   774→                        </span>
   775→                      </div>
   776→                      {n.body ? (
   777→                        <p className="mt-0.5 line-clamp-2 text-xs text-muted-foreground">
   778→                          {n.body}
   779→                        </p>
   780→                      ) : null}
   781→                      <div className="mt-1 flex items-center gap-2">
   782→                        {typeLabel ? (
   783→                          <Badge
   784→                            variant="outline"
   785→                            className="h-4 px-1.5 text-[10px] font-medium text-muted-foreground"
   786→                          >
   787→                            {typeLabel}
   788→                          </Badge>
   789→                        ) : null}
   790→                        {!n.read ? (
   791→                          <button
   792→                            type="button"
   793→                            onClick={() => onMarkRead(n.id)}
   794→                            className="inline-flex items-center gap-1 text-[10px] font-medium text-primary hover:underline"
   795→                          >
   796→                            <Check className="size-3" />
   797→                            Marcar como lida
   798→                          </button>
   799→                        ) : null}
   800→                      </div>
   801→                    </div>
   802→                  </li>
   803→                )
   804→              })}
   805→            </ul>
   806→          )}
   807→        </ScrollArea>
   808→      </DropdownMenuContent>
   809→    </DropdownMenu>
   810→  )
   811→}

   855→export function StatCard({
   856→  icon: Icon,
   857→  label,
   858→  value,
   859→  hint,
   860→  tone = "primary",
   861→  index = 0,
   862→  trend,

   865→  label: string
   866→  value: React.ReactNode

   868→  tone?: "primary" | "amber" | "sky" | "rose" | "zinc"
   869→  /** Index for staggered mount animation (0-based). */
   870→  index?: number

   872→  trend?: { direction: "up" | "down"; label: string }

   904→                    : "bg-rose-100 text-rose-700 dark:bg-rose-900/40 dark:text-rose-200",
