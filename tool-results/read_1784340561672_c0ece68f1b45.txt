   138→   138→// ---------------------------------------------------------------------------
   139→   139→// Validation helpers
   140→   140→// ---------------------------------------------------------------------------
   141→   141→
   142→   142→function cardNameValid(v: string) { return v.trim().length >= 3 }
   143→   143→function cardNumberValid(v: string) { return v.replace(/\s/g, "").length >= 13 }
   144→   144→function cardExpiryValid(v: string) { return /^\d{2}\/\d{2}$/.test(v) }
   145→   145→function cardCvvValid(v: string) { return /^\d{3,4}$/.test(v) }

   169→   169→export function BookingModal() {
   170→   170→  const open = useUIStore((s) => s.bookingModal.open)
   171→   171→  const providerIdPreset = useUIStore((s) => s.bookingModal.providerId)
   172→   172→  const serviceIdPreset = useUIStore((s) => s.bookingModal.serviceId)
   173→   173→  const close = useUIStore((s) => s.closeBooking)
   174→   174→  const openAuth = useUIStore((s) => s.openAuth)
   175→   175→  const isMobile = useIsMobile()
   176→   176→  const navigate = useViewStore((s) => s.navigate)
   177→   177→  const user = useAuthStore((s) => s.user)
   178→   178→
   179→   179→  const [step, setStep] = React.useState<Step>(1)
   180→   180→  const [state, setState] = React.useState<BookingFormState>(initialState())
   181→   181→  const [submitting, setSubmitting] = React.useState(false)
   182→   182→  const [touched, setTouched] = React.useState<Set<string>>(new Set())
   183→   183→
   184→   184→  // Reset when modal opens
   185→   185→  React.useEffect(() => {
   186→   186→    if (open) {
   187→   187→      setStep(1)
   188→   188→      setState(initialState())
   189→   189→      setTouched(new Set())
   190→   190→    }
   191→   191→  }, [open])
   192→   192→
   193→   193→  // Fetch provider + services
   194→   194→  const providerQuery = useQuery({
   195→   195→    queryKey: ["provider", providerIdPreset],
   196→   196→    queryFn: () =>
   197→   197→      apiGet<ProviderDetail>(`/api/providers/${providerIdPreset}`),
   198→   198→    enabled: open && !!providerIdPreset,
   199→   199→    staleTime: 60 * 1000,
   200→   200→  })
   201→   201→
   202→   202→  const servicesQuery = useQuery({
   203→   203→    queryKey: ["services-by-provider", providerIdPreset],
   204→   204→    queryFn: () =>
   205→   205→      apiGet<ProviderService[]>("/api/services", { providerId: providerIdPreset }),
   206→   206→    enabled: open && !!providerIdPreset,
   207→   207→    staleTime: 60 * 1000,
   208→   208→  })
   209→   209→
   210→   210→  const provider = providerQuery.data
   211→   211→  const services = servicesQuery.data ?? []
   212→   212→  const selectedService =
   213→   213→    services.find((s) => s.id === serviceIdPreset) ?? services[0]
   214→   214→
   215→   215→  React.useEffect(() => {
   216→   216→    if (open && selectedService) {
   217→   217→      setState((s) => ({ ...s, quantity: 1 }))
   218→   218→    }
   219→   219→  }, [open, selectedService])
   220→   220→
   221→   221→  const set = <K extends keyof BookingFormState>(
   222→   222→    key: K,
   223→   223→    value: BookingFormState[K],
   224→   224→  ) => setState((s) => ({ ...s, [key]: value }))
   225→   225→
   226→   226→  const markTouched = (field: string) =>
   227→   227→    setTouched((prev) => new Set(prev).add(field))
   228→   228→
   229→   229→  // ── Step validation ──
   230→   230→  const step1Valid = !!state.date && !!state.time
   231→   231→  const step2Valid =
   232→   232→    !!state.address.cep &&
   233→   233→    state.address.cep.replace(/\D/g, "").length === 8 &&
   234→   234→    !!state.address.street &&
   235→   235→    !!state.address.number &&
   236→   236→    !!state.address.city &&
   237→   237→    !!state.address.state
   238→   238→  const step3Valid =
   239→   239→    state.paymentMethod === "PIX" ||
   240→   240→    (state.paymentMethod === "CARD" &&
   241→   241→      cardNameValid(state.cardName) &&
   242→   242→      cardNumberValid(state.cardNumber) &&
   243→   243→      cardExpiryValid(state.cardExpiry) &&
   244→   244→      cardCvvValid(state.cardCvv))
   245→   245→  // Step 4 is always valid (it's a review)
   246→   246→
   247→   247→  const validSteps: Record<number, boolean> = {
   248→   248→    1: step1Valid,
   249→   249→    2: step2Valid,
   250→   250→    3: step3Valid,
   251→   251→    4: true,
   252→   252→  }
   253→   253→
   254→   254→  // ── Navigation ──
   255→   255→  const handleStepClick = (target: Step) => {
   256→   256→    if (target < step) {
   257→   257→      setStep(target)
   258→   258→      return
   259→   259→    }
   260→   260→    for (let i = 1; i < target; i++) {
   261→   261→      if (!validSteps[i as Step]) {
   262→   262→        toast.error("Complete os passos anteriores primeiro.")
   263→   263→        return
   264→   264→      }
   265→   265→    }
   266→   266→    setStep(target)
   267→   267→  }
   268→   268→
   269→   269→  const handleNext = () => {
   270→   270→    if (step === 1 && !step1Valid) {
   271→   271→      toast.error("Selecione data e horário para continuar.")
   272→   272→      return
   273→   273→    }
   274→   274→    if (step === 2 && !step2Valid) {
   275→   275→      toast.error("Preencha o endereço completo.")
   276→   276→      return
   277→   277→    }
   278→   278→    if (step === 3 && !step3Valid) {
   279→   279→      toast.error("Verifique os dados de pagamento.")
   280→   280→      return
   281→   281→    }
   282→   282→    setStep((s) => Math.min(4, s + 1) as Step)
   283→   283→  }
   284→   284→
   285→   285→  const handleBack = () => setStep((s) => Math.max(1, s - 1) as Step)
   286→   286→
   287→   287→  const handleSubmit = async () => {
   288→   288→    if (!user) {
   289→   289→      toast.info("Faça cadastro gratuito para agendar serviços.")
   290→   290→      openAuth("register", "CLIENT")
   291→   291→      return
   292→   292→    }
   293→   293→    if (!provider || !selectedService || !state.date || !state.time) {
   294→   294→      toast.error("Dados incompletos. Revise o agendamento.")
   295→   295→      return
   296→   296→    }
   297→   297→
   298→   298→    const [h, m] = state.time.split(":").map(Number)
   299→   299→    const scheduledAt = new Date(state.date)
   300→   300→    scheduledAt.setHours(h ?? 0, m ?? 0, 0, 0)
   301→   301→    const amount = (selectedService.basePrice || 0) * (state.quantity || 1)
   302→   302→
   303→   303→    setSubmitting(true)
   304→   304→    try {
   305→   305→      await apiPost("/api/bookings", {
   306→   306→        providerId: provider.id,
   307→   307→        serviceId: selectedService.id,
   308→   308→        scheduledAt: scheduledAt.toISOString(),
   309→   309→        address: [
   310→   310→          state.address.street,
   311→   311→          state.address.number,
   312→   312→          state.address.complement,
   313→   313→          state.address.district,
   314→   314→        ]
   315→   315→          .filter(Boolean)
   316→   316→          .join(", "),
   317→   317→        cep: state.address.cep,
   318→   318→        lat: state.address.lat,
   319→   319→        lng: state.address.lng,
   320→   320→        amount,
   321→   321→        paymentMethod: state.paymentMethod,
   322→   322→        notes: state.notes,
   323→   323→        quantity: state.quantity,
   324→   324→        unit: selectedService.unit,
   325→   325→      })
   326→   326→      toast.success("Agendamento confirmado! Acompanhe em seus agendamentos.")
   327→   327→      close()
   328→   328→      navigate("client.bookings")
   329→   329→    } catch (e) {
   330→   330→      const msg =
   331→   331→        (e as { message?: string })?.message ??
   332→   332→        "Não foi possível concluir o agendamento. Tente novamente."
   333→   333→      toast.error(msg)
   334→   334→    } finally {
   335→   335→      setSubmitting(false)
   336→   336→    }
   337→   337→  }
   338→   338→
   339→   339→  // ── Step content ──
   340→   340→  const stepContent = (() => {
   341→   341→    switch (step) {
   342→   342→      case 1:
   343→   343→        return (
   344→   344→          <Step1Schedule
   345→   345→            state={state}
   346→   346→            set={set}
   347→   347→            availability={provider?.availability ?? []}
   348→   348→            selectedService={selectedService}
   349→   349→            provider={provider}
   350→   350→            loading={providerQuery.isLoading || servicesQuery.isLoading}
   351→   351→            isDesktop={!isMobile}
   352→   352→          />
   353→   353→        )
   354→   354→      case 2:
   355→   355→        return (
   356→   356→          <Step2Details
   357→   357→            state={state}
   358→   358→            set={set}
   359→   359→            provider={provider}
   360→   360→            selectedService={selectedService}
   361→   361→          />
   362→   362→        )
   363→   363→      case 3:
   364→   364→        return (
   365→   365→          <Step3Payment
   366→   366→            state={state}
   367→   367→            set={set}
   368→   368→            markTouched={markTouched}
   369→   369→            touched={touched}
   370→   370→            selectedService={selectedService}
   371→   371→          />
   372→   372→        )
   373→   373→      case 4:
   374→   374→        return (
   375→   375→          <Step4Confirmation
   376→   376→            state={state}
   377→   377→            provider={provider}
   378→   378→            selectedService={selectedService}
   379→   379→            goToStep={setStep}
   380→   380→          />
   381→   381→        )
   382→   382→      default:
   383→   383→        return null
   384→   384→    }
   385→   385→  })()
   386→   386→
   387→   387→  // Dynamic dialog sizing: wider on Step 1 for side-by-side calendar layout
   388→   388→  const dialogSizeClass = step === 1
   389→   389→    ? "sm:max-w-2xl"
   390→   390→    : "sm:max-w-lg"
   391→   391→
   392→   392→  // ── Step indicator (reused for desktop custom layout) ──
   393→   393→  const stepIndicator = (
   394→   394→    <div className="border-b px-4 sm:px-5 py-3">
   395→   395→      <div className="flex items-center justify-between">
   396→   396→        {STEPS.map((s, i) => {
   397→   397→          const active = step === s.id
   398→   398→          const done = validSteps[s.id] && step > s.id
   399→   399→          const Icon = s.icon
   400→   400→          return (
   401→   401→            <React.Fragment key={s.id}>
   402→   402→              <button
   403→   403→                type="button"
   404→   404→                onClick={() => handleStepClick(s.id)}
   405→   405→                disabled={!done && s.id > step}
   406→   406→                className={cn(
   407→   407→                  "flex items-center gap-1.5 text-xs transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring rounded-md px-1 py-0.5",
   408→   408→                  active
   409→   409→                    ? "text-emerald-700 dark:text-emerald-400 font-semibold"
   410→   410→                    : done
   411→   411→                      ? "text-emerald-600 cursor-pointer hover:text-emerald-700"
   412→   412→                      : "text-muted-foreground cursor-default",
   413→   413→                )}
   414→   414→              >
   415→   415→                <span
   416→   416→                  className={cn(
   417→   417→                    "inline-flex size-7 items-center justify-center rounded-full border-2 text-xs font-bold transition-all",
   418→   418→                    active &&
   419→   419→                      "border-emerald-600 bg-emerald-600 text-white shadow-sm shadow-emerald-600/25",
   420→   420→                    done &&
   421→   421→                      "border-emerald-600 bg-emerald-600 text-white cursor-pointer",
   422→   422→                    !active &&
   423→   423→                      !done &&
   424→   424→                      "border-muted-foreground/20 text-muted-foreground",
   425→   425→                  )}
   426→   426→                >
   427→   427→                  {done ? (
   428→   428→                    <Check className="size-3.5" />
   429→   429→                  ) : Icon ? (
   430→   430→                    <Icon className="size-3.5" />
   431→   431→                  ) : (
   432→   432→                    s.id
   433→   433→                  )}
   434→   434→                </span>
   435→   435→                <span className="hidden sm:inline">
   436→   436→                  {s.shortLabel ?? s.label}
   437→   437→                </span>
   438→   438→              </button>
   439→   439→              {i < STEPS.length - 1 && (
   440→   440→                <div className="flex-1 h-px bg-muted-foreground/15 mx-1 sm:mx-2 relative">
   441→   441→                  <div
   442→   442→                    className="absolute inset-0 bg-emerald-500 transition-transform origin-left duration-300"
   443→   443→                    style={{
   444→   444→                      transform:
   445→   445→                        validSteps[STEPS[i + 1]?.id] || step > s.id
   446→   446→                          ? "scaleX(1)"
   447→   447→                          : step === s.id && validSteps[s.id]
   448→   448→                            ? "scaleX(0.5)"
   449→   449→                            : "scaleX(0)",
   450→   450→                    }}
   451→   451→                  />
   452→   452→                </div>
   453→   453→              )}
   454→   454→            </React.Fragment>
   455→   455→          )
   456→   456→        })}
   457→   457→      </div>
   458→   458→    </div>
   459→   459→  )
   460→   460→
   461→   461→  // ── Footer buttons ──
   462→   462→  const footerButtons = (
   463→   463→    <div className="border-t bg-background/95 backdrop-blur px-4 sm:px-5 py-2.5 sticky bottom-0">
   464→   464→      <div className="flex items-center justify-between gap-2">
   465→   465→        {step > 1 ? (
   466→   466→          <Button
   467→   467→            type="button"
   468→   468→            variant="ghost"
   469→   469→            size="sm"
   470→   470→            onClick={handleBack}
   471→   471→            disabled={submitting}
   472→   472→            className="text-muted-foreground hover:text-foreground h-9 gap-1"
   473→   473→          >
   474→   474→            <ChevronLeft className="size-4" />
   475→   475→            Voltar
   476→   476→          </Button>
   477→   477→        ) : (
   478→   478→          <div />
   479→   479→        )}
   480→   480→        {step < 4 ? (
   481→   481→          <Button
   482→   482→            type="button"
   483→   483→            size="sm"
   484→   484→            onClick={handleNext}
   485→   485→            disabled={!validSteps[step]}
   486→   486→            className="h-9 bg-emerald-600 hover:bg-emerald-700 gap-1.5"
   487→   487→          >
   488→   488→            Continuar
   489→   489→            <ChevronRight className="size-4" />
   490→   490→          </Button>
   491→   491→        ) : (
   492→   492→          <Button
   493→   493→            type="button"
   494→   494→            size="sm"
   495→   495→            onClick={handleSubmit}
   496→   496→            disabled={submitting || !validSteps[step]}
   497→   497→            className="h-9 bg-emerald-600 hover:bg-emerald-700 gap-1.5"
   498→   498→          >
   499→   499→            {submitting && <Loader2 className="size-3.5 animate-spin" />}
   500→   500→            <Send className="size-3.5" />
   501→   501→            Confirmar agendamento
   502→   502→          </Button>
   503→   503→        )}
   504→   504→      </div>
   505→   505→    </div>
   506→   506→  )
   507→   507→
   508→   508→  const wizardBody = (
   509→   509→    <StepWizard
   510→   510→      steps={STEPS}
   511→   511→      currentStep={step}
   512→   512→      validSteps={validSteps}
