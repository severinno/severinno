   148→   148→function initials(name?: string | null): string {
   149→   149→  if (!name) return "?"
   150→   150→  return name
   151→   151→    .split(" ")
   152→   152→    .map((p) => p[0])
   153→   153→    .filter(Boolean)
   154→   154→    .slice(0, 2)
   155→   155→    .join("")
   156→   156→    .toUpperCase()
   157→   157→}
