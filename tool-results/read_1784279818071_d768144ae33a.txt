   700→   700→function RevenueChart({
   701→   701→  recentBookings,
   702→   702→}: {
   703→   703→  recentBookings: AdminStats["recentBookings"]
   704→   704→}) {
   705→   705→  // Série dos últimos 6 meses sintetizada a partir de recentBookings pagos.
   706→   706→  // Substituir por endpoint dedicado quando disponível.
   707→   707→  const months: { label: string; revenue: number }[] = []
   708→   708→  const now = new Date()
   709→   709→  for (let i = 5; i >= 0; i--) {
   710→   710→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1)
   711→   711→    const label = d.toLocaleDateString("pt-BR", {
   712→   712→      month: "short",
   713→   713→      year: "2-digit",
   714→   714→    })
   715→   715→    months.push({ label, revenue: 0 })
   716→   716→  }
   717→   717→  for (const b of recentBookings) {
   718→   718→    if (b.paymentStatus !== "PAID") continue
   719→   719→    const created = new Date(b.createdAt)
   720→   720→    const diffMonths =
   721→   721→      (now.getFullYear() - created.getFullYear()) * 12 +
   722→   722→      (now.getMonth() - created.getMonth())
   723→   723→    const idx = months.length - 1 - diffMonths
   724→   724→    if (idx >= 0 && idx < months.length) {
   725→   725→      months[idx].revenue += b.amount
   726→   726→    }
   727→   727→  }
   728→   728→
   729→   729→  return (
   730→   730→    <div className="h-[200px] w-full">
   731→   731→      <ResponsiveContainer width="100%" height="100%">
   732→   732→        <BarChart data={months} margin={{ left: 0, right: 8, top: 8, bottom: 0 }}>
   733→   733→          <CartesianGrid
   734→   734→            vertical={false}
   735→   735→            stroke="var(--border)"
   736→   736→            strokeDasharray="3 3"
   737→   737→          />
   738→   738→          <XAxis
   739→   739→            dataKey="label"
   740→   740→            tickLine={false}
   741→   741→            axisLine={false}
   742→   742→            tick={{ fontSize: 11, fill: "var(--muted-foreground)" }}
   743→   743→          />
   744→   744→          <YAxis
   745→   745→            tickLine={false}
   746→   746→            axisLine={false}
   747→   747→            width={56}
   748→   748→            tickFormatter={(v) =>
   749→   749→              v >= 1000 ? `R$ ${(v / 1000).toFixed(1)}k` : `R$ ${v}`
   750→   750→            }
   751→   751→            tick={{ fontSize: 11, fill: "var(--muted-foreground)" }}
   752→   752→          />
   753→   753→          <RTooltip
   754→   754→            cursor={{ fill: "var(--accent)", opacity: 0.5 }}
   755→   755→            formatter={(v: number) => [formatBRL(v), "Receita"]}
   756→   756→            contentStyle={CHART_TOOLTIP_STYLE}
   757→   757→          />
   758→   758→          <Bar
   759→   759→            dataKey="revenue"
