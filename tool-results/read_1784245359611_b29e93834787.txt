    26→import { toast } from "sonner"
    27→

    29→import {

    59→} from "@/components/ui/dropdown-menu"

    67→} from "@/components/ui/select"
    68→import { Switch } from "@/components/ui/switch"
    69→import {

    79→type CategoryOption = {

    88→
    89→type AdminService = {

    92→  description: string
    93→  basePrice: number
    94→  unit: ServiceUnit
    95→  active: boolean
    96→  createdAt: string
    97→  photos?: string[] | null
    98→  category?: CategoryOption | null
    99→  provider: {

   101→    name: string

   103→    city?: string | null
   104→    state?: string | null
   105→    verified: boolean
   106→    active: boolean
   107→  }
   108→  _count?: { bookings: number; reviews: number }

   110→
   111→type AdminServicesResponse = {
   112→  items: AdminService[]
   113→  total: number

   115→
   116→type ProviderOption = {

   122→
   123→export function AdminServices() {
   124→  const queryClient = useQueryClient()
   125→  const [q, setQ] = React.useState("")
   126→  const [debouncedQ, setDebouncedQ] = React.useState("")
   127→  const [active, setActive] = React.useState<ActiveFilter>("ALL")
   128→  const [categoryFilter, setCategoryFilter] = React.useState<string>("ALL")
   129→  const [providerFilter, setProviderFilter] = React.useState<string>("ALL")
   130→  const [deleteTarget, setDeleteTarget] = React.useState<AdminService | null>(
   131→    null,
   132→  )
   133→
   134→  React.useEffect(() => {
   135→    const t = setTimeout(() => setDebouncedQ(q.trim()), 350)
   136→    return () => clearTimeout(t)
   137→  }, [q])
   138→
   139→  // Full category list (includes parentId/level so we can build breadcrumb paths)
   140→  const { data: categories } = useQuery({
   141→    queryKey: ["categories", "all-flat"],
   142→    queryFn: () => apiGet<CategoryOption[]>("/api/categories"),
   143→    staleTime: 60_000,
   144→  })
   145→
   146→  // Provider options for the filter dropdown (lightweight)
   147→  const { data: providersList } = useQuery({
   148→    queryKey: ["admin", "providers", "options"],
   149→    queryFn: () =>
   150→      apiGet<{ items: ProviderOption[]; total: number }>(
   151→        "/api/admin/users",
   152→        {
   153→          role: "PROVIDER",
   154→          limit: 100,
   155→        },
   156→      ),
   157→    staleTime: 60_000,
   158→  })
   159→
   160→  const { data, isLoading, isError } = useQuery({
   161→    queryKey: [
   162→      "admin",
   163→      "services",
   164→      { debouncedQ, active, categoryFilter, providerFilter },
   165→    ],
   166→    queryFn: () =>
   167→      apiGet<AdminServicesResponse>("/api/admin/services", {
   168→        ...(debouncedQ ? { q: debouncedQ } : {}),
   169→        ...(active !== "ALL" ? { active } : {}),
   170→        ...(categoryFilter !== "ALL" ? { categoryId: categoryFilter } : {}),
   171→        ...(providerFilter !== "ALL" ? { providerId: providerFilter } : {}),
   172→      }),
   173→    staleTime: 15_000,
   174→  })
   175→
   176→  const invalidate = () =>
   177→    queryClient.invalidateQueries({ queryKey: ["admin", "services"] })
   178→
   179→  const patchMutation = useMutation({
   180→    mutationFn: ({
   181→      id,
   182→      patch,
   183→    }: {
   184→      id: string
   185→      patch: { active?: boolean }
   186→    }) => apiPatch(`/api/services/${id}`, patch),
   187→    onSuccess: () => invalidate(),
   188→    onError: (e: unknown) =>
   189→      toast.error(errMsg(e, "Falha ao atualizar serviço.")),
   190→  })
   191→
   192→  const deleteMutation = useMutation({
   193→    mutationFn: (id: string) => apiDelete(`/api/services/${id}`),
   194→    onSuccess: () => {
   195→      toast.success("Serviço excluído.")
   196→      invalidate()
   197→      setDeleteTarget(null)
   198→    },
   199→    onError: (e: unknown) =>
   200→      toast.error(errMsg(e, "Não foi possível excluir.")),
   201→  })
   202→
   203→  const items = data?.items ?? []
   204→
   205→  const activeFilterCount =
   206→    (debouncedQ ? 1 : 0) +
   207→    (active !== "ALL" ? 1 : 0) +
   208→    (categoryFilter !== "ALL" ? 1 : 0) +
   209→    (providerFilter !== "ALL" ? 1 : 0)
   210→
   211→  const clearFilters = () => {
   212→    setQ("")
   213→    setDebouncedQ("")
   214→    setActive("ALL")
   215→    setCategoryFilter("ALL")
   216→    setProviderFilter("ALL")
   217→  }
   218→
   219→  return (
   220→    <div className="flex flex-col gap-4">
   221→      {/* Quick stat pill */}
   222→      <div className="flex flex-wrap items-center justify-end gap-2">
   223→        <span className="inline-flex items-center gap-1.5 rounded-full border bg-card px-2.5 py-1 text-xs font-medium text-muted-foreground shadow-sm">
   224→          <Wrench className="size-3 text-primary" />
   225→          {(data?.total ?? 0).toLocaleString("pt-BR")} serviço(s) no total
   226→        </span>
   227→      </div>
   228→
   229→      {/* Filter bar */}
   230→      <div className="flex flex-wrap items-center gap-3 rounded-xl border bg-card p-3 shadow-sm">
   231→        <div className="relative min-w-[200px] flex-1">
   232→          <Search className="absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
   233→          <Input
   234→            id="s-search"
   235→            value={q}
   236→            onChange={(e) => setQ(e.target.value)}
   237→            placeholder="Buscar por título ou descrição"
   238→            className="h-9 pl-8"
   239→          />
   240→        </div>
   241→        <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   242→          <SelectTrigger className="h-9 w-auto min-w-[160px]">
   243→            <SelectValue placeholder="Categoria" />
   244→          </SelectTrigger>
   245→          <SelectContent>
   246→            <SelectItem value="ALL">Todas as categorias</SelectItem>
   247→            {(categories ?? [])
   248→              .slice()
   249→              .sort(
   250→                (a, b) =>
   251→                  a.level - b.level ||
   252→                  (a.order ?? 0) - (b.order ?? 0) ||
   253→                  a.name.localeCompare(b.name),
   254→              )
   255→              .map((c) => (
   256→                <SelectItem key={c.id} value={c.id}>
   257→                  <span className="text-muted-foreground">
   258→                    {"—".repeat(c.level)}
   259→                  </span>{" "}
   260→                  {c.name}
   261→                  <span className="ml-1.5 text-[10px] uppercase tracking-wide text-muted-foreground">
   262→                    {c.level === 0
   263→                      ? "pai"
   264→                      : c.level === 1
   265→                        ? "filha"
   266→                        : "sub"}
   267→                  </span>
   268→                </SelectItem>
   269→              ))}
   270→          </SelectContent>
   271→        </Select>
   272→        <Select
   273→          value={providerFilter}
   274→          onValueChange={setProviderFilter}
   275→        >
   276→          <SelectTrigger className="h-9 w-auto min-w-[160px]">
   277→            <SelectValue placeholder="Prestador" />
   278→          </SelectTrigger>
   279→          <SelectContent>
   280→            <SelectItem value="ALL">Todos os prestadores</SelectItem>
   281→            {(providersList?.items ?? []).map((p) => (
   282→              <SelectItem key={p.id} value={p.id}>
   283→                {p.name}
   284→              </SelectItem>
   285→            ))}
   286→          </SelectContent>
   287→        </Select>
   288→        <Select
   289→          value={active}
   290→          onValueChange={(v) => setActive(v as ActiveFilter)}
   291→        >
   292→          <SelectTrigger className="h-9 w-auto min-w-[140px]">
   293→            <SelectValue placeholder="Status" />
   294→          </SelectTrigger>
   295→          <SelectContent>
   296→            <SelectItem value="ALL">Status: todos</SelectItem>
   297→            <SelectItem value="true">Apenas ativos</SelectItem>
   298→            <SelectItem value="false">Apenas inativos</SelectItem>
   299→          </SelectContent>
   300→        </Select>
   301→        <Button
   302→          variant="ghost"
   303→          size="sm"
   304→          onClick={clearFilters}
   305→          disabled={activeFilterCount === 0}
   306→          className="h-9 gap-1.5 text-muted-foreground"
   307→        >
   308→          <X className="size-3.5" />
   309→          Limpar filtros
   310→        </Button>
   311→        {activeFilterCount > 0 ? (
   312→          <Badge
   313→            variant="outline"
   314→            className="gap-1 border-primary/30 bg-primary/5 text-primary"
   315→          >
   316→            <span className="size-1.5 rounded-full bg-primary" />
   317→            {activeFilterCount} filtro(s) ativo(s)
   318→          </Badge>
   319→        ) : null}
   320→      </div>
   321→
   322→      {/* Result count */}
   323→      <div className="flex items-center justify-between gap-2">
   324→        <p className="text-xs text-muted-foreground">
   325→          {items.length > 0
   326→            ? `Mostrando ${items.length} de ${(data?.total ?? 0).toLocaleString(
   327→                "pt-BR",
   328→              )} serviço(s)`
   329→            : "Nenhum serviço"}
   330→        </p>
   331→        {patchMutation.isPending || deleteMutation.isPending ? (
   332→          <span className="inline-flex items-center gap-1.5 text-[11px] text-muted-foreground">
   333→            <Loader2 className="size-3 animate-spin" />
   334→            Salvando...
   335→          </span>
   336→        ) : null}
   337→      </div>
   338→
   339→      <Card className="overflow-hidden">
   340→        <CardContent className="p-0">
   341→          {isError ? (
   342→            <div className="flex flex-col items-center gap-3 px-4 py-16 text-center">
   343→              <div className="flex size-14 items-center justify-center rounded-full bg-rose-50 text-rose-600 dark:bg-rose-950/30 dark:text-rose-300">
   344→                <Wrench className="size-6" />
   345→              </div>
   346→              <div>
   347→                <p className="text-sm font-semibold">
   348→                  Não foi possível carregar os serviços
   349→                </p>
   350→                <p className="mt-0.5 text-xs text-muted-foreground">
   351→                  Verifique sua conexão e tente novamente.
   352→                </p>
   353→              </div>
   354→            </div>
   355→          ) : (
   356→            <div className="overflow-x-auto">
   357→              <Table>
   358→                <TableHeader>
   359→                  <TableRow className="h-11 bg-muted/50 hover:bg-muted/50">
   360→                    <TableHead className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   361→                      Serviço
   362→                    </TableHead>
   363→                    <TableHead className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   364→                      Prestador
   365→                    </TableHead>
   366→                    <TableHead className="hidden text-xs font-semibold uppercase tracking-wide text-muted-foreground md:table-cell">
   367→                      Categoria
   368→                    </TableHead>
   369→                    <TableHead className="text-right text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   370→                      Preço
   371→                    </TableHead>
   372→                    <TableHead className="text-center text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   373→                      Ativo
   374→                    </TableHead>
   375→                    <TableHead className="hidden text-xs font-semibold uppercase tracking-wide text-muted-foreground sm:table-cell">
   376→                      Criado
   377→                    </TableHead>
   378→                    <TableHead className="text-right text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   379→                      Ações
   380→                    </TableHead>
   381→                  </TableRow>
   382→                </TableHeader>
   383→                <TableBody>
   384→                  {isLoading ? (
   385→                    Array.from({ length: 6 }).map((_, i) => (
   386→                      <TableRow key={i} className="h-14">
   387→                        <TableCell colSpan={7}>
   388→                          <Skeleton className="h-9 w-full" />
   389→                        </TableCell>
   390→                      </TableRow>
   391→                    ))
   392→                  ) : items.length === 0 ? (
   393→                    <TableRow className="h-14 hover:bg-transparent">
   394→                      <TableCell colSpan={7} className="py-12">
   395→                        <div className="flex flex-col items-center gap-3 text-center">
   396→                          <div className="flex size-16 items-center justify-center rounded-full bg-muted text-muted-foreground">
   397→                            <Wrench className="size-7" />
   398→                          </div>
   399→                          <div>
   400→                            <p className="text-base font-semibold">
   401→                              Nenhum serviço encontrado
   402→                            </p>
   403→                            <p className="mt-0.5 text-sm text-muted-foreground">
   404→                              Ajuste os filtros ou aguarde novos cadastros.
   405→                            </p>
   406→                          </div>
   407→                          {activeFilterCount > 0 ? (
   408→                            <Button
   409→                              variant="outline"
   410→                              size="sm"
   411→                              onClick={clearFilters}
   412→                              className="gap-1.5"
   413→                            >
   414→                              <X className="size-3.5" />
   415→                              Limpar filtros
   416→                            </Button>
   417→                          ) : null}
   418→                        </div>
   419→                      </TableCell>
   420→                    </TableRow>
   421→                  ) : (
   422→                    items.map((s) => {
   423→                      const path = resolveCategoryPath(s.category, categories)
   424→                      const photo = firstPhoto(s.photos)
   425→                      return (
   426→                        <TableRow
   427→                          key={s.id}
   428→                          className={cn(
   429→                            "h-14 border-b transition-colors last:border-0 hover:bg-muted/30",
   430→                            !s.active && "opacity-70",
   431→                          )}
   432→                        >
   433→                          <TableCell className="px-4 py-3">
   434→                            <div className="flex items-center gap-2.5">
   435→                              <div className="flex size-10 shrink-0 items-center justify-center overflow-hidden rounded-md border bg-muted">
   436→                                {photo ? (
   437→                                  <img
   438→                                    src={photo}
   439→                                    alt={s.title}
   440→                                    className="size-full object-cover"
   441→                                  />
   442→                                ) : (
   443→                                  <ImageIcon className="size-4 text-muted-foreground" />
   444→                                )}
   445→                              </div>
   446→                              <div className="min-w-0">
   447→                                <p className="truncate text-sm font-medium">
   448→                                  {s.title}
   449→                                </p>
   450→                                <p className="truncate text-xs text-muted-foreground">
   451→                                  {s.description.slice(0, 80)}
   452→                                  {s.description.length > 80 ? "…" : ""}
   453→                                </p>
   454→                              </div>
   455→                            </div>
   456→                          </TableCell>
   457→                          <TableCell className="px-4 py-3">
   458→                            <div className="flex items-center gap-2">
   459→                              <Avatar className="size-7 shrink-0">
   460→                                {s.provider.avatarUrl ? (
   461→                                  <AvatarImage
   462→                                    src={s.provider.avatarUrl}
   463→                                    alt={s.provider.name}
   464→                                  />
   465→                                ) : null}
   466→                                <AvatarFallback className="bg-primary/10 text-[10px] font-semibold text-primary">
   467→                                  {initials(s.provider.name)}
   468→                                </AvatarFallback>
   469→                              </Avatar>
   470→                              <div className="min-w-0">
   471→                                <p className="truncate text-xs font-medium">
   472→                                  {s.provider.name}
   473→                                </p>
   474→                                <p className="truncate text-[10px] text-muted-foreground">
   475→                                  {s.provider.city ?? "—"}
   476→                                  {s.provider.state
   477→                                    ? `/${s.provider.state}`
   478→                                    : ""}
   479→                                </p>
   480→                              </div>
   481→                            </div>
   482→                          </TableCell>
   483→                          <TableCell className="hidden px-4 py-3 md:table-cell">
   484→                            {path.length > 0 ? (
   485→                              <div className="flex flex-wrap items-center gap-1">
   486→                                {path.map((c, i) => (
   487→                                  <React.Fragment key={c.id}>
   488→                                    {i > 0 ? (
   489→                                      <ChevronRight className="size-3 text-muted-foreground/60" />
   490→                                    ) : null}
   491→                                    <Badge
   492→                                      variant="outline"
   493→                                      className={cn(
   494→                                        "h-5 px-2 text-[10px] font-medium",
   495→                                        c.level === 0
   496→                                          ? "border-emerald-200 bg-emerald-50 text-emerald-700 dark:border-emerald-900/40 dark:bg-emerald-950/30 dark:text-emerald-300"
   497→                                          : c.level === 1
   498→                                            ? "border-teal-200 bg-teal-50 text-teal-700 dark:border-teal-900/40 dark:bg-teal-950/30 dark:text-teal-300"
   499→                                            : "border-slate-200 bg-slate-50 text-slate-600 dark:border-slate-700/40 dark:bg-slate-800/40 dark:text-slate-300",
   500→                                      )}
   501→                                    >
   502→                                      {c.name}
   503→                                    </Badge>
   504→                                  </React.Fragment>
   505→                                ))}
   506→                              </div>
   507→                            ) : (

   509→                                —
   510→                              </span>
   511→                            )}
   512→                          </TableCell>
   513→                          <TableCell className="px-4 py-3 text-right">
   514→                            <span className="text-sm font-semibold tabular-nums text-foreground">
   515→                              {formatBRL(s.basePrice)}
   516→                            </span>
   517→                            <span className="ml-1 text-[10px] text-muted-foreground">
   518→                              /{SERVICE_UNIT_SHORT[s.unit]}
   519→                            </span>
   520→                          </TableCell>
   521→                          <TableCell className="px-4 py-3 text-center">
   522→                            <Switch
   523→                              checked={s.active}
   524→                              onCheckedChange={(v) =>
   525→                                patchMutation.mutate({

   533→                            />
   534→                          </TableCell>
   535→                          <TableCell className="hidden px-4 py-3 text-xs text-muted-foreground tabular-nums sm:table-cell">
   536→                            {formatDate(s.createdAt)}
   537→                          </TableCell>
   538→                          <TableCell className="px-4 py-3 text-right">
   539→                            <DropdownMenu>
   540→                              <DropdownMenuTrigger asChild>
   541→                                <Button
   542→                                  variant="ghost"
   543→                                  size="icon"

   546→                                >

   550→                              <DropdownMenuContent align="end">

   576→                      )
   577→                    })
   578→                  )}
   579→                </TableBody>

   630→/** Resolve the parent chain (pai › filha › sub) from a leaf category. */
   631→function resolveCategoryPath(
   632→  leaf: CategoryOption | null | undefined,
   633→  all: CategoryOption[] | undefined,
   634→): CategoryOption[] {
   635→  if (!leaf || !all) return []
   636→  const byId = new Map(all.map((c) => [c.id, c]))
   637→  const path: CategoryOption[] = []
   638→  let current: CategoryOption | undefined = leaf
   639→  let safety = 0
   640→  while (current && safety < 5) {
   641→    path.unshift(current)
   642→    current = current.parentId ? byId.get(current.parentId) : undefined
   643→    safety++
   644→  }
   645→  return path
   646→}

   654→function initials(name: string): string {
   655→  if (!name) return "?"
   656→  const parts = name.trim().split(/\s+/)
   657→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase()
   658→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase()
   659→}
