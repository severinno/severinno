     4→ * AdminTaxonomy — gerenciador da árvore de categorias em 3 níveis.
     5→ *
     6→ * Heurísticas de Nielsen aplicadas neste redesign:
     7→ *   H1  Visibilidade do status  → PageSectionHeader + contagem + SavingPill
     8→ *   H2  Correspondência c/ mundo real → labels pt-BR; remoção do drag-handle

    23→ *   DELETE /api/categories/[id]       (409 se tiver filhos ou serviços)
    24→ *
    25→ * Observação H7: a API atual não expõe endpoint de reordenação. Para não enganar
    26→ * o administrador (H2), mantemos os botões "Mover p/ cima/baixo" visíveis mas

   200→type PendingToggle = {

   203→  currentValue: boolean

   206→export function AdminTaxonomy() {
   207→  const queryClient = useQueryClient()
   208→  const [createOpen, setCreateOpen] = React.useState(false)
   209→  const [editTarget, setEditTarget] = React.useState<CategoryNode | null>(null)
   210→  const [deleteTarget, setDeleteTarget] = React.useState<CategoryNode | null>(
   211→    null,
   212→  )
   213→  const [expanded, setExpanded] = React.useState<Set<string>>(new Set())
   214→  const [pendingToggle, setPendingToggle] = React.useState<PendingToggle>(null)
   215→
   216→  const { data: flat, isLoading, isError, refetch } = useQuery({
   217→    queryKey: ["categories", "all"],
   218→    queryFn: () => apiGet<CategoryRow[]>("/api/categories"),
   219→    staleTime: 30_000,
   220→  })
   221→
   222→  // Build the tree (from a flat list)
   223→  const tree = React.useMemo<CategoryNode[]>(() => {
   224→    if (!flat) return []
   225→    const byParent = new Map<string | null, CategoryNode[]>()
   226→    for (const c of flat) {
   227→      const key = c.parentId ?? null
   228→      const arr = byParent.get(key) ?? []
   229→      arr.push({ ...c, children: [] })
   230→      byParent.set(key, arr)
   231→    }
   232→    const roots = byParent.get(null) ?? []
   233→    const build = (nodes: CategoryNode[]): CategoryNode[] =>
   234→      nodes
   235→        .sort(
   236→          (a, b) =>
   237→            (a.order ?? 0) - (b.order ?? 0) || a.name.localeCompare(b.name),
   238→        )
   239→        .map((n) => {
   240→          const children = byParent.get(n.id) ?? []
   241→          n.children = build(children)
   242→          return n
   243→        })
   244→    return build(roots)
   245→  }, [flat])
   246→
   247→  // H1/H8 — service count por categoria não está disponível no MVP (o endpoint
   248→  // /api/categories não retorna contagem de serviços). Em vez de manter um
   249→  // badge que nunca mostra nada (H1 violado), simplesmente NÃO renderizamos.
   250→  // TODO(N1/Svcs): quando /api/services?categoryId=X existir, alimentar aqui.
   251→
   252→  // Auto-expand all level-0 nodes on first load
   253→  React.useEffect(() => {
   254→    if (tree.length && expanded.size === 0) {
   255→      setExpanded(new Set(tree.map((n) => n.id)))
   256→    }
   257→  }, [tree, expanded.size])
   258→
   259→  const invalidate = () =>
   260→    queryClient.invalidateQueries({ queryKey: ["categories"] })
   261→
   262→  const createMutation = useMutation({
   263→    mutationFn: (payload: CategoryPayload) =>
   264→      apiPost<{ category: CategoryRow }>("/api/categories", payload),
   265→    onSuccess: () => {
   266→      toast.success("Categoria criada com sucesso.")
   267→      invalidate()
   268→      setCreateOpen(false)
   269→    },
   270→    onError: (e: unknown) => {
   271→      toast.error(errMsg(e, "Não foi possível criar a categoria."))
   272→    },
   273→  })
   274→
   275→  const patchMutation = useMutation({
   276→    mutationFn: ({ id, patch }: { id: string; patch: Partial<CategoryPayload> }) =>
   277→      apiPatch<{ category: CategoryRow }>(`/api/categories/${id}`, patch),
   278→    onSuccess: () => {
   279→      toast.success("Categoria atualizada.")
   280→      invalidate()
   281→      setEditTarget(null)
   282→    },
   283→    onError: (e: unknown) => {
   284→      toast.error(errMsg(e, "Não foi possível atualizar a categoria."))
   285→    },
   286→  })
   287→
   288→  const toggleActiveMutation = useMutation({
   289→    mutationFn: ({ id, active }: { id: string; active: boolean }) =>
   290→      apiPatch<{ category: CategoryRow }>(`/api/categories/${id}`, { active }),
   291→    onSuccess: (_d, vars) => {
   292→      toast.success(
   293→        vars.active
   294→          ? "Categoria ativada."
   295→          : "Categoria desativada.",
   296→      )
   297→      invalidate()
   298→      setPendingToggle(null)
   299→    },
   300→    onError: (e: unknown) => {
   301→      toast.error(errMsg(e, "Falha ao alternar o estado ativo."))
   302→      setPendingToggle(null)
   303→    },
   304→  })
   305→
   306→  const deleteMutation = useMutation({
   307→    mutationFn: (id: string) => apiDelete(`/api/categories/${id}`),
   308→    onSuccess: () => {
   309→      toast.success("Categoria excluída.")
   310→      invalidate()
   311→      setDeleteTarget(null)
   312→    },
   313→    onError: (e: unknown) => {
   314→      const msg = errMsg(e, "Não foi possível excluir.")
   315→      // H9 — mensagem específica para 409 (vínculos existentes)
   316→      toast.error(msg, {
   317→        description:
   318→          "Remova os vínculos (filhos ou serviços) antes de tentar novamente.",
   319→      })
   320→    },
   321→  })
   322→
   323→  const toggleExpand = (id: string) =>
   324→    setExpanded((prev) => {
   325→      const next = new Set(prev)
   326→      if (next.has(id)) next.delete(id)
   327→      else next.add(id)
   328→      return next
   329→    })
   330→
   331→  // H5 — handler do confirm-toggle
   332→  const handleToggleConfirm = () => {
   333→    if (!pendingToggle) return
   334→    toggleActiveMutation.mutate({
   335→      id: pendingToggle.id,
   336→      active: !pendingToggle.currentValue,
   337→    })
   338→  }
   339→
   340→  const totalNodes = flat?.length ?? 0
   341→  const expandAll = () => {
   342→    if (!flat) return
   343→    setExpanded(new Set(flat.map((c) => c.id)))
   344→  }
   345→  const collapseAll = () => setExpanded(new Set())
   346→
   347→  return (
   348→    <div className="flex flex-col gap-4">
   349→      <PageSectionHeader
   350→        title="Taxonomia de categorias"
   351→        description="Organize a árvore em até 3 níveis: pai, filha e subcategoria."
   352→      />
   353→
   354→      {/* Toolbar — H8 minimalismo, agrupa contagem + ações de árvore */}
   355→      <div className="flex flex-wrap items-center justify-between gap-3 rounded-xl border bg-card p-3 shadow-sm">
   356→        <div className="flex flex-wrap items-center gap-2">
   357→          <span className="inline-flex items-center gap-1.5 rounded-full border bg-muted/40 px-2.5 py-1 text-xs font-medium text-muted-foreground">
   358→            <FolderTree className="size-3.5 text-primary" />
   359→            {totalNodes} {totalNodes === 1 ? "categoria" : "categorias"}
   360→          </span>
   361→          <span className="hidden items-center gap-1.5 text-xs text-muted-foreground sm:flex">
   362→            Níveis:
   363→            {([0, 1, 2] as const).map((lv) => (
   364→              <StatusBadge key={lv} tone={LEVEL_META[lv].tone}>
   365→                <span className={cn("size-1.5 rounded-full", LEVEL_META[lv].dot)} />
   366→                {LEVEL_META[lv].label}
   367→              </StatusBadge>
   368→            ))}
   369→          </span>
   370→        </div>
   371→        <div className="flex flex-wrap items-center gap-2">
   372→          <Button
   373→            variant="outline"
   374→            size="sm"
   375→            onClick={expandAll}
   376→            disabled={!flat || flat.length === 0}
   377→            className="h-8 gap-1.5"
   378→          >
   379→            <ChevronsUpDown className="size-3.5" />
   380→            Expandir tudo
   381→          </Button>
   382→          <Button
   383→            variant="outline"
   384→            size="sm"
   385→            onClick={collapseAll}
   386→            disabled={expanded.size === 0}
   387→            className="h-8 gap-1.5"
   388→          >
   389→            <ChevronsDownUp className="size-3.5" />
   390→            Recolher tudo
   391→          </Button>
   392→          <Button
   393→            size="sm"
   394→            onClick={() => setCreateOpen(true)}
   395→            className="h-8 gap-1.5"
   396→          >
   397→            <Plus className="size-4" />
   398→            Nova categoria
   399→          </Button>
   400→        </div>
   401→      </div>
   402→
   403→      {/* H9 — error recovery */}
   404→      {isError ? (
   405→        <ErrorRetry onRetry={() => refetch()} />
   406→      ) : (
   407→        <Card className="overflow-hidden">
   408→          <CardContent className="p-0">
   409→            {isLoading ? (
   410→              <div className="flex flex-col gap-2 p-4">
   411→                {Array.from({ length: 4 }).map((_, i) => (
   412→                  <Skeleton key={i} className="h-14 w-full" />
   413→                ))}
   414→              </div>
   415→            ) : tree.length === 0 ? (
   416→              <div className="flex flex-col items-center gap-3 px-4 py-16 text-center">
   417→                <div className="flex size-16 items-center justify-center rounded-full bg-muted text-muted-foreground">
   418→                  <FolderTree className="size-7" />
   419→                </div>
   420→                <div>
   421→                  <p className="text-lg font-semibold">
   422→                    Nenhuma categoria cadastrada
   423→                  </p>
   424→                  <p className="mt-0.5 text-sm text-muted-foreground">
   425→                    Crie a primeira categoria pai para iniciar a taxonomia.
   426→                  </p>
   427→                </div>
   428→                <Button
   429→                  size="sm"
   430→                  onClick={() => setCreateOpen(true)}
   431→                  className="gap-1.5"
   432→                >
   433→                  <Plus className="size-4" />
   434→                  Criar primeira categoria
   435→                </Button>
   436→              </div>
   437→            ) : (
   438→              <ul className="flex flex-col" role="tree">
   439→                {tree.map((node) => (
   440→                  <TreeNode
   441→                    key={node.id}
   442→                    node={node}
   443→                    level={0}
   444→                    expanded={expanded}
   445→                    onToggleExpand={toggleExpand}
   446→                    onEdit={(n) => setEditTarget(n)}
   447→                    onDelete={(n) => setDeleteTarget(n)}
   448→                    onRequestToggleActive={(n) =>
   449→                      setPendingToggle({
   450→                        id: n.id,
   451→                        name: n.name,
   452→                        currentValue: n.active,
   453→                      })
   454→                    }
   455→                  />
   456→                ))}
   457→              </ul>
   458→            )}
   459→          </CardContent>
   460→        </Card>
   461→      )}
   462→
   463→      {/* Create dialog */}
   464→      <CategoryDialog
   465→        open={createOpen}
   466→        onOpenChange={setCreateOpen}
   467→        title="Nova categoria"
   468→        description="Defina o nome, slug e o pai. O nível é determinado automaticamente pelo pai escolhido."
   469→        allCategories={(flat as CategoryRow[] | undefined) ?? []}
   470→        submitting={createMutation.isPending}
   471→        onSubmit={(p) => createMutation.mutate(p)}
   472→      />
   473→
   474→      {/* Edit dialog */}
   475→      <CategoryDialog
   476→        open={!!editTarget}
   477→        onOpenChange={(open) => !open && setEditTarget(null)}
   478→        title="Editar categoria"
   479→        description="Altere os campos abaixo. Mudar o pai pode alterar o nível da categoria."
   480→        allCategories={(flat as CategoryRow[] | undefined) ?? []}
   481→        initial={editTarget ?? undefined}
   482→        submitting={patchMutation.isPending}
   483→        onSubmit={(p) => {
   484→          if (!editTarget) return
   485→          patchMutation.mutate({ id: editTarget.id, patch: p })
   486→        }}
   487→      />
   488→
   489→      {/* H5 — confirmação de toggle (não-instantâneo) */}
   490→      <ConfirmToggleDialog
   491→        open={!!pendingToggle}
   492→        onOpenChange={(open) => !open && setPendingToggle(null)}
   493→        targetLabel={pendingToggle?.name ?? ""}
   494→        field="active"
   495→        currentValue={pendingToggle?.currentValue ?? false}
   496→        onConfirm={handleToggleConfirm}
   497→      />
   498→
   499→      {/* H3/H5 — confirmação de exclusão */}
   500→      <ConfirmDialog
   501→        open={!!deleteTarget}
   502→        onOpenChange={(open) => !open && setDeleteTarget(null)}
   503→        title="Excluir categoria?"
   504→        description={
   505→          <>
   506→            Você está prestes a excluir{" "}
   507→            <strong className="text-foreground">
   508→              {deleteTarget?.name}
   509→            </strong>
   510→            . Esta ação não pode ser desfeita. Categorias com filhos ou serviços
   511→            vinculados não podem ser excluídas.
   512→          </>
   513→        }
   514→        confirmLabel={deleteMutation.isPending ? "Excluindo…" : "Excluir"}
   515→        onConfirm={() => deleteTarget && deleteMutation.mutate(deleteTarget.id)}
   516→        variant="destructive"
   517→      />
   518→    </div>
   519→  )
   520→}

   522→// ---------------------------------------------------------------------------
   523→// TreeNode — recursivo
   524→// ---------------------------------------------------------------------------
   525→function TreeNode({
   526→  node,
   527→  level,
   528→  expanded,
   529→  onToggleExpand,
   530→  onEdit,
   531→  onDelete,
   532→  onRequestToggleActive,
   533→}: {
   534→  node: CategoryNode
   535→  level: number
   536→  expanded: Set<string>
   537→  onToggleExpand: (id: string) => void
   538→  onEdit: (n: CategoryNode) => void
   539→  onDelete: (n: CategoryNode) => void
   540→  onRequestToggleActive: (n: CategoryNode) => void
   541→}) {
   542→  const meta = LEVEL_META[level] ?? LEVEL_META[0]
   543→  const hasChildren = !!node.children && node.children.length > 0
   544→  const isOpen = expanded.has(node.id)
   545→  const hasIcon = !!node.icon && ICON_BY_NAME.has(node.icon)
   546→
   547→  return (
   548→    <li
   549→      role="treeitem"
   550→      aria-expanded={hasChildren ? isOpen : undefined}
   551→      aria-selected={false}
   552→    >
   553→      <div
   554→        className={cn(
   555→          "group relative flex flex-wrap items-center gap-2 border-b px-3 py-2.5 transition-colors hover:bg-muted/30 last:border-0",
   556→          meta.rowAccent,
   557→          !node.active && "opacity-60",
   558→        )}
   559→        style={{ paddingLeft: `${12 + level * 22}px` }}
   560→      >
   561→        {/* Connecting line (visual, níveis aninhados) */}
   562→        {level > 0 ? (
   563→          <span
   564→            aria-hidden
   565→            className="pointer-events-none absolute left-0 top-0 h-full border-l border-dashed border-border"
   566→            style={{ marginLeft: `${12 + (level - 1) * 22 + 8}px` }}
   567→          />
   568→        ) : null}
   569→
   570→        {/* Expand toggle */}
   571→        <button
   572→          type="button"

   760→function CategoryDialog({
   761→  open,

   768→  onSubmit,
   769→}: {

   771→  onOpenChange: (open: boolean) => void
   772→  title: string
   773→  description: string
   774→  allCategories: CategoryRow[]
   775→  initial?: CategoryNode

   778→}) {
   779→  const [name, setName] = React.useState("")
   780→  const [slug, setSlug] = React.useState("")
   781→  const [slugTouched, setSlugTouched] = React.useState(false)

   783→  const [icon, setIcon] = React.useState<string>("__none__")

   786→
   787→  // Reset form when opening
   788→  React.useEffect(() => {
   789→    if (!open) return
   790→    if (initial) {
   791→      setName(initial.name)
   792→      setSlug(initial.slug)
   793→      setSlugTouched(true)
   794→      setParentId(initial.parentId ?? "__none__")
   795→      setIcon(initial.icon && initial.icon.length > 0 ? initial.icon : "__none__")
   796→      setOrder(initial.order ?? 0)

   801→      setSlugTouched(false)

   827→    !submitting

   831→  const hasCurrentIcon = currentIconName.length > 0 && ICON_BY_NAME.has(currentIconName)
   832→  const isCustomIcon =
   833→    currentIconName.length > 0 && !ICON_BY_NAME.has(currentIconName)
   834→

   836→    e.preventDefault()
   837→    if (!canSubmit) return
   838→    onSubmit({
   839→      name: name.trim(),
   840→      slug,
   841→      parentId: parentId === "__none__" ? null : parentId,
   842→      level,
   843→      icon: currentIconName,

   861→              <Label htmlFor="cat-name">
   862→                Nome <span className="text-red-500">*</span>
   863→              </Label>
   864→              <Input
   865→                id="cat-name"
   866→                value={name}
   867→                onChange={(e) => setName(e.target.value)}
   868→                placeholder="Ex.: Reparos"
   869→                required
   870→                maxLength={80}
   871→              />
   872→            </div>
   873→            <div className="flex flex-col gap-1.5">
   874→              <div className="flex items-center gap-1">
   875→                <Label htmlFor="cat-slug">
   876→                  Slug <span className="text-red-500">*</span>
   877→                </Label>
   878→                <Tooltip>
   879→                  <TooltipTrigger asChild>
   880→                    <button
   881→                      type="button"
   882→                      className="text-muted-foreground transition-colors hover:text-foreground"
   883→                      aria-label="O que é um slug?"
   884→                    >
   885→                      <Info className="size-3.5" />
   886→                    </button>
   887→                  </TooltipTrigger>
   888→                  <TooltipContent
   889→                    side="right"
   890→                    className="max-w-xs text-xs leading-relaxed"
   891→                  >
   892→                    Identificador único usado nas URLs. Gerado
   893→                    automaticamente a partir do nome.
   894→                  </TooltipContent>
   895→                </Tooltip>
   896→              </div>
   897→              <Input
   898→                id="cat-slug"
   899→                value={slug}
   900→                onChange={(e) => {
   901→                  setSlug(e.target.value)

   905→                required
   906→                maxLength={80}

   910→              </p>
   911→            </div>

   915→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
