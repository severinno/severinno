     3→/**
     4→ * AdminSettings — editor dinâmico de configurações (console tipo .env).
     5→ *
     6→ * Heurísticas de Nielsen aplicadas neste redesign:
     7→ *   H1  Visibilidade do status  → FreshnessLabel + SavingPill no sticky bar
     8→ *   H3  Controle e liberdade    → Undo por campo (pilha de edições) + Descartar tudo
     9→ *   H4  Consistência            → UMA cor por grupo; StatusBadge p/ "Secreto"
    10→ *   H5  Prevenção de erros      → Eye/EyeOff p/ revelar segredos; warning banner

    16→ *        caption "Para remover, limpe o valor e salve."
    17→ *
    18→ * Data source: GET /api/admin/settings    → { items: Setting[], total }
    19→ *              POST /api/admin/settings   { settings: [{ key, value }] } (upsert)

   142→function lookupHelp(key: string): string | null {
   143→  if (SETTING_HELP[key]) return SETTING_HELP[key]
   144→  const upper = key.toUpperCase()
   145→  // Heurísticas por prefixo
   146→  if (upper.startsWith("PAYMENT_"))
   147→    return "Configuração relacionada ao gateway de pagamento."
   148→  if (upper.startsWith("SMTP_"))
   149→    return "Configuração do servidor de envio de e-mails (SMTP)."
   150→  if (upper.startsWith("EMAIL_"))
   151→    return "Configuração de remetente e provedor de e-mail."
   152→  if (upper.startsWith("NOMINATIM_") || upper.startsWith("GEO_"))
   153→    return "Configuração de geolocalização (endereço, CEP, mapa)."
   154→  if (upper.startsWith("SITE_"))
   155→    return "Configuração geral do site (branding, URL, nome)."
   156→  if (upper.startsWith("SUPPORT_"))
   157→    return "Canal de atendimento/suporte ao cliente."
   158→  if (upper.startsWith("WEATHER_"))
   159→    return "Integração com serviço de previsão do tempo."
   160→  return null
   161→}

   166→export function AdminSettings() {
   167→  const queryClient = useQueryClient()
   168→  const [createOpen, setCreateOpen] = React.useState(false)
   169→  const [draft, setDraft] = React.useState<Record<string, string>>({})
   170→  const [dirty, setDirty] = React.useState<Set<string>>(new Set())
   171→  // H3 — pilha de edições (para "Desfazer" reversível por campo)
   172→  const [editHistory, setEditHistory] = React.useState<string[]>([])
   173→  // H5 — chaves reveladas (segredos)
   174→  const [revealedKeys, setRevealedKeys] = React.useState<Set<string>>(new Set())
   175→  // H7 — busca por chave
   176→  const [query, setQuery] = React.useState("")
   177→  // H1 — última atualização
   178→  const [lastFetched, setLastFetched] = React.useState<Date | null>(null)
   179→
   180→  const { data, isLoading, isError, refetch, dataUpdatedAt } = useQuery({
   181→    queryKey: ["admin", "settings"],
   182→    queryFn: () => apiGet<SettingsResponse>("/api/admin/settings"),
   183→    staleTime: 30_000,
   184→  })
   185→
   186→  // H1 — sincroniza freshness label
   187→  React.useEffect(() => {
   188→    if (dataUpdatedAt) setLastFetched(new Date(dataUpdatedAt))
   189→  }, [dataUpdatedAt])
   190→
   191→  const items = data?.items ?? []
   192→
   193→  // Group by prefix (first segment before "_"); fallback to "Geral"
   194→  const groups = React.useMemo(() => {
   195→    const map = new Map<string, Setting[]>()
   196→    for (const s of items) {
   197→      const idx = s.key.indexOf("_")
   198→      const group = idx > 0 ? s.key.slice(0, idx) : "Geral"
   199→      const arr = map.get(group) ?? []
   200→      arr.push(s)
   201→      map.set(group, arr)
   202→    }
   203→    // Sort groups alphabetically, but keep "Geral" last
   204→    return Array.from(map.entries()).sort((a, b) => {
   205→      if (a[0] === "Geral") return 1
   206→      if (b[0] === "Geral") return -1
   207→      return a[0].localeCompare(b[0])
   208→    })
   209→  }, [items])
   210→
   211→  // H7 — quando buscando, lista flat filtrada por chave (case-insensitive)
   212→  const flatFiltered = React.useMemo(() => {
   213→    const q = query.trim().toLowerCase()
   214→    if (!q) return null
   215→    return items.filter((s) => s.key.toLowerCase().includes(q))
   216→  }, [items, query])
   217→
   218→  // Initialize draft when items arrive
   219→  React.useEffect(() => {
   220→    if (items.length === 0) return
   221→    setDraft((prev) => {
   222→      const next: Record<string, string> = { ...prev }
   223→      for (const s of items) {
   224→        if (!(s.key in next)) next[s.key] = s.value
   225→      }
   226→      return next
   227→    })
   228→  }, [items])
   229→
   230→  const upsertMutation = useMutation({
   231→    mutationFn: (settings: Array<{ key: string; value: string }>) =>
   232→      apiPost<{ items: Setting[] }>("/api/admin/settings", { settings }),
   233→    onSuccess: (_data, vars) => {
   234→      toast.success(
   235→        vars.length === 1
   236→          ? `Configuração "${vars[0].key}" salva.`
   237→          : `${vars.length} configurações salvas.`,
   238→      )
   239→      setDirty(new Set())
   240→      setEditHistory([])
   241→      queryClient.invalidateQueries({ queryKey: ["admin", "settings"] })
   242→    },
   243→    onError: (e: unknown) =>
   244→      toast.error(errMsg(e, "Não foi possível salvar as configurações.")),
   245→  })
   246→
   247→  const createMutation = useMutation({
   248→    mutationFn: (setting: { key: string; value: string }) =>
   249→      apiPost<{ items: Setting[] }>("/api/admin/settings", {
   250→        settings: [setting],
   251→      }),
   252→    onSuccess: (_d, vars) => {
   253→      toast.success(`Configuração "${vars.key}" criada.`)
   254→      queryClient.invalidateQueries({ queryKey: ["admin", "settings"] })
   255→      setCreateOpen(false)
   256→    },
   257→    onError: (e: unknown) =>
   258→      toast.error(errMsg(e, "Não foi possível criar a configuração.")),
   259→  })
   260→
   261→  // H3 — change tracking registra ordem das edições p/ undo por campo
   262→  const handleValueChange = (key: string, value: string) => {
   263→    setDraft((prev) => ({ ...prev, [key]: value }))
   264→    setDirty((prev) => {
   265→      const next = new Set(prev)
   266→      const original = items.find((s) => s.key === key)?.value
   267→      if (original !== value) next.add(key)
   268→      else next.delete(key)
   269→      return next
   270→    })
   271→    setEditHistory((prev) => {
   272→      // move key to end (most recent)
   273→      const next = prev.filter((k) => k !== key)
   274→      next.push(key)
   275→      return next
   276→    })
   277→  }
   278→
   279→  const saveAll = () => {
   280→    if (dirty.size === 0) return
   281→    const pairs = Array.from(dirty).map((key) => ({
   282→      key,
   283→      value: draft[key] ?? "",
   284→    }))
   285→    upsertMutation.mutate(pairs)
   286→  }
   287→
   288→  // H3 — desfaz apenas a última edição (mais controle que "Descartar tudo")
   289→  const undoLastEdit = () => {
   290→    if (editHistory.length === 0) return
   291→    const lastKey = editHistory[editHistory.length - 1]
   292→    const original = items.find((s) => s.key === lastKey)?.value ?? ""
   293→    setDraft((prev) => ({ ...prev, [lastKey]: original }))
   294→    setDirty((prev) => {
   295→      const next = new Set(prev)
   296→      next.delete(lastKey)
   297→      return next
   298→    })
   299→    setEditHistory((prev) => prev.slice(0, -1))
   300→    toast.success(`Alteração em "${lastKey}" desfeita.`)
   301→  }
   302→
   303→  // H3 — descarta TODAS as alterações
   304→  const resetAll = () => {
   305→    const next: Record<string, string> = {}
   306→    for (const s of items) next[s.key] = s.value
   307→    setDraft(next)
   308→    setDirty(new Set())
   309→    setEditHistory([])
   310→    toast.success("Todas as alterações foram descartadas.")
   311→  }
   312→
   313→  // H5 — toggle de revelar segredo
   314→  const toggleReveal = (key: string) => {
   315→    setRevealedKeys((prev) => {
   316→      const next = new Set(prev)
   317→      if (next.has(key)) next.delete(key)
   318→      else next.add(key)
   319→      return next
   320→    })
   321→  }
   322→
   323→  return (
   324→    <div className="flex flex-col gap-4 pb-24">
   325→      <PageSectionHeader
   326→        title="Configurações"
   327→        description="Chaves dinâmicas (estilo .env) que controlam pagamentos, e-mail, geolocalização e mais."
   328→        action={
   329→          lastFetched ? (
   330→            <FreshnessLabel updatedAt={lastFetched} />
   331→          ) : null
   332→        }
   333→      />
   334→
   335→      {/* Warning banner — H5 prevenção */}
   336→      <Alert className="border-amber-200 bg-amber-50 text-amber-900 dark:border-amber-800/50 dark:bg-amber-950/30 dark:text-amber-100">
   337→        <AlertTriangle className="size-4 text-amber-600 dark:text-amber-400" />
   338→        <AlertTitle className="text-sm font-semibold">
   339→          Estas configurações afetam todo o sistema. Edite com cuidado.
   340→        </AlertTitle>
   341→        <AlertDescription className="text-xs">
   342→          Alterações são salvas no banco e aplicadas instantaneamente. Erros
   343→          podem afetar pagamentos, e-mails e geolocalização.
   344→        </AlertDescription>
   345→      </Alert>
   346→
   347→      {/* Toolbar — H7 busca + contagem + H1 freshness + nova */}
   348→      <div className="flex flex-col gap-3 rounded-xl border bg-card p-3 shadow-sm sm:flex-row sm:items-center">
   349→        <div className="flex flex-1 flex-wrap items-center gap-2">
   350→          <span className="inline-flex items-center gap-1.5 text-xs text-muted-foreground">
   351→            {isLoading
   352→              ? "Carregando configurações..."
   353→              : `${items.length} configuração(ões) em ${groups.length} grupo(s)`}
   354→            {dirty.size > 0 ? (
   355→              <span className="ml-1 inline-flex items-center gap-1 font-medium text-primary">
   356→                · {dirty.size} pendente(s)
   357→              </span>
   358→            ) : null}
   359→          </span>
   360→          <SearchInput
   361→            value={query}
   362→            onChange={setQuery}
   363→            placeholder="Buscar por chave (ex.: PAYMENT_)"
   364→            className="min-w-[200px] flex-1"
   365→          />
   366→        </div>
   367→        <div className="flex shrink-0 items-center gap-2">
   368→          {lastFetched ? (
   369→            <span className="hidden sm:inline-flex">
   370→              <FreshnessLabel updatedAt={lastFetched} />
   371→            </span>
   372→          ) : null}
   373→          <Button
   374→            variant="outline"
   375→            size="sm"
   376→            onClick={() => setCreateOpen(true)}
   377→            className="gap-1.5"
   378→          >
   379→            <Plus className="size-3.5" />
   380→            Nova configuração
   381→          </Button>
   382→        </div>
   383→      </div>
   384→
   385→      {/* Body — error / loading / empty / list */}
   386→      {isError ? (
   387→        <ErrorState
   388→          title="Não foi possível carregar as configurações"
   389→          description="Verifique sua conexão e tente novamente."
   390→          onRetry={() => refetch()}
   391→        />
   392→      ) : isLoading ? (
   393→        <div className="flex flex-col gap-3">
   394→          {Array.from({ length: 3 }).map((_, i) => (
   395→            <Skeleton key={i} className="h-40 w-full" />
   396→          ))}
   397→        </div>
   398→      ) : items.length === 0 ? (
   399→        <EmptyState
   400→          icon={SettingsIcon}
   401→          title="Nenhuma configuração"
   402→          description="Crie a primeira configuração para começar."
   403→          action={
   404→            <Button
   405→              size="sm"
   406→              onClick={() => setCreateOpen(true)}
   407→              className="gap-1.5"
   408→            >
   409→              <Plus className="size-4" />
   410→              Nova configuração
   411→            </Button>
   412→          }
   413→        />
   414→      ) : flatFiltered ? (
   415→        // H7 — modo busca: flat list
   416→        <FlatResultsList
   417→          items={flatFiltered}
   418→          draft={draft}
   419→          dirty={dirty}
   420→          revealedKeys={revealedKeys}
   421→          onValueChange={handleValueChange}
   422→          onToggleReveal={toggleReveal}
   423→        />
   424→      ) : groups.length === 0 ? (
   425→        <EmptyState
   426→          icon={SettingsIcon}
   427→          title="Nenhum grupo encontrado"
   428→          description="Ajuste a busca ou crie uma nova configuração."
   429→        />
   430→      ) : (
   431→        <div className="grid grid-cols-1 gap-4 lg:grid-cols-2">
   432→          {groups.map(([group, groupItems]) => {
   433→            const meta = GROUP_META[group] ?? GROUP_META.default
   434→            const GroupIcon = meta.icon
   435→            return (
   436→              <Card key={group} className="overflow-hidden">
   437→                <div className="flex items-center justify-between border-b p-5">
   438→                  <div className="flex items-center gap-3">
   439→                    <span
   440→                      className={cn(
   441→                        "flex size-9 shrink-0 items-center justify-center rounded-lg",
   442→                        meta.tone,
   443→                      )}
   444→                    >
   445→                      <GroupIcon className="size-4" />
   446→                    </span>
   447→                    <div>
   448→                      <p className="text-base font-semibold">{meta.label}</p>
   449→                      <p className="text-xs text-muted-foreground">
   450→                        {meta.description}
   451→                      </p>
   452→                    </div>
   453→                  </div>
   454→                  <StatusBadge tone="zinc">
   455→                    {groupItems.length}{" "}
   456→                    {groupItems.length === 1 ? "chave" : "chaves"}
   457→                  </StatusBadge>
   458→                </div>
   459→                <CardContent className="flex flex-col gap-2.5 p-4">
   460→                  {groupItems.map((s) => (
   461→                    <SettingRow
   462→                      key={s.key}
   463→                      setting={s}
   464→                      draftValue={draft[s.key] ?? ""}
   465→                      isDirty={dirty.has(s.key)}
   466→                      isRevealed={revealedKeys.has(s.key)}
   467→                      onValueChange={handleValueChange}
   468→                      onToggleReveal={toggleReveal}
   469→                    />
   470→                  ))}
   471→                </CardContent>
   472→              </Card>
   473→            )
   474→          })}
   475→        </div>
   476→      )}
   477→
   478→      {/* H8 — UM único caminho de salvar: sticky bar (sem botão por linha) */}
   479→      {dirty.size > 0 ? (
   480→        <div className="fixed inset-x-0 bottom-0 z-30 border-t bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/80">
   481→          <div className="mx-auto flex max-w-5xl flex-wrap items-center justify-between gap-3 px-4 py-3">
   482→            <p className="text-sm text-muted-foreground">
   483→              <span className="font-medium text-foreground tabular-nums">
   484→                {dirty.size}
   485→              </span>{" "}
   486→              alteração(ões) não salva(s)
   487→              {editHistory.length > 0 ? (
   488→                <span className="ml-1 text-xs text-muted-foreground">
   489→                  · última:{" "}
   490→                  <code className="font-mono">{editHistory[editHistory.length - 1]}</code>
   491→                </span>
   492→              ) : null}
   493→            </p>
   494→            <div className="flex flex-wrap items-center gap-2">
   495→              {upsertMutation.isPending ? (
   496→                <span className="inline-flex items-center gap-1.5 text-xs text-amber-700 dark:text-amber-300">
   497→                  <Loader2 className="size-3.5 animate-spin" />
   498→                  Salvando…
   499→                </span>
   500→              ) : null}
   501→              <Button
   502→                variant="outline"
   503→                size="sm"
   504→                onClick={undoLastEdit}
   505→                disabled={
   506→                  upsertMutation.isPending || editHistory.length === 0
   507→                }
   508→                className="gap-1.5"
   509→              >
   510→                <Undo2 className="size-3.5" />
   511→                Desfazer
   512→                {editHistory.length > 0 ? (
   513→                  <span className="ml-0.5 rounded bg-muted px-1 text-[10px] tabular-nums">
   514→                    {editHistory.length}
   515→                  </span>
   516→                ) : null}
   517→              </Button>
   518→              <Button
   519→                variant="outline"
   520→                size="sm"
   521→                onClick={resetAll}
   522→                disabled={upsertMutation.isPending}
   523→                className="gap-1.5 text-muted-foreground hover:text-foreground"
   524→              >
   525→                <Trash2 className="size-3.5" />
   526→                Descartar tudo
   527→              </Button>
   528→              <Button
   529→                size="sm"
   530→                onClick={saveAll}
   531→                disabled={upsertMutation.isPending}

   749→// CreateSettingDialog
   750→// ---------------------------------------------------------------------------
   751→function CreateSettingDialog({
   752→  open,
   753→  onOpenChange,
   754→  existingKeys,
   755→  submitting,
   756→  onSubmit,
   757→}: {
   758→  open: boolean
   759→  onOpenChange: (open: boolean) => void
   760→  existingKeys: string[]
   761→  submitting: boolean
   762→  onSubmit: (payload: { key: string; value: string }) => void
   763→}) {
   764→  const [key, setKey] = React.useState("")
   765→  const [value, setValue] = React.useState("")
   766→
   767→  React.useEffect(() => {
   768→    if (open) {
   769→      setKey("")
   770→      setValue("")
   771→    }
   772→  }, [open])
   773→
   774→  const keyValid = /^[A-Z0-9_]+$/.test(key) && key.length >= 1
   775→  const dupe = existingKeys.includes(key.toUpperCase())
   776→  const help = keyValid && !dupe ? lookupHelp(key.toUpperCase()) : null
   777→
   778→  const canSubmit = keyValid && !dupe && !submitting
   779→
   780→  return (
   781→    <Dialog open={open} onOpenChange={onOpenChange}>
   782→      <DialogContent className="sm:max-w-md">
   783→        <DialogHeader>
   784→          <DialogTitle>Nova configuração</DialogTitle>
   785→          <DialogDescription>
   786→            Adicione um novo par chave/valor. A chave deve ter apenas letras
   787→            maiúsculas, números e underline.
   788→          </DialogDescription>
   789→        </DialogHeader>
   790→
   791→        <form
   792→          onSubmit={(e) => {
   793→            e.preventDefault()
   794→            if (!canSubmit) return
   795→            onSubmit({ key: key.toUpperCase(), value })
   796→          }}
   797→          className="flex flex-col gap-3 py-1"
   798→        >
   799→          <div className="flex flex-col gap-1.5">
   800→            <Label htmlFor="set-key">Chave</Label>
   801→            <Input
   802→              id="set-key"
   803→              value={key}
   804→              onChange={(e) =>
   805→                setKey(e.target.value.toUpperCase().replace(/[^A-Z0-9_]/g, ""))
   806→              }
   807→              placeholder="EX.: PAYMENT_API_KEY"
   808→              className="font-mono"
   809→              autoFocus
   810→            />
   811→            {dupe ? (
   812→              <p className="text-[11px] text-red-600">
   813→                Esta chave já existe.
   814→              </p>
   815→            ) : help ? (
   816→              <p className="flex items-start gap-1 text-[11px] text-muted-foreground">
   817→                <Info className="mt-0.5 size-3 shrink-0" />
   818→                {help}
   819→              </p>
   820→            ) : (
   821→              <p className="text-[11px] text-muted-foreground">
   822→                Convenção: <code>GRUPO_NOME</code> (ex.:{" "}
   823→                <code>PAYMENT_PIX_KEY</code>).
   824→              </p>
   825→            )}
   826→          </div>
   827→          <div className="flex flex-col gap-1.5">
   828→            <Label htmlFor="set-value">Valor</Label>
   829→            <Input
   830→              id="set-value"
   831→              value={value}
   832→              onChange={(e) => setValue(e.target.value)}
   833→              placeholder="Valor da configuração"
   834→              className="font-mono text-xs"
   835→            />
   836→          </div>
   837→          <DialogFooter>
   838→            <Button
   839→              type="button"
   840→              variant="outline"
   841→              onClick={() => onOpenChange(false)}
   842→              disabled={submitting}
   843→            >
   844→              Cancelar
   845→            </Button>
   846→            <Button type="submit" disabled={!canSubmit} className="gap-1.5">
   847→              {submitting ? (
   848→                <Loader2 className="size-4 animate-spin" />
   849→              ) : (
   850→                <Plus className="size-4" />
   851→              )}
   852→              Criar
   853→            </Button>
   854→          </DialogFooter>
   855→        </form>
   856→      </DialogContent>
   857→    </Dialog>
   858→  )
   859→}
