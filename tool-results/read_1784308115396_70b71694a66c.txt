   283→// ---------------------------------------------------------------------------
   284→// Hook: fetch public stats
   285→// ---------------------------------------------------------------------------
   286→
   287→function usePublicStats() {
   288→  const [stats, setStats] = React.useState<PublicStats | null>(null)
   289→
   290→  React.useEffect(() => {
   291→    let cancelled = false
   292→    async function load() {
   293→      try {
   294→        const res = await fetch("/api/stats/public")
   295→        if (!res.ok) return
   296→        const data: PublicStats = await res.json()
   297→        if (!cancelled) setStats(data)
   298→      } catch {
   299→        // silently ignore — fallback values used
   300→      }
   301→    }
   302→    load()
   303→    return () => { cancelled = true }
   304→  }, [])
   305→
   306→  return stats
   307→}
