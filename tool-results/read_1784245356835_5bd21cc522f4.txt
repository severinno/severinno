   875→function RevenueChart({
   876→  recentBookings,
   877→}: {
   878→  recentBookings: AdminStats["recentBookings"]
   879→}) {
   880→  // Build a 6-month series from the recentBookings sample (only 5 available —
   881→  // for MVP we synthesize a simple distribution around them so the chart isn't
   882→  // empty. Replace with a dedicated endpoint when available.)
   883→  const months: { label: string; revenue: number }[] = []
   884→  const now = new Date()
   885→  for (let i = 5; i >= 0; i--) {
   886→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1)
   887→    const label = d.toLocaleDateString("pt-BR", {
   888→      month: "short",
   889→      year: "2-digit",
   890→    })
   891→    months.push({ label, revenue: 0 })
   892→  }
   893→  // Distribute recent paid bookings across last 2 months as a stub.
   894→  for (const b of recentBookings) {
   895→    if (b.paymentStatus !== "PAID") continue
   896→    const created = new Date(b.createdAt)
   897→    const diffMonths =
   898→      (now.getFullYear() - created.getFullYear()) * 12 +
   899→      (now.getMonth() - created.getMonth())
   900→    const idx = months.length - 1 - diffMonths
   901→    if (idx >= 0 && idx < months.length) {
   902→      months[idx].revenue += b.amount
   903→    }
   904→  }
   905→
   906→  return (
   907→    <div className="h-[200px] w-full">
   908→      <ResponsiveContainer width="100%" height="100%">
   909→        <BarChart data={months} margin={{ left: 0, right: 8, top: 8, bottom: 0 }}>

   911→            vertical={false}
   912→            stroke="var(--border)"
   913→            strokeDasharray="3 3"

  1030→function initials(name: string): string {
  1031→  if (!name) return "?"
  1032→  const parts = name.trim().split(/\s+/)
  1033→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase()
  1034→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase()
