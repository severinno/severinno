   136→   136→export function AdminTaxonomy() {
   137→   137→  const queryClient = useQueryClient()
   138→   138→  const [createOpen, setCreateOpen] = React.useState(false)
   139→   139→  const [editTarget, setEditTarget] = React.useState<CategoryNode | null>(null)
   140→   140→  const [deleteTarget, setDeleteTarget] = React.useState<CategoryNode | null>(null)
   141→   141→  const [expanded, setExpanded] = React.useState<Set<string>>(new Set())
   142→   142→
   143→   143→  const { data: flat, isLoading, isError } = useQuery({
   144→   144→    queryKey: ["categories", "all"],
   145→   145→    queryFn: () => apiGet<CategoryRow[]>("/api/categories"),
   146→   146→    staleTime: 30_000,
   147→   147→  })
   148→   148→
   149→   149→  // Build the tree + counts (from a flat list)
   150→   150→  const tree = React.useMemo<CategoryNode[]>(() => {
   151→   151→    if (!flat) return []
   152→   152→    const byParent = new Map<string | null, CategoryNode[]>()
   153→   153→    for (const c of flat) {
   154→   154→      const key = c.parentId ?? null
   155→   155→      const arr = byParent.get(key) ?? []
   156→   156→      arr.push({ ...c, children: [] })
   157→   157→      byParent.set(key, arr)
   158→   158→    }
   159→   159→    const roots = byParent.get(null) ?? []
   160→   160→    const build = (nodes: CategoryNode[]): CategoryNode[] =>
   161→   161→      nodes
   162→   162→        .sort((a, b) => (a.order ?? 0) - (b.order ?? 0) || a.name.localeCompare(b.name))
   163→   163→        .map((n) => {
   164→   164→          const children = byParent.get(n.id) ?? []
   165→   165→          n.children = build(children)
   166→   166→          return n
   167→   167→        })
   168→   168→    return build(roots)
   169→   169→  }, [flat])
   170→   170→
   171→   171→  const serviceCounts = React.useMemo(() => {
   172→   172→    // Service count is fetched per-category lazily via a separate query —
   173→   173→    // but for MVP we only have the flat /api/categories (no counts). We'll
   174→   174→    // render counts if present on the node (admin extensions may add later).
   175→   175→    return new Map<string, number>()
   176→   176→  }, [])
   177→   177→
   178→   178→  // Auto-expand all level-0 nodes on first load
   179→   179→  React.useEffect(() => {
   180→   180→    if (tree.length && expanded.size === 0) {
   181→   181→      setExpanded(new Set(tree.map((n) => n.id)))
   182→   182→    }
   183→   183→  }, [tree, expanded.size])
   184→   184→

   186→   186→    queryClient.invalidateQueries({ queryKey: ["categories"] })
   187→   187→

   189→   189→    mutationFn: (payload: CategoryPayload) =>
   190→   190→      apiPost<{ category: CategoryRow }>("/api/categories", payload),
   191→   191→    onSuccess: () => {
   192→   192→      toast.success("Categoria criada com sucesso.")
   193→   193→      invalidate()
   194→   194→      setCreateOpen(false)
   195→   195→    },
   196→   196→    onError: (e: unknown) => {
   197→   197→      toast.error(errMsg(e, "Não foi possível criar a categoria."))
   198→   198→    },
   199→   199→  })
   200→   200→
   201→   201→  const patchMutation = useMutation({
   202→   202→    mutationFn: ({ id, patch }: { id: string; patch: Partial<CategoryPayload> }) =>
   203→   203→      apiPatch<{ category: CategoryRow }>(`/api/categories/${id}`, patch),
   204→   204→    onSuccess: () => {
   205→   205→      toast.success("Categoria atualizada.")
   206→   206→      invalidate()
   207→   207→      setEditTarget(null)
   208→   208→    },
   209→   209→    onError: (e: unknown) => {
   210→   210→      toast.error(errMsg(e, "Não foi possível atualizar a categoria."))
   211→   211→    },
   212→   212→  })
   213→   213→
   214→   214→  const toggleActiveMutation = useMutation({
   215→   215→    mutationFn: ({ id, active }: { id: string; active: boolean }) =>
   216→   216→      apiPatch<{ category: CategoryRow }>(`/api/categories/${id}`, { active }),
   217→   217→    onSuccess: () => {
   218→   218→      invalidate()
   219→   219→    },
   220→   220→    onError: (e: unknown) => {
   221→   221→      toast.error(errMsg(e, "Falha ao alternar o estado ativo."))
   222→   222→    },
   223→   223→  })
   224→   224→
   225→   225→  const deleteMutation = useMutation({
   226→   226→    mutationFn: (id: string) => apiDelete(`/api/categories/${id}`),
   227→   227→    onSuccess: () => {
   228→   228→      toast.success("Categoria excluída.")
   229→   229→      invalidate()
   230→   230→      setDeleteTarget(null)
   231→   231→    },
   232→   232→    onError: (e: unknown) => {
   233→   233→      const msg = errMsg(e, "Não foi possível excluir.")
   234→   234→      // Special 409 message
   235→   235→      toast.error(msg, {
   236→   236→        description:
   237→   237→          "Remova os vínculos (filhos ou serviços) antes de tentar novamente.",
   238→   238→      })
   239→   239→    },
   240→   240→  })
   241→   241→
   242→   242→  const toggleExpand = (id: string) =>
   243→   243→    setExpanded((prev) => {
   244→   244→      const next = new Set(prev)
   245→   245→      if (next.has(id)) next.delete(id)
   246→   246→      else next.add(id)
   247→   247→      return next
   248→   248→    })
   249→   249→
   250→   250→  if (isError) {
   251→   251→    return (
   252→   252→      <Card>
   253→   253→        <CardContent className="py-12 text-center text-sm text-muted-foreground">
   254→   254→          Não foi possível carregar a árvore de categorias.
   255→   255→        </CardContent>
   256→   256→      </Card>
   257→   257→    )
   258→   258→  }
   259→   259→
   260→   260→  const totalNodes = flat?.length ?? 0
   261→   261→  const expandAll = () => {
   262→   262→    if (!flat) return
   263→   263→    setExpanded(new Set(flat.map((c) => c.id)))
   264→   264→  }
   265→   265→  const collapseAll = () => setExpanded(new Set())
   266→   266→
   267→   267→  return (
   268→   268→    <div className="flex flex-col gap-4">
   269→   269→      {/* Toolbar */}
   270→   270→      <div className="flex flex-wrap items-center justify-between gap-3 rounded-xl border bg-card p-3 shadow-sm">
   271→   271→        <div className="flex flex-wrap items-center gap-2">
   272→   272→          <span className="inline-flex items-center gap-1.5 rounded-full border bg-muted/40 px-2.5 py-1 text-xs font-medium text-muted-foreground">
   273→   273→            <FolderTree className="size-3.5 text-primary" />
   274→   274→            {totalNodes} categoria(s)
   275→   275→          </span>
   276→   276→          <span className="hidden items-center gap-1.5 text-xs text-muted-foreground sm:flex">
   277→   277→            Níveis:
   278→   278→            <Badge
   279→   279→              variant="secondary"
   280→   280→              className="gap-1 bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-200"
   281→   281→            >
   282→   282→              <span className="size-1.5 rounded-full bg-emerald-500" />
   283→   283→              Pai
   284→   284→            </Badge>
   285→   285→            <Badge
   286→   286→              variant="secondary"
   287→   287→              className="gap-1 bg-teal-100 text-teal-800 dark:bg-teal-900/40 dark:text-teal-200"
   288→   288→            >
   289→   289→              <span className="size-1.5 rounded-full bg-teal-500" />
   290→   290→              Filha
   291→   291→            </Badge>
   292→   292→            <Badge
   293→   293→              variant="secondary"
   294→   294→              className="gap-1 bg-slate-200 text-slate-700 dark:bg-slate-700/40 dark:text-slate-200"
   295→   295→            >
   296→   296→              <span className="size-1.5 rounded-full bg-slate-400" />
   297→   297→              Sub
   298→   298→            </Badge>
   299→   299→          </span>
   300→   300→        </div>
   301→   301→        <div className="flex flex-wrap items-center gap-2">
   302→   302→          <Button
   303→   303→            variant="outline"
   304→   304→            size="sm"
   305→   305→            onClick={expandAll}
   306→   306→            disabled={!flat || flat.length === 0}
   307→   307→            className="h-8 gap-1.5"
   308→   308→          >
   309→   309→            <ChevronsUpDown className="size-3.5" />
   310→   310→            Expandir tudo
   311→   311→          </Button>
   312→   312→          <Button
   313→   313→            variant="outline"
   314→   314→            size="sm"
   315→   315→            onClick={collapseAll}
   316→   316→            disabled={expanded.size === 0}
   317→   317→            className="h-8 gap-1.5"
   318→   318→          >
   319→   319→            <ChevronsDownUp className="size-3.5" />
   320→   320→            Recolher tudo
   321→   321→          </Button>
   322→   322→          <Button
   323→   323→            size="sm"
   324→   324→            onClick={() => setCreateOpen(true)}
   325→   325→            className="h-8 gap-1.5"
   326→   326→          >
   327→   327→            <Plus className="size-4" />
   328→   328→            Nova categoria
   329→   329→          </Button>
   330→   330→        </div>
   331→   331→      </div>
   332→   332→
   333→   333→      {/* Tree */}
   334→   334→      <Card className="overflow-hidden">
   335→   335→        <CardContent className="p-0">
   336→   336→          {isLoading ? (
   337→   337→            <div className="flex flex-col gap-2 p-4">
   338→   338→              {Array.from({ length: 4 }).map((_, i) => (
   339→   339→                <Skeleton key={i} className="h-14 w-full" />
   340→   340→              ))}
   341→   341→            </div>
   342→   342→          ) : tree.length === 0 ? (
   343→   343→            <div className="flex flex-col items-center gap-3 px-4 py-16 text-center">
   344→   344→              <div className="flex size-16 items-center justify-center rounded-full bg-muted text-muted-foreground">
   345→   345→                <FolderTree className="size-7" />
   346→   346→              </div>
   347→   347→              <div>
   348→   348→                <p className="text-lg font-semibold">Nenhuma categoria cadastrada</p>
   349→   349→                <p className="mt-0.5 text-sm text-muted-foreground">
   350→   350→                  Crie a primeira categoria pai para iniciar a taxonomia.
   351→   351→                </p>
   352→   352→              </div>
   353→   353→              <Button size="sm" onClick={() => setCreateOpen(true)} className="gap-1.5">
   354→   354→                <Plus className="size-4" />
   355→   355→                Criar primeira categoria
   356→   356→              </Button>
   357→   357→            </div>
   358→   358→          ) : (
   359→   359→            <ul className="flex flex-col" role="tree">
   360→   360→              {tree.map((node) => (
   361→   361→                <TreeNode
   362→   362→                  key={node.id}
   363→   363→                  node={node}
   364→   364→                  level={0}
   365→   365→                  expanded={expanded}
   366→   366→                  onToggleExpand={toggleExpand}
   367→   367→                  onEdit={(n) => setEditTarget(n)}
   368→   368→                  onDelete={(n) => setDeleteTarget(n)}
   369→   369→                  onToggleActive={(n, active) =>
   370→   370→                    toggleActiveMutation.mutate({ id: n.id, active })
   371→   371→                  }
   372→   372→                  serviceCounts={serviceCounts}
   373→   373→                />
   374→   374→              ))}
   375→   375→            </ul>
   376→   376→          )}
   377→   377→        </CardContent>
   378→   378→      </Card>
   379→   379→
   380→   380→      {/* Create dialog */}
   381→   381→      <CategoryDialog
   382→   382→        open={createOpen}
   383→   383→        onOpenChange={setCreateOpen}
   384→   384→        title="Nova categoria"
   385→   385→        description="Defina o nome, slug e o pai. O nível é determinado automaticamente pelo pai escolhido."
   386→   386→        allCategories={(flat as CategoryRow[] | undefined) ?? []}
   387→   387→        submitting={createMutation.isPending}
   388→   388→        onSubmit={(p) => createMutation.mutate(p)}
   389→   389→      />
   390→   390→
   391→   391→      {/* Edit dialog */}
   392→   392→      <CategoryDialog
   393→   393→        open={!!editTarget}
   394→   394→        onOpenChange={(open) => !open && setEditTarget(null)}
   395→   395→        title="Editar categoria"
   396→   396→        description="Altere os campos abaixo. Mudar o pai pode alterar o nível da categoria."
   397→   397→        allCategories={(flat as CategoryRow[] | undefined) ?? []}
   398→   398→        initial={editTarget ?? undefined}
   399→   399→        submitting={patchMutation.isPending}
   400→   400→        onSubmit={(p) => {
   401→   401→          if (!editTarget) return
   402→   402→          patchMutation.mutate({ id: editTarget.id, patch: p })
   403→   403→        }}
   404→   404→      />
   405→   405→
   406→   406→      {/* Delete confirmation */}
   407→   407→      <AlertDialog
   408→   408→        open={!!deleteTarget}
   409→   409→        onOpenChange={(open) => !open && setDeleteTarget(null)}
   410→   410→      >
   411→   411→        <AlertDialogContent>
   412→   412→          <AlertDialogHeader>
   413→   413→            <AlertDialogTitle>Excluir categoria?</AlertDialogTitle>
   414→   414→            <AlertDialogDescription>
   415→   415→              Você está prestes a excluir{" "}
   416→   416→              <strong className="text-foreground">
   417→   417→                {deleteTarget?.name}
   418→   418→              </strong>
   419→   419→              . Esta ação não pode ser desfeita. Categorias com filhos ou
   420→   420→              serviços vinculados não podem ser excluídas.
   421→   421→            </AlertDialogDescription>
   422→   422→          </AlertDialogHeader>
   423→   423→          <AlertDialogFooter>
   424→   424→            <AlertDialogCancel disabled={deleteMutation.isPending}>
   425→   425→              Cancelar
   426→   426→            </AlertDialogCancel>
   427→   427→            <AlertDialogAction
   428→   428→              onClick={() => deleteTarget && deleteMutation.mutate(deleteTarget.id)}
   429→   429→              disabled={deleteMutation.isPending}
   430→   430→              className="gap-1.5 bg-red-600 hover:bg-red-700 focus-visible:ring-red-600"
   431→   431→            >
   432→   432→              {deleteMutation.isPending ? (
   433→   433→                <Loader2 className="size-4 animate-spin" />
   434→   434→              ) : (
   435→   435→                <Trash2 className="size-4" />
   436→   436→              )}
   437→   437→              Excluir
   438→   438→            </AlertDialogAction>
   439→   439→          </AlertDialogFooter>
   440→   440→        </AlertDialogContent>
   441→   441→      </AlertDialog>
   442→   442→    </div>
   443→   443→  )
   444→   444→}

   447→   447→// TreeNode — recursive
   448→   448→// ---------------------------------------------------------------------------
   449→   449→function TreeNode({
   450→   450→  node,
   451→   451→  level,
   452→   452→  expanded,
   453→   453→  onToggleExpand,
   454→   454→  onEdit,
   455→   455→  onDelete,
   456→   456→  onToggleActive,
   457→   457→  serviceCounts,
   458→   458→}: {
   459→   459→  node: CategoryNode
   460→   460→  level: number
   461→   461→  expanded: Set<string>
   462→   462→  onToggleExpand: (id: string) => void
   463→   463→  onEdit: (n: CategoryNode) => void
   464→   464→  onDelete: (n: CategoryNode) => void
   465→   465→  onToggleActive: (n: CategoryNode, active: boolean) => void
   466→   466→  serviceCounts: Map<string, number>
   467→   467→}) {
   468→   468→  const meta = LEVEL_META[level] ?? LEVEL_META[0]
   469→   469→  const hasChildren = !!node.children && node.children.length > 0
   470→   470→  const isOpen = expanded.has(node.id)
   471→   471→  const services = serviceCounts.get(node.id) ?? 0
   472→   472→
   473→   473→  return (
   474→   474→    <li
   475→   475→      role="treeitem"
   476→   476→      aria-expanded={hasChildren ? isOpen : undefined}
   477→   477→      aria-selected={false}
   478→   478→    >
   479→   479→      <div
   480→   480→        className={cn(
   481→   481→          "group flex items-center gap-2 border-b px-3 py-2.5 transition-colors hover:bg-muted/30 last:border-0",
   482→   482→          meta.rowAccent,
   483→   483→          !node.active && "opacity-60",
   484→   484→        )}
   485→   485→        style={{ paddingLeft: `${12 + level * 22}px` }}

   648→   648→function slugify(value: string): string {
   649→   649→  return value
   650→   650→    .toLowerCase()
   651→   651→    .normalize("NFD")
   652→   652→    .replace(/[\u0300-\u036f]/g, "")
   653→   653→    .replace(/[^a-z0-9]+/g, "-")
   654→   654→    .replace(/^-+|-+$/g, "")
   655→   655→    .slice(0, 80)
   656→   656→}

   658→   658→function CategoryDialog({
   659→   659→  open,
   660→   660→  onOpenChange,
   661→   661→  title,
   662→   662→  description,
   663→   663→  allCategories,

   665→   665→  submitting,
   666→   666→  onSubmit,
   667→   667→}: {
   668→   668→  open: boolean
   669→   669→  onOpenChange: (open: boolean) => void
