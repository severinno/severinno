   142→function cardNameValid(v: string) { return v.trim().length >= 3 }

   169→export function BookingModal() {
   170→  const open = useUIStore((s) => s.bookingModal.open)
   171→  const providerIdPreset = useUIStore((s) => s.bookingModal.providerId)
   172→  const serviceIdPreset = useUIStore((s) => s.bookingModal.serviceId)
   173→  const close = useUIStore((s) => s.closeBooking)
   174→  const openAuth = useUIStore((s) => s.openAuth)
   175→  const isMobile = useIsMobile()
   176→  const navigate = useViewStore((s) => s.navigate)
   177→  const user = useAuthStore((s) => s.user)
   178→
   179→  const [step, setStep] = React.useState<Step>(1)
   180→  const [state, setState] = React.useState<BookingFormState>(initialState())
   181→  const [submitting, setSubmitting] = React.useState(false)
   182→  const [touched, setTouched] = React.useState<Set<string>>(new Set())
   183→
   184→  // Reset when modal opens
   185→  React.useEffect(() => {
   186→    if (open) {
   187→      setStep(1)
   188→      setState(initialState())
   189→      setTouched(new Set())
   190→    }
   191→  }, [open])
   192→
   193→  // Fetch provider + services
   194→  const providerQuery = useQuery({
   195→    queryKey: ["provider", providerIdPreset],
   196→    queryFn: () =>
   197→      apiGet<ProviderDetail>(`/api/providers/${providerIdPreset}`),
   198→    enabled: open && !!providerIdPreset,
   199→    staleTime: 60 * 1000,
   200→  })
   201→
   202→  const servicesQuery = useQuery({
   203→    queryKey: ["services-by-provider", providerIdPreset],
   204→    queryFn: () =>
   205→      apiGet<ProviderService[]>("/api/services", { providerId: providerIdPreset }),
   206→    enabled: open && !!providerIdPreset,
   207→    staleTime: 60 * 1000,
   208→  })
   209→
   210→  const provider = providerQuery.data
   211→  const services = servicesQuery.data ?? []
   212→  const selectedService =
   213→    services.find((s) => s.id === serviceIdPreset) ?? services[0]
   214→
   215→  React.useEffect(() => {
   216→    if (open && selectedService) {
   217→      setState((s) => ({ ...s, quantity: 1 }))
   218→    }
   219→  }, [open, selectedService])
   220→
   221→  const set = <K extends keyof BookingFormState>(
   222→    key: K,
   223→    value: BookingFormState[K],
   224→  ) => setState((s) => ({ ...s, [key]: value }))
   225→
   226→  const markTouched = (field: string) =>
   227→    setTouched((prev) => new Set(prev).add(field))
   228→
   229→  // ── Step validation ──
   230→  const step1Valid = !!state.date && !!state.time
   231→  const step2Valid =
   232→    !!state.address.cep &&
   233→    state.address.cep.replace(/\D/g, "").length === 8 &&
   234→    !!state.address.street &&
   235→    !!state.address.number &&
   236→    !!state.address.city &&
   237→    !!state.address.state
   238→  const step3Valid =
   239→    state.paymentMethod === "PIX" ||
   240→    (state.paymentMethod === "CARD" &&
   241→      cardNameValid(state.cardName) &&
   242→      cardNumberValid(state.cardNumber) &&
   243→      cardExpiryValid(state.cardExpiry) &&
   244→      cardCvvValid(state.cardCvv))
   245→  // Step 4 is always valid (it's a review)
   246→
   247→  const validSteps: Record<number, boolean> = {
   248→    1: step1Valid,
   249→    2: step2Valid,
   250→    3: step3Valid,
   251→    4: true,
   252→  }
   253→
   254→  // ── Navigation ──
   255→  const handleStepClick = (target: Step) => {
   256→    if (target < step) {
   257→      setStep(target)
   258→      return
   259→    }
   260→    for (let i = 1; i < target; i++) {
   261→      if (!validSteps[i as Step]) {
   262→        toast.error("Complete os passos anteriores primeiro.")
   263→        return
   264→      }
   265→    }
   266→    setStep(target)
   267→  }
   268→
   269→  const handleNext = () => {
   270→    if (step === 1 && !step1Valid) {
   271→      toast.error("Selecione data e horário para continuar.")
   272→      return
   273→    }
   274→    if (step === 2 && !step2Valid) {
   275→      toast.error("Preencha o endereço completo.")
   276→      return
   277→    }
   278→    if (step === 3 && !step3Valid) {
   279→      toast.error("Verifique os dados de pagamento.")
   280→      return
   281→    }
   282→    setStep((s) => Math.min(4, s + 1) as Step)
   283→  }
   284→
   285→  const handleBack = () => setStep((s) => Math.max(1, s - 1) as Step)
   286→
   287→  const handleSubmit = async () => {
   288→    if (!user) {
   289→      toast.info("Faça cadastro gratuito para agendar serviços.")
   290→      openAuth("register", "CLIENT")
   291→      return
   292→    }
   293→    if (!provider || !selectedService || !state.date || !state.time) {
   294→      toast.error("Dados incompletos. Revise o agendamento.")
   295→      return
   296→    }
   297→
   298→    const [h, m] = state.time.split(":").map(Number)
   299→    const scheduledAt = new Date(state.date)
   300→    scheduledAt.setHours(h ?? 0, m ?? 0, 0, 0)
   301→    const amount = (selectedService.basePrice || 0) * (state.quantity || 1)
   302→
   303→    setSubmitting(true)
   304→    try {
   305→      await apiPost("/api/bookings", {
   306→        providerId: provider.id,
   307→        serviceId: selectedService.id,
   308→        scheduledAt: scheduledAt.toISOString(),
   309→        address: [
   310→          state.address.street,
   311→          state.address.number,
   312→          state.address.complement,
   313→          state.address.district,
   314→        ]
   315→          .filter(Boolean)
   316→          .join(", "),
   317→        cep: state.address.cep,
   318→        lat: state.address.lat,
   319→        lng: state.address.lng,
   320→        amount,
   321→        paymentMethod: state.paymentMethod,
   322→        notes: state.notes,
   323→        quantity: state.quantity,
   324→        unit: selectedService.unit,
   325→      })
   326→      toast.success("Agendamento confirmado! Acompanhe em seus agendamentos.")
   327→      close()
   328→      navigate("client.bookings")
   329→    } catch (e) {
   330→      const msg =
   331→        (e as { message?: string })?.message ??
   332→        "Não foi possível concluir o agendamento. Tente novamente."
   333→      toast.error(msg)
   334→    } finally {
   335→      setSubmitting(false)
   336→    }
   337→  }
   338→
   339→  // ── Step content ──
   340→  const stepContent = (() => {
   341→    switch (step) {
   342→      case 1:
   343→        return (
   344→          <Step1Schedule
   345→            state={state}
   346→            set={set}
   347→            availability={provider?.availability ?? []}
   348→            selectedService={selectedService}
   349→            provider={provider}
   350→            loading={providerQuery.isLoading || servicesQuery.isLoading}
   351→            isDesktop={!isMobile}
   352→          />
   353→        )
   354→      case 2:
   355→        return (
   356→          <Step2Details
   357→            state={state}
   358→            set={set}
   359→            provider={provider}
   360→            selectedService={selectedService}
   361→          />
   362→        )
   363→      case 3:
   364→        return (
   365→          <Step3Payment
   366→            state={state}
   367→            set={set}
   368→            markTouched={markTouched}
   369→            touched={touched}
   370→            selectedService={selectedService}
   371→          />
   372→        )
   373→      case 4:
   374→        return (
   375→          <Step4Confirmation
   376→            state={state}
   377→            provider={provider}
   378→            selectedService={selectedService}
   379→            goToStep={setStep}
   380→          />
   381→        )
   382→      default:
   383→        return null
   384→    }
   385→  })()
   386→
   387→  // Dynamic dialog sizing: wider on Step 1 for side-by-side calendar layout
   388→  const dialogSizeClass = step === 1
   389→    ? "sm:max-w-2xl"
   390→    : "sm:max-w-lg"
   391→
   392→  // ── Step indicator (reused for desktop custom layout) ──
   393→  const stepIndicator = (
   394→    <div className="border-b px-4 sm:px-5 py-3">
   395→      <div className="flex items-center justify-between">
   396→        {STEPS.map((s, i) => {
   397→          const active = step === s.id
   398→          const done = validSteps[s.id] && step > s.id
   399→          const Icon = s.icon
   400→          return (
   401→            <React.Fragment key={s.id}>
   402→              <button
   403→                type="button"
   404→                onClick={() => handleStepClick(s.id)}
   405→                disabled={!done && s.id > step}
   406→                className={cn(
   407→                  "flex items-center gap-1.5 text-xs transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring rounded-md px-1 py-0.5",
   408→                  active
   409→                    ? "text-emerald-700 dark:text-emerald-400 font-semibold"
   410→                    : done
   411→                      ? "text-emerald-600 cursor-pointer hover:text-emerald-700"
   412→                      : "text-muted-foreground cursor-default",
   413→                )}
   414→              >
   415→                <span
   416→                  className={cn(
   417→                    "inline-flex size-7 items-center justify-center rounded-full border-2 text-xs font-bold transition-all",
   418→                    active &&
   419→                      "border-emerald-600 bg-emerald-600 text-white shadow-sm shadow-emerald-600/25",
   420→                    done &&
   421→                      "border-emerald-600 bg-emerald-600 text-white cursor-pointer",
   422→                    !active &&
   423→                      !done &&
   424→                      "border-muted-foreground/20 text-muted-foreground",
   425→                  )}
   426→                >
   427→                  {done ? (
   428→                    <Check className="size-3.5" />
   429→                  ) : Icon ? (
   430→                    <Icon className="size-3.5" />
   431→                  ) : (
   432→                    s.id
   433→                  )}
   434→                </span>
   435→                <span className="hidden sm:inline">
   436→                  {s.shortLabel ?? s.label}
   437→                </span>
   438→              </button>
   439→              {i < STEPS.length - 1 && (
   440→                <div className="flex-1 h-px bg-muted-foreground/15 mx-1 sm:mx-2 relative">
   441→                  <div
   442→                    className="absolute inset-0 bg-emerald-500 transition-transform origin-left duration-300"
   443→                    style={{
   444→                      transform:
   445→                        validSteps[STEPS[i + 1]?.id] || step > s.id
   446→                          ? "scaleX(1)"
   447→                          : step === s.id && validSteps[s.id]
   448→                            ? "scaleX(0.5)"
   449→                            : "scaleX(0)",
   450→                    }}
   451→                  />
   452→                </div>
   453→              )}
   454→            </React.Fragment>
   455→          )
   456→        })}
   457→      </div>
   458→    </div>
   459→  )
   460→
   461→  // ── Footer buttons ──
   462→  const footerButtons = (
   463→    <div className="border-t bg-background/95 backdrop-blur px-4 sm:px-5 py-2.5 sticky bottom-0">
   464→      <div className="flex items-center justify-between gap-2">
   465→        {step > 1 ? (
   466→          <Button
   467→            type="button"
   468→            variant="ghost"
   469→            size="sm"
   470→            onClick={handleBack}
   471→            disabled={submitting}
   472→            className="text-muted-foreground hover:text-foreground h-9 gap-1"
   473→          >
   474→            <ChevronLeft className="size-4" />
   475→            Voltar
   476→          </Button>
   477→        ) : (
   478→          <div />
   479→        )}
   480→        {step < 4 ? (
   481→          <Button
   482→            type="button"
   483→            size="sm"
   484→            onClick={handleNext}
   485→            disabled={!validSteps[step]}
   486→            className="h-9 bg-emerald-600 hover:bg-emerald-700 gap-1.5"
   487→          >
   488→            Continuar
   489→            <ChevronRight className="size-4" />
   490→          </Button>
   491→        ) : (
   492→          <Button
   493→            type="button"
   494→            size="sm"
   495→            onClick={handleSubmit}
   496→            disabled={submitting || !validSteps[step]}
   497→            className="h-9 bg-emerald-600 hover:bg-emerald-700 gap-1.5"
   498→          >
   499→            {submitting && <Loader2 className="size-3.5 animate-spin" />}
   500→            <Send className="size-3.5" />
   501→            Confirmar agendamento
   502→          </Button>
   503→        )}
   504→      </div>
   505→    </div>
   506→  )
   507→
   508→  const wizardBody = (
   509→    <StepWizard
   510→      steps={STEPS}
   511→      currentStep={step}
   512→      validSteps={validSteps}
   513→      onStepClick={handleStepClick}
   514→      onBack={handleBack}
   515→      onNext={handleNext}
   516→      onSubmit={handleSubmit}
   517→      submitting={submitting}
   518→      currentStepValid={validSteps[step] ?? true}
   519→      submitLabel={
   520→        <>
   521→          <Send className="size-3.5" />
   522→          Confirmar agendamento
   523→        </>
   524→      }
   525→    >
   526→      {stepContent}
   527→    </StepWizard>
   528→  )
   529→
   530→  if (isMobile) {
   531→    return (
   532→      <Sheet open={open} onOpenChange={(o) => !o && close()}>
   533→        <SheetContent
   534→          side="bottom"
   535→          className="h-[100dvh] max-h-[100dvh] w-full p-0 sm:max-w-full gap-0 flex flex-col"
   536→          onInteractOutside={(e) => e.preventDefault()}
   537→        >
   538→          <SheetHeader className="px-4 pt-4 pb-2 shrink-0">
   539→            <SheetTitle className="flex items-center gap-2 text-base">
   540→              <CalendarDays className="size-4 text-emerald-600" />
   541→              Agendar serviço
   542→            </SheetTitle>
   543→            <SheetDescription className="text-xs">
   544→              Escolha data, detalhes e pagamento
   545→            </SheetDescription>
   546→          </SheetHeader>
   547→          <div className="flex-1 overflow-hidden">{wizardBody}</div>
   548→        </SheetContent>
   549→      </Sheet>
   550→    )
   551→  }
   552→
   553→  // ── Desktop: Custom layout with proper scrolling ──
   554→  return (
   555→    <Dialog open={open} onOpenChange={(o) => !o && close()}>
   556→      <DialogContent
   557→        className={cn(
   558→          "p-0 gap-0 flex flex-col max-h-[90vh] transition-[max-width] duration-300",
   559→          dialogSizeClass,
   560→        )}
   561→        onInteractOutside={(e) => e.preventDefault()}
   562→      >
   563→        {/* Header — Nielsen #3: User control */}
   564→        <DialogHeader className="px-5 pt-4 pb-2 border-b shrink-0">
   565→          <DialogTitle className="flex items-center gap-2 text-base">
   566→            <CalendarDays className="size-4 text-emerald-600" />
   567→            Agendar serviço
   568→          </DialogTitle>
   569→          <DialogDescription className="text-xs">
   570→            Escolha data, detalhes e pagamento
   571→          </DialogDescription>
   572→        </DialogHeader>
   573→
   574→        {/* Step indicator — always visible */}
   575→        {stepIndicator}
   576→
   577→        {/* Step content — scrollable */}
   578→        <div className="flex-1 overflow-y-auto min-h-0 px-4 sm:px-5 py-4">
   579→          <AnimatePresence mode="wait">
   580→            <motion.div
   581→              key={step}
   582→              initial={{ opacity: 0, x: 12 }}
   583→              animate={{ opacity: 1, x: 0 }}
   584→              exit={{ opacity: 0, x: -12 }}
   585→              transition={{ duration: 0.18 }}
   586→            >
   587→              {stepContent}
   588→            </motion.div>
   589→          </AnimatePresence>

   591→
   592→        {/* Footer — always visible at bottom */}
   593→        {footerButtons}
   594→      </DialogContent>
   595→    </Dialog>
   596→  )
   597→}

   603→// Nielsen #1: Visibility of system status — selected date/time highlighted
   604→// Nielsen #6: Recognition over recall — visible calendar, time slot buttons
   605→// Nielsen #8: Minimalist — clean two-column layout
   606→// Reference: shadcnspace.com Calendar 03 - Time Calendar
   607→// ---------------------------------------------------------------------------
   608→
   609→function Step1Schedule({
   610→  state,
   611→  set,
   612→  availability,
   613→  selectedService,
   614→  provider,
   615→  loading,
   616→  isDesktop,
   617→}: {
   618→  state: BookingFormState
   619→  set: <K extends keyof BookingFormState>(
   620→    key: K,
   621→    value: BookingFormState[K],
   622→  ) => void
   623→  availability: ProviderDetail["availability"]
   624→  selectedService?: ProviderService
   625→  provider?: ProviderDetail
   626→  loading: boolean
   627→  isDesktop: boolean
   628→}) {
   629→  // Generate slot list for selected date
